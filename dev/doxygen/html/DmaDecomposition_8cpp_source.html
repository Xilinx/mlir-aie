27int64_t maxLegalInputSizeForDim(const AIE::AIETargetModel &tm, int col, int row,
28 unsigned dim, uint64_t elemWidth,
29 uint32_t gran) {
30 uint32_t wrapBits = tm.getDmaBdWrapBits(col, row);
31 if (wrapBits == 0)
32 return 0;
33
34 if (dim == 0) {
35 int64_t maxHw = (1LL << wrapBits) - 1;
36 int64_t maxInput = maxHw * (int64_t)gran / (int64_t)elemWidth;
37 int64_t divisor = bdGranuleDivisor(elemWidth, gran);
38 if (divisor > 1)
39 maxInput = (maxInput / divisor) * divisor;
40 return maxInput;
41 }
42 if (dim == 3)
43 return 1LL << 6; // iteration wrap is 6 bits
44 return (1LL << wrapBits) - 1;
45}

47/// Enumerate divisors b of n in descending order (largest inner factor first).
48void divisorsDescending(int64_t n, SmallVectorImpl<int64_t> &out) {
49 out.clear();
50 if (n <= 1)
51 return;
52 SmallVector<int64_t, 32> small;
53 for (int64_t b = 2; b * b <= n; ++b) {
54 if (n % b != 0)
55 continue;
56 out.push_back(n / b);
57 if (b != n / b)
58 small.push_back(b);
59 }
60 llvm::sort(out);
61 std::reverse(out.begin(), out.end());
62 out.append(small.rbegin(), small.rend());
63}

65FailureOr<SmallVector<NdDmaPattern>>
66decomposeRecursive(Operation *forOp, BaseMemRefType bufType,
67 const AIE::AIETargetModel &tm, int col, int row,
68 const NdDmaPattern &pattern) {
69 if (patternPassesVerification(forOp, bufType, tm, col, row, pattern))
70 return SmallVector<NdDmaPattern>{pattern};
71
72 DataLayout dataLayout = DataLayout::closest(forOp);
73 uint64_t elemWidth = dataLayout.getTypeSizeInBits(bufType.getElementType());
74 uint32_t gran = tm.getAddressGenGranularity();
75
76 // The hardware BD emits elements in lexicographic order of the loop indices
77 // with d0 the innermost (fastest) and d3 the outermost (slowest) dimension.
78 // A decomposition is only correct if the concatenation of the sub-transfers'
79 // emitted element sequences is IDENTICAL (same order, not just same set) to
80 // the original. Both transformations below are order-preserving by
81 // construction.
82
83 // Outermost active dimension (highest index with size > 1).
84 int outermost = -1;
85 for (int i = 3; i >= 0; --i)
86 if (pattern.sizes[i] > 1) {
87 outermost = i;
88 break;
89 }
90 if (outermost < 0)
91 return failure();
92
93 // (1) Order-preserving dimension factoring: split dim d (size N = a*b,
94 // stride s) into an inner dim (b, s) kept at position d and an outer dim
95 // (a, b*s) inserted at position d+1, shifting the higher dims outward. The
96 // factored pair stays adjacent so the sub-traversal of dim d is contiguous
97 // and its place in the overall nesting is unchanged => element order is
98 // preserved. Requires the outermost slot to be free so no dim is dropped.
99 if (pattern.sizes[3] == 1) {
100 for (unsigned d = 0; d < 3; ++d) {
101 int64_t n = pattern.sizes[d];
102 if (n <= 1)
103 continue;
104 int64_t s = pattern.strides[d];
105
106 SmallVector<int64_t, 32> divisors;
107 divisorsDescending(n, divisors);
108 for (int64_t b : divisors) {
109 int64_t a = n / b;
110 if (a <= 1 || b <= 1)
111 continue;
112
113 // Both factors must remain granule-realizable on the innermost dim.
114 if (d == 0 && (!isConstMultipleOfGranule(b, elemWidth, gran)))
115 continue;
116
117 NdDmaPattern factored = pattern;
118 // Shift dims (d+1 .. 2) outward to (d+2 .. 3).
119 for (unsigned i = 3; i > d + 1; --i) {
120 factored.sizes[i] = pattern.sizes[i - 1];
121 factored.strides[i] = pattern.strides[i - 1];
122 factored.offsets[i] = pattern.offsets[i - 1];
123 }
124 factored.sizes[d] = b; // inner factor
125 factored.strides[d] = s; // inner keeps original stride/offset
126 factored.sizes[d + 1] = a; // outer factor
127 factored.strides[d + 1] = b * s; // outer stride
128 factored.offsets[d + 1] = 0;
129
130 auto sub = decomposeRecursive(forOp, bufType, tm, col, row, factored);
131 if (succeeded(sub))
132 return sub;
133 }
134 }
135 }
136
137 // (2) Order-preserving slicing: only the OUTERMOST active dimension may be
138 // split into contiguous index ranges emitted in order. Slicing an inner
139 // dimension would interleave the outer iterations and reorder the emitted
140 // element stream, so it is not allowed.
141 {
142 unsigned d = static_cast<unsigned>(outermost);
143 int64_t n = pattern.sizes[d];
144 int64_t chunkSize =
145 maxLegalInputSizeForDim(tm, col, row, d, elemWidth, gran);
146 if (chunkSize > 0 && chunkSize < n) {
147 int64_t numChunks = (n + chunkSize - 1) / chunkSize;
148 SmallVector<NdDmaPattern> combined;
149 for (int64_t i = 0; i < numChunks; ++i) {
150 NdDmaPattern slice = pattern;
151 slice.sizes[d] = std::min(chunkSize, n - i * chunkSize);
152 slice.offsets[d] = pattern.offsets[d] + i * chunkSize;
153
154 auto sub = decomposeRecursive(forOp, bufType, tm, col, row, slice);
155 // failed() above already guards this deref; the checker just doesn't
156 // associate FailureOr's failed()/succeeded() with its optional base.
157 if (failed(sub))
158 return failure();
159 SmallVector<NdDmaPattern> &subPatterns =
160 *sub; // NOLINT(bugprone-unchecked-optional-access)
161 combined.append(subPatterns.begin(), subPatterns.end());
162 }
163 return combined;
164 }
165 }
166
167 return failure();
168}

189bool AIEX::isDecomposableNdDmaPattern(Operation *forOp,
190 BaseMemRefType referencedBufType,
191 const AIE::AIETargetModel &targetModel,
192 int tileCol, int tileRow,
193 ArrayRef<int64_t> offsetsInnermostFirst,
194 ArrayRef<int64_t> sizesInnermostFirst,
195 ArrayRef<int64_t> stridesInnermostFirst) {
196 if (offsetsInnermostFirst.size() != kNdDmaDims ||
197 sizesInnermostFirst.size() != kNdDmaDims ||
198 stridesInnermostFirst.size() != kNdDmaDims)
199 return false;
200
201 if (isContiguousTransfer(sizesInnermostFirst, stridesInnermostFirst))
202 return false;
203
204 NdDmaPattern pattern;
205 pattern.offsets.assign(offsetsInnermostFirst.begin(),
206 offsetsInnermostFirst.end());
207 pattern.sizes.assign(sizesInnermostFirst.begin(), sizesInnermostFirst.end());
208 pattern.strides.assign(stridesInnermostFirst.begin(),
209 stridesInnermostFirst.end());
210
211 if (patternPassesVerification(forOp, referencedBufType, targetModel, tileCol,
212 tileRow, pattern))
213 return false;
214
215 auto decomposed = decomposeNdDmaPattern(forOp, referencedBufType, pattern,
216 targetModel, tileCol, tileRow);
217 // failed() above already guards this deref; the checker just doesn't
218 // associate FailureOr's failed()/succeeded() with its optional base.
219 if (failed(decomposed))
220 return false;
221 SmallVector<NdDmaPattern> &bds =
222 *decomposed; // NOLINT(bugprone-unchecked-optional-access)
223 return !bds.empty();
224}

226FailureOr<SmallVector<NdDmaPattern>>
227AIEX::decomposeNdDmaPattern(Operation *forOp, BaseMemRefType referencedBufType,
228 const NdDmaPattern &pattern,
229 const AIE::AIETargetModel &targetModel, int tileCol,
230 int tileRow) {
231 if (pattern.offsets.size() != kNdDmaDims ||
232 pattern.sizes.size() != kNdDmaDims ||
233 pattern.strides.size() != kNdDmaDims)
234 return failure();
235
236 if (isContiguousTransfer(pattern.sizes, pattern.strides))
237 return failure();
238
239 if (patternPassesVerification(forOp, referencedBufType, targetModel, tileCol,
240 tileRow, pattern))
241 return failure();
242
243 return decomposeRecursive(forOp, referencedBufType, targetModel, tileCol,
244 tileRow, pattern);

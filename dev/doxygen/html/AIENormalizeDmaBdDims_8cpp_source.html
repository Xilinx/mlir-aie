31 void runOnOperation() override {
32 DeviceOp device = getOperation();
33
34 device.walk([&](DMABDOp op) {
35 if (op.getMixedSizes().empty())
36 return;
37
38 for (OpFoldResult s : op.getMixedSizes())
39 if (!getConstantIntValue(s))
40 return;
41 for (OpFoldResult s : op.getMixedStrides())
42 if (!getConstantIntValue(s))
43 return;
44
45 if (op.getPadDimensions() && !op.getPadDimensions()->empty())
46 return;
47
48 std::optional<SmallVector<BDDimLayoutAttr>> maybeDims =
49 op.getConstantDimensions();
50 if (!maybeDims || maybeDims->empty())
51 return;
52 SmallVector<BDDimLayoutAttr> &origDims = *maybeDims;
53
54 SmallVector<BDDimLayoutAttr> newDims;
55 for (BDDimLayoutAttr dim : origDims) {
56 if (dim.getSize() != 1)
57 newDims.push_back(dim);
58 }
59
60 if (newDims.size() == origDims.size())
61 return;
62
63 int64_t product = 1;
64 for (BDDimLayoutAttr dim : origDims)
65 product *= dim.getSize();
66
67 if (!op.hasLen() && !op.getBuffer().getType().hasStaticShape())
68 return;
69 std::optional<int32_t> lenVal = op.getConstantLen();
70 if (op.getLen() && !lenVal)
71 return;
72 if (lenVal.has_value() && static_cast<int64_t>(*lenVal) != product)
73 return;
74
75 if (newDims.empty() || isContiguousBDTransfer(newDims)) {
76 int32_t len = static_cast<int32_t>(product);
77 op.getLenMutable().clear();
78 op.setStaticLen(len);
79 op.getSizesMutable().clear();
80 op.getStridesMutable().clear();
81 op.setStaticSizes(std::nullopt);
82 op.setStaticStrides(std::nullopt);
83 return;
84 }
85
86 SmallVector<int64_t> sizes, strides;
87 sizes.reserve(newDims.size());
88 strides.reserve(newDims.size());
89 for (BDDimLayoutAttr dim : newDims) {
90 sizes.push_back(static_cast<int64_t>(dim.getSize()));
91 strides.push_back(static_cast<int64_t>(dim.getStride()));
92 }
93
94 op.getSizesMutable().clear();
95 op.getStridesMutable().clear();
96 op.setStaticSizes(sizes);
97 op.setStaticStrides(strides);
98 });
99 }
100};

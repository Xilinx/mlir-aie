25// Named apart from xilinx::AIE::AIEArch, which several AIEVec sources also see.
26enum class AIEVecArch { AIE1, AIE2, AIE2P, AIE2PS };
27
28struct AIEVecTargetInfo {
29 AIEVecArch arch = AIEVecArch::AIE1;
30
31 bool isAIE1() const { return arch == AIEVecArch::AIE1; }
32 bool isAIE2() const { return arch == AIEVecArch::AIE2; }
33 bool isAIE2P() const { return arch == AIEVecArch::AIE2P; }
34 bool isAIE2PS() const { return arch == AIEVecArch::AIE2PS; }
35
36 // Whether vector-to-AIEVec conversion and AIEVec-to-LLVM lowering exist for
37 // this target. AIE2PS is accepted as a target, but has neither.
38 bool hasAIEVecLowering() const { return !isAIE2PS(); }
39};
40
41// Accepts "" and "aie" (AIE1), "aieml" and "aie2" (AIE2), "aie2p" and "aie2ps".
42// Returns std::nullopt for any other spelling.

45// As above, but also reports an unknown spelling as an error on `op`.

49// As above, and also reports an error on `op` if the target has no AIEVec

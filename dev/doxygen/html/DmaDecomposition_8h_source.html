29/// Number of ND addressing dimensions in the NPU DMA instruction encoding
30/// (d0..d2 wrap/stride + d3 iteration/repeat).
31static constexpr unsigned kNdDmaDims = 4;
32
33/// Innermost-first ND access pattern (d0..d3 / repeat), matching the
34/// convention used by verifyStridesWraps and NpuDmaMemcpyNdOp verification.
35struct NdDmaPattern {
36 llvm::SmallVector<int64_t, kNdDmaDims> offsets;
37 llvm::SmallVector<int64_t, kNdDmaDims> sizes;
38 llvm::SmallVector<int64_t, kNdDmaDims> strides;
39};

60/// Decompose an illegal pattern into one or more legal sub-patterns that move

55// The C++ that declines a build with a reason. `why` becomes a string
56// literal (quotes and backslashes escaped, line breaks flattened). When the
57// snippet goes through an emitc.verbatim with operands, `{}` is a
58// placeholder and `{{` its escape, so a `{` in the message is doubled; a
59// verbatim without operands is emitted literally.
60static std::string refuse(StringRef why, bool inFormat) {
61 std::string lit;
62 for (char ch : why) {
63 if (ch == '\n' || ch == '\r')
64 lit += ' ';
65 else if (ch == '"' || ch == '\\')
66 (lit += '\\') += ch;
67 else if (ch == '{' && inFormat)
68 lit += "{{";
69 else
70 lit += ch;
71 }
72 return "return aie_runtime::txn_refused(\"" + lit + "\");";
73}

86// The C++ variable name of a runtime BD pool, declared in the generated
87// function's prologue. One pool per tile and BD partition: channels sharing a
88// partition must share one pool, or they would hand out the same id twice.
89template <typename PoolOp>
90std::string bdPoolName(PoolOp op) {
91 return "bd_pool_" + std::to_string(op.getColumn()) + "_" +
92 std::to_string(op.getRow()) + "_" + std::to_string(op.getBdBegin()) +
93 "_" + std::to_string(op.getBdEnd());
94}

96// BD ids the static allocator has already placed on (col, row), read off the
97// placed aie.mem / aie.memtile_dma / aie.shim_dma chains. Sorted so the emitted
98// stream is deterministic.
99llvm::SmallVector<uint32_t, 8> staticBdIdsOnTile(AIE::DeviceOp device,
100 uint32_t col, uint32_t row) {
101 llvm::SmallVector<uint32_t, 8> ids;
102 for (AIE::DmaBody program : device.getOps<AIE::DmaBody>()) {
103 auto id = program.getTileID();
104 if (id.col != static_cast<int>(col) || id.row != static_cast<int>(row))
105 continue;
106 llvm::append_range(ids, AIE::getAssignedBdIds(program));
107 }
108 llvm::sort(ids);
109 ids.erase(llvm::unique(ids), ids.end());
110 return ids;
111}

121// Emit a call to a TxnEncoding.h free function on the txn vector. Scalar
122// operands pass through as-is; convert-arith-to-emitc later lowers them to C++
123// literals or function parameters, byte-identical to the static path.
124void emitTxnCall(OpBuilder &b, Location loc, StringRef fn, Value txnVec,
125 ValueRange args) {
126 SmallVector<Value> callArgs;
127 callArgs.push_back(txnVec);
128 callArgs.append(args.begin(), args.end());
129 emitc::CallOpaqueOp::create(b, loc, TypeRange{}, ("aie_runtime::" + fn).str(),
130 callArgs);

140class AIEXToEmitCConverter {
141public:
142 AIEXToEmitCConverter(emitc::FuncOp funcOp, Value txnVec,
143 const DeviceResolved &resolved, Value opCountVar,
144 bool foldDDRAddrOffset)
145 : funcOp(funcOp), txnVec(txnVec), resolved(resolved),
146 opCountVar(opCountVar), foldDDRAddrOffset(foldDDRAddrOffset) {}

148 // Convert every npu/pool op in the body, recursing into scf regions. Returns
149 // the compile-time op count for a straight-line body (the literal op_count),
150 // or nullopt on error. With a runtime op-count var (a loop is present) each
151 // op increments it instead and the returned count is unused.
152 std::optional<uint32_t> run() {
153 // DMA lowering consumes buffer views but can leave dead subview/cast
154 // chains behind (notably after configure/run materialization). They are
155 // not transaction operations and must not reach the C++ converter.
156 bool erased;
157 do {
158 erased = false;
159 funcOp.walk([&](Operation *op) {
160 if (isa<memref::MemRefDialect>(op->getDialect()) &&
161 isOpTriviallyDead(op)) {
162 op->erase();
163 erased = true;
164 }
165 });
166 } while (erased);
167 uint32_t count = 0;
168 SmallVector<Operation *> consumed;
169 convertBlockRecursive(funcOp.getBlocks().front(), count, consumed);
170 if (!ok)
171 return std::nullopt;
172
173 // Erase the converted npu ops, then DCE their now-dead operand clones (a
174 // get_global feeding a blockwrite, a folded-away address constant). Region-
175 // scoped runRegionDCE, not a module canonicalizer: the latter would also
176 // drop the shared memref.global the blockwrite data was inlined from, which
177 // other sequences in the module still reference.
178 for (Operation *op : llvm::reverse(consumed))
179 op->erase();
180 IRRewriter rewriter(funcOp.getContext());
181 (void)runRegionDCE(rewriter, funcOp.getFunctionBody());
182 return count;
183 }
184
185private:
186 // Convert the npu/pool ops in `block`, recursing through scf op regions so a
187 // rolled loop's body is lowered in place (the scf.for itself is left for the
188 // later convert-scf-to-emitc step). Collects converted ops into `consumed`.
189 void convertBlockRecursive(Block &block, uint32_t &count,
190 SmallVector<Operation *> &consumed) {
191 SmallVector<Operation *> work;
192 for (Operation &op : block)
193 work.push_back(&op);
194 for (Operation *op : work) {
195 if (!ok)
196 return;
197 // Structural emitc ops and arith (constant / runtime-value math) are left
198 // for the later arith-to-emitc step. Recurse into scf ops (and any other
199 // region-carrying control flow) to convert their bodies; the control-flow
200 // op itself is handled by convert-scf-to-emitc afterwards.
201 if (isa<emitc::EmitCDialect, arith::ArithDialect>(op->getDialect()))
202 continue;
203 if (isa<scf::SCFDialect>(op->getDialect())) {
204 for (Region &r : op->getRegions())
205 for (Block &b : r)
206 convertBlockRecursive(b, count, consumed);
207 continue;
208 }
209 convertOne(op, count);
210 if (!isa<memref::GetGlobalOp>(op))
211 consumed.push_back(op);
212 }
213 }

215private:
216 // Record one emitted txn op: bump the compile-time count and, when a runtime
217 // op-count variable exists (the sequence has a loop, so the count is not
218 // known at compile time), emit a `++__opcount;` so the header gets the true
219 // runtime total.
220 void countOp(OpBuilder &b, Location loc, uint32_t &count) {
221 ++count;
222 if (opCountVar)
223 emitc::VerbatimOp::create(b, loc, "++{};", ValueRange{opCountVar});
224 }

226 // Convert one op, appending to the txn vector and incrementing `count`.
227 // Scalar operands flow through as SSA (lowered later by
228 // convert-arith-to-emitc); only the address field is replaced by its
229 // device-resolved literal. Unsupported ops set `ok` false.
230 void convertOne(Operation *op, uint32_t &count) {
231 OpBuilder b(op);
232 Location loc = op->getLoc();
233
234 llvm::TypeSwitch<Operation *, void>(op)
235 .Case<AIEX::NpuWrite32Op>([&](auto w) {
236 // A compile-time address folds to a literal; a runtime address (the
237 // dynamic BD pool's bdBase + wordIdx, arith over a popped id) is
238 // passed through as SSA and lowered by convert-arith-to-emitc.
239 Value addrV = runtimeOrResolvedAddr(b, loc, w, w.getAddress());
240 if (!addrV)
241 return fail(w, "cannot convert a symbolic/unresolved write32 "
242 "address to the C++ TXN target");
243 emitTxnCall(b, loc, "txn_append_write32", txnVec,
244 {addrV, w.getValue()});
245 countOp(b, loc, count);
246 })
247 .Case<AIEX::NpuMaskWrite32Op>([&](auto mw) {
248 Value addrV = runtimeOrResolvedAddr(b, loc, mw, mw.getAddress());
249 if (!addrV)
250 return fail(mw, "cannot convert a symbolic/unresolved maskwrite32 "
251 "address to the C++ TXN target");
252 emitTxnCall(b, loc, "txn_append_maskwrite32", txnVec,
253 {addrV, mw.getValue(), mw.getMask()});
254 countOp(b, loc, count);
255 })
256 .Case<AIEX::NpuMaskPollOp>([&](auto mp) {
257 Value addrV = runtimeOrResolvedAddr(b, loc, mp, mp.getAddress());
258 if (!addrV)
259 return fail(mp, "cannot convert a symbolic/unresolved maskpoll "
260 "address to the C++ TXN target");
261 emitTxnCall(b, loc, "txn_append_maskpoll32", txnVec,
262 {addrV, mp.getValue(), mp.getMask()});
263 countOp(b, loc, count);
264 })
265 .Case<AIEX::NpuSyncOp>([&](auto s) {
266 emitTxnCall(b, loc, "txn_append_sync", txnVec,
267 {s.getColumn(), s.getRow(), s.getDirection(),
268 s.getChannel(), s.getColumnNum(), s.getRowNum()});
269 countOp(b, loc, count);
270 })
271 .Case<AIEX::NpuLoadPdiOp>([&](auto pdi) {
272 Value address = emitc::ConstantOp::create(
273 b, loc, emitc::OpaqueType::get(b.getContext(), "uint64_t"),
274 emitc::OpaqueAttr::get(b.getContext(),
275 std::to_string(pdi.getAddress()) + "ULL"));
276 emitTxnCall(b, loc, "txn_append_loadpdi", txnVec,
277 {u32Literal(b, loc, pdi.getId()),
278 u32Literal(b, loc, pdi.getSize()), address});
279 countOp(b, loc, count);
280 })
281 .Case<AIEX::NpuPreemptOp>([&](auto preempt) {
282 emitTxnCall(b, loc, "txn_append_preempt", txnVec,
283 {u32Literal(b, loc, preempt.getLevel())});
284 countOp(b, loc, count);
285 })
286 .Case<AIEX::NpuAddressPatchOp>([&](auto ap) {
287 // A runtime bd_id makes the patched register address runtime: prefer
288 // the SSA addr_val (flows through arith-to-emitc) over the constant.
289 Value addrV = ap.getAddrVal() ? ap.getAddrVal()
290 : u32Literal(b, loc, ap.getAddr());
291 std::optional<uint32_t> argIdx = ap.getArgIdx();
292 if (!argIdx)
293 return fail(ap, "address_patch still names its host buffer by SSA "
294 "value; run -aie-resolve-address-patch-buffers");
295 Value idxV = emitc::ConstantOp::create(
296 b, loc, emitc::OpaqueType::get(b.getContext(), "int32_t"),
297 emitc::OpaqueAttr::get(b.getContext(), std::to_string(*argIdx)));
298 // txn_append_arg_patch applies the DDR-aperture fold itself, so the
299 // rule lives only in TxnEncoding.h. Both operands are literals here,
300 // so -O2 folds the branch away.
301 Value foldV = emitc::ConstantOp::create(
302 b, loc, emitc::OpaqueType::get(b.getContext(), "bool"),
303 emitc::OpaqueAttr::get(b.getContext(),
304 foldDDRAddrOffset ? "true" : "false"));
305 Value argPlus = ap.getArgPlus();
306 // i32 offsets are unsigned bit patterns in the static emitter.
307 // Cast before widening so bit 31 is not sign-extended to 64 bits.
308 if (argPlus.getType().isInteger(32))
309 argPlus = emitc::CastOp::create(b, loc, getU32Type(b.getContext()),
310 argPlus);
311 emitTxnCall(b, loc, "txn_append_arg_patch", txnVec,
312 {addrV, idxV, argPlus, foldV});
313 countOp(b, loc, count);
314 })
315 .Case<AIEX::NpuBlockWriteOp>([&](auto bw) {
316 convertBlockWrite(b, loc, bw);
317 countOp(b, loc, count);
318 })
319 .Case<AIEX::NpuBlockWriteValuesOp>([&](auto bw) {
320 convertBlockWriteValues(b, loc, bw);
321 countOp(b, loc, count);
322 })
323 .Case<cf::AssertOp>([&](auto a) {
324 // Host-side guard: a violated condition yields no stream
325 // (std::nullopt), and its message is what the host reports. Appends
326 // nothing, so not counted.
327 emitc::VerbatimOp::create(
328 b, loc, "if (!({})) " + refuse(a.getMsg(), /*inFormat=*/true),
329 ValueRange{a.getArg()});
330 })
331 .Case<AIEX::DMABdPoolPopOp>([&](AIEX::DMABdPoolPopOp pop) {
332 // Draw a BD id from the tile's runtime pool (declared in the
333 // prologue) into a fresh C++ variable; nullopt if the pool is empty.
334 // Downstream ops consume the id, so replace the SSA result with a
335 // literal naming the variable.
336 std::string var = "bd_" + std::to_string(nextPoolVar++);
337 emitc::VerbatimOp::create(
338 b, loc,
339 "uint32_t " + var + "; if (!aie_runtime::bd_pool_pop(" +
340 bdPoolName(pop) + ", " + var + ")) " +
341 refuse("no free buffer descriptor on tile (" +
342 std::to_string(pop.getColumn()) + ", " +
343 std::to_string(pop.getRow()) +
344 "): the sequence keeps more transfers in flight "
345 "than the tile has descriptors",
346 /*inFormat=*/false));
347 Value ref =
348 emitc::LiteralOp::create(b, loc, pop.getBdId().getType(), var);
349 pop.getBdId().replaceAllUsesWith(ref);
350 })
351 .Case<AIEX::DMABdPoolPushOp>([&](AIEX::DMABdPoolPushOp push) {
352 // Return a BD id to the tile's runtime pool.
353 emitc::CallOpaqueOp::create(
354 b, loc, TypeRange{}, "aie_runtime::bd_pool_push",
355 ValueRange{poolRef(b, loc, bdPoolName(push)), push.getBdId()});
356 })
357 // XRT allocates the scratchpad only for a full-ELF context, which has
358 // no per-call instruction buffer to carry a generated stream.
359 .Case<AIEX::NpuCreateScratchpadOp, AIEX::NpuUpdateFromScratchpadOp>(
360 [&](Operation *sp) {
361 fail(sp, "needs the scratchpad of a full-ELF context, which the "
362 "C++ TXN target's per-call streams run without; use a "
363 "runtime sequence argument instead of a scratchpad "
364 "parameter");
365 })
366 // memref.get_global feeding a blockwrite is consumed by
367 // convertBlockWrite (data inlined); the now-dead op is erased later.
368 .Case<memref::GetGlobalOp>([&](auto) {})
369 .Default([&](Operation *unsupported) {
370 if (isa<scf::SCFDialect>(unsupported->getDialect()))
371 fail(unsupported,
372 "control flow in runtime sequences is not yet supported by "
373 "the C++ TXN target");
374 else
375 fail(unsupported, "is not supported by the C++ TXN target");
376 });
377 }

379 // The compile-time-resolved absolute address (folding buffer/col/row) as a
380 // uint32_t literal, or null if it could not be resolved statically. Address
381 // resolution depends on the parent device, so it is precomputed before the op
382 // is cloned out (see recordDeviceResolved).
383 Value resolvedAddr(OpBuilder &b, Location loc, Operation *clone) {
384 auto it = resolved.absoluteAddr.find(clone);
385 if (it == resolved.absoluteAddr.end())
386 return {};
387 return u32Literal(b, loc, it->second);
388 }
389
390 // The address for a write32/maskwrite32: the device-resolved literal, else a
391 // genuine runtime address (arith-derived, e.g. the pool's bdBase + wordIdx).
392 // A bare block argument selects no register and is rejected; null for a
393 // symbolic/unresolved address.
394 Value runtimeOrResolvedAddr(OpBuilder &b, Location loc, Operation *clone,
395 Value addrOperand) {
396 if (Value lit = resolvedAddr(b, loc, clone))
397 return lit;
398 Operation *def = addrOperand ? addrOperand.getDefiningOp() : nullptr;
399 if (def && !isa<arith::ConstantOp>(def))
400 return addrOperand;
401 return {};
402 }

418 void convertBlockWrite(OpBuilder &b, Location loc, AIEX::NpuBlockWriteOp bw) {
419 auto addrIt = resolved.absoluteAddr.find(bw);
420 auto dataIt = resolved.blockWriteData.find(bw);
421 if (addrIt == resolved.absoluteAddr.end() ||
422 dataIt == resolved.blockWriteData.end() || !dataIt->second) {
423 bw.emitOpError("cannot convert blockwrite without a constant address and "
424 "data to the C++ TXN target");
425 ok = false;
426 return;
427 }
428 uint32_t addr = addrIt->second;
429 DenseIntElementsAttr data = dataIt->second;
430
431 // Emit the payload as a typed EmitC array variable rather than a verbatim
432 // C++ string, so the generated code carries real types.
433 MLIRContext *ctx = b.getContext();
434 int64_t n = data.size();
435 auto arrTy =
436 emitc::ArrayType::get(ctx, SmallVector<int64_t>{n}, getU32Type(ctx));
437 std::string init = "{";
438 llvm::raw_string_ostream ss(init);
439 bool first = true;
440 for (APInt word : data.getValues<APInt>()) {
441 if (!first)
442 ss << ", ";
443 ss << llvm::format("0x%08xu", static_cast<unsigned>(word.getZExtValue()));
444 first = false;
445 }
446 ss << "}";
447 auto arrVar = emitc::VariableOp::create(b, loc, arrTy,
448 emitc::OpaqueAttr::get(ctx, init));
449
450 uint32_t colVal = 0, rowVal = 0;
451 if (auto col = bw.getColumn(), row = bw.getRow(); col && row) {
452 colVal = *col;
453 rowVal = *row;
454 }
455 emitTxnCall(b, loc, "txn_append_blockwrite", txnVec,
456 {u32Literal(b, loc, addr), arrVar.getResult(),
457 u32Literal(b, loc, static_cast<uint32_t>(n)),
458 u32Literal(b, loc, colVal), u32Literal(b, loc, rowVal)});
459 }
460
461 // A blockwrite whose payload is computed at TXN-build time:
462 // stage the words into a local C++ array, then hand that array
463 // to one txn_append_blockwrite. Unlike convertBlockWrite there is no constant
464 // memref to inline, so the slots are filled by assignment from the operand
465 // values, which convert-arith-to-emitc lowers in place afterwards.
466 void convertBlockWriteValues(OpBuilder &b, Location loc,
467 AIEX::NpuBlockWriteValuesOp bw) {
468 MLIRContext *ctx = b.getContext();
469 Value addrV = runtimeOrResolvedAddr(b, loc, bw, bw.getAddress());
470 if (!addrV) {
471 fail(bw, "cannot convert a symbolic/unresolved blockwrite_values address "
472 "to the C++ TXN target");
473 return;
474 }
475 ValueRange words = bw.getValues();
476 int64_t n = static_cast<int64_t>(words.size());
477
478 // Declare the staging array (zero-initialized; every slot is assigned
479 // below). Typed emitc.variable, matching convertBlockWrite's array.
480 auto arrTy =
481 emitc::ArrayType::get(ctx, SmallVector<int64_t>{n}, getU32Type(ctx));
482 auto arrVar = emitc::VariableOp::create(b, loc, arrTy,
483 emitc::OpaqueAttr::get(ctx, "{}"));
484
485 // Fill the slots. Emitted as verbatim assignments (the same mechanism the
486 // runtime guards above use to reference a not-yet-lowered SSA value): the
487 // i32 words convert to the array's uint32_t elementwise, which is the
488 // intended two's-complement reinterpretation of the packed BD fields.
489 for (int64_t i = 0; i < n; i++)
490 emitc::VerbatimOp::create(b, loc, "{}[" + std::to_string(i) + "] = {};",
491 ValueRange{arrVar.getResult(), words[i]});
492
493 // col/row are 0: the address is already flat (as for npu.blockwrite).
494 emitTxnCall(b, loc, "txn_append_blockwrite", txnVec,
495 {addrV, arrVar.getResult(),
496 u32Literal(b, loc, static_cast<uint32_t>(n)),
497 u32Literal(b, loc, 0), u32Literal(b, loc, 0)});
498 }

503 // Runtime op-count variable (the C++ `__opcount`), or null when the sequence
504 // is straight-line and the count is a compile-time literal.
505 Value opCountVar;
506 bool ok = true;
507 // Counter for unique popped-BD-id C++ variable names.
508 unsigned nextPoolVar = 0;
509 bool foldDDRAddrOffset;
510};

512struct ConvertAIEXToEmitCPass
513 : xilinx::impl::ConvertAIEXToEmitCBase<ConvertAIEXToEmitCPass> {
514 ConvertAIEXToEmitCPass() = default;
515 // foldDDRAddrOffset is `protected` on the generated base, settable only via
516 // a -pass-pipeline string; this constructor is the programmatic equivalent.
517 explicit ConvertAIEXToEmitCPass(const ConvertAIEXToEmitCOptions &options)
518 : ConvertAIEXToEmitCBase(options) {}
519
520 void runOnOperation() override {
521 ModuleOp moduleOp = getOperation();
522 MLIRContext *ctx = &getContext();
523
524 struct SeqInfo {
525 AIE::RuntimeSequenceOp seq;
526 AIE::DeviceOp device;
527 };
528 SmallVector<SeqInfo> sequences;
529 moduleOp.walk([&](AIE::RuntimeSequenceOp seq) {
530 sequences.push_back({seq, seq->getParentOfType<AIE::DeviceOp>()});
531 });
532 if (emitDispatchShim && sequences.size() != 1) {
533 moduleOp.emitOpError()
534 << "emit-dispatch-shim needs exactly one aie.runtime_sequence to "
535 "wrap, found "
536 << sequences.size();
537 return signalPassFailure();
538 }
539 if (sequences.empty())
540 return;
541
542 OpBuilder builder(ctx);
543 for (auto &[seqOp, deviceOp] : sequences) {
544 if (!deviceOp) {
545 seqOp.emitOpError("must be nested inside an aie.device");
546 return signalPassFailure();
547 }
548 std::optional<emitc::FuncOp> gen =
549 emitFunction(builder, moduleOp, seqOp, deviceOp);
550 if (!gen)
551 return signalPassFailure();
552 if (emitDispatchShim) {
553 emitc::FuncOp generated = *gen;
554 if (failed(emitDispatchShimFuncs(builder, moduleOp, generated)))
555 return signalPassFailure();
556 }
557 }
558
559 // Replace the module body with just the generated emitc funcs + includes.
560 SmallVector<Operation *> toErase;
561 for (Operation &op : *moduleOp.getBody())
562 if (!isa<emitc::FuncOp, emitc::IncludeOp>(op))
563 toErase.push_back(&op);
564 for (Operation *op : llvm::reverse(toErase))
565 op->erase();
566
567 // Lower the arith ops (constant scalar fields, runtime-value arithmetic
568 // feeding npu ops) and any scf control flow (a rolled dynamic loop) left in
569 // the function bodies to emitc.
570 if (failed(lowerArithAndScfToEmitC(moduleOp)))
571 return signalPassFailure();
572
573 builder.setInsertionPointToStart(moduleOp.getBody());
574 emitc::IncludeOp::create(builder, moduleOp.getLoc(),
575 "aie/Runtime/TxnEncoding.h",
576 /*is_standard_include=*/false);
577 emitc::IncludeOp::create(builder, moduleOp.getLoc(), "cstdint",
578 /*is_standard_include=*/true);
579 emitc::IncludeOp::create(builder, moduleOp.getLoc(), "vector",
580 /*is_standard_include=*/true);
581 // The builder returns std::optional: a runtime scalar that would overflow a
582 // narrow BD field yields std::nullopt instead of a truncated stream.
583 emitc::IncludeOp::create(builder, moduleOp.getLoc(), "optional",
584 /*is_standard_include=*/true);
585 }
586
587 // Run the upstream arith-to-emitc and scf-to-emitc conversions over the
588 // generated functions in one pass: a rolled dynamic loop is scf.for, and its
589 // bounds / iter_args are arith, so both must lower together.
590 LogicalResult lowerArithAndScfToEmitC(ModuleOp moduleOp) {
591 // ArithToEmitC has no patterns for the integer ops the Python bindings
592 // emit for `//`, ceildiv, min and max on staged scalars. Expand them to
593 // the divsi/cmpi/select forms it does lower before converting.
594 {
595 RewritePatternSet expand(moduleOp.getContext());
596 arith::populateCeilFloorDivExpandOpsPatterns(expand);
597 arith::populateExpandMinMaxIPatterns(expand);
598 // Rewrite only: folding or CSE here would reshape the sequence the
599 // per-op conversion below expects (and the C++ its tests check).
600 GreedyRewriteConfig config;
601 config.enableFolding(false).enableConstantCSE(false);
602 if (failed(applyPatternsGreedily(moduleOp, std::move(expand), config)))
603 return failure();
604 }
605
606 TypeConverter typeConverter;
607 typeConverter.addConversion([](Type t) { return t; });
608 populateEmitCSizeTTypeConversions(typeConverter);
609
610 RewritePatternSet patterns(moduleOp.getContext());
611 populateArithToEmitCPatterns(typeConverter, patterns);
612 populateSCFToEmitCConversionPatterns(patterns, typeConverter);
613
614 ConversionTarget target(*moduleOp.getContext());
615 target.addLegalDialect<emitc::EmitCDialect>();
616 target.addIllegalDialect<arith::ArithDialect>();
617 target.addIllegalDialect<scf::SCFDialect>();
618 return applyPartialConversion(moduleOp, target, std::move(patterns));
619 }

622 // The C spelling emitc's translator emits for a parameter type, for the ABI
623 // string only. Must mirror CppEmitter::emitType down to its signedness rule,
624 // or the reported ABI disagrees with the emitted signature.
625 static std::optional<std::string> cTypeName(Type t) {
626 if (isa<emitc::SizeTType>(t))
627 return std::string("size_t");
628 auto intTy = dyn_cast<IntegerType>(t);
629 if (!intTy || !emitc::isSupportedIntegerType(intTy))
630 return std::nullopt;
631 if (intTy.getWidth() == 1)
632 return std::string("bool");
633 return (intTy.isUnsigned() ? "uint" : "int") +
634 std::to_string(intTy.getWidth()) + "_t";
635 }
636
637 // Emit the ctypes-callable entry points: dispatch_abi() reporting the
638 // parameter types, and dispatch_generate() handing back a pointer + word
639 // count into thread-local storage, or -2 when the builder declined.
640 LogicalResult emitDispatchShimFuncs(OpBuilder &builder, ModuleOp moduleOp,
641 emitc::FuncOp gen) {
642 MLIRContext *ctx = moduleOp.getContext();
643 Location loc = gen.getLoc();
644 TypeRange genParams = gen.getFunctionType().getInputs();
645
646 SmallVector<std::string> abiTypes;
647 for (Type t : genParams) {
648 std::optional<std::string> name = cTypeName(t);
649 if (!name)
650 return gen.emitOpError()
651 << "runtime-sequence parameter type " << t
652 << " has no dispatch ABI spelling; it cannot be passed from the "
653 "host at dispatch time";
654 abiTypes.push_back(*name);
655 }
656
657 builder.setInsertionPointToEnd(moduleOp.getBody());
658 // Python resolves these two by name out of the built shared library. ELF
659 // exports every extern symbol unless told otherwise, but MSVC-target
660 // linking exports nothing without an explicit marker, so on Windows the
661 // DLL would load and then fail to resolve dispatch_abi. The macro comes
662 // from TxnEncoding.h, which the emitted file already includes, and expands
663 // to nothing off Windows.
664 auto externC =
665 builder.getStrArrayAttr({"extern \"C\"", "AIE_DISPATCH_EXPORT"});
666
667 // Pointers must be emitc.ptr: an opaque type may not have a pointer as its
668 // outer type.
669 auto charPtrTy =
670 emitc::PointerType::get(emitc::OpaqueType::get(ctx, "const char"));
671 auto abiFn = emitc::FuncOp::create(builder, loc, "dispatch_abi",
672 FunctionType::get(ctx, {}, {charPtrTy}));
673 abiFn.setSpecifiersAttr(externC);
674 OpBuilder ab = OpBuilder::atBlockBegin(abiFn.addEntryBlock());
675 Value abiStr = emitc::LiteralOp::create(
676 ab, loc, charPtrTy, "\"" + llvm::join(abiTypes, ",") + "\"");
677 emitc::ReturnOp::create(ab, loc, abiStr);
678
679 SmallVector<Type> shimParams(genParams);
680 shimParams.push_back(
681 emitc::PointerType::get(emitc::PointerType::get(getU32Type(ctx))));
682 auto i64Ty = emitc::OpaqueType::get(ctx, "int64_t");
683 auto shimFn =
684 emitc::FuncOp::create(builder, loc, "dispatch_generate",
685 FunctionType::get(ctx, shimParams, {i64Ty}));
686 shimFn.setSpecifiersAttr(externC);
687 Block *shimBlock = shimFn.addEntryBlock();
688 OpBuilder sb = OpBuilder::atBlockBegin(shimBlock);
689
690 SmallVector<Value> genArgs(shimBlock->getArguments().drop_back());
691 SmallVector<StringRef> holes(genArgs.size(), "{}");
692 std::string call = "auto __result = " + gen.getSymName().str() + "(" +
693 llvm::join(holes, ", ") + ");";
694 emitc::VerbatimOp::create(
695 sb, loc, "static thread_local " + kTxnVecType.str() + " __txn;");
696 emitc::VerbatimOp::create(sb, loc, "aie_runtime::txn_refusal = nullptr;");
697 emitc::VerbatimOp::create(sb, loc, call, genArgs);
698 emitc::VerbatimOp::create(sb, loc, "if (!__result) return -2;");
699 emitc::VerbatimOp::create(sb, loc, "__txn = std::move(*__result);");
700 emitc::VerbatimOp::create(sb, loc, "*{} = __txn.data();",
701 ValueRange{shimBlock->getArguments().back()});
702 emitc::ReturnOp::create(
703 sb, loc,
704 emitc::LiteralOp::create(sb, loc, i64Ty,
705 "static_cast<int64_t>(__txn.size())"));
706
707 // dispatch_last_refusal(): why the last dispatch_generate() on this
708 // thread returned -2 (a string literal in this library), or null.
709 auto whyFn = emitc::FuncOp::create(builder, loc, "dispatch_last_refusal",
710 FunctionType::get(ctx, {}, {charPtrTy}));
711 whyFn.setSpecifiersAttr(externC);
712 OpBuilder wb = OpBuilder::atBlockBegin(whyFn.addEntryBlock());
713 emitc::ReturnOp::create(
714 wb, loc,
715 emitc::LiteralOp::create(wb, loc, charPtrTy,
716 "aie_runtime::txn_refusal"));
717 return success();
718 }

720 // Resolve `orig`'s device-dependent values (still under the device) keyed by
721 // its detached `clone`.
722 static void recordDeviceResolved(Operation *orig, Operation *clone,
723 DeviceResolved &resolved) {
724 if (auto w = dyn_cast<AIEX::NpuWrite32Op>(orig)) {
725 if (auto a = w.getAbsoluteAddress())
726 resolved.absoluteAddr[clone] = *a;
727 } else if (auto mw = dyn_cast<AIEX::NpuMaskWrite32Op>(orig)) {
728 if (auto a = mw.getAbsoluteAddress())
729 resolved.absoluteAddr[clone] = *a;
730 } else if (auto mp = dyn_cast<AIEX::NpuMaskPollOp>(orig)) {
731 if (auto a = mp.getAbsoluteAddress())
732 resolved.absoluteAddr[clone] = *a;
733 } else if (auto bw = dyn_cast<AIEX::NpuBlockWriteOp>(orig)) {
734 if (auto a = bw.getAbsoluteAddress())
735 resolved.absoluteAddr[clone] = *a;
736 if (auto d = bw.getDataWords())
737 resolved.blockWriteData[clone] = d;
738 } else if (auto bwv = dyn_cast<AIEX::NpuBlockWriteValuesOp>(orig)) {
739 // Already absolute: no buffer/col/row to fold in.
740 if (auto a = AIEX::getConstantIntOperand(bwv.getAddress()))
741 resolved.absoluteAddr[clone] = *a;
742 }
743 }

745 FailureOr<emitc::FuncOp> emitFunction(OpBuilder &builder, ModuleOp moduleOp,
746 AIE::RuntimeSequenceOp seqOp,
747 AIE::DeviceOp deviceOp) {
748 Location loc = seqOp.getLoc();
749 Block &entry = seqOp.getBody().front();
750
751 // The generated function takes the sequence's non-memref args (the runtime
752 // scalars) and returns the assembled txn vector. Memref args address host
753 // buffers resolved by address_patch; they are not function parameters here.
754 SmallVector<Type> paramTypes;
755 auto txnVecType =
756 emitc::OpaqueType::get(moduleOp.getContext(), kTxnVecType);
757 // The builder can fail at TXN-build time (a runtime scalar overflowing a
758 // narrow BD field), so it returns std::optional<std::vector<uint32_t>>:
759 // std::nullopt on a failed guard, the assembled vector otherwise.
760 auto txnRetType = emitc::OpaqueType::get(
761 moduleOp.getContext(),
762 (llvm::Twine("std::optional<") + kTxnVecType + ">").str());
763 // A runtime scalar of type `index` (e.g. a loop trip count) has no C++
764 // spelling on its own; emit it as emitc.size_t so it flows straight into a
765 // converted scf.for bound with no leftover cast. Other scalar types pass
766 // through unchanged.
767 auto sizeT = emitc::SizeTType::get(moduleOp.getContext());
768 auto paramTypeFor = [&](Type t) -> Type {
769 return isa<IndexType>(t) ? Type(sizeT) : t;
770 };
771 for (BlockArgument arg : entry.getArguments())
772 if (!isa<BaseMemRefType>(arg.getType()))
773 paramTypes.push_back(paramTypeFor(arg.getType()));
774
775 builder.setInsertionPointToEnd(moduleOp.getBody());
776 std::string funcName = "generate_txn_" + deviceOp.getSymName().str() + "_" +
777 seqOp.getSymName().str();
778 auto funcOp = emitc::FuncOp::create(
779 builder, loc, funcName,
780 FunctionType::get(moduleOp.getContext(), paramTypes,
781 TypeRange{txnRetType}));
782 funcOp.setSpecifiersAttr(builder.getStrArrayAttr({"inline"}));
783 Block *funcBlock = funcOp.addEntryBlock();
784 OpBuilder fb = OpBuilder::atBlockBegin(funcBlock);
785
786 // EmitC has no native std::vector construction, so declare the accumulator
787 // with a verbatim statement and reference it by name via an emitc.literal
788 // (which translates inline, no extra copy). All txn_append_* calls take
789 // this value, so the generated C++ operates on the single `txn` vector.
790 emitc::VerbatimOp::create(fb, loc, "std::vector<uint32_t> txn;");
791 Value txnVec = emitc::LiteralOp::create(fb, loc, txnVecType, "txn");
792 emitTxnCall(fb, loc, "txn_init", txnVec, {});
793
794 // Declare each runtime BD free-list pool the sequence draws ids from,
795 // with the bounds its ops name.
796 llvm::StringSet<> declaredPools;
797 auto declarePool = [&](auto poolOp) {
798 std::string name = bdPoolName(poolOp);
799 if (!declaredPools.insert(name).second)
800 return;
801 uint32_t lo = poolOp.getBdBegin(), hi = poolOp.getBdEnd();
802 emitc::VerbatimOp::create(fb, loc,
803 "aie_runtime::BdPool " + name +
804 " = aie_runtime::bd_pool_init_range(" +
805 std::to_string(lo) + ", " +
806 std::to_string(hi) + ");");
807 // Hand back the ids the static allocator already placed on this tile.
808 for (uint32_t id :
809 staticBdIdsOnTile(deviceOp, poolOp.getColumn(), poolOp.getRow()))
810 if (id >= lo && id < hi)
811 emitc::VerbatimOp::create(fb, loc,
812 "aie_runtime::bd_pool_reserve(" + name +
813 ", " + std::to_string(id) + ");");
814 };
815 seqOp.walk([&](Operation *op) {
816 if (auto pop = dyn_cast<AIEX::DMABdPoolPopOp>(op))
817 declarePool(pop);
818 else if (auto push = dyn_cast<AIEX::DMABdPoolPushOp>(op))
819 declarePool(push);
820 });
821
822 // A rolled loop makes the op count a runtime quantity: declare a __opcount
823 // accumulator the converter bumps per op. A straight-line sequence keeps
824 // the compile-time literal count (byte-identical golden).
825 bool hasControlFlow = false;
826 seqOp.walk([&](Operation *op) {
827 if (isa<scf::SCFDialect>(op->getDialect()))
828 hasControlFlow = true;
829 });
830 Value opCountVar;
831 if (hasControlFlow) {
832 emitc::VerbatimOp::create(fb, loc, "uint32_t __opcount = 0;");
833 opCountVar = emitc::LiteralOp::create(
834 fb, loc, getU32Type(moduleOp.getContext()), "__opcount");
835 }
836
837 // Clone the sequence body (whole regions, so a rolled scf.for comes along),
838 // mapping non-memref block args to function params. Resolve device-relative
839 // values (address, blockwrite data) while the originals are still under the
840 // device, keyed by clone, since the accessors don't work once detached.
841 IRMapping mapping;
842 unsigned p = 0;
843 for (BlockArgument arg : entry.getArguments())
844 if (!isa<BaseMemRefType>(arg.getType()))
845 mapping.map(arg, funcBlock->getArgument(p++));
846 DeviceResolved resolved;
847 for (Operation &op : entry.without_terminator()) {
848 fb.clone(op, mapping);
849 // Record device-resolved values for the clone and any nested op (a BD
850 // inside a rolled loop resolves the same way as a top-level one).
851 op.walk([&](Operation *orig) {
852 Operation *c = mapping.lookupOrNull(orig);
853 if (c)
854 recordDeviceResolved(orig, c, resolved);
855 });
856 }
857
858 AIEXToEmitCConverter conv(funcOp, txnVec, resolved, opCountVar,
859 foldDDRAddrOffset);
860 std::optional<uint32_t> count = conv.run();
861 if (!count)
862 return failure();
863
864 // Finalize the header with the op count + device info. The count is the
865 // runtime `__opcount` when the sequence has a loop, else the compile-time
866 // literal.
867 const AIE::AIETargetModel &tm = deviceOp.getTargetModel();
868 OpBuilder eb(funcBlock, funcBlock->end());
869 std::string countStr =
870 hasControlFlow ? "__opcount" : (std::to_string(*count) + "u");
871 // txn_device_info assigns TxnDeviceInfo by field name, so a field added or
872 // reordered there cannot silently mis-encode this generated header.
873 std::string header = "aie_runtime::txn_prepend_header(txn, " + countStr +
874 ", aie_runtime::txn_device_info(" +
875 std::to_string(AIE::txnDeviceGen(tm)) + ", " +
876 std::to_string(tm.rows()) + ", " +
877 std::to_string(tm.columns()) + ", " +
878 std::to_string(tm.getNumMemTileRows()) + "));";
879 emitc::VerbatimOp::create(eb, loc, header);
880 // Return the vector as the optional result. The literal text differs from
881 // the "txn" accumulator literal (so the two are not deduplicated) and is
882 // typed as the optional return so emitc's return-type check passes; it
883 // relies on the implicit std::vector -> std::optional conversion in C++.
884 Value ret = emitc::LiteralOp::create(eb, loc, txnRetType, "std::move(txn)");
885 emitc::ReturnOp::create(eb, loc, ret);
886 return funcOp;
887 }

15Value DMAChannelAnalysis::getTileKey(Value tile) {

17 auto col = tileOp.tryGetCol();
18 auto row = tileOp.tryGetRow();
19 if (!col || !row)
20 return tile;

24DMAChannelAnalysis::DMAChannelAnalysis(DeviceOp &device) {
25 for (auto program : device.getOps<DmaBody>()) {
26 for (Block &block : program.getDmaBody()) {
27 for (auto start : block.getOps<DMAStartOp>()) {
28 usedChannels.try_emplace(std::make_tuple(getTileKey(program.getTile()),
29 start.getChannelDir(),
30 start.getChannelIndex()),
31 start.getOperation());
32 }
33 }
34 }
35
36 for (auto flowOp : device.getOps<FlowOp>()) {
37 if (flowOp.getSourceBundle() == WireBundle::Core) {
38 usedStreams.insert({getTileKey(flowOp.getSource()), DMAChannelDir::MM2S,
39 flowOp.getSourceChannel()});
40 }
41 if (flowOp.getDestBundle() == WireBundle::Core) {
42 usedStreams.insert({getTileKey(flowOp.getDest()), DMAChannelDir::S2MM,
43 flowOp.getDestChannel()});
44 }
45 }
46
47 // Shim allocations reserve channels outside the DMA bodies above.
48 for (auto allocOp : device.getOps<ShimDMAAllocationOp>()) {
49 usedChannels.try_emplace(std::make_tuple(getTileKey(allocOp.getTile()),
50 allocOp.getChannelDir(),
51 (int)allocOp.getChannelIndex()),
52 allocOp.getOperation());
53 }
54}

56int DMAChannelAnalysis::getDMAChannelLimit(
57 TileLike tile, DMAChannelDir dir, bool requiresAdjacentTileAccessChannels) {
58 int maxChannelNum = (dir == DMAChannelDir::MM2S)
59 ? tile.getNumSourceConnections(WireBundle::DMA)
60 : tile.getNumDestConnections(WireBundle::DMA);
61
62 if (!requiresAdjacentTileAccessChannels)
63 return maxChannelNum;
64
65 std::optional<int> col = tile.tryGetCol();
66 std::optional<int> row = tile.tryGetRow();
67 const auto &targetModel = getTargetModel(tile);
68 if (col && row)
69 return std::min<int>(
70 maxChannelNum,
71 targetModel.getMaxChannelNumForAdjacentMemTile(*col, *row));
72
73 // Assignment before placement must be safe at every compatible position,
74 // not assume that unresolved coordinates remove neighbor restrictions.
75 bool foundPosition = false;
76 for (int c = 0; c < targetModel.columns(); ++c) {
77 if (col && c != *col)
78 continue;
79 for (int r = 0; r < targetModel.rows(); ++r) {
80 if ((row && r != *row) ||
81 targetModel.getTileType(c, r) != tile.getTileType())
82 continue;
83 foundPosition = true;
84 maxChannelNum = std::min<int>(
85 maxChannelNum, targetModel.getMaxChannelNumForAdjacentMemTile(c, r));
86 }
87 }
88
89 return foundPosition ? maxChannelNum : 0;
90}

92int DMAChannelAnalysis::getDMAChannelIndex(
93 TileLike tile, DMAChannelDir dir, bool requiresAdjacentTileAccessChannels,
94 Operation *owner) {
95 int limit = getDMAChannelLimit(tile, dir, requiresAdjacentTileAccessChannels);
96 for (int i = 0; i < limit; i++) {
97 if (reservePinnedChannel(tile, dir, i, owner) >= 0) {
98 return i;
99 }
100 }
101 return -1;

104int DMAChannelAnalysis::reservePinnedChannel(TileLike tile, DMAChannelDir dir,
105 int channel, Operation *owner) {
106 int maxChannelNum = getDMAChannelLimit(tile, dir, false);
107 if (channel < 0 || channel >= maxChannelNum) {
108 return -1;
109 }
110 return usedChannels
111 .try_emplace(std::make_tuple(getTileKey(tile->getResult(0)),
112 dir, channel),
113 owner)
114 .second
115 ? channel
116 : -1;
117}

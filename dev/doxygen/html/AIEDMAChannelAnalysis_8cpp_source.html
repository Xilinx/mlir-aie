24DMAChannelAnalysis::DMAChannelAnalysis(DeviceOp &device) {
25 for (auto program : device.getOps<DmaBody>()) {
26 for (Block &block : program.getDmaBody()) {
27 for (auto start : block.getOps<DMAStartOp>()) {
28 usedChannels.try_emplace(std::make_tuple(getTileKey(program.getTile()),
29 start.getChannelDir(),
30 start.getChannelIndex()),
31 start.getOperation());
32 }
33 }
34 }
35
36 for (auto flowOp : device.getOps<FlowOp>()) {
37 if (flowOp.getSourceBundle() == WireBundle::Core) {
38 usedStreams.try_emplace({getTileKey(flowOp.getSource()),
39 DMAChannelDir::MM2S, flowOp.getSourceChannel()},
40 StreamClaim{flowOp, false});
41 }
42 if (flowOp.getDestBundle() == WireBundle::Core) {
43 usedStreams.try_emplace({getTileKey(flowOp.getDest()),
44 DMAChannelDir::S2MM, flowOp.getDestChannel()},
45 StreamClaim{flowOp, false});
46 }
47 }
48
49 for (auto flow : device.getOps<PacketFlowOp>()) {
50 Block &ports = flow.getPorts().front();
51 for (auto source : ports.getOps<PacketSourceOp>())
52 if (source.getBundle() == WireBundle::Core)
53 usedStreams.try_emplace({getTileKey(source.getTile()),
54 DMAChannelDir::MM2S, source.channelIndex()},
55 StreamClaim{source, true});
56 for (auto dest : ports.getOps<PacketDestOp>())
57 if (dest.getBundle() == WireBundle::Core)
58 usedStreams.try_emplace({getTileKey(dest.getTile()),
59 DMAChannelDir::S2MM, dest.channelIndex()},
60 StreamClaim{dest, true});
61 }
62
63 // Shim allocations reserve channels outside the DMA bodies above.
64 for (auto allocOp : device.getOps<ShimDMAAllocationOp>()) {
65 usedChannels.try_emplace(std::make_tuple(getTileKey(allocOp.getTile()),
66 allocOp.getChannelDir(),
67 (int)allocOp.getChannelIndex()),
68 allocOp.getOperation());
69 }
70}

72int DMAChannelAnalysis::getDMAChannelLimit(
73 TileLike tile, DMAChannelDir dir, bool requiresAdjacentTileAccessChannels) {
74 int maxChannelNum = (dir == DMAChannelDir::MM2S)
75 ? tile.getNumSourceConnections(WireBundle::DMA)
76 : tile.getNumDestConnections(WireBundle::DMA);
77
78 if (!requiresAdjacentTileAccessChannels)
79 return maxChannelNum;
80
81 std::optional<int> col = tile.tryGetCol();
82 std::optional<int> row = tile.tryGetRow();
83 const auto &targetModel = getTargetModel(tile);
84 if (col && row)
85 return std::min<int>(
86 maxChannelNum,
87 targetModel.getMaxChannelNumForAdjacentMemTile(*col, *row));
88
89 // Assignment before placement must be safe at every compatible position,
90 // not assume that unresolved coordinates remove neighbor restrictions.
91 bool foundPosition = false;
92 for (int c = 0; c < targetModel.columns(); ++c) {
93 if (col && c != *col)
94 continue;
95 for (int r = 0; r < targetModel.rows(); ++r) {
96 if ((row && r != *row) ||
97 targetModel.getTileType(c, r) != tile.getTileType())
98 continue;
99 foundPosition = true;
100 maxChannelNum = std::min<int>(
101 maxChannelNum, targetModel.getMaxChannelNumForAdjacentMemTile(c, r));
102 }
103 }
104
105 return foundPosition ? maxChannelNum : 0;
106}

120int DMAChannelAnalysis::reservePinnedChannel(TileLike tile, DMAChannelDir dir,
121 int channel, Operation *owner) {
122 int maxChannelNum = getDMAChannelLimit(tile, dir, false);
123 if (channel < 0 || channel >= maxChannelNum) {
124 return -1;
125 }
126 return usedChannels
127 .try_emplace(std::make_tuple(getTileKey(tile->getResult(0)),
128 dir, channel),
129 owner)
130 .second
131 ? channel
132 : -1;

15DMAChannelAnalysis::DMAChannelAnalysis(DeviceOp &device) {
16 for (auto program : device.getOps<DmaBody>()) {
17 for (Block &block : program.getDmaBody()) {
18 for (auto start : block.getOps<DMAStartOp>()) {
19 usedChannels.insert({program.getTile(), start.getChannelDir(),
20 start.getChannelIndex()});
21 }
22 }
23 }
24
25 for (auto flowOp : device.getOps<FlowOp>()) {
26 if (flowOp.getSourceBundle() == WireBundle::Core) {
27 usedStreams.insert(
28 {flowOp.getSource(), DMAChannelDir::MM2S, flowOp.getSourceChannel()});
29 }
30 if (flowOp.getDestBundle() == WireBundle::Core) {
31 usedStreams.insert(
32 {flowOp.getDest(), DMAChannelDir::S2MM, flowOp.getDestChannel()});
33 }
34 }
35
36 // Shim allocations reserve channels outside the DMA bodies above.
37 for (auto allocOp : device.getOps<ShimDMAAllocationOp>()) {
38 auto tile = allocOp.getTileOp();
39 if (!tile) {
40 continue;
41 }
42 usedChannels.insert({tile.getResult(), allocOp.getChannelDir(),
43 (int)allocOp.getChannelIndex()});
44 }
45}

47int DMAChannelAnalysis::getDMAChannelIndex(
48 TileLike tile, DMAChannelDir dir, bool requiresAdjacentTileAccessChannels) {
49 int maxChannelNum = (dir == DMAChannelDir::MM2S)
50 ? tile.getNumSourceConnections(WireBundle::DMA)
51 : tile.getNumDestConnections(WireBundle::DMA);
52
53 // Reaching a neighbor's memory restricts the range, and which neighbor a
54 // tile has is only known once it is placed.
55 std::optional<int> col = tile.tryGetCol();
56 std::optional<int> row = tile.tryGetRow();
57 if (requiresAdjacentTileAccessChannels && col && row) {
58 const auto &targetModel = getTargetModel(tile);
59 maxChannelNum = std::min<int>(
60 maxChannelNum,
61 targetModel.getMaxChannelNumForAdjacentMemTile(*col, *row));
62 }
63
64 for (int i = 0; i < maxChannelNum; i++) {
65 if (reservePinnedChannel(tile, dir, i) >= 0) {
66 return i;
67 }
68 }
69 return -1;
70}

72int DMAChannelAnalysis::reservePinnedChannel(TileLike tile, DMAChannelDir dir,
73 int channel) {
74 int maxChannelNum = (dir == DMAChannelDir::MM2S)
75 ? tile.getNumSourceConnections(WireBundle::DMA)
76 : tile.getNumDestConnections(WireBundle::DMA);
77 if (channel < 0 || channel >= maxChannelNum) {
78 return -1;
79 }
80 return usedChannels.insert({tile->getResult(0), dir, channel}).second
81 ? channel
82 : -1;
83}

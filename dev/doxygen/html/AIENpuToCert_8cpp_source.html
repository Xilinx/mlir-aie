55// True if `seq` holds its device's generated configuration, i.e. carries the
56// aie.generated_configure tag, either from convert-aie-to-transaction or from
57// tagConfigureByName.
58static bool isGeneratedConfigure(AIE::RuntimeSequenceOp seq) {
59 return AIE::AIEDialect::GeneratedConfigureAttrHelper(seq.getContext())
60 .isAttrPresent(seq);
61}
62
63// Hand-written IR may have no tag: there the sequence named `configure` is the
64// configuration. Tag it, unless a sequence of the device is already tagged (the
65// generated one was then named `configure0`, and `configure` is the user's).
66// Runs on every device before any sequence is converted, so the choice does
67// not depend on conversion order.
68static void tagConfigureByName(AIE::DeviceOp device) {
69 auto seqs = device.getBody()->getOps<AIE::RuntimeSequenceOp>();
70 if (llvm::any_of(seqs, isGeneratedConfigure))
71 return;
72 for (auto seq : seqs)
73 if (seq.getSymName() == "configure") {
74 AIE::AIEDialect::GeneratedConfigureAttrHelper(device.getContext())
75 .setAttr(seq, UnitAttr::get(device.getContext()));
76 return;
77 }
78}

80// The generated configuration sequence of `device`, or null if it has none.
81static AIE::RuntimeSequenceOp findGeneratedConfigure(AIE::DeviceOp device) {
82 for (auto seq : device.getBody()->getOps<AIE::RuntimeSequenceOp>())
83 if (isGeneratedConfigure(seq))
84 return seq;
85 return nullptr;
86}

88// Returns the PDI id to use for a cert.load_pdi that references `deviceSymName`
89// within `parentDevice`. If a load_pdi for that symbol already exists, its id
90// is reused so repeated loads of the same PDI share an id; otherwise a fresh id
91// (1-based, one past the current maximum) is assigned.
92static uint32_t getOrAssignPdiId(AIE::DeviceOp parentDevice,
93 StringRef deviceSymName) {
94 uint32_t pdiId = 1;
95 bool foundExisting = false;
96 parentDevice.walk([&](AIEX::CertLoadPdiOp loadPdiOp) {
97 if (loadPdiOp.getSymbol() == deviceSymName) {
98 pdiId = loadPdiOp.getPdiId();
99 foundExisting = true;
100 } else if (!foundExisting) {
101 pdiId = std::max(pdiId, loadPdiOp.getPdiId() + 1);
102 }
103 });
104 return pdiId;
105}

107struct RuntimeSequenceToCertJob : OpConversionPattern<AIE::RuntimeSequenceOp> {
108 using OpConversionPattern::OpConversionPattern;
109
110 RuntimeSequenceToCertJob(MLIRContext *context, PatternBenefit benefit = 1)
111 : OpConversionPattern(context, benefit) {}
112
113 LogicalResult
114 matchAndRewrite(AIE::RuntimeSequenceOp op, OpAdaptor adaptor,
115 ConversionPatternRewriter &rewriter) const override {
116
117 bool isConfigure = isGeneratedConfigure(op);
118 uint32_t newJobId = 1;
119 if (!isConfigure) {
120 uint32_t maxJobId = 1;
121 op->getParentOp()->walk([&](AIEX::CertJobOp certJobOp) {
122 maxJobId = std::max(maxJobId, certJobOp.getJobId());
123 });
124 newJobId = maxJobId + 1;
125 }
126
127 // Replace the runtime sequence with a job directly inside aie.device.
128 // CertJobOp lists DeviceOp as an allowed parent in its ParentOneOf trait.
129 // cert-legalize-pages later groups contiguous jobs into implicit pages.
130 rewriter.setInsertionPoint(op);
131 auto jobOp = AIEX::CertJobOp::create(rewriter, op.getLoc(), newJobId);
132
133 // Tag the implicit configuration job so a later step can force it to be the
134 // first page on uC0 and keep it out of user implicit pages.
135 if (isConfigure)
136 jobOp->setAttr("cert.configure", rewriter.getUnitAttr());
137
138 // Clone runtime sequence body into job
139 // Note: This preserves block arguments from the runtime sequence, which
140 // will be present in the MLIR IR but are not emitted in the final assembly
141 IRMapping remap;
142 op.getRegion().cloneInto(&jobOp.getBody(), remap);
143 AIEX::CertJobOp::ensureTerminator(jobOp.getBody(), rewriter, op->getLoc());
144
145 // Erase the original runtime sequence
146 rewriter.eraseOp(op);
147
148 return success();
149 }

152struct NpuWrite32ToCertWrite32 : OpConversionPattern<AIEX::NpuWrite32Op> {
153 using OpConversionPattern::OpConversionPattern;
154
155 LogicalResult
156 matchAndRewrite(AIEX::NpuWrite32Op op, OpAdaptor adaptor,
157 ConversionPatternRewriter &rewriter) const override {
158
159 std::optional<uint32_t> address =
160 AIEX::getConstantIntOperand(op.getAddress());
161 std::optional<uint32_t> value = AIEX::getConstantIntOperand(op.getValue());
162 if (!address || !value)
163 return op.emitOpError(
164 "cannot lower to cert.write32 with non-constant address or value");
165
166 // Get the absolute address, which properly handles row/col if present
167 std::optional<uint32_t> absAddress = op.getAbsoluteAddress();
168 if (!absAddress)
169 return failure();
170
171 uint32_t absoluteAddr = *absAddress;
172
173 // If row and col are specified, validate that the address upper bits match
174 std::optional<uint32_t> col = op.getColumn();
175 std::optional<uint32_t> row = op.getRow();
176 if (col && row) {
177 const auto &tm = AIE::getTargetModel(op);
178 uint32_t expectedUpperBits = ((*col & 0xff) << tm.getColumnShift()) |
179 ((*row & 0xff) << tm.getRowShift());
180
181 // Warn if the original address had non-zero upper bits that don't match
182 uint32_t origAddress = *address;
183 uint32_t origUpperBits = origAddress & ~0xfffff;
184 if (origUpperBits != 0 && origUpperBits != expectedUpperBits) {
185 op.emitWarning() << "address upper bits (0x"
186 << llvm::utohexstr(origUpperBits)
187 << ") don't match row=" << *row << " col=" << *col
188 << " computed bits (0x"
189 << llvm::utohexstr(expectedUpperBits) << ")";
190 }
191 }
192
193 rewriter.replaceOpWithNewOp<AIEX::CertWrite32Op>(op, absoluteAddr, *value);
194 return success();
195 }

272struct NpuBlockWriteToCertUcDma : OpConversionPattern<AIEX::NpuBlockWriteOp> {
273 // nextChainId is owned by the caller and shared by every rewrite in one
274 // conversion, so each name search resumes after the last name allocated.
275 NpuBlockWriteToCertUcDma(MLIRContext *ctx, unsigned &nextChainId)
276 : OpConversionPattern(ctx), nextChainId(nextChainId) {}
277
278 LogicalResult
279 matchAndRewrite(AIEX::NpuBlockWriteOp op, OpAdaptor adaptor,
280 ConversionPatternRewriter &rewriter) const override {
281
282 memref::GetGlobalOp dataOperand =
283 dyn_cast_or_null<memref::GetGlobalOp>(op.getData().getDefiningOp());
284 if (!dataOperand)
285 return failure();
286 MemRefType dataType = cast<MemRefType>(dataOperand.getResult().getType());
287 uint32_t dataSize = dataType.getNumElements();
288 // An empty copy is a no-op, and AIEBU rejects a descriptor whose payload
289 // label has no data.
290 if (dataSize == 0) {
291 rewriter.eraseOp(op);
292 return success();
293 }
294
295 std::string symbolName = "chain_" + std::to_string(nextChainId++);
296 while (op->getParentOfType<AIE::DeviceOp>().lookupSymbol(symbolName))
297 symbolName = "chain_" + std::to_string(nextChainId++);
298
299 // Create a new uc_dma_write_des_sync operation
300 rewriter.replaceOpWithNewOp<AIEX::CertUcDmaWriteDesSyncOp>(op, symbolName);
301
302 // Create the uc_dma_chain operation
303 // Find the nearest device to insert the chain
304 auto parentDevice = op->getParentOfType<AIE::DeviceOp>();
305 if (!parentDevice) {
306 // No parent device - this shouldn't happen but handle gracefully
307 return failure();
308 }
309
310 // Insert after the last existing uc_dma_chain (before any pages/jobs),
311 // preserving the order of blockwrite ops. Globals are allowed to precede
312 // the chains, so skip over them too.
313 Block *deviceBody = parentDevice.getBody();
314 Operation *insertAfter = nullptr;
315 for (Operation &bodyOp : *deviceBody) {
316 if (isa<AIEX::CertUcDmaChainOp>(bodyOp) || isa<memref::GlobalOp>(bodyOp))
317 insertAfter = &bodyOp;
318 else
319 break;
320 }
321 if (insertAfter)
322 rewriter.setInsertionPointAfter(insertAfter);
323 else
324 rewriter.setInsertionPointToStart(deviceBody);
325
326 auto symbolAttr = rewriter.getStringAttr(symbolName);
327 auto chainOp =
328 AIEX::CertUcDmaChainOp::create(rewriter, op.getLoc(), symbolAttr);
329
330 Block *bb = new Block();
331 chainOp.getRegion().push_back(bb);
332 rewriter.setInsertionPointToStart(bb);
333 AIEX::CertUcDmaBdOp::create(rewriter, op.getLoc(), dataOperand.getName(),
334 op.getAddress(), dataSize, false);
335
336 AIEX::CertUcDmaChainOp::ensureTerminator(chainOp.getBody(), rewriter,
337 op->getLoc());
338 return success();
339 }
340
341 unsigned &nextChainId;

344struct NpuSyncToCertWaitTCTS : OpConversionPattern<AIEX::NpuSyncOp> {
345 using OpConversionPattern::OpConversionPattern;
346
347 LogicalResult
348 matchAndRewrite(AIEX::NpuSyncOp op, OpAdaptor adaptor,
349 ConversionPatternRewriter &rewriter) const override {
350 std::optional<uint32_t> rowOpt = AIEX::getConstantIntOperand(op.getRow());
351 std::optional<uint32_t> colOpt =
352 AIEX::getConstantIntOperand(op.getColumn());
353 std::optional<uint32_t> channelOpt =
354 AIEX::getConstantIntOperand(op.getChannel());
355 std::optional<uint32_t> directionOpt =
356 AIEX::getConstantIntOperand(op.getDirection());
357 if (!rowOpt || !colOpt || !channelOpt || !directionOpt)
358 return op.emitOpError(
359 "cannot lower to cert.wait_tcts with non-constant sync parameters");
360 uint32_t row = *rowOpt;
361 uint32_t col = *colOpt;
362
363 // These are the shift amounts from the tct packet format in the
364 // architecture spec. The firmware expects the row and column packed and
365 // shifted down to zero.
366 const int row_id_shift = 16;
367 const int col_id_shift = 21;
368 uint16_t tile_id = col << (col_id_shift - row_id_shift) | row;
369 uint32_t channel = *channelOpt;
370 uint32_t direction = *directionOpt;
371
372 const std::vector<int> chan2actor_shim_s2mm = {0, 2, 3, 4};
373 const std::vector<int> chan2actor_shim_mm2s = {6, 7, 8, 9, 10, 11, 12, 13};
374 const std::vector<int> chan2actor_mem_s2mm = {1, 2, 3, 4, 5, 6, 7};
375 const std::vector<int> chan2actor_mem_mm2s = {16, 17, 18, 19, 20,
376 22, 23, 24, 25, 26};
377 const std::vector<int> chan2actor_tile_s2mm = {0, 1};
378 const std::vector<int> chan2actor_tile_mm2s = {6};
379
380 const auto &tm = AIE::getTargetModel(op);
381 const bool isS2MM =
382 direction == static_cast<std::underlying_type_t<AIE::DMAChannelDir>>(
383 AIE::DMAChannelDir::S2MM);
384
385 const std::vector<int> *chan2actor = nullptr;
386 if (tm.isCoreTile(col, row))
387 chan2actor = isS2MM ? &chan2actor_tile_s2mm : &chan2actor_tile_mm2s;
388 else if (tm.isMemTile(col, row))
389 chan2actor = isS2MM ? &chan2actor_mem_s2mm : &chan2actor_mem_mm2s;
390 else
391 chan2actor = isS2MM ? &chan2actor_shim_s2mm : &chan2actor_shim_mm2s;
392
393 size_t chanIdx = static_cast<size_t>(channel);
394 if (chanIdx >= chan2actor->size()) {
395 op.emitError("invalid DMA channel ")
396 << channel << " for " << (isS2MM ? "S2MM" : "MM2S")
397 << " direction in NpuSyncToCertWaitTCTS conversion";
398 return failure();
399 }
400
401 uint8_t actor_id = static_cast<uint8_t>((*chan2actor)[chanIdx]);
402 uint8_t num_tcts = 1;
403 rewriter.replaceOpWithNewOp<AIEX::CertWaitTCTSOp>(op, tile_id, actor_id,
404 num_tcts);
405 return success();
406 }

439struct RunOpInlining : OpRewritePattern<AIEX::RunOp> {
440 using OpRewritePattern::OpRewritePattern;
441
442 LogicalResult matchAndRewrite(AIEX::RunOp op,
443 PatternRewriter &rewriter) const override {
444 // Get the callee runtime sequence
445 AIE::RuntimeSequenceOp calleeRuntimeSequence =
446 op.getCalleeRuntimeSequenceOp();
447 if (!calleeRuntimeSequence)
448 return failure();
449
450 // Get the callee body region
451 Region &calleeBody = calleeRuntimeSequence.getBody();
452
453 // Create argument mapping from run op arguments to callee parameters
454 IRMapping argMap;
455 ValueRange values = op.getArgs();
456 if (values.size() != calleeBody.getNumArguments()) {
457 op.emitError("number of run op arguments (")
458 << values.size()
459 << ") does not match number of callee runtime sequence "
460 "arguments ("
461 << calleeBody.getNumArguments() << ")";
462 return failure();
463 }
464 for (unsigned i = 0, n = calleeBody.getNumArguments(); i < n; i++) {
465 BlockArgument arg = calleeBody.getArgument(i);
466 Value val = values[i];
467 argMap.map(arg, val);
468 }
469
470 // Clone operations from callee into current location
471 rewriter.setInsertionPoint(op);
472 for (Operation &o : calleeBody.front().getOperations()) {
473 // Skip the terminator
474 if (isa<AIE::EndOp>(o))
475 continue;
476 rewriter.clone(o, argMap);
477 }
478
479 // Erase the run op
480 rewriter.eraseOp(op);
481
482 return success();
483 }

486struct ConfigureOpToCertSection : OpRewritePattern<AIEX::ConfigureOp> {
487 using OpRewritePattern::OpRewritePattern;
488
489 LogicalResult matchAndRewrite(AIEX::ConfigureOp op,
490 PatternRewriter &rewriter) const override {
491 // Get the referenced device
492 AIE::DeviceOp referencedDevice = op.getReferencedDeviceOp();
493 if (!referencedDevice)
494 return failure();
495
496 // Get the device's symbol name for the section
497 StringRef deviceSymName = referencedDevice.getSymName();
498
499 // Find the parent DeviceOp to insert the section into
500 auto parentDevice = op->getParentOfType<AIE::DeviceOp>();
501 if (!parentDevice)
502 return failure();
503
504 // Check if section already exists (avoid creating duplicates)
505 auto existingSection = dyn_cast_if_present<AIEX::CertSectionOp>(
506 parentDevice.lookupSymbol(deviceSymName));
507 if (existingSection) {
508 // Section already exists, just create load_pdi at call site, reusing the
509 // PDI ID already assigned to this section (or assigning a new one).
510 uint32_t pdiId = getOrAssignPdiId(parentDevice, deviceSymName);
511
512 rewriter.setInsertionPoint(op);
513 AIEX::CertLoadPdiOp::create(
514 rewriter, op.getLoc(), rewriter.getUI32IntegerAttr(pdiId),
515 FlatSymbolRefAttr::get(rewriter.getContext(), deviceSymName));
516
517 // Clone configure body operations after the load_pdi
518 IRMapping bodyMapper;
519 for (Operation &o : op.getRegion().front().getOperations()) {
520 if (isa<AIE::EndOp>(o))
521 continue;
522 rewriter.clone(o, bodyMapper);
523 }
524
525 rewriter.eraseOp(op);
526 return success();
527 }
528
529 // Create cert.section with the device symbol name
530 rewriter.setInsertionPoint(parentDevice.getBody()->getTerminator());
531 auto sectionOp = AIEX::CertSectionOp::create(
532 rewriter, op.getLoc(), rewriter.getStringAttr(deviceSymName));
533
534 // Create the section body with a page containing a job
535 Block *sectionBlock = new Block();
536 sectionOp.getBody().push_back(sectionBlock);
537 rewriter.setInsertionPointToStart(sectionBlock);
538
539 // Create page within section
540 auto pageOp = AIEX::CertPageOp::create(rewriter, op.getLoc());
541 Block *pageBlock = new Block();
542 pageOp.getBody().push_back(pageBlock);
543 rewriter.setInsertionPointToStart(pageBlock);
544
545 // Create job within page - assign unique job ID across all devices
546 // Find the maximum job ID in the parent device
547 uint32_t maxJobId = 0;
548 parentDevice.walk([&](AIEX::CertJobOp certJobOp) {
549 maxJobId = std::max(maxJobId, certJobOp.getJobId());
550 });
551 uint32_t sectionJobId = maxJobId + 1;
552
553 auto jobOp = AIEX::CertJobOp::create(rewriter, op.getLoc(), sectionJobId);
554 Block *jobBlock = new Block();
555 jobOp.getBody().push_back(jobBlock);
556 rewriter.setInsertionPointToStart(jobBlock);
557
558 // Find the "configure" sequence content from the referenced device
559 // It could be either:
560 // 1. Still the generated configuration sequence (if that device hasn't
561 // been processed yet)
562 // 2. Already converted to cert.page/cert.job (if that device was processed
563 // first)
564
565 AIE::RuntimeSequenceOp configureSeq =
566 findGeneratedConfigure(referencedDevice);
567
568 if (configureSeq) {
569 // Clone operations from the "configure" runtime_sequence
570 IRMapping mapper;
571 for (Operation &o : configureSeq.getRegion().front().getOperations()) {
572 if (isa<AIE::EndOp>(o))
573 continue;
574 rewriter.clone(o, mapper);
575 }
576 } else {
577 // "configure" was already converted - find the first cert.page
578 // (configuration page)
579 AIEX::CertPageOp configPage = nullptr;
580 for (auto &op : referencedDevice.getBody()->getOperations()) {
581 if (auto page = dyn_cast<AIEX::CertPageOp>(op)) {
582 configPage = page;
583 break; // Take first page (configure)
584 }
585 }
586
587 if (configPage) {
588 // Clone operations from the config page's job
589 IRMapping mapper;
590 for (auto &op : configPage.getBody().front().getOperations()) {
591 if (auto job = dyn_cast<AIEX::CertJobOp>(op)) {
592 for (Operation &jobOp : job.getBody().front().getOperations()) {
593 if (isa<AIE::EndOp>(jobOp))
594 continue;
595 rewriter.clone(jobOp, mapper);
596 }
597 break; // Only clone first job
598 }
599 }
600 }
601 }
602
603 // Ensure terminators
604 AIEX::CertJobOp::ensureTerminator(jobOp.getBody(), rewriter, op.getLoc());
605 AIEX::CertPageOp::ensureTerminator(pageOp.getBody(), rewriter, op.getLoc());
606 AIEX::CertSectionOp::ensureTerminator(sectionOp.getBody(), rewriter,
607 op.getLoc());
608
609 // At the configure call site, replace with load_pdi, reusing the PDI ID
610 // already assigned to this device if present (or assigning a new one).
611 uint32_t pdiId = getOrAssignPdiId(parentDevice, deviceSymName);
612
613 rewriter.setInsertionPoint(op);
614 AIEX::CertLoadPdiOp::create(
615 rewriter, op.getLoc(), rewriter.getUI32IntegerAttr(pdiId),
616 FlatSymbolRefAttr::get(rewriter.getContext(), deviceSymName));
617
618 // Clone configure body operations after the load_pdi
619 // Operations that reference external values were not cloned into the
620 // section, so we clone them here into the main control flow
621 IRMapping bodyMapper;

634// The blockwrite that loads the BD an address_patch patches: the nearest
635// preceding blockwrite, when it targets the patched BD.
636static AIEX::NpuBlockWriteOp
637findPatchedBlockWrite(AIEX::NpuAddressPatchOp patch) {
638 const auto &tm = AIE::getTargetModel(patch);
639 uint32_t addr = patch.getAddr();
640 int col = (addr >> tm.getColumnShift()) & 0x1f;
641 int row = (addr >> tm.getRowShift()) & 0x1f;
642 if (!tm.isValidTile({col, row}))
643 return {};
644 for (Block::iterator it(patch); it != patch->getBlock()->begin();) {
645 auto blockWrite = dyn_cast<AIEX::NpuBlockWriteOp>(*--it);
646 if (!blockWrite)
647 continue;
648 if (blockWrite.getAddress() + tm.getDmaBdAddressOffset(col, row) != addr)
649 return {};
650 return blockWrite;
651 }
652 return {};
653}

655// An address_patch writes buffer address + arg_plus into a BD register, but
656// cert.apply_offset_57 adds the buffer address to the BD template in memory.
657// So before conversion the patched template word must hold arg_plus, and the
658// template must belong to its blockwrite alone: dma-to-npu shares identical
659// templates between BDs, and a shared one would accumulate every BD's address.
660static LogicalResult prepareAddressPatchTemplates(AIE::DeviceOp device) {
661 LogicalResult result = success();
662 SymbolTable symbols(device);
663 // Count each global's symbol uses once, then keep the counts current as
664 // getters are added and erased. Asking SymbolTable per patch walks the whole
665 // device each time, which is quadratic in the number of patched BDs. With no
666 // counts (an op with unknown symbol uses), every patched template is copied.
667 std::optional<llvm::DenseMap<StringAttr, unsigned>> symbolUseCounts;
668 if (auto uses = SymbolTable::getSymbolUses(&device.getRegion())) {
669 symbolUseCounts.emplace();
670 for (const SymbolTable::SymbolUse &use : *uses)
671 ++(*symbolUseCounts)[use.getSymbolRef().getRootReference()];
672 }
673 SmallVector<AIEX::NpuAddressPatchOp> patches;
674 device.walk([&](AIEX::NpuAddressPatchOp patch) { patches.push_back(patch); });
675 for (AIEX::NpuAddressPatchOp patch : patches) {
676 AIEX::NpuBlockWriteOp blockWrite = findPatchedBlockWrite(patch);
677 if (!blockWrite)
678 continue;
679 auto getGlobal = dyn_cast_or_null<memref::GetGlobalOp>(
680 blockWrite.getData().getDefiningOp());
681 if (!getGlobal)
682 continue;
683 auto global = symbols.lookup<memref::GlobalOp>(getGlobal.getName());
684 if (!global)
685 continue;
686 std::optional<Attribute> initialValue = global.getInitialValue();
687 auto words = initialValue ? dyn_cast<DenseIntElementsAttr>(*initialValue)
688 : DenseIntElementsAttr();
689 if (!words)
690 continue;
691
692 if (!patch.getArgIdx()) {
693 patch.emitError("cert lowering requires an address_patch with arg_idx");
694 result = failure();
695 continue;
696 }
697 APInt argPlus;
698 if (!matchPattern(patch.getArgPlus(), m_ConstantInt(&argPlus))) {
699 patch.emitError("cert lowering requires a constant arg_plus: "
700 "apply_offset_57 cannot add a runtime offset");
701 result = failure();
702 continue;
703 }
704 // An i32 arg_plus is unsigned, so any value fits; a negative i64 has 64
705 // active bits and is rejected.
706 if (argPlus.getActiveBits() > 32) {
707 patch.emitError("arg_plus ")
708 << argPlus.getSExtValue()
709 << " does not fit the BD's 32-bit address word";
710 result = failure();
711 continue;
712 }
713 unsigned word = (patch.getAddr() - blockWrite.getAddress()) / 4;
714 if (word >= words.getNumElements())
715 continue;
716
717 // Give this blockwrite a private template if anything else uses it,
718 // whether through another getter or through this one (after CSE).
719 if (!symbolUseCounts ||
720 symbolUseCounts->lookup(global.getSymNameAttr()) != 1 ||
721 !getGlobal->hasOneUse()) {
722 OpBuilder builder(global);
723 auto copy = cast<memref::GlobalOp>(builder.clone(*global));
724 // insert renames the copy, so count it under its new name.
725 StringAttr copyName = symbols.insert(copy);
726 OpBuilder atBlockWrite(blockWrite);
727 auto privateGet =
728 memref::GetGlobalOp::create(atBlockWrite, getGlobal.getLoc(),
729 getGlobal.getType(), copy.getSymName());
730 blockWrite.getDataMutable().assign(privateGet);
731 if (symbolUseCounts)
732 (*symbolUseCounts)[copyName] = 1;
733 if (getGlobal->use_empty()) {
734 if (symbolUseCounts)
735 --(*symbolUseCounts)[global.getSymNameAttr()];
736 getGlobal->erase();
737 }
738 global = copy;
739 }
740
741 auto values = llvm::to_vector(words.getValues<APInt>());
742 values[word] = argPlus.zextOrTrunc(values[word].getBitWidth());
743 global.setInitialValueAttr(
744 DenseIntElementsAttr::get(words.getType(), values));
745 }
746 return result;
747}

771struct MergeConsecutiveCertUcDmaWriteDesSyncOps
772 : OpRewritePattern<AIEX::CertUcDmaWriteDesSyncOp> {
773 using OpRewritePattern::OpRewritePattern;
774
775 LogicalResult matchAndRewrite(AIEX::CertUcDmaWriteDesSyncOp op,
776 PatternRewriter &rewriter) const override {
777 // Fusing the two enqueues sinks the earlier one to the later one's position
778 // in the job, so every op scanned over is an op the earlier DMA is moved
779 // past. Only ops with no architectural effect at all may be crossed; this
780 // is an allowlist rather than a denylist so a cert op added to this dialect
781 // later blocks the merge by default instead of silently becoming
782 // reorderable with respect to a DMA enqueue. cert.nop is the only such op
783 // today: it emits padding and carries no ordering, sync or side effect.
784 auto isTransparentToMerge = [](Operation *o) {
785 return isa<AIEX::CertNopOp>(o);
786 };
787
788 // Get the previous operation in the block
789 Block::iterator it(op);
790 AIEX::CertUcDmaWriteDesSyncOp prevWriteDesSync = nullptr;
791 while (it != op->getBlock()->begin() && !prevWriteDesSync) {
792 --it;
793 Operation *prevOp = &*it;
794 prevWriteDesSync = dyn_cast<AIEX::CertUcDmaWriteDesSyncOp>(prevOp);
795 if (!prevWriteDesSync && !isTransparentToMerge(prevOp))
796 return failure();
797 }
798 if (!prevWriteDesSync)
799 return failure();
800
801 // find the uc_dma_chain
802 StringRef sym_name = op.getSymbol();
803 StringRef prev_sym_name = prevWriteDesSync.getSymbol();
804 auto chain = dyn_cast_if_present<AIEX::CertUcDmaChainOp>(
805 op->getParentOfType<AIE::DeviceOp>().lookupSymbol(sym_name));
806 auto prevChain = dyn_cast_if_present<AIEX::CertUcDmaChainOp>(
807 prevWriteDesSync->getParentOfType<AIE::DeviceOp>().lookupSymbol(
808 prev_sym_name));
809 if (!chain || !prevChain)
810 return failure();
811
812 // The merge rewrites @chain's BD list in place and deletes @prevChain, so
813 // both symbols must be private to this pair of enqueues. Cert IR that is
814 // hand-authored rather than lowered from npu ops may enqueue one chain from
815 // several places: folding into a shared @chain would silently change those
816 // other enqueues, and erasing a shared @prevChain would leave a dangling
817 // symbol reference that neither the verifier nor the page cost model
818 // notices. Decline the merge instead.
819 auto deviceOp = op->getParentOfType<AIE::DeviceOp>();
820 auto isSingleUse = [&](AIEX::CertUcDmaChainOp c) {
821 auto uses = SymbolTable::getSymbolUses(c, deviceOp);
822 return uses && llvm::hasSingleElement(*uses);
823 };
824 if (!isSingleUse(chain) || !isSingleUse(prevChain))
825 return failure();
826
827 // Compute the size of the current and previous chains. If their combined
828 // data size is greater than the cert page size, then we cannot merge them.
829 //
830 // A chain costs 16 bytes of descriptor per BD on top of the BD's payload --
831 // the same accounting updateCostForOp (the page splitter's authoritative
832 // cost model) uses. Charging payload only under-reports a chain by 16 bytes
833 // per BD, which for many small BDs is most of its real size.
834 auto chainSize = [](AIEX::CertUcDmaChainOp c) {
835 uint32_t size = 0;
836 for (auto bdOp : c.getBody().front().getOps<AIEX::CertUcDmaBdOp>())
837 size += bdOp.getLength() * sizeof(int) + 16; // payload + bd descriptor
838 return size;
839 };
840 if ((chainSize(chain) + chainSize(prevChain)) >= cert_page_size)
841 return failure();
842
843 IRMapping map;
844 rewriter.setInsertionPointToStart(&chain.getBody().front());
845 for (auto &o : prevChain.getBody().front().getOperations()) {
846 auto bdOp = dyn_cast<AIEX::CertUcDmaBdOp>(o);
847 if (!bdOp)
848 continue;
849 AIEX::CertUcDmaBdOp::create(
850 rewriter, bdOp.getLoc(), bdOp.getRemoteAddress(),
851 bdOp.getLocalAddress(), bdOp.getLength(), true);
852 }
853 rewriter.eraseOp(prevChain);
854 rewriter.eraseOp(prevWriteDesSync);
855 return success();
856 }
857};

859struct SplitNpuBlockWriteOpPattern : OpRewritePattern<AIEX::NpuBlockWriteOp> {
860 using OpRewritePattern::OpRewritePattern;
861
862 LogicalResult matchAndRewrite(AIEX::NpuBlockWriteOp op,
863 PatternRewriter &rewriter) const override {
864
865 memref::GetGlobalOp dataOperand =
866 dyn_cast_or_null<memref::GetGlobalOp>(op.getData().getDefiningOp());
867 if (!dataOperand)
868 return failure();
869
870 MemRefType dataType = cast<MemRefType>(dataOperand.getResult().getType());
871 uint32_t dataSize = dataType.getNumElements();
872
873 uint32_t dataSizeBytes = dataSize * sizeof(int);
874 if (dataSizeBytes < cert_page_size)
875 return failure();
876
877 auto loc = op.getLoc();
878
879 // Calculate split point (split roughly in half)
880 uint32_t splitElements = dataSize / 2;
881 uint32_t firstChunkSize = splitElements;
882 uint32_t secondChunkSize = dataSize - splitElements;
883
884 // Find the original memref.global operation
885 auto deviceOp = op->getParentOfType<AIE::DeviceOp>();
886 auto originalGlobal = dyn_cast_if_present<memref::GlobalOp>(
887 deviceOp.lookupSymbol(dataOperand.getName()));
888 if (!originalGlobal)
889 return failure();
890
891 // Get the original data attribute
892 auto originalData = originalGlobal.getInitialValue();
893 if (!originalData)
894 return failure();
895
896 auto denseData = dyn_cast<DenseIntElementsAttr>(*originalData);
897 if (!denseData)
898 return failure();
899
900 // Split the data into two chunks
901 auto dataValues = denseData.getValues<APInt>();
902 std::vector<APInt> firstChunkData(dataValues.begin(),
903 dataValues.begin() + firstChunkSize);
904 std::vector<APInt> secondChunkData(dataValues.begin() + firstChunkSize,
905 dataValues.end());
906
907 // Create new global operations for the split data
908 auto elementType = rewriter.getI32Type();
909 auto firstChunkType = MemRefType::get({firstChunkSize}, elementType);
910 auto secondChunkType = MemRefType::get({secondChunkSize}, elementType);
911 TensorType firstTensorType =
912 RankedTensorType::get({firstChunkSize}, elementType);
913 TensorType secondTensorType =
914 RankedTensorType::get({secondChunkSize}, elementType);
915
916 auto firstChunkAttr =
917 DenseIntElementsAttr::get(firstTensorType, firstChunkData);
918 auto secondChunkAttr =
919 DenseIntElementsAttr::get(secondTensorType, secondChunkData);
920
921 // Generate unique names for the new globals
922 std::string firstName = dataOperand.getName().str() + "_split_0";
923 std::string secondName = dataOperand.getName().str() + "_split_1";
924
925 // Ensure unique names
926 int counter = 0;
927 while (deviceOp.lookupSymbol(firstName)) {
928 firstName =
929 dataOperand.getName().str() + "_split_0_" + std::to_string(counter++);
930 }
931 counter = 0;
932 while (deviceOp.lookupSymbol(secondName)) {
933 secondName =
934 dataOperand.getName().str() + "_split_1_" + std::to_string(counter++);
935 }
936
937 // Create the new global operations
938 rewriter.setInsertionPoint(originalGlobal);
939 memref::GlobalOp::create(rewriter, loc, firstName,
940 rewriter.getStringAttr("private"), firstChunkType,
941 firstChunkAttr, true, nullptr);
942
943 memref::GlobalOp::create(rewriter, loc, secondName,
944 rewriter.getStringAttr("private"), secondChunkType,
945 secondChunkAttr, true, nullptr);
946
947 // Create get_global operations for the new data
948 rewriter.setInsertionPoint(op);
949
950 auto firstGetGlobal =
951 memref::GetGlobalOp::create(rewriter, loc, firstChunkType, firstName);
952 auto secondGetGlobal =
953 memref::GetGlobalOp::create(rewriter, loc, secondChunkType, secondName);
954
955 uint32_t baseAddr = op.getAddress();
956
957 AIEX::NpuBlockWriteOp::create(rewriter, loc, baseAddr,
958 firstGetGlobal.getResult(), nullptr, nullptr,
959 nullptr);
960
961 AIEX::NpuBlockWriteOp::create(rewriter, loc, baseAddr + firstChunkSize * 4,
962 secondGetGlobal.getResult(), nullptr, nullptr,
963 nullptr);
964
965 // Replace the original operation
966 rewriter.eraseOp(op);
967
968 LLVM_DEBUG(llvm::outs()
969 << "Split NpuBlockWriteOp with data size: " << dataSizeBytes
970 << " bytes into chunks of " << firstChunkSize << " and "
971 << secondChunkSize << " elements\n");
972
973 return success();
974 }

977struct AIENpuToCertPass
978 : xilinx::AIEX::impl::AIENpuToCertBase<AIENpuToCertPass> {
979 void runOnOperation() override {
980 auto moduleOp = getOperation();
981 if (failed(AIE::inlineWriteConfigs(moduleOp)))
982 return signalPassFailure();
983
984 // Collect all devices
985 llvm::SmallVector<AIE::DeviceOp, 4> devices;
986 for (auto &op : moduleOp.getBody()->getOperations()) {
987 if (auto deviceOp = dyn_cast<AIE::DeviceOp>(op)) {
988 devices.push_back(deviceOp);
989 }
990 }
991
992 for (auto dev : devices)
993 tagConfigureByName(dev);
994
995 // Process main device first (named "main" or specified by --device-name)
996 // so referenced devices' "configure" sequences can be found before they
997 // get converted to cert.job
998 AIE::DeviceOp mainDevice = nullptr;
999 for (auto dev : devices) {
1000 if (dev.getSymName() == deviceName) {
1001 mainDevice = dev;
1002 break;
1003 }
1004 }
1005
1006 if (!mainDevice) {
1007 moduleOp.emitError("no device found matching --device-name \"")
1008 << deviceName << "\"";
1009 signalPassFailure();
1010 return;
1011 }
1012
1013 migrateGlobalsIntoMainDevice(mainDevice, devices);
1014 processDevice(mainDevice);
1015
1016 // Then process other devices
1017 for (auto dev : devices) {
1018 if (dev != mainDevice) {
1019 processDevice(dev);
1020 }
1021 }
1022
1023 // Remove absorbed devices (keep only main device)
1024 // Referenced devices have been absorbed as cert.sections, so remove them
1025 llvm::SmallVector<AIE::DeviceOp, 4> devicesToRemove;
1026 for (auto dev : devices) {
1027 if (dev != mainDevice) {
1028 devicesToRemove.push_back(dev);
1029 }
1030 }
1031
1032 for (auto dev : devicesToRemove) {
1033 dev.erase();
1034 }
1035 }
1036
1037 // Move globals into the surviving device and rename payload references before
1038 // cloning configure bodies. Reserve device names for sections, including when
1039 // renaming the main device's globals: processDevice skips names that resolve.
1040 void migrateGlobalsIntoMainDevice(AIE::DeviceOp mainDevice,
1041 ArrayRef<AIE::DeviceOp> devices) {
1042 auto isDeviceName = [&](StringRef candidate) {
1043 return llvm::any_of(devices, [&](AIE::DeviceOp device) {
1044 return device.getSymName() == candidate;
1045 });
1046 };
1047
1048 for (auto global : llvm::to_vector(mainDevice.getOps<memref::GlobalOp>())) {
1049 StringRef oldName = global.getSymName();
1050 if (!isDeviceName(oldName))
1051 continue;
1052 std::string newName;
1053 unsigned counter = 0;
1054 do
1055 newName = oldName.str() + "_" + std::to_string(counter++);
1056 while (nameTaken(mainDevice, newName) || isDeviceName(newName));
1057 renamePayloadUses(mainDevice, oldName, newName);
1058 global.setSymName(newName);
1059 }
1060
1061 for (auto dev : devices) {
1062 if (dev == mainDevice)
1063 continue;
1064 auto globals = llvm::to_vector(dev.getOps<memref::GlobalOp>());
1065 for (auto global : globals) {
1066 StringRef oldName = global.getSymName();
1067 std::string newName = oldName.str();
1068 unsigned counter = 0;
1069 // Avoid a temporary collision with another source global: rewriting
1070 // its uses later would otherwise redirect both payloads to one symbol.
1071 auto taken = [&](StringRef candidate) {
1072 return nameTaken(mainDevice, candidate) || isDeviceName(candidate) ||
1073 (candidate != oldName && nameTaken(dev, candidate));
1074 };
1075 while (taken(newName))
1076 newName = oldName.str() + "_" + std::to_string(counter++);
1077 if (newName != oldName) {
1078 renamePayloadUses(dev, oldName, newName);
1079 global.setSymName(newName);
1080 }
1081 global->moveBefore(mainDevice.getBody(), mainDevice.getBody()->begin());
1082 }
1083 }
1084 }

1086 // The device verifier rejects duplicate sym_names among all body ops,
1087 // including SSA ops such as aie.buffer that the symbol table does not index.
1088 static bool nameTaken(AIE::DeviceOp dev, StringRef name) {
1089 return llvm::any_of(dev.getBody()->getOperations(), [&](Operation &op) {
1090 auto symName =
1091 op.getAttrOfType<StringAttr>(SymbolTable::getSymbolAttrName());
1092 return symName && symName.getValue() == name;
1093 });
1094 }
1095
1096 // Rewrite only references that name a payload global. A generic symbol-use
1097 // rewrite would also rename same-named device references, such as
1098 // load_pdi's device_ref, which resolve against the module. Nested symbol
1099 // tables resolve their own names, so their uses are left alone.
1100 static void renamePayloadUses(AIE::DeviceOp dev, StringRef oldName,
1101 StringRef newName) {
1102 auto newRef = FlatSymbolRefAttr::get(dev.getContext(), newName);
1103 dev.walk<WalkOrder::PreOrder>([&](Operation *op) {
1104 if (op != dev && op->hasTrait<OpTrait::SymbolTable>())
1105 return WalkResult::skip();
1106 if (auto get = dyn_cast<memref::GetGlobalOp>(op)) {
1107 if (get.getName() == oldName)
1108 get.setNameAttr(newRef);
1109 } else if (auto bd = dyn_cast<AIEX::CertUcDmaBdOp>(op)) {
1110 if (bd.getRemoteAddress() == oldName)
1111 bd.setRemoteAddressAttr(newRef);
1112 } else if (auto patch = dyn_cast<AIEX::CertApplyOffset57Op>(op)) {
1113 if (patch.getSymbol() == oldName)
1114 patch.setSymbolAttr(newRef);
1115 }
1116 return WalkResult::advance();
1117 });
1118 }

1120 void processDevice(AIE::DeviceOp currentDevice) {
1121 auto moduleOp = currentDevice->getParentOfType<ModuleOp>();
1122
1123 // Identify and convert referenced "configure" sequences early, before the
1124 // RuntimeSequenceOp conversion runs on other devices.
1125 if (moduleOp) {
1126
1127 // Collect npu.load_pdi operations (before they are converted below)
1128 llvm::SmallVector<StringRef, 4> referencedDeviceSyms;
1129 currentDevice.walk([&](AIEX::NpuLoadPdiOp loadPdiOp) {
1130 if (loadPdiOp.getDeviceRef())
1131 referencedDeviceSyms.push_back(*loadPdiOp.getDeviceRef());
1132 });
1133
1134 if (!referencedDeviceSyms.empty()) {
1135 OpBuilder builder(&getContext());
1136
1137 for (StringRef refSymName : referencedDeviceSyms) {
1138 // Skip if section already exists
1139 if (currentDevice.lookupSymbol(refSymName))
1140 continue;
1141
1142 // Find referenced device
1143 AIE::DeviceOp refDevice = nullptr;
1144 for (auto &op : moduleOp.getBody()->getOperations()) {
1145 if (auto dev = dyn_cast<AIE::DeviceOp>(op)) {
1146 if (dev.getSymName() == refSymName) {
1147 refDevice = dev;
1148 break;
1149 }
1150 }
1151 }
1152
1153 if (!refDevice)
1154 continue;
1155
1156 // Find the generated configuration sequence
1157 AIE::RuntimeSequenceOp configureSeq =
1158 findGeneratedConfigure(refDevice);
1159
1160 if (!configureSeq)
1161 continue;
1162
1163 // Create cert.section in current device
1164 builder.setInsertionPoint(currentDevice.getBody()->getTerminator());
1165 auto sectionOp =
1166 AIEX::CertSectionOp::create(builder, configureSeq.getLoc(),
1167 builder.getStringAttr(refSymName));
1168
1169 Block *sectionBlock = new Block();
1170 sectionOp.getBody().push_back(sectionBlock);
1171 builder.setInsertionPointToStart(sectionBlock);
1172
1173 // Create page
1174 auto pageOp =
1175 AIEX::CertPageOp::create(builder, configureSeq.getLoc());
1176 Block *pageBlock = new Block();
1177 pageOp.getBody().push_back(pageBlock);
1178 builder.setInsertionPointToStart(pageBlock);
1179
1180 // Get unique job ID
1181 uint32_t maxJobId = 0;
1182 currentDevice.walk([&](AIEX::CertJobOp certJobOp) {
1183 maxJobId = std::max(maxJobId, certJobOp.getJobId());
1184 });
1185
1186 // Create job
1187 auto jobOp = AIEX::CertJobOp::create(builder, configureSeq.getLoc(),
1188 maxJobId + 1);
1189
1190 // Clone configure sequence body into job
1191 IRMapping mapper;
1192 configureSeq.getRegion().cloneInto(&jobOp.getBody(), mapper);
1193
1194 // Ensure terminators
1195 AIEX::CertJobOp::ensureTerminator(jobOp.getBody(), builder,
1196 configureSeq.getLoc());
1197 AIEX::CertPageOp::ensureTerminator(pageOp.getBody(), builder,
1198 configureSeq.getLoc());
1199 AIEX::CertSectionOp::ensureTerminator(sectionOp.getBody(), builder,
1200 configureSeq.getLoc());
1201 }
1202 }
1203 }

1206 RewritePatternSet p_run_inline(&getContext());
1207 p_run_inline.insert<RunOpInlining>(&getContext());
1208 if (failed(applyPatternsGreedily(currentDevice, std::move(p_run_inline))))
1209 return signalPassFailure();

1211 // Then convert ConfigureOps to cert.section + cert.load_pdi
1212 RewritePatternSet p_configure(&getContext());
1213 p_configure.insert<ConfigureOpToCertSection>(&getContext());
1214 if (failed(applyPatternsGreedily(currentDevice, std::move(p_configure))))
1215 return signalPassFailure();
1216
1217 ConversionTarget target(getContext());
1218 target.addIllegalOp<AIE::RuntimeSequenceOp>();
1219
1220 target.addLegalOp<AIEX::CertApplyOffset57Op>();
1221 target.addLegalOp<AIEX::CertJobOp>();
1222 target.addLegalOp<AIEX::CertPageOp>();
1223 target.addLegalOp<AIEX::CertSectionOp>();
1224 target.addLegalOp<AIEX::CertLoadPdiOp>();
1225 target.addLegalOp<AIEX::CertMaskWrite32Op>();
1226 target.addLegalOp<AIEX::CertMaskPoll32Op>();
1227 target.addLegalOp<AIEX::CertUcDmaWriteDesSyncOp>();
1228 target.addLegalOp<AIEX::CertUcDmaChainOp>();
1229 target.addLegalOp<AIEX::CertUcDmaBdOp>();
1230 target.addLegalOp<AIEX::CertWrite32Op>();
1231 target.addLegalOp<AIEX::CertWaitTCTSOp>();
1232 target.addLegalOp<AIEX::ConfigureOp>(); // TODO: Convert in separate pass
1233 target.addLegalDialect<AIE::AIEDialect>();
1234
1235 RewritePatternSet p0(&getContext());
1236 p0.insert<RuntimeSequenceToCertJob>(&getContext());
1237
1238 if (failed(applyPartialConversion(currentDevice, target, std::move(p0))))
1239 return signalPassFailure();
1240
1241 if (failed(prepareAddressPatchTemplates(currentDevice)))
1242 return signalPassFailure();
1243
1244 target.addIllegalOp<AIEX::NpuAddressPatchOp>();
1245
1246 // patch conversion must come before blockwrite conversion
1247 RewritePatternSet p1(&getContext());
1248 p1.insert<NpuAddressPatchToCertApplyOffset57>(&getContext());
1249
1250 if (failed(applyPartialConversion(currentDevice, target, std::move(p1))))
1251 return signalPassFailure();
1252
1253 // Split oversized NpuBlockWriteOps before lowering them to cert ops
1254 {
1255 RewritePatternSet p(&getContext());
1256 p.insert<SplitNpuBlockWriteOpPattern>(&getContext());
1257 if (failed(applyPatternsGreedily(getOperation(), std::move(p))))
1258 signalPassFailure();
1259 }
1260
1261 target.addIllegalOp<AIEX::NpuBlockWriteOp>();
1262 target.addIllegalOp<AIEX::NpuBlockWriteValuesOp>();
1263 target.addIllegalOp<AIEX::NpuMaskWrite32Op>();
1264 target.addIllegalOp<AIEX::NpuMaskPollOp>();
1265 target.addIllegalOp<AIEX::NpuSyncOp>();
1266 target.addIllegalOp<AIEX::NpuWrite32Op>();
1267 target.addIllegalOp<AIEX::NpuLoadPdiOp>();
1268
1269 unsigned nextChainId = 0;
1270 RewritePatternSet p2(&getContext());
1271 p2.insert<NpuBlockWriteToCertUcDma>(&getContext(), nextChainId);
1272 p2.insert<NpuMaskWrite32ToCertMaskWrite32>(&getContext());
1273 p2.insert<NpuMaskPollToCertMaskPoll32>(&getContext());
1274 p2.insert<NpuWrite32ToCertWrite32>(&getContext());
1275 p2.insert<NpuSyncToCertWaitTCTS>(&getContext());
1276 p2.insert<NpuLoadPdiToCertLoadPdi>(&getContext());
1277
1278 if (failed(applyPartialConversion(currentDevice, target, std::move(p2))))
1279 return signalPassFailure();
1280
1281 // Run the greedy driver with no patterns, purely for its DCE: p2
1282 // leaves each converted blockwrite's memref.get_global (and any constant
1283 // feeding it) dead, since the new op references the global by symbol
1284 // rather than by SSA value. CertJobOp's verifier rejects those ops
1285 RewritePatternSet cleanup(&getContext());
1286 if (failed(applyPatternsGreedily(getOperation(), std::move(cleanup))))
1287 return signalPassFailure();
1288
1289 // Convert referenced devices to cert.sections
1290 // Now that cert.load_pdi ops exist, find referenced devices and convert
1291 // their cert jobs into sections in this device
1292 {
1293 OpBuilder builder(&getContext());
1294
1295 // Find all cert.load_pdi in current device
1296 llvm::SetVector<StringRef> referencedDeviceNames;
1297 currentDevice.walk([&](AIEX::CertLoadPdiOp loadPdiOp) {
1298 StringRef refSymName = loadPdiOp.getSymbol();
1299 if (!currentDevice.lookupSymbol(refSymName))
1300 referencedDeviceNames.insert(refSymName);
1301 });
1302
1303 // For each referenced device, create a section with its cert jobs
1304 for (StringRef refSymName : referencedDeviceNames) {
1305 // Find the referenced device
1306 AIE::DeviceOp refDevice = nullptr;
1307 for (auto &op : moduleOp.getBody()->getOperations()) {
1308 if (auto dev = dyn_cast<AIE::DeviceOp>(op)) {
1309 if (dev.getSymName() == refSymName) {
1310 refDevice = dev;
1311 break;
1312 }
1313 }
1314 }
1315
1316 if (!refDevice)
1317 continue;
1318
1319 // Find all cert.page ops in the referenced device that are direct
1320 // children
1321 llvm::SmallVector<AIEX::CertPageOp, 4> certPages;
1322 for (auto &op : refDevice.getBody()->getOperations()) {
1323 if (auto pageOp = dyn_cast<AIEX::CertPageOp>(op)) {
1324 certPages.push_back(pageOp);
1325 }
1326 }
1327
1328 if (certPages.empty())
1329 continue;
1330
1331 // Create cert.section in current device
1332 builder.setInsertionPoint(currentDevice.getBody()->getTerminator());
1333 auto sectionOp = AIEX::CertSectionOp::create(
1334 builder, refDevice.getLoc(), builder.getStringAttr(refSymName));
1335
1336 Block *sectionBlock = new Block();
1337 sectionOp.getBody().push_back(sectionBlock);
1338
1339 // Clone all cert.page operations into the section
1340 IRMapping mapper;
1341 for (auto pageOp : certPages) {
1342 builder.setInsertionPointToEnd(sectionBlock);
1343 builder.clone(*pageOp.getOperation(), mapper);
1344 }
1345
1346 // Ensure section terminator
1347 AIEX::CertSectionOp::ensureTerminator(sectionOp.getBody(), builder,
1348 refDevice.getLoc());
1349 }
1350 }

1352 // Clean up unused block arguments from cert.job operations
1353 currentDevice.walk([&](AIEX::CertJobOp jobOp) {
1354 Block &jobBlock = jobOp.getBody().front();
1355
1356 // Remove unused block arguments (in reverse order)
1357 for (int i = jobBlock.getNumArguments() - 1; i >= 0; --i) {
1358 BlockArgument arg = jobBlock.getArgument(i);
1359 if (arg.use_empty()) {
1360 jobBlock.eraseArgument(i);
1361 }
1362 }
1363 });
1364 }

1378// Registers r0..r7 are local to a job. Splitting a job, or moving some of its
1379// ops into a new job, gives each part its own local registers, so a value
1380// defined in one part is gone in the other. r8..r23 are shared by all jobs and
1381// survive both.
1382static bool isLocalRegister(uint32_t r) { return r < 8; }
1383
1384// The local registers `o` reads (uses) and writes (defs).
1385static void collectLocalRegisterDefsUses(Operation &o,
1386 SmallVectorImpl<uint32_t> &uses,
1387 SmallVectorImpl<uint32_t> &defs) {
1388 auto add = [](SmallVectorImpl<uint32_t> &regs, uint32_t r) {
1389 if (isLocalRegister(r))
1390 regs.push_back(r);
1391 };
1392 if (auto op = dyn_cast<AIEX::CertUcDmaWriteDesOp>(o)) {
1393 add(defs, op.getWaitHandle());
1394 } else if (auto op = dyn_cast<AIEX::CertWaitUcDmaOp>(o)) {
1395 add(uses, op.getWaitHandle());
1396 } else if (auto op = dyn_cast<AIEX::CertRead32Op>(o)) {
1397 add(defs, op.getValue());
1398 } else if (auto op = dyn_cast<AIEX::CertRead32DOp>(o)) {
1399 add(uses, op.getAddress());
1400 add(defs, op.getValue());
1401 } else if (auto op = dyn_cast<AIEX::CertWrite32DOp>(o)) {
1402 if (op.getAddressIsReg())
1403 add(uses, op.getAddress());
1404 if (op.getValueIsReg())
1405 add(uses, op.getValue());
1406 }
1407}

1409// A job touching any local register is never split, which keeps every
1410// definition and use of those registers in one job.
1411static bool jobUsesLocalRegister(AIEX::CertJobOp job) {
1412 SmallVector<uint32_t> uses, defs;
1413 for (Operation &o : job.getBody().front())
1414 collectLocalRegisterDefsUses(o, uses, defs);
1415 return !uses.empty() || !defs.empty();
1416}

1418// The first local register `ops` reads before defining it: its value comes
1419// from earlier in the job, so moving `ops` into a new job would lose it.
1420static std::optional<uint32_t>
1421findLocalRegisterCrossingIn(ArrayRef<Operation *> ops) {
1422 llvm::SmallDenseSet<uint32_t> defined;
1423 SmallVector<uint32_t> uses, defs;
1424 for (Operation *o : ops) {
1425 uses.clear();
1426 defs.clear();
1427 collectLocalRegisterDefsUses(*o, uses, defs);
1428 for (uint32_t r : uses)
1429 if (!defined.contains(r))
1430 return r;
1431 defined.insert(defs.begin(), defs.end());
1432 }
1433 return std::nullopt;
1434}

1436// AIEBU lays down each chain and each payload global once per page, however
1437// many pushes on that page name them, so data is charged once per page.
1438struct PageDataSeen {
1439 llvm::DenseSet<Operation *> chains;
1440 llvm::DenseSet<Operation *> payloads;
1441};

1443// A pushed chain's BDs and payloads, charged the first time the page names
1444// them.
1445static void addChainDataCost(StringRef sym_name, AIE::DeviceOp deviceOp,
1446 uint32_t &data_cost, PageDataSeen &seen) {
1447 auto chain = dyn_cast_if_present<AIEX::CertUcDmaChainOp>(
1448 deviceOp.lookupSymbol(sym_name));
1449 if (!chain || !seen.chains.insert(chain).second)
1450 return;
1451 for (auto bdOp : chain.getBody().front().getOps<AIEX::CertUcDmaBdOp>()) {
1452 data_cost += 16; // bd op
1453 auto global = dyn_cast_if_present<memref::GlobalOp>(
1454 deviceOp.lookupSymbol(bdOp.getRemoteAddress()));
1455 if (!global || !seen.payloads.insert(global).second)
1456 continue;
1457 auto initVal = global.getInitialValue();
1458 if (!initVal)
1459 continue;
1460 auto data = dyn_cast<DenseIntElementsAttr>(*initVal);
1461 if (!data)
1462 continue;
1463 data_cost += data.getNumElements() * 4; // 4 bytes per element
1464 }
1465}

1467// Instruction sizes follow the AIEBU CERT ISA specification so the running
1468// total here matches the bytes emitJob actually produces. Keep the cost model
1469// and emitter in sync when adding a cert op:
1470// an op the emitter can emit but this function ignores is counted as free,
1471// which makes the page look smaller than it is.
1472static void updateCostForOp(Operation &o, AIE::DeviceOp deviceOp,
1473 uint32_t &text_cost, uint32_t &data_cost,
1474 PageDataSeen &seen) {
1475 // Several distinct op kinds share an encoded instruction size below --
1476 // an instruction-size table, not a copy-paste clone.
1477 // NOLINTBEGIN(bugprone-branch-clone)
1478 if (isa<AIEX::CertLocalBarrierOp>(o)) {
1479 text_cost += 4; // ISA_OPSIZE_LOCAL_BARRIER
1480 } else if (isa<AIEX::CertRemoteBarrierOp>(o)) {
1481 text_cost += 8; // ISA_OPSIZE_REMOTE_BARRIER
1482 } else if (isa<AIEX::CertRelAcqSyncOp>(o)) {
1483 text_cost += 12; // encoded REL_ACQ_SYNC instruction size
1484 } else if (isa<AIEX::CertWaitTCTSOp>(o)) {
1485 text_cost += 8; // wait tct
1486 } else if (isa<AIEX::CertMaskWrite32Op, AIEX::CertMaskPoll32Op>(o)) {
1487 text_cost += 16; // mask write / poll
1488 } else if (isa<AIEX::CertWrite32Op>(o)) {
1489 text_cost += 12; // ISA_OPSIZE_WRITE_32
1490 } else if (isa<AIEX::CertApplyOffset57Op>(o)) {
1491 text_cost += 8; // ISA_OPSIZE_APPLY_OFFSET_57
1492 } else if (isa<AIEX::CertNopOp>(o)) {
1493 text_cost += 4; // ISA_OPSIZE_NOP
1494 } else if (isa<AIEX::CertPreemptOp>(o)) {
1495 text_cost += 8; // ISA_OPSIZE_PREEMPT
1496 } else if (isa<AIEX::CertLoadPdiOp>(o)) {
1497 text_cost += 12; // ISA_OPSIZE_LOAD_PDI
1498 // NOLINTEND(bugprone-branch-clone)
1499 } else if (auto syncOp = dyn_cast<AIEX::CertUcDmaWriteDesSyncOp>(o)) {
1500 text_cost += 4; // ISA_OPSIZE_UC_DMA_WRITE_DES_SYNC
1501 addChainDataCost(syncOp.getSymbol(), deviceOp, data_cost, seen);
1502 } else if (auto desOp = dyn_cast<AIEX::CertUcDmaWriteDesOp>(o)) {
1503 text_cost += 8; // ISA_OPSIZE_UC_DMA_WRITE_DES
1504 addChainDataCost(desOp.getSymbol(), deviceOp, data_cost, seen);
1505 } else if (isa<AIEX::CertWaitUcDmaOp>(o)) {
1506 text_cost += 4; // ISA_OPSIZE_WAIT_UC_DMA
1507 } else if (isa<AIEX::CertRead32Op>(o)) {
1508 text_cost += 8; // ISA_OPSIZE_READ_32
1509 } else if (isa<AIEX::CertRead32DOp>(o)) {
1510 text_cost += 4; // ISA_OPSIZE_READ_32_D
1511 } else if (isa<AIEX::CertWrite32DOp>(o)) {
1512 text_cost += 12; // ISA_OPSIZE_WRITE_32_D
1513 } else if (isa<AIEX::CertSaveTimestampsOp>(o)) {
1514 text_cost += 8; // ISA_OPSIZE_SAVE_TIMESTAMPS
1515 } else if (isa<AIEX::CertSaveRegisterOp>(o)) {
1516 text_cost += 12; // ISA_OPSIZE_SAVE_REGISTER
1517 }
1518}

1520static uint32_t estimateCost(AIEX::CertPageOp op, uint32_t split_target,
1521 AIEX::CertJobOp &split_job,
1522 Block::iterator &split_iter,
1523 bool &found_split_point) {
1524 uint32_t text_cost = 32; // page header: `.align 16` padding plus `.eop`
1525 uint32_t data_cost = 0;
1526 PageDataSeen seen;
1527 found_split_point = false;
1528 AIE::DeviceOp deviceOp = op->getParentOfType<AIE::DeviceOp>();
1529
1530 for (auto job : op.getBody().front().getOps<AIEX::CertJobOp>()) {
1531 // Cuts inside a job that uses local registers would lose their values.
1532 bool splittable = !jobUsesLocalRegister(job);
1533 // START_JOB precedes the body, so it is already spent at every candidate
1534 // split point inside this job; END_JOB is charged after the body below.
1535 text_cost += cert_job_start_cost;
1536 for (auto &o : job.getBody().front().getOperations()) {
1537 Block::iterator current(&o);
1538 if (splittable && !found_split_point && !isa<AIE::EndOp>(o) &&
1539 current != job.getBody().front().begin() &&
1540 (text_cost + data_cost) >= split_target) {
1541 split_job = job;
1542 split_iter = current;
1543 found_split_point = true;
1544 }
1545
1546 updateCostForOp(o, deviceOp, text_cost, data_cost, seen);
1547
1548 if (splittable && !found_split_point &&
1549 (text_cost + data_cost) >= split_target) {
1550 Block::iterator next = current;
1551 ++next;
1552 if (next != job.getBody().front().end() && !isa<AIE::EndOp>(*next)) {
1553 split_job = job;
1554 split_iter = next;
1555 found_split_point = true;
1556 } else if (!isa<AIE::EndOp>(o) &&
1557 current != job.getBody().front().begin()) {
1558 // Cutting after the last real op would leave an empty second half.
1559 // Cut before it; exclude the terminator to guarantee progress.
1560 split_job = job;
1561 split_iter = current;
1562 found_split_point = true;
1563 }
1564 }
1565 }
1566 text_cost += cert_job_end_cost;
1567 }
1568 return text_cost + data_cost;
1569}

1571// A cut before flat index p separates a span [lo, hi] when lo < p <= hi.
1572// Local barrier participants must share a page (see
1573// checkLocalBarrierColocation). A payload patch must share a page with every
1574// later uC-DMA push of that payload: each page carries its own payload copy,
1575// so a push on another page would use the unpatched addresses.
1576enum class SpanKind { LocalBarrier, PatchedPayload };
1577
1578struct IndivisibleSpan {
1579 int lo, hi;
1580 SpanKind kind;
1581};
1582
1583// Index non-terminator ops in textual order and record indivisible spans.
1584static void computeIndivisibleSpans(AIEX::CertPageOp page,
1585 std::map<Operation *, int> &flatIndex,
1586 SmallVectorImpl<IndivisibleSpan> &spans) {
1587 AIE::DeviceOp deviceOp = page->getParentOfType<AIE::DeviceOp>();
1588 std::map<int, std::pair<int, int>> barriers;
1589 // Patches accumulate and survive pushes, so later pushes depend on the
1590 // earliest patch of their payload.
1591 llvm::StringMap<int> firstPatch;
1592 auto payloadsOf = [&](StringRef chainSym) {
1593 SmallVector<StringRef> payloads;
1594 auto chain = dyn_cast_if_present<AIEX::CertUcDmaChainOp>(
1595 deviceOp ? deviceOp.lookupSymbol(chainSym) : nullptr);
1596 if (!chain || chain.getBody().empty())
1597 return payloads;
1598 for (auto bd : chain.getBody().front().getOps<AIEX::CertUcDmaBdOp>())
1599 payloads.push_back(bd.getRemoteAddress());
1600 return payloads;
1601 };
1602
1603 int idx = 0;
1604 for (auto job : page.getBody().front().getOps<AIEX::CertJobOp>()) {
1605 for (auto &o : job.getBody().front().getOperations()) {
1606 if (isa<AIE::EndOp>(o))
1607 continue;
1608 flatIndex[&o] = idx;
1609 if (auto bar = dyn_cast<AIEX::CertLocalBarrierOp>(o)) {
1610 int id = static_cast<int>(bar.getLocalBarrierId());
1611 auto it = barriers.find(id);
1612 if (it == barriers.end())
1613 barriers[id] = {idx, idx};
1614 else {
1615 it->second.first = std::min(it->second.first, idx);
1616 it->second.second = std::max(it->second.second, idx);
1617 }
1618 } else if (auto patch = dyn_cast<AIEX::CertApplyOffset57Op>(o)) {
1619 firstPatch.try_emplace(patch.getSymbol(), idx);
1620 } else if (StringRef pushed = pushedChain(o); !pushed.empty()) {
1621 for (StringRef payload : payloadsOf(pushed)) {
1622 auto it = firstPatch.find(payload);
1623 if (it != firstPatch.end())
1624 spans.push_back({it->second, idx, SpanKind::PatchedPayload});
1625 }
1626 }
1627 ++idx;
1628 }
1629 }
1630 for (auto &[id, span] : barriers)
1631 spans.push_back({span.first, span.second, SpanKind::LocalBarrier});
1632}

1634// Return whether the cut separates a span, optionally reporting its kind.
1635static bool cutSeparatesSpan(ArrayRef<IndivisibleSpan> spans, int p,
1636 SpanKind *kind = nullptr) {
1637 const auto *it = llvm::find_if(
1638 spans, [p](const IndivisibleSpan &s) { return s.lo < p && s.hi >= p; });
1639 if (it == spans.end())
1640 return false;
1641 if (kind)
1642 *kind = it->kind;
1643 return true;
1644}

1646// Choose the span-safe cut closest to split_target, allowing both interior
1647// and job-boundary cuts. Each half needs content so repeated splitting
1648// progresses.
1649static bool findLegalSplitPoint(AIEX::CertPageOp page, uint32_t split_target,
1650 const std::map<Operation *, int> &flatIndex,
1651 ArrayRef<IndivisibleSpan> spans,
1652 AIEX::CertJobOp &split_job,
1653 Block::iterator &split_iter) {
1654 AIE::DeviceOp deviceOp = page->getParentOfType<AIE::DeviceOp>();
1655 uint32_t text_cost = 32; // page header, mirrors estimateCost
1656 uint32_t data_cost = 0;
1657 PageDataSeen seen;
1658 bool found = false;
1659 uint32_t best_delta = 0;
1660 // Whether some job already visited holds a real op, i.e. whether cutting at
1661 // the current job's first op would leave content on the earlier page.
1662 bool sawOpInEarlierJob = false;
1663 for (auto job : page.getBody().front().getOps<AIEX::CertJobOp>()) {
1664 text_cost += cert_job_start_cost; // mirrors estimateCost
1665 bool splittable = !jobUsesLocalRegister(job); // mirrors estimateCost
1666 bool sawOpInJob = false;
1667 for (auto &o : job.getBody().front().getOperations()) {
1668 if (isa<AIE::EndOp>(o))
1669 continue;
1670 Block::iterator current(&o);
1671 // Candidate: cut before `o`, legal as an interior cut when an op of this
1672 // job precedes it, or as a job-boundary cut when an earlier job does.
1673 // Admitting the latter matters for a page built of small jobs whose sizes
1674 // only sum over the limit: no job has an interior point worth cutting at,
1675 // yet a cut between two whole jobs splits it perfectly well.
1676 if (sawOpInJob ? splittable : sawOpInEarlierJob) {
1677 auto fi = flatIndex.find(&o);
1678 int p = fi == flatIndex.end() ? 0 : fi->second;
1679 if (!cutSeparatesSpan(spans, p)) {
1680 uint32_t cost = text_cost + data_cost;
1681 uint32_t delta =
1682 cost > split_target ? cost - split_target : split_target - cost;
1683 if (!found || delta < best_delta) {
1684 found = true;
1685 best_delta = delta;
1686 split_job = job;
1687 split_iter = current;
1688 }
1689 }
1690 }
1691 updateCostForOp(o, deviceOp, text_cost, data_cost, seen);
1692 sawOpInJob = true;
1693 }
1694 text_cost += cert_job_end_cost; // mirrors estimateCost
1695 sawOpInEarlierJob |= sawOpInJob;
1696 }
1697 return found;
1698}

1702// Pattern to isolate load_pdi and preempt operations into their own job and
1703// page According to CERT spec: "load_pdi and preempt should take one whole job
1704// which in turn should take one whole page"
1705struct IsolateFullPageOpsPattern : OpRewritePattern<AIEX::CertJobOp> {
1706 // `hadError` is raised with an error diagnostic, as in
1707 // SplitCertPageOpPattern.
1708 IsolateFullPageOpsPattern(MLIRContext *context, bool *hadError)
1709 : OpRewritePattern(context), hadError(hadError) {}
1710
1711 // Create an empty, terminated cert.page directly after `after`, in the same
1712 // block. Staying in `after`'s block is what keeps a page that was moved into
1713 // a cert.attach_to_group by placement lowering on its uC.
1714 static AIEX::CertPageOp makePageAfter(PatternRewriter &rewriter,
1715 Operation *after, Location loc) {
1716 rewriter.setInsertionPointAfter(after);
1717 auto page = AIEX::CertPageOp::create(rewriter, loc);
1718 page.getBody().push_back(new Block());
1719 AIEX::CertPageOp::ensureTerminator(page.getBody(), rewriter, loc);
1720 return page;
1721 }
1722
1723 LogicalResult matchAndRewrite(AIEX::CertJobOp jobOp,
1724 PatternRewriter &rewriter) const override {
1725
1726 // Find first load_pdi or preempt in this job
1727 Operation *fullPageOp = nullptr;
1728 Block::iterator fullPageOpIter;
1729 for (Block::iterator it = jobOp.getBody().front().begin();
1730 it != jobOp.getBody().front().end(); ++it) {
1731 if (isa<AIEX::CertLoadPdiOp, AIEX::CertPreemptOp>(*it)) {
1732 fullPageOp = &*it;
1733 fullPageOpIter = it;
1734 break;
1735 }
1736 }
1737
1738 if (!fullPageOp)
1739 return failure(); // No full-page op in this job
1740
1741 auto parentPage = jobOp->getParentOfType<AIEX::CertPageOp>();
1742 if (!parentPage)
1743 return failure(); // No parent page (unusual but skip)
1744
1745 // Real ops in this job (the terminator does not count).
1746 size_t opCount = 0;
1747 for (Operation &op : jobOp.getBody().front().getOperations())
1748 if (!isa<AIE::EndOp>(op))
1749 opCount++;
1750
1751 // Pages execute in order, so the rewrite has to preserve this job's
1752 // position among its *siblings*, not just relative to the parent page:
1753 // anchoring every new page on the parent page's boundary hoists the jobs
1754 // that surround this one across the isolated load_pdi/preempt. Nothing
1755 // downstream checks op order, so that reordering is silent.
1756 SmallVector<Operation *> leadingSiblings, trailingSiblings;
1757 bool pastSelf = false;
1758 for (Operation &op : parentPage.getBody().front()) {
1759 if (isa<AIE::EndOp>(op))
1760 continue;
1761 if (&op == jobOp.getOperation()) {
1762 pastSelf = true;
1763 continue;
1764 }
1765 (pastSelf ? trailingSiblings : leadingSiblings).push_back(&op);
1766 }
1767
1768 // Already properly isolated: sole op of the sole job of its page.
1769 if (opCount == 1 && leadingSiblings.empty() && trailingSiblings.empty())
1770 return failure();
1771
1772 auto loc = jobOp.getLoc();
1773
1774 // The parent page stays where it is and keeps the leading siblings; every
1775 // page created below is chained after it, in program order.
1776 Operation *anchor = parentPage.getOperation();
1777
1778 if (opCount == 1) {
1779 // The job is exactly the full-page op: it just needs a page of its own.
1780 auto isolatedPage = makePageAfter(rewriter, anchor, loc);
1781 jobOp->moveBefore(isolatedPage.getBody().front().getTerminator());
1782 anchor = isolatedPage;
1783 } else {
1784 // Mixed job: split it into up to three jobs, each on its own page, in
1785 // source order (ops before the full-page op, it, then the ops after).
1786 auto parentDevice = jobOp->getParentOfType<AIE::DeviceOp>();
1787 uint32_t maxJobId = 0;
1788 parentDevice.walk([&](AIEX::CertJobOp certJobOp) {
1789 maxJobId = std::max(maxJobId, certJobOp.getJobId());
1790 });
1791 uint32_t beforeJobId = jobOp.getJobId();
1792 uint32_t fullPageJobId = maxJobId + 1;
1793 uint32_t afterJobId = maxJobId + 2;
1794
1795 SmallVector<Operation *> beforeOps, afterOps;
1796 for (Block::iterator it = jobOp.getBody().front().begin();
1797 it != fullPageOpIter; ++it)
1798 if (!isa<AIE::EndOp>(*it))
1799 beforeOps.push_back(&*it);

1837 }

1840 // the isolated page, so they move to a page of their own behind it.

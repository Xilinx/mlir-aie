20// The name of a mode, t<elemBits>_<rows>x<cols>[_lo|_hi|_flip], read as a
21// matrix of rows x cols elements.
22struct ShuffleModeShape {
23 unsigned elemBits;
24 unsigned rows;
25 unsigned cols;
26 // Reverses each group of `cols` elements instead of transposing.
27 bool flip;
28 // Returns the upper 512 bits of the 1024-bit result.
29 bool hi;
30
31 unsigned blockBits() const { return elemBits * rows * cols; }
32 // The matrix spans lhs:rhs, so the mode reads both operands.
33 bool isTwoInput() const { return blockBits() == 1024; }
34};

41// Byte i of the 512-bit result is byte perm[i] of the 128-byte lhs:rhs. The

43// 512 bits is transposed in each block of lhs of its size.

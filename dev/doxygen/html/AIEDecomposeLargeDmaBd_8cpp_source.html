45static bool allConstant(NpuDmaMemcpyNdOp op) {
46 return llvm::all_of(op.getMixedSizes(),
47 [](OpFoldResult s) {
48 return getConstantIntValue(s).has_value();
49 }) &&
50 llvm::all_of(op.getMixedStrides(),
51 [](OpFoldResult s) {
52 return getConstantIntValue(s).has_value();
53 }) &&
54 llvm::all_of(op.getMixedOffsets(), [](OpFoldResult s) {
55 return getConstantIntValue(s).has_value();
56 });
57}

59static bool allConstant(AIE::DMABDOp op) {
60 if (op.getPadDimensions().has_value())
61 return false;
62 if (op.getOffset() && !op.getConstantOffset())
63 return false;
64 if (op.getLen() && !op.getConstantLen())
65 return false;
66 if (op.getMixedSizes().empty())
67 return false;
68 return llvm::all_of(op.getMixedSizes(),
69 [](OpFoldResult s) {
70 return getConstantIntValue(s).has_value();
71 }) &&
72 llvm::all_of(op.getMixedStrides(), [](OpFoldResult s) {
73 return getConstantIntValue(s).has_value();
74 });
75}

77static NdDmaPattern patternFromOp(NpuDmaMemcpyNdOp op) {
78 NdDmaPattern pattern;
79 pattern.offsets = llvm::map_to_vector(
80 llvm::reverse(op.getMixedOffsets()),
81 [](OpFoldResult s) { return getConstantIntValue(s).value(); });
82 pattern.sizes = llvm::map_to_vector(
83 llvm::reverse(op.getMixedSizes()),
84 [](OpFoldResult s) { return getConstantIntValue(s).value(); });
85 pattern.strides = llvm::map_to_vector(
86 llvm::reverse(op.getMixedStrides()),
87 [](OpFoldResult s) { return getConstantIntValue(s).value(); });
88 return pattern;
89}

91// Only called after allConstant(op) confirmed every size/stride resolves to
92// a constant, so the has_value() checks below are redundant in practice --
93// asserted rather than re-verified to keep that invariant visible here too.
94static NdDmaPattern patternFromDmaBd(AIE::DMABDOp op) {
95 SmallVector<int64_t, kNdDmaDims> outerSizes;
96 SmallVector<int64_t, kNdDmaDims> outerStrides;
97 for (OpFoldResult s : op.getMixedSizes()) {
98 auto c = getConstantIntValue(s);
99 assert(c && "size must be constant (already checked by allConstant)");
100 outerSizes.push_back(*c);
101 }
102 for (OpFoldResult s : op.getMixedStrides()) {
103 auto c = getConstantIntValue(s);
104 assert(c && "stride must be constant (already checked by allConstant)");
105 outerStrides.push_back(*c);
106 }
107 while (outerSizes.size() < kNdDmaDims) {
108 outerSizes.insert(outerSizes.begin(), 1);
109 outerStrides.insert(outerStrides.begin(), 0);
110 }
111
112 NdDmaPattern pattern;
113 pattern.offsets.assign(outerSizes.size(), 0);
114 pattern.sizes = llvm::map_to_vector(llvm::reverse(outerSizes),
115 [](int64_t v) { return v; });
116 pattern.strides = llvm::map_to_vector(llvm::reverse(outerStrides),
117 [](int64_t v) { return v; });
118 return pattern;
119}

125static int64_t flatOffsetFromPattern(int64_t baseFlatOffset,
126 const NdDmaPattern &pattern) {
127 int64_t flat = baseFlatOffset + pattern.baseOffset;
128 for (unsigned k = 0; k < 4; ++k)
129 flat += pattern.offsets[k] * pattern.strides[k];
130 return flat;

192// The factor by which decomposition scaled the executions of a pass, num/den
193// in lowest terms: a pass over the original takes `iterations` executions,
194// and over `after` one per index of its iteration dimension. Below one, the
195// executions merged, `after`'s inner dimensions taking in one the original
196// iterated over.
197struct IterationScale {
198 int64_t num = 1;
199 int64_t den = 1;
200
201 bool isOne() const { return num == den; }
202 // Whether `runs` executions of the original are whole executions of the
203 // rewrite.
204 bool divides(int64_t runs) const { return runs % den == 0; }
205 int64_t apply(int64_t runs) const { return runs / den * num; }
206 std::string str() const {
207 return den == 1 ? std::to_string(num)
208 : std::to_string(num) + "/" + std::to_string(den);
209 }
210};
211
212static IterationScale iterationScale(int64_t iterations,
213 const NdDmaPattern &after) {
214 int64_t common = std::gcd(after.sizes[3], iterations);
215 return {after.sizes[3] / common, iterations / common};
216}

255 if (!tile)
256 return std::nullopt;
257 return std::make_pair(tile, cast<DMAConfigureTaskLike>(*cfgFor));
258 }
259 return std::nullopt;

298// Why the slices of `bd` cannot go out as one task each, or nullopt if they
299// can. Every use of the task has to be one splitIntoTasks rewrites, and every
300// start has to run whole passes over the pattern, a pass being one execution
301// per index of the `iterations`-long iteration dimension: the slices of a pass
302// are issued in turn, so a partial pass has no slice boundary to stop at.
303static std::optional<std::string>
304whyNotSeparateTasks(AIE::DMABDOp bd, DMAConfigureTaskLike taskOp,
305 int64_t iterations) {
306 std::string why;
307 llvm::raw_string_ostream os(why);
308 if (bd.getBdIdVal()) {
309 os << "the descriptor's bd_id is a runtime value";
310 return why;
311 }
312 if (taskOp.getRepeatCountVal()) {
313 os << "the task's repeat count is a runtime value";
314 return why;
315 }
316 for (Operation *user : taskOp->getResult(0).getUsers()) {
317 if (!isa<DMAStartTaskOp, DMAAwaitTaskOp, DMAFreeTaskOp>(user)) {
318 os << "the task is used by '" << user->getName() << "'";
319 return why;
320 }
321 if (user->getBlock() != taskOp->getBlock()) {
322 os << "the task is used outside the block that configures it";
323 return why;
324 }
325 auto start = dyn_cast<DMAStartTaskOp>(user);
326 if (!start)
327 continue;
328 if (start.getRepeatCountVal()) {
329 os << "a start's repeat count is a runtime value";
330 return why;
331 }
332 std::optional<uint32_t> rc = start.getRepeatCount();
333 int64_t runs = static_cast<int64_t>(rc ? *rc : taskOp.getRepeatCount());
334 ++runs;
335 if (runs % iterations != 0) {
336 os << "a start runs it " << runs
337 << " times, not a whole number of passes over its " << iterations
338 << "-long iteration dimension";
339 return why;
340 }
341 }
342 return std::nullopt;
343}

345// Issue the slices of `bd` as one single-descriptor task each. The task
346// becomes the first slice and the others are configured right after it, until
347// orderSlices moves each up to its first start. Only the last slice issues the
348// task's token, so an await of the task becomes an await of the last slice,
349// and a free frees every slice. Each start becomes a
350// group of consecutive starts, every slice once per pass, which orderSlices
351// then interleaves with the starts around it.
352static void splitIntoTasks(RewriterBase &rewriter, AIE::DMABDOp bd,
353 DMAConfigureTaskLike taskOp,
354 ArrayRef<NdDmaPattern> slices,
355 int64_t baseFlatOffset, int64_t iterations,
356 Slices &state) {
357 bool token = taskOp.getIssueToken();
358 int64_t taskRuns = static_cast<int64_t>(taskOp.getRepeatCount()) + 1;
359 SmallVector<Operation *> users(taskOp->getResult(0).getUsers());
360
361 SmallVector<DMAConfigureTaskLike> tasks{taskOp};
362 rewriter.setInsertionPointAfter(taskOp);
363 for (size_t i = 1; i < slices.size(); ++i) {
364 tasks.push_back(cast<DMAConfigureTaskLike>(rewriter.clone(*taskOp)));
365 state.lateTasks.insert(tasks.back());
366 }
367
368 for (auto it : llvm::enumerate(slices)) {
369 size_t i = it.index();
370 const NdDmaPattern &sub = it.value();
371 DMAConfigureTaskLike task = tasks[i];
372 AIE::DMABDOp sliceBd = bd;
373 if (i > 0)
374 task->walk([&](AIE::DMABDOp b) { sliceBd = b; });
375 int64_t flatOffset = flatOffsetFromPattern(baseFlatOffset, sub);
376 int32_t len = static_cast<int32_t>(lenFromInnermost3(sub.sizes));
377 rewriter.modifyOpInPlace(sliceBd, [&]() {
378 updateTaskBdInPlace(sliceBd, static_cast<int32_t>(flatOffset), len,
379 toOuter(sub.sizes), toOuter(sub.strides));
380 // A pinned id stays with the first slice, as in a chain.
381 if (i > 0)
382 sliceBd.removeBdIdAttr();
383 });
384 rewriter.modifyOpInPlace(task, [&]() {
385 // One pass over the slice is one execution per index of its own
386 // iteration dimension.
387 task.setRepeatCount(static_cast<uint32_t>(sub.sizes[3] - 1));
388 task.setIssueToken(token && i + 1 == slices.size());
389 });
390 }
391
392 for (Operation *user : users) {
393 if (auto await = dyn_cast<DMAAwaitTaskOp>(user)) {
394 rewriter.modifyOpInPlace(await, [&]() {
395 await.getTaskMutable().assign(tasks.back()->getResult(0));
396 });
397 continue;
398 }
399 if (auto free = dyn_cast<DMAFreeTaskOp>(user)) {
400 rewriter.setInsertionPointAfter(free);
401 for (DMAConfigureTaskLike task : llvm::drop_begin(tasks))
402 DMAFreeTaskOp::create(rewriter, free.getLoc(), task->getResult(0));
403 continue;
404 }
405 auto start = cast<DMAStartTaskOp>(user);
406 std::optional<uint32_t> rc = start.getRepeatCount();
407 int64_t runs = rc ? static_cast<int64_t>(*rc) + 1 : taskRuns;
408 int64_t count = runs / iterations * static_cast<int64_t>(slices.size());
409 int64_t group = state.nextGroup++;
410 // Only the last start of the last slice may issue the token: an earlier
411 // pass's would complete an await before the transfer has.
412 bool startToken = token && !start.getNoToken();
413 auto mark = [&](DMAStartTaskOp s, int64_t index) {
414 state.starts[s] = {group, index, count};
415 };
416 rewriter.modifyOpInPlace(start, [&]() { start.removeRepeatCountAttr(); });
417 mark(start, 0);
418 state.blocks.insert(start->getBlock());
419 rewriter.setInsertionPointAfter(start);
420 for (int64_t k = 1; k < count; ++k) {
421 size_t i = static_cast<size_t>(k) % slices.size();
422 bool withholds =
423 token && i + 1 == slices.size() && (!startToken || k + 1 != count);
424 auto s = DMAStartTaskOp::create(
425 rewriter, start.getLoc(), tasks[i]->getResult(0),
426 /*repeat_count=*/nullptr, /*repeat_count_val=*/nullptr,
427 /*no_token=*/withholds ? rewriter.getUnitAttr() : nullptr);
428 mark(s, k);
429 }
430 }
431}

435// The channel a start pushes onto, or nullopt when its task is not a configure
436// on a placed tile this pass can read.
437static std::optional<ChannelKey> channelOf(DMAStartTaskOp start) {
438 Operation *def = start.getTask().getDefiningOp();
439 if (auto cfg = dyn_cast_or_null<DMAConfigureTaskOp>(def)) {
440 if (AIE::TileOp tile = cfg.tryGetTileOp())
441 return DmaQueueModel::keyOf(tile, cfg.getDirection(), cfg.getChannel());
442 return std::nullopt;
443 }
444 auto cfgFor = dyn_cast_or_null<DMAConfigureTaskForOp>(def);
445 if (!cfgFor)
446 return std::nullopt;
447 auto dev = cfgFor->getParentOfType<AIE::DeviceOp>();
448 auto alloc = dev ? AIE::ShimDMAAllocationOp::getForSymbol(
449 dev, cfgFor.getAlloc().getRootReference())
450 : AIE::ShimDMAAllocationOp();
451 AIE::TileOp tile = alloc ? alloc.getTileOp() : AIE::TileOp();
452 if (!tile)
453 return std::nullopt;
454 return DmaQueueModel::keyOf(tile, alloc.getChannelDir(),
455 alloc.getChannelIndex());
456}

458// Interleave the slices splitIntoTasks emitted with the starts around them.
459// Later slices wait for earlier ones to finish (see
460// aie-assign-runtime-sequence-bd-ids), so a counterpart transfer left behind
461// all of them (a core's input for an output, say) would deadlock.
462//
463// A round runs from one start up to the next start on a channel it already
464// used. Each start's first slice keeps its place; later ones follow the
465// round's last start, ordered by index / size, ties in program order. Slices
466// move past configures, RTP writes and lock sets, which never wait. Any other
467// effect ends a round: an await, sync, poll, untracked push, free of an
468// unplaced slice, or a write32/blockwrite (either may be a push).
469static void orderSlices(Block &block, Slices &state) {
470 struct Pending {
471 DMAStartTaskOp start;
472 int64_t index, count, head;
473 };
474 SmallVector<Pending> pending;
475 llvm::DenseSet<Value> pendingTasks;
476 std::set<ChannelKey> round;
477 llvm::DenseMap<int64_t, int64_t> headOfGroup;
478 Operation *anchor = nullptr;
479 int64_t heads = 0;
480
481 auto flush = [&]() {
482 llvm::stable_sort(pending, [](const Pending &a, const Pending &b) {
483 int64_t lhs = a.index * b.count;
484 int64_t rhs = b.index * a.count;
485 return lhs != rhs ? lhs < rhs : a.head < b.head;
486 });
487 Operation *after = anchor;
488 for (Pending &p : pending) {
489 p.start->moveAfter(after);
490 after = p.start;
491 // Configure a slice only once it is due, so that the descriptors it
492 // takes are not held while earlier slices wait: the BD-ID pass can take
493 // descriptors back only from started tasks.
494 Operation *task = p.start.getTask().getDefiningOp();
495 if (task && state.lateTasks.erase(task))
496 task->moveBefore(p.start);
497 }
498 pending.clear();
499 pendingTasks.clear();
500 round.clear();
501 headOfGroup.clear();
502 anchor = nullptr;
503 };
504
505 for (Operation &op : llvm::make_early_inc_range(block)) {
506 if (auto start = dyn_cast<DMAStartTaskOp>(op)) {
507 std::optional<ChannelKey> channel = channelOf(start);
508 auto it = state.starts.find(start);
509 std::optional<Slices::Start> slice;
510 if (it != state.starts.end())
511 slice = it->second;
512 if (channel && slice && slice->index > 0) {
513 auto head = headOfGroup.find(slice->group);
514 if (head != headOfGroup.end()) {
515 pending.push_back({start, slice->index, slice->count, head->second});
516 pendingTasks.insert(start.getTask());
517 continue;
518 }
519 }
520 if (!channel || round.count(*channel))
521 flush();
522 if (!channel)
523 continue;
524 round.insert(*channel);
525 if (slice)
526 headOfGroup[slice->group] = heads;
527 ++heads;
528 anchor = &op;
529 continue;
530 }
531 if (auto free = dyn_cast<DMAFreeTaskOp>(op)) {
532 if (pendingTasks.contains(free.getTask()))
533 flush();
534 continue;
535 }
536 if (isa<DMAConfigureTaskOp, DMAConfigureTaskForOp, NpuWriteRTPOp,
537 SetLockOp>(op) ||
538 isMemoryEffectFree(&op))
539 continue;
540 flush();
541 }
542 flush();
543}

545static void decomposeMemcpy(RewriterBase &rewriter, NpuDmaMemcpyNdOp op) {
546 if (!allConstant(op))
547 return;
548 // A runtime length stays whole; see DMABDOp::verify.
549 if (op.getLengthParameterAttr() || op.getLengthStateTableIdxAttr())
550 return;
551
552 NdDmaPattern pattern = patternFromOp(op);
553 if (isContiguousTransfer(pattern.sizes, pattern.strides))
554 return;
555
556 AIE::DeviceOp dev = op->getParentOfType<AIE::DeviceOp>();
557 if (!dev)
558 return;
559
560 auto allocOp = AIE::ShimDMAAllocationOp::getForSymbol(
561 dev, op.getMetadata().getRootReference());
562 if (!allocOp)
563 return;
564
565 AIE::TileOp tile = allocOp.getTileOp();
566 if (!tile)
567 return;
568
569 int col = tile.getCol();
570 int row = tile.getRow();
571 const AIE::AIETargetModel &targetModel = AIE::getTargetModel(op);
572 auto bufferType = cast<BaseMemRefType>(op.getMemref().getType());
573
574 if (patternPassesVerification(op, bufferType, targetModel, col, row, pattern))
575 return;
576
577 auto decomposed =
578 decomposeNdDmaPattern(op, bufferType, pattern, targetModel, col, row);
579 // failed() already guards both dereferences below via short-circuit /
580 // prior control flow; the checker just doesn't associate FailureOr's
581 // failed()/succeeded() idiom with the std::optional base it derives from.
582 if (failed(decomposed) ||
583 decomposed->empty()) // NOLINT(bugprone-unchecked-optional-access)
584 return;
585 // Bind a plain reference now that decomposed is known non-failed and
586 // non-empty, so nothing past this point looks like an optional access.
587 SmallVector<NdDmaPattern> &bds =
588 *decomposed; // NOLINT(bugprone-unchecked-optional-access)
589 if (bds.size() > targetModel.getNumBDs(col, row))
590 return;
591
592 rewriter.setInsertionPoint(op);
593 if (bds.size() == 1) {
594 rewriter.replaceOpWithNewOp<NpuDmaMemcpyNdOp>(
595 op, op, toOuter(bds.front().offsets), toOuter(bds.front().sizes),
596 toOuter(bds.front().strides));
597 return;
598 }
599
600 llvm::DenseSet<int64_t> usedIds;
601 if (auto seq = op->getParentOfType<AIE::RuntimeSequenceOp>()) {
602 seq.walk([&](NpuDmaMemcpyNdOp other) {
603 if (other == op)
604 return;
605 if (other.getMetadata() == op.getMetadata())
606 usedIds.insert(other.getId());
607 });
608 }
609
610 int64_t nextId = op.getId();
611 for (auto [idx, subPattern] : llvm::enumerate(bds)) {
612 bool last = idx + 1 == bds.size();
613 int64_t id = allocateNextId(op, nextId, usedIds);
614 nextId = id + 1;
615 createDecomposedOp(rewriter, op, subPattern, id,
616 last && op.getIssueToken());
617 }
618 rewriter.eraseOp(op);
619}

621// Rewrites a task's descriptor into legal ones. Fails only after reporting why
622// it cannot.
623static LogicalResult decomposeTaskBd(RewriterBase &rewriter, AIE::DMABDOp op,
624 Slices &slices) {
625 auto taskAndTile = resolveTaskAndTile(op);
626 if (!taskAndTile)
627 return success();
628
629 AIE::TileOp tile = taskAndTile->first;
630 DMAConfigureTaskLike taskOp = taskAndTile->second;
631
632 // A descriptor with more dimensions than a BD has cannot be lowered as it
633 // is, so where this pass cannot reduce it, it says why.
634 bool tooManyDims = op.getMixedSizes().size() > kNdDmaDims;
635 auto cannotReduce = [&]() {
636 return op.emitOpError()
637 << "has " << op.getMixedSizes().size()
638 << " dimensions, and a buffer descriptor holds " << kNdDmaDims
639 << "; the extra ones can only be split off ";
640 };
641 if (!allConstant(op)) {
642 if (tooManyDims)
643 return cannotReduce()
644 << "a descriptor whose offset, length, sizes and strides are "
645 "all constant and that has no padding";
646 return success();
647 }
648 if (countTaskBds(taskOp) != 1) {
649 if (tooManyDims)
650 return cannotReduce() << "a task's only descriptor";
651 return success();
652 }
653 // A descriptor takes its locks once per execution, which decomposition
654 // regroups, and a slice of its own would take none.
655 bool takesLocks = false;
656 taskOp->walk([&](AIE::UseLockOp) { takesLocks = true; });
657 if (takesLocks) {
658 if (tooManyDims)
659 return cannotReduce() << "a descriptor that takes no locks";
660 return success();
661 }
662 // A runtime length stays whole; see DMABDOp::verify.
663 if (op.getLengthParameterAttr() || op.getLengthStateTableIdxAttr())
664 return success();
665
666 int col = tile.getCol();
667 int row = tile.getRow();
668 const AIE::AIETargetModel &targetModel = AIE::getTargetModel(op);
669 auto bufferType = cast<BaseMemRefType>(op.getBuffer().getType());
670 // See contiguousAndFits for why only d3 can make one too long.
671 int64_t maxIterations = 1LL << targetModel.getDmaBdIterBits(col, row);
672 auto lowerable = [&](const NdDmaPattern &p) {
673 if (p.sizes.size() != kNdDmaDims)
674 return false;
675 if (isContiguousTransfer(p.sizes, p.strides))
676 return p.sizes[3] <= maxIterations;
677 return patternPassesVerification(op, bufferType, targetModel, col, row, p);
678 };
679
680 NdDmaPattern pattern = patternFromDmaBd(op);
681 if (lowerable(pattern))
682 return success();
683
684 // Outer iteration dimensions that re-read the same data only repeat what
685 // is inside them, as the task's repeat count does. They go, and a pass
686 // shrinks to what they repeat: where that is all the pattern needs it
687 // lowers as it is, and otherwise the pieces it splits into are restarted
688 // rather than configured once per repeat.
689 NdDmaPattern unrepeated = pattern;
690 for (unsigned d = unrepeated.sizes.size();
691 d-- > 3 && (unrepeated.sizes[d] == 1 || unrepeated.strides[d] == 0);)
692 unrepeated.sizes[d] = 1;
693 bool dropRepeats = lowerable(unrepeated);
694 pattern = unrepeated;
695 // One pass over the pattern, in executions of the descriptor.
696 int64_t iterations = iterationCount(pattern);
697
698 auto decomposed = dropRepeats ? FailureOr<SmallVector<NdDmaPattern>>(
699 SmallVector<NdDmaPattern>{pattern})
700 : decomposeNdDmaPattern(op, bufferType, pattern,
701 targetModel, col, row);
702 // failed() already guards both dereferences below via short-circuit /
703 // prior control flow; the checker just doesn't associate FailureOr's
704 // failed()/succeeded() idiom with the std::optional base it derives from.
705 if (failed(decomposed) ||
706 decomposed->empty()) { // NOLINT(bugprone-unchecked-optional-access)
707 if (tooManyDims)
708 return op.emitOpError()
709 << "has " << op.getMixedSizes().size()
710 << " dimensions, and no split into at most " << kMaxNdDmaPieces
711 << " descriptors of " << kNdDmaDims << " fits this tile";
712 return success();
713 }
714 // Bind a plain reference now that decomposed is known non-failed and
715 // non-empty, so nothing past this point looks like an optional access.
716 SmallVector<NdDmaPattern> &bds =
717 *decomposed; // NOLINT(bugprone-unchecked-optional-access)
718
719 int64_t baseFlatOffset = op.getConstantOffset().value_or(0);
720
721 if (bds.size() == 1) {
722 const NdDmaPattern &sub = bds.front();
723 int64_t flatOffset = flatOffsetFromPattern(baseFlatOffset, sub);
724 auto outerSizes = toOuter(sub.sizes);
725 auto outerStrides = toOuter(sub.strides);
726 // len covers one BD invocation, so it tracks the innermost three
727 // dimensions only. Rewriting in place can move extent into the fourth
728 // (repeat) dimension -- e.g. an innermost run too long for the wrap
729 // field factors out an extra dimension and pushes the outermost one into
730 // the repeat slot -- so len has to be recomputed alongside the shape.
731 int32_t len = static_cast<int32_t>(lenFromInnermost3(sub.sizes));
732
733 IterationScale scale = iterationScale(iterations, sub);
734 int64_t runs = 0;
735 if (!scale.isOne()) {
736 if (taskOp.getRepeatCountVal())
737 return op.emitOpError()
738 << "cannot decompose a buffer descriptor whose repeat count "
739 "is a runtime value: decomposition needs to scale it by "
740 << scale.str();
741 // A scaled count past what one queue push carries is fine: the BD-ID
742 // pass issues it as several starts. Only the attribute width limits
743 // it. Widen before multiplying: the accessor returns int32_t, so the
744 // addition alone would overflow in int and wrap past this check.
745 // Merged executions need every start to run whole ones.
746 auto check = [&](Operation *at, int64_t repeat, StringRef runner,
747 StringRef count) -> LogicalResult {
748 int64_t before = repeat + 1;
749 if (!scale.divides(before))
750 return at->emitOpError()
751 << "cannot decompose: it merges every " << scale.den
752 << " executions of the buffer descriptor into " << scale.num
753 << ", and " << runner << " runs it " << before << " times";
754 int64_t after = scale.apply(before);
755 if (after - 1 > std::numeric_limits<int32_t>::max())
756 return at->emitOpError()
757 << "decomposition scales " << count << " by " << scale.str()
758 << " to " << (after - 1) << ", beyond a 32-bit repeat_count";
759 return success();
760 };
761 runs = taskOp.getRepeatCount() + int64_t{1};
762 if (failed(check(op, runs - 1, "the task", "the repeat count")))
763 return failure();
764 // A start that overrides the task's count repeats the same BD, so it
765 // scales by the same factor.
766 for (Operation *user : taskOp->getResult(0).getUsers()) {
767 auto start = dyn_cast<DMAStartTaskOp>(user);
768 if (!start)
769 continue;
770 if (start.getRepeatCountVal())
771 return start.emitOpError()
772 << "cannot decompose a buffer descriptor this start repeats "
773 "a runtime number of times: decomposition needs to scale "
774 "the count by "
775 << scale.str();
776 if (IntegerAttr rc = start.getRepeatCountAttr())
777 if (failed(check(start, rc.getInt(), "this start",
778 "this start's repeat count")))
779 return failure();
780 }
781 runs = scale.apply(runs);
782 }
783
784 rewriter.modifyOpInPlace(op, [&]() {
785 updateTaskBdInPlace(op, static_cast<int32_t>(flatOffset), len, outerSizes,
786 outerStrides);
787 });
788 if (!scale.isOne()) {
789 rewriter.modifyOpInPlace(taskOp, [&]() {
790 taskOp.setRepeatCount(static_cast<uint32_t>(runs - 1));
791 });
792 for (Operation *user : taskOp->getResult(0).getUsers())
793 if (auto start = dyn_cast<DMAStartTaskOp>(user))
794 if (IntegerAttr rc = start.getRepeatCountAttr())
795 rewriter.modifyOpInPlace(start, [&]() {
796 start.setRepeatCountAttr(
797 rewriter.getI32IntegerAttr(scale.apply(rc.getInt() + 1) - 1));
798 });
799 }
800 return success();
801 }
802
803 // A split would make every slice reuse the one out_of_order_id slot.
804 // TODO: BD iteration plus padding may enable this feature.
805 if (op.getOutOfOrderId().has_value())
806 return op.emitOpError() << "splitting an out-of-order buffer descriptor "
807 "into multiple descriptors is not implemented";
808
809 // Every member of a chain runs once per task execution, so the one
810 // queue-push repeat count is shared by all of them. A slice that also wants
811 // its own iteration factor needs a private repeat, which only a task of
812 // its own has. A chain also holds all of its descriptors until it
813 // finishes, so past the queue depth the slices go out as separate tasks,
814 // whose descriptors aie-assign-runtime-sequence-bd-ids can recycle one by
815 // one.
816 bool sharedRepeat = llvm::all_of(bds, [&](const NdDmaPattern &sub) {
817 return iterationScale(iterations, sub).isOne();
818 });
819 bool chainFits =
820 sharedRepeat && bds.size() <= targetModel.getNumBDs(col, row);
821 uint32_t depth = targetModel.getDmaTaskQueueDepth();
822 if (!chainFits || (depth > 0 && bds.size() > depth)) {
823 // The dynamic BD pool allocates a chain under runtime control flow, but
824 // separate tasks there are left whole.
825 if (isUnderRuntimeControlFlow(op)) {
826 if (tooManyDims)
827 return cannotReduce()
828 << "outside runtime control flow, since its " << bds.size()
829 << " pieces have to be separate tasks";
830 op.emitRemark() << "deferring decomposition into " << bds.size()
831 << " separate tasks under runtime control flow";
832 return success();
833 }
834 std::optional<std::string> why =
835 whyNotSeparateTasks(op, taskOp, iterations);
836 if (!why) {
837 splitIntoTasks(rewriter, op, taskOp, bds, baseFlatOffset, iterations,
838 slices);
839 return success();
840 }
841 if (!sharedRepeat)
842 return op.emitOpError()
843 << "cannot split this buffer descriptor: its slices need "
844 "per-descriptor repeat counts, which a chain shares, and "
845 "they cannot be separate tasks because "
846 << *why;
847 if (!chainFits)
848 return op.emitOpError()
849 << "cannot split this buffer descriptor: its " << bds.size()
850 << " slices outnumber the tile's "
851 << targetModel.getNumBDs(col, row)
852 << " buffer descriptors, and they cannot be separate tasks "
853 "because "
854 << *why;
855 }
856
857 Region *body = &taskOp.getBody();
858 if (body->empty())
859 return success();
860
861 SmallVector<Block *> blocks;
862 blocks.push_back(op->getBlock());
863 for (unsigned i = 1; i < bds.size(); ++i)
864 blocks.push_back(rewriter.createBlock(body));
865
866 for (auto [idx, subPattern] : llvm::enumerate(bds)) {
867 Block *block = blocks[idx];
868 int64_t flatOffset = flatOffsetFromPattern(baseFlatOffset, subPattern);
869 auto outerSizes = toOuter(subPattern.sizes);
870 auto outerStrides = toOuter(subPattern.strides);
871 int32_t len = static_cast<int32_t>(lenFromInnermost3(subPattern.sizes));
872
873 if (idx == 0) {
874 rewriter.modifyOpInPlace(op, [&]() {
875 updateTaskBdInPlace(op, static_cast<int32_t>(flatOffset), len,
876 outerSizes, outerStrides);
877 });
878 Operation *oldTerm = block->getTerminator();
879 rewriter.setInsertionPoint(oldTerm);
880 if (idx + 1 < bds.size())
881 AIE::NextBDOp::create(rewriter, op.getLoc(), blocks[idx + 1]);
882 else
883 AIE::EndOp::create(rewriter, op.getLoc());
884 rewriter.eraseOp(oldTerm);
885 } else {
886 rewriter.setInsertionPointToStart(block);
887 createTaskBd(rewriter, op.getLoc(), op, static_cast<int32_t>(flatOffset),
888 len, outerSizes, outerStrides);
889 rewriter.setInsertionPointToEnd(block);
890 if (idx + 1 < bds.size())
891 AIE::NextBDOp::create(rewriter, op.getLoc(), blocks[idx + 1]);
892 else
893 AIE::EndOp::create(rewriter, op.getLoc());
894 }
895 }
896
897 return success();
898}

921 return signalPassFailure();

923 orderSlices(*block, slices);
924 }

37static bool allConstant(NpuDmaMemcpyNdOp op) {
38 return llvm::all_of(op.getMixedSizes(),
39 [](OpFoldResult s) {
40 return getConstantIntValue(s).has_value();
41 }) &&
42 llvm::all_of(op.getMixedStrides(),
43 [](OpFoldResult s) {
44 return getConstantIntValue(s).has_value();
45 }) &&
46 llvm::all_of(op.getMixedOffsets(), [](OpFoldResult s) {
47 return getConstantIntValue(s).has_value();
48 });
49}

51static bool allConstant(AIE::DMABDOp op) {
52 if (op.getPadDimensions().has_value())
53 return false;
54 if (op.getOffset() && !op.getConstantOffset())
55 return false;
56 if (op.getLen() && !op.getConstantLen())
57 return false;
58 if (op.getMixedSizes().empty())
59 return false;
60 return llvm::all_of(op.getMixedSizes(),
61 [](OpFoldResult s) {
62 return getConstantIntValue(s).has_value();
63 }) &&
64 llvm::all_of(op.getMixedStrides(), [](OpFoldResult s) {
65 return getConstantIntValue(s).has_value();
66 });
67}

69static NdDmaPattern patternFromOp(NpuDmaMemcpyNdOp op) {
70 NdDmaPattern pattern;
71 pattern.offsets = llvm::map_to_vector(
72 llvm::reverse(op.getMixedOffsets()),
73 [](OpFoldResult s) { return getConstantIntValue(s).value(); });
74 pattern.sizes = llvm::map_to_vector(
75 llvm::reverse(op.getMixedSizes()),
76 [](OpFoldResult s) { return getConstantIntValue(s).value(); });
77 pattern.strides = llvm::map_to_vector(
78 llvm::reverse(op.getMixedStrides()),
79 [](OpFoldResult s) { return getConstantIntValue(s).value(); });
80 return pattern;
81}

83static NdDmaPattern patternFromDmaBd(AIE::DMABDOp op) {
84 SmallVector<int64_t, 4> outerSizes;
85 SmallVector<int64_t, 4> outerStrides;
86 for (OpFoldResult s : op.getMixedSizes())
87 outerSizes.push_back(getConstantIntValue(s).value());
88 for (OpFoldResult s : op.getMixedStrides())
89 outerStrides.push_back(getConstantIntValue(s).value());
90 while (outerSizes.size() < 4) {
91 outerSizes.insert(outerSizes.begin(), 1);
92 outerStrides.insert(outerStrides.begin(), 0);
93 }
94
95 NdDmaPattern pattern;
96 pattern.offsets = {0, 0, 0, 0};
97 pattern.sizes = llvm::map_to_vector(llvm::reverse(outerSizes),
98 [](int64_t v) { return v; });
99 pattern.strides = llvm::map_to_vector(llvm::reverse(outerStrides),
100 [](int64_t v) { return v; });
101 return pattern;
102}

104static SmallVector<int64_t, 4> toOuter(ArrayRef<int64_t> inner) {
105 return llvm::map_to_vector(llvm::reverse(inner), [](int64_t v) { return v; });
106}

108static int64_t flatOffsetFromPattern(int64_t baseFlatOffset,
109 const NdDmaPattern &pattern) {
110 int64_t flat = baseFlatOffset;
111 for (unsigned k = 0; k < 4; ++k)
112 flat += pattern.offsets[k] * pattern.strides[k];
113 return flat;
114}

116static int64_t lenFromInnermost3(ArrayRef<int64_t> sizesInnermostFirst) {
117 int64_t len = 1;
118 for (unsigned i = 0; i < 3; ++i)
119 len *= sizesInnermostFirst[i];
120 return len;
121}

123static AIE::BDDimLayoutArrayAttr outerDimsAttr(MLIRContext *ctx,
124 ArrayRef<int64_t> outerSizes,
125 ArrayRef<int64_t> outerStrides) {
126 SmallVector<AIE::BDDimLayoutAttr> dims;
127 dims.reserve(outerSizes.size());
128 for (auto [s, t] : llvm::zip(outerSizes, outerStrides))
129 dims.push_back(AIE::BDDimLayoutAttr::get(ctx, static_cast<uint32_t>(s),
130 static_cast<uint32_t>(t)));
131 return AIE::BDDimLayoutArrayAttr::get(ctx, dims);
132}

134static void updateTaskBdInPlace(AIE::DMABDOp bd, int32_t offset, int32_t len,
135 ArrayRef<int64_t> outerSizes,
136 ArrayRef<int64_t> outerStrides) {
137 bd.getOffsetMutable().clear();
138 bd.setStaticOffset(offset);
139 bd.getLenMutable().clear();
140 bd.setStaticLen(len);
141 bd.getSizesMutable().clear();
142 bd.getStridesMutable().clear();
143 bd.setStaticSizes(DenseI64ArrayAttr::get(bd.getContext(), outerSizes));
144 bd.setStaticStrides(DenseI64ArrayAttr::get(bd.getContext(), outerStrides));
145}

147static AIE::DMABDOp createTaskBd(PatternRewriter &rewriter, Location loc,
148 AIE::DMABDOp tmpl, int32_t offset, int32_t len,
149 ArrayRef<int64_t> outerSizes,
150 ArrayRef<int64_t> outerStrides) {
151 auto dims = outerDimsAttr(rewriter.getContext(), outerSizes, outerStrides);
152 auto bd =
153 AIE::DMABDOp::create(rewriter, loc, tmpl.getBuffer(), offset, len, dims);
154 if (tmpl.getPacketAttr())
155 bd.setPacketAttr(tmpl.getPacketAttr());
156 if (tmpl.getBurstLengthAttr())
157 bd.setBurstLengthAttr(tmpl.getBurstLengthAttr());
158 if (tmpl.getOffsetParameterAttr())
159 bd.setOffsetParameterAttr(tmpl.getOffsetParameterAttr());
160 return bd;
161}

177static bool isUnderRuntimeControlFlow(AIE::DMABDOp op) {
178 auto seq = op->getParentOfType<AIE::RuntimeSequenceOp>();
179 if (!seq)
180 return false;
181 for (Operation *parent = op->getParentOp(); parent && parent != seq;
182 parent = parent->getParentOp()) {
183 if (isa<scf::ForOp, scf::WhileOp, scf::IfOp>(parent))
184 return true;
185 }
186 return false;
187}

189static std::optional<std::pair<AIE::TileOp, Operation *>>
190resolveTaskAndTile(AIE::DMABDOp op) {
191 if (auto cfg = op->getParentOfType<DMAConfigureTaskOp>()) {
192 AIE::TileOp tile = cfg.getTileOp();
193 if (!tile)
194 return std::nullopt;
195 return std::make_pair(tile, cfg.getOperation());
196 }
197 if (auto cfgFor = op->getParentOfType<DMAConfigureTaskForOp>()) {
198 AIE::DeviceOp dev = op->getParentOfType<AIE::DeviceOp>();
199 if (!dev)
200 return std::nullopt;
201 auto allocOp = AIE::ShimDMAAllocationOp::getForSymbol(
202 dev, cfgFor.getAlloc().getRootReference());
203 if (!allocOp)
204 return std::nullopt;
205 AIE::TileOp tile = allocOp.getTileOp();
206 if (!tile)
207 return std::nullopt;
208 return std::make_pair(tile, cfgFor.getOperation());
209 }
210 return std::nullopt;
211}

230 op.getD1ZeroBeforeAttr(), op.getD2ZeroBeforeAttr(),
231 op.getD0ZeroAfterAttr(), op.getD1ZeroAfterAttr(), op.getD2ZeroAfterAttr(),
232 op.getBurstLengthAttr(), op.getOffsetParameterAttr(),

236static int64_t allocateNextId(NpuDmaMemcpyNdOp op, int64_t startId,
237 llvm::DenseSet<int64_t> &usedIds) {
238 int64_t id = startId;
239 while (usedIds.contains(id))
240 ++id;
241 usedIds.insert(id);
242 return id;

245struct DecomposeLargeDmaBdPattern : OpRewritePattern<NpuDmaMemcpyNdOp> {
246 using OpRewritePattern::OpRewritePattern;
247
248 LogicalResult matchAndRewrite(NpuDmaMemcpyNdOp op,
249 PatternRewriter &rewriter) const override {
250 if (!allConstant(op))
251 return failure();
252
253 NdDmaPattern pattern = patternFromOp(op);
254 if (isContiguousTransfer(pattern.sizes, pattern.strides))
255 return failure();
256
257 AIE::DeviceOp dev = op->getParentOfType<AIE::DeviceOp>();
258 if (!dev)
259 return failure();
260
261 auto allocOp = AIE::ShimDMAAllocationOp::getForSymbol(
262 dev, op.getMetadata().getRootReference());
263 if (!allocOp)
264 return failure();
265
266 AIE::TileOp tile = allocOp.getTileOp();
267 if (!tile)
268 return failure();
269
270 int col = tile.getCol();
271 int row = tile.getRow();
272 const AIE::AIETargetModel &targetModel = AIE::getTargetModel(op);
273 auto bufferType = cast<BaseMemRefType>(op.getMemref().getType());
274
275 if (patternPassesVerification(op, bufferType, targetModel, col, row,
276 pattern))
277 return failure();
278
279 auto decomposed =
280 decomposeNdDmaPattern(op, bufferType, pattern, targetModel, col, row);
281 if (failed(decomposed) || decomposed->empty())
282 return failure();
283 if (decomposed->size() > targetModel.getNumBDs(col, row))
284 return failure();
285
286 if (decomposed->size() == 1) {
287 rewriter.replaceOpWithNewOp<NpuDmaMemcpyNdOp>(
288 op, op.getMemref(), ValueRange{}, ValueRange{}, ValueRange{},
289 DenseI64ArrayAttr::get(op.getContext(),
290 toOuter(decomposed->front().offsets)),
291 DenseI64ArrayAttr::get(op.getContext(),
292 toOuter(decomposed->front().sizes)),
293 DenseI64ArrayAttr::get(op.getContext(),
294 toOuter(decomposed->front().strides)),
295 op.getPacketAttr(), op.getMetadata(), op.getIdAttr(),
296 op.getIssueTokenAttr(), op.getD0ZeroBeforeAttr(),
297 op.getD1ZeroBeforeAttr(), op.getD2ZeroBeforeAttr(),
298 op.getD0ZeroAfterAttr(), op.getD1ZeroAfterAttr(),
299 op.getD2ZeroAfterAttr(), op.getBurstLengthAttr(),
300 op.getOffsetParameterAttr(), op.getOffsetStateTableIdxAttr());
301 return success();
302 }
303
304 llvm::DenseSet<int64_t> usedIds;
305 if (auto seq = op->getParentOfType<AIE::RuntimeSequenceOp>()) {
306 seq.walk([&](NpuDmaMemcpyNdOp other) {
307 if (other == op)
308 return;
309 if (other.getMetadata() == op.getMetadata())
310 usedIds.insert(other.getId());
311 });
312 }
313
314 int64_t nextId = op.getId();
315 rewriter.setInsertionPoint(op);
316 for (auto [idx, subPattern] : llvm::enumerate(*decomposed)) {
317 bool last = idx + 1 == decomposed->size();
318 int64_t id = allocateNextId(op, nextId, usedIds);
319 nextId = id + 1;
320 createDecomposedOp(rewriter, op, subPattern, id,
321 last && op.getIssueToken());
322 }
323 rewriter.eraseOp(op);
324 return success();
325 }
326};

328struct DecomposeLargeDmaBdTaskPattern : OpRewritePattern<AIE::DMABDOp> {
329 using OpRewritePattern::OpRewritePattern;
330
331 LogicalResult matchAndRewrite(AIE::DMABDOp op,
332 PatternRewriter &rewriter) const override {
333 if (op->getParentOfType<AIE::MemOp>() ||
334 op->getParentOfType<AIE::ShimDMAOp>() ||
335 op->getParentOfType<AIE::MemTileDMAOp>() ||
336 op->getParentOfType<AIE::DMAOp>())
337 return failure();
338
339 auto taskAndTile = resolveTaskAndTile(op);
340 if (!taskAndTile)
341 return failure();
342
343 AIE::TileOp tile = taskAndTile->first;
344 Operation *taskOp = taskAndTile->second;
345
346 if (!allConstant(op))
347 return failure();
348 if (countTaskBds(taskOp) != 1)
349 return failure();
350
351 NdDmaPattern pattern = patternFromDmaBd(op);
352 if (isContiguousTransfer(pattern.sizes, pattern.strides))
353 return failure();
354
355 int col = tile.getCol();
356 int row = tile.getRow();
357 const AIE::AIETargetModel &targetModel = AIE::getTargetModel(op);
358 auto bufferType = cast<BaseMemRefType>(op.getBuffer().getType());
359
360 if (patternPassesVerification(op, bufferType, targetModel, col, row,
361 pattern))
362 return failure();
363
364 auto decomposed =
365 decomposeNdDmaPattern(op, bufferType, pattern, targetModel, col, row);
366 if (failed(decomposed) || decomposed->empty())
367 return failure();
368 if (decomposed->size() > targetModel.getNumBDs(col, row))
369 return failure();
370
371 if (decomposed->size() > 1 && isUnderRuntimeControlFlow(op)) {
372 op.emitRemark()
373 << "deferring multi-BD decomposition under runtime control flow "
374 "(dynamic BD pool supports single-BD tasks only)";
375 return failure();
376 }
377
378 int64_t baseFlatOffset = op.getConstantOffset().value_or(0);
379
380 if (decomposed->size() == 1) {
381 const NdDmaPattern &sub = decomposed->front();
382 int64_t flatOffset = flatOffsetFromPattern(baseFlatOffset, sub);
383 auto outerSizes = toOuter(sub.sizes);
384 auto outerStrides = toOuter(sub.strides);
385 rewriter.modifyOpInPlace(op, [&]() {
386 op.getOffsetMutable().clear();
387 op.setStaticOffset(static_cast<int32_t>(flatOffset));
388 op.getSizesMutable().clear();
389 op.getStridesMutable().clear();
390 op.setStaticSizes(DenseI64ArrayAttr::get(op.getContext(), outerSizes));
391 op.setStaticStrides(
392 DenseI64ArrayAttr::get(op.getContext(), outerStrides));
393 });
394 return success();
395 }
396
397 Region *body = getTaskBody(taskOp);
398 if (!body || body->empty())
399 return failure();
400
401 SmallVector<Block *> blocks;
402 blocks.push_back(op->getBlock());
403 for (unsigned i = 1; i < decomposed->size(); ++i)
404 blocks.push_back(rewriter.createBlock(body));
405
406 for (auto [idx, subPattern] : llvm::enumerate(*decomposed)) {
407 Block *block = blocks[idx];
408 int64_t flatOffset = flatOffsetFromPattern(baseFlatOffset, subPattern);
409 auto outerSizes = toOuter(subPattern.sizes);
410 auto outerStrides = toOuter(subPattern.strides);
411 int32_t len = static_cast<int32_t>(lenFromInnermost3(subPattern.sizes));
412
413 if (idx == 0) {
414 rewriter.modifyOpInPlace(op, [&]() {
415 updateTaskBdInPlace(op, static_cast<int32_t>(flatOffset), len,
416 outerSizes, outerStrides);
417 });
418 Operation *oldTerm = block->getTerminator();
419 rewriter.setInsertionPoint(oldTerm);
420 if (idx + 1 < decomposed->size())
421 AIE::NextBDOp::create(rewriter, op.getLoc(), blocks[idx + 1]);
422 else
423 AIE::EndOp::create(rewriter, op.getLoc());
424 rewriter.eraseOp(oldTerm);
425 } else {
426 rewriter.setInsertionPointToStart(block);
427 createTaskBd(rewriter, op.getLoc(), op,
428 static_cast<int32_t>(flatOffset), len, outerSizes,
429 outerStrides);
430 rewriter.setInsertionPointToEnd(block);
431 if (idx + 1 < decomposed->size())
432 AIE::NextBDOp::create(rewriter, op.getLoc(), blocks[idx + 1]);
433 else
434 AIE::EndOp::create(rewriter, op.getLoc());
435 }
436 }
437
438 return success();
439 }
440};

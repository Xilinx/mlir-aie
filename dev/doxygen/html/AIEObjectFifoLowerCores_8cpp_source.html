35// Marks the `memref.alloca`s emitted here for bookkeeping only (objects held,
36// current object index). Memrefs thread through control flow more easily than
37// SSA values; mem2reg turns them back into SSA at the end of the pass, and the
38// marker keeps that sweep off allocas this pass did not create.
39constexpr llvm::StringLiteral kBookkeepingSlotAttrName =
40 "aie.objectfifo.bookkeeping_slot";

42/// Build an `scf.index_switch` on `idx` with `n` cases (0..n-1) plus a default;
43/// case k (and the default) yields `elem(k)`. The runtime selection folds to a
44/// constant once the enclosing loops are unrolled. Leaves the builder after the
45/// switch and returns its result.
46Value buildRotatingSwitch(OpBuilder &builder, Location loc, Value idx,
47 Type resultTy, int n,
48 llvm::function_ref<Value(int)> elem) {
49 SmallVector<int64_t, 4> caseValues;
50 for (int c = 0; c < n; ++c) {
51 caseValues.push_back(c);
52 }
53 auto cases = DenseI64ArrayAttr::get(builder.getContext(), caseValues);
54 auto switchOp = scf::IndexSwitchOp::create(builder, loc, TypeRange{resultTy},
55 idx, cases, n);
56 builder.createBlock(&switchOp.getDefaultRegion());
57 builder.setInsertionPointToStart(&switchOp.getDefaultBlock());
58 scf::YieldOp::create(builder, loc, elem(0));
59 for (int c = 0; c < n; ++c) {
60 builder.createBlock(&switchOp.getCaseRegions()[c]);
61 builder.setInsertionPointToStart(&switchOp.getCaseBlock(c));
62 scf::YieldOp::create(builder, loc, elem(c));
63 }
64 builder.setInsertionPointAfter(switchOp);
65 return switchOp.getResult(0);
66}

68/// Advance a rotating object-index counter in its slot:
69/// counter = (counter + released) mod depth.
70void emitAdvanceObjectIndex(OpBuilder &builder, Location loc, Value slot,
71 int depth, int released) {
72 Value size =
73 arith::ConstantOp::create(builder, loc, builder.getI32IntegerAttr(depth));
74 Value old = memref::LoadOp::create(builder, loc, slot, ValueRange{});
75 Value step = arith::ConstantOp::create(builder, loc,
76 builder.getI32IntegerAttr(released));
77 Value sum = arith::AddIOp::create(builder, loc, old, step);
78 Value wrapping =
79 arith::CmpIOp::create(builder, loc, arith::CmpIPredicate::sge, sum, size);
80 // The counter stays in [0, depth), so releasing a single object can only ever
81 // wrap to exactly 0.
82 Value wrapped = released == 1
83 ? Value(arith::ConstantOp::create(
84 builder, loc, builder.getI32IntegerAttr(0)))
85 : Value(arith::SubIOp::create(builder, loc, sum, size));
86 Value next = arith::SelectOp::create(builder, loc, wrapping, wrapped, sum);
87 memref::StoreOp::create(builder, loc, next, slot, ValueRange{});
88}

90/// Locks a core endpoint toggles for `action`, one per segment it works: a
91/// filler waits on free objects and hands back full ones, a drainer the
92/// reverse.
93SmallVector<FlatSymbolRefAttr>
94segmentLocksFor(ObjectFifoCoreEndpointOp endpoint, LockAction action) {
95 bool acquiring = action == LockAction::AcquireGreaterEqual;
96 bool wantsProduce = acquiring != endpoint.drains();
97
98 SmallVector<FlatSymbolRefAttr> locks;
99 for (ObjectFifoSegmentOp segment : endpoint.getSelectedSegments()) {
100 if (auto lock = wantsProduce ? segment.getProduceLockAttr()
101 : segment.getConsumeLockAttr()) {
102 locks.push_back(lock);
103 }
104 }
105 return locks;
106}

108/// Everything a rewrite pattern needs that is not local to its op, resolved
109/// before the driver runs so that no matchAndRewrite scans the IR.
110struct LoweringContext {
111 DeviceOp device;
112 bool usesSemaphoreLocks;
113 /// Per-(core, endpoint) runtime bookkeeping: the rotating object index and,
114 /// where locks count, the number of objects currently held.
115 DenseMap<std::pair<Operation *, Operation *>, memref::AllocaOp> objectIndex{};
116 DenseMap<std::pair<Operation *, Operation *>, memref::AllocaOp> heldCount{};
117 bool sawError = false;
118
119 ObjectFifoCoreEndpointOp endpointOf(Operation *op, StringRef name) {
120 return SymbolTable::lookupNearestSymbolFrom<ObjectFifoCoreEndpointOp>(
121 device, StringAttr::get(op->getContext(), name));
122 }
123
124 LockOp lockOf(FlatSymbolRefAttr name) {
125 return SymbolTable::lookupNearestSymbolFrom<LockOp>(device, name);
126 }

129/// Emit the rotating UseLocks for a binary-lock (AIE1) acquire or release. One
130/// lock travels with each object, and the starting index is only known at
131/// runtime, so each of the `count` locks is selected by an scf.index_switch on
132/// the rotation counter -- folded to a concrete lock once the loops unroll.
133void emitBinaryUseLocks(OpBuilder &builder, Location loc,
134 MutableArrayRef<LockOp> locks, int depth, bool drains,
135 Value counterSlot, int count, LockAction action) {
136 if (count == 0 || locks.empty()) {
137 return;
138 }
139 // The value carries the direction: 1 for a filler's release or a drainer's
140 // acquire, 0 for the opposite.
141 bool full = (!drains && action == LockAction::Release) ||
142 (drains && action == LockAction::Acquire);
143 Value counter =
144 memref::LoadOp::create(builder, loc, counterSlot, ValueRange{});
145 Value index =
146 arith::IndexCastOp::create(builder, loc, builder.getIndexType(), counter);
147 Type lockTy = locks[0].getType();
148 for (int i = 0; i < count; i++) {
149 Value lock =
150 buildRotatingSwitch(builder, loc, index, lockTy, depth, [&](int c) {
151 return locks[(i + c) % depth].getResult();
152 });
153 UseLockOp::create(builder, loc, lock, action, full ? 1 : 0);
154 }
155}

157struct LowerRelease : OpRewritePattern<ObjectFifoReleaseOp> {
158 LoweringContext &ctx;
159 LowerRelease(MLIRContext *context, LoweringContext &ctx)
160 : OpRewritePattern(context), ctx(ctx) {}
161
162 LogicalResult matchAndRewrite(ObjectFifoReleaseOp releaseOp,
163 PatternRewriter &rewriter) const override {
164 auto endpoint = ctx.endpointOf(releaseOp, releaseOp.getObjFifoName());
165 if (!endpoint) {
166 return failure();
167 }
168 ObjectFifoPoolOp pool = endpoint.getPoolOp();
169 CoreOp core = releaseOp->getParentOfType<CoreOp>();
170 auto key = std::make_pair(core.getOperation(), endpoint.getOperation());
171 memref::AllocaOp index = ctx.objectIndex.lookup(key);
172
173 int count = releaseOp.relNumber() * pool.getRepeatCount().value_or(1);
174 rewriter.setInsertionPointAfter(releaseOp);
175 Location loc = releaseOp.getLoc();
176
177 if (!ctx.usesSemaphoreLocks) {
178 // The rotation counter is created by the matching acquire, so a release
179 // without one is releasing objects never acquired on this tile.
180 if (!index) {
181 releaseOp->emitOpError(
182 "objectFifo release has no corresponding acquire on this tile");
183 ctx.sawError = true;
184 return failure();
185 }
186 SmallVector<LockOp> locks = pool.getLockOps();
187 emitBinaryUseLocks(rewriter, loc, locks, pool.getDepth(),
188 endpoint.drains(), index.getResult(), count,
189 LockAction::Release);
190 } else {
191 Value amount = arith::ConstantOp::create(
192 rewriter, loc, rewriter.getI32IntegerAttr(count));
193 for (FlatSymbolRefAttr name :
194 segmentLocksFor(endpoint, LockAction::Release)) {
195 UseLockOp::create(rewriter, loc, ctx.lockOf(name), LockAction::Release,
196 amount);
197 }
198
199 memref::AllocaOp held = ctx.heldCount.lookup(key);
200 assert(held && "a held counter is created for every released endpoint");
201 Value current =
202 memref::LoadOp::create(rewriter, loc, held.getResult(), ValueRange{});
203 Value next = arith::SubIOp::create(rewriter, loc, current, amount);
204 memref::StoreOp::create(rewriter, loc, next, held.getResult(),
205 ValueRange{});
206 }
207
208 if (index) {
209 emitAdvanceObjectIndex(rewriter, loc, index.getResult(), pool.getDepth(),
210 releaseOp.getSize());
211 }
212 return success();
213 }

216struct LowerAcquire : OpRewritePattern<ObjectFifoAcquireOp> {
217 LoweringContext &ctx;
218 LowerAcquire(MLIRContext *context, LoweringContext &ctx)
219 : OpRewritePattern(context), ctx(ctx) {}
220
221 LogicalResult matchAndRewrite(ObjectFifoAcquireOp acquireOp,
222 PatternRewriter &rewriter) const override {
223 auto endpoint = ctx.endpointOf(acquireOp, acquireOp.getObjFifoName());
224 if (!endpoint) {
225 return failure();
226 }
227 ObjectFifoPoolOp pool = endpoint.getPoolOp();
228 CoreOp core = acquireOp->getParentOfType<CoreOp>();
229 auto key = std::make_pair(core.getOperation(), endpoint.getOperation());
230
231 int wanted = acquireOp.acqNumber();
232 int repeat = pool.getRepeatCount().value_or(1);
233 rewriter.setInsertionPointAfter(acquireOp);
234 Location loc = acquireOp.getLoc();
235
236 if (!ctx.usesSemaphoreLocks) {
237 // Each object has its own rotating binary lock; while the loops are
238 // rolled the offset of the held locks is unknown, so acquire the whole
239 // window [counter, counter + wanted). Redundant re-acquires of
240 // still-held locks are pruned after unrolling.
241 memref::AllocaOp index = ctx.objectIndex.lookup(key);
242 assert(index && "a rotation counter is created for every acquire");
243 SmallVector<LockOp> locks = pool.getLockOps();
244 emitBinaryUseLocks(rewriter, loc, locks, pool.getDepth(),
245 endpoint.drains(), index.getResult(), wanted * repeat,
246 LockAction::Acquire);
247 return replaceWithObjects(acquireOp, endpoint, key, rewriter);
248 }
249
250 // An acquire names every object the core wants to hold, so only the ones
251 // not already held are taken.
252 memref::AllocaOp held = ctx.heldCount.lookup(key);
253 assert(held && "a held counter is created for every acquire");
254 Value current =
255 memref::LoadOp::create(rewriter, loc, held.getResult(), ValueRange{});
256 Value target = arith::ConstantOp::create(
257 rewriter, loc, rewriter.getI32IntegerAttr(wanted));
258 Value zero =
259 arith::ConstantOp::create(rewriter, loc, rewriter.getI32IntegerAttr(0));
260 Value shortfall = arith::SubIOp::create(rewriter, loc, target, current);
261 Value delta = arith::MaxSIOp::create(rewriter, loc, shortfall, zero);
262 if (repeat > 1) {
263 Value repeatVal = arith::ConstantOp::create(
264 rewriter, loc, rewriter.getI32IntegerAttr(repeat));
265 delta = arith::MulIOp::create(rewriter, loc, delta, repeatVal);
266 }
267 for (FlatSymbolRefAttr name :
268 segmentLocksFor(endpoint, LockAction::AcquireGreaterEqual)) {
269 UseLockOp::create(rewriter, loc, ctx.lockOf(name),
270 LockAction::AcquireGreaterEqual, delta);
271 }
272
273 Value next = arith::AddIOp::create(rewriter, loc, current, delta);
274 memref::StoreOp::create(rewriter, loc, next, held.getResult(),
275 ValueRange{});
276 return replaceWithObjects(acquireOp, endpoint, key, rewriter);
277 }
278
279 /// Hand back the objects the rotating index selects, one per result.
280 LogicalResult replaceWithObjects(ObjectFifoAcquireOp acquireOp,
281 ObjectFifoCoreEndpointOp endpoint,
282 std::pair<Operation *, Operation *> key,
283 PatternRewriter &rewriter) const {
284 ObjectFifoPoolOp pool = endpoint.getPoolOp();
285 SmallVector<Value> buffers;
286 for (BufferLike buffer : pool.getBufferOps()) {
287 buffers.push_back(buffer.getBuffer());
288 }
289 if (buffers.empty()) {
290 return success();
291 }
292 memref::AllocaOp index = ctx.objectIndex.lookup(key);
293 assert(index && "a rotation counter is created for every acquire");
294
295 Location loc = acquireOp.getLoc();
296
297 // An endpoint holding one side of a join or distribute reaches only its
298 // own run of each object.
299 MemRefType accessType = endpoint.getAccessType();
300 if (accessType != pool.getElemType()) {
301 auto [offset, size] = endpoint.getExtent();
302 for (Value &buffer : buffers) {
303 buffer = memref::SubViewOp::create(
304 rewriter, loc, accessType, buffer,
305 ArrayRef<OpFoldResult>{rewriter.getIndexAttr(offset)},
306 {rewriter.getIndexAttr(size)}, {rewriter.getIndexAttr(1)});
307 }
308 }
309 Value counter =
310 memref::LoadOp::create(rewriter, loc, index.getResult(), ValueRange{});
311 Value idx = arith::IndexCastOp::create(rewriter, loc,
312 rewriter.getIndexType(), counter);
313 int depth = pool.getDepth();
314 for (auto [base, result] : llvm::enumerate(acquireOp.getObjects())) {
315 Value object = buildRotatingSwitch(
316 rewriter, loc, idx, buffers[0].getType(), depth,
317 [&, base = base](int c) { return buffers[(base + c) % depth]; });
318 rewriter.replaceAllUsesWith(result, object);
319 }
320 return success();
321 }
322};

328 /// A rank-0 `memref.alloca` initialized to 0, which mem2reg later threads
329 /// through the enclosing loops as an iter_arg.
330 memref::AllocaOp makeSlot(OpBuilder &builder, Location loc, Value zero) {
331 auto scalarTy =
332 MemRefType::get(SmallVector<int64_t>{}, builder.getI32Type());
333 auto slot = memref::AllocaOp::create(builder, loc, scalarTy);
334 slot->setAttr(kBookkeepingSlotAttrName, builder.getUnitAttr());
335 memref::StoreOp::create(builder, loc, zero, slot.getResult(), ValueRange{});
336 return slot;
337 }
338
339 void emitCounters(CoreOp coreOp, LoweringContext &ctx, OpBuilder &builder) {
340 builder.setInsertionPointToStart(&coreOp.getBody().front());
341 Operation *core = coreOp.getOperation();
342 Value zero;
343
344 auto record = [&](DenseMap<std::pair<Operation *, Operation *>,
345 memref::AllocaOp> &slots,
346 StringRef name) {
347 auto endpoint = ctx.endpointOf(coreOp, name);
348 if (!endpoint) {
349 return;
350 }
351 auto key = std::make_pair(core, endpoint.getOperation());
352 if (slots.count(key)) {
353 return;
354 }
355 if (!zero) {
356 zero = arith::ConstantOp::create(builder, coreOp.getLoc(),
357 builder.getI32IntegerAttr(0));
358 }
359 slots[key] = makeSlot(builder, coreOp.getLoc(), zero);
360 };
361
362 coreOp.walk([&](ObjectFifoAcquireOp a) {
363 record(ctx.objectIndex, a.getObjFifoName());
364 });
365 if (!ctx.usesSemaphoreLocks) {
366 return;
367 }
368 coreOp.walk([&](ObjectFifoAcquireOp a) {
369 record(ctx.heldCount, a.getObjFifoName());
370 });
371 coreOp.walk([&](ObjectFifoReleaseOp r) {
372 record(ctx.heldCount, r.getObjFifoName());
373 });
374 }
375
376 /// Tell `--aie-objectFifo-unroll` how far a loop must be unrolled for the
377 /// rotating indices inside it to fold: the least common multiple of the
378 /// depths it touches. Accesses nested in a child loop drive that child, not
379 /// this one, so an ancestor is not over-unrolled.
380 void annotateUnrollHints(DeviceOp device, LoweringContext &ctx,
381 OpBuilder &builder) {
382 for (auto coreOp : device.getOps<CoreOp>()) {
383 coreOp.walk([&](scf::ForOp forOp) {
384 int64_t period = 1;
385 bool touched = false;
386 auto account = [&](Operation *op, StringRef name) {
387 if (op->getParentOfType<scf::ForOp>() != forOp) {
388 return;
389 }
390 auto endpoint = ctx.endpointOf(op, name);
391 if (!endpoint) {
392 return;
393 }
394 touched = true;
395 period = std::lcm(period, (int64_t)endpoint.getPoolOp().getDepth());
396 };
397 forOp.getBody()->walk(
398 [&](ObjectFifoAcquireOp a) { account(a, a.getObjFifoName()); });
399 forOp.getBody()->walk(
400 [&](ObjectFifoReleaseOp r) { account(r, r.getObjFifoName()); });
401 if (touched) {
402 forOp->setAttr(kObjectFifoUnrollHintAttrName,
403 builder.getI64IntegerAttr(period));
404 }
405 });
406 }
407 }
408
409 LogicalResult promoteBookkeepingSlots(DeviceOp device) {
410 SmallVector<PromotableAllocationOpInterface> allocators;
411 device.walk([&](memref::AllocaOp allocaOp) {
412 if (allocaOp->hasAttr(kBookkeepingSlotAttrName)) {
413 allocators.push_back(
414 cast<PromotableAllocationOpInterface>(allocaOp.getOperation()));
415 }
416 });
417 if (allocators.empty()) {
418 return success();
419 }
420
421 DataLayout dataLayout = DataLayout::closest(device);
422 DominanceInfo dominance(device);
423 OpBuilder builder(device.getContext());
424 (void)tryToPromoteMemorySlots(allocators, builder, dataLayout, dominance);
425
426 WalkResult leftover = device.walk([&](memref::AllocaOp allocaOp) {
427 if (!allocaOp->hasAttr(kBookkeepingSlotAttrName)) {
428 return WalkResult::advance();
429 }
430 allocaOp.emitOpError()
431 << "objectFifo bookkeeping slot could not be promoted to SSA "
432 "(mem2reg left it in place); the objectFifo lowering requires "
433 "all bookkeeping counters to become loop-carried SSA values";
434 return WalkResult::interrupt();
435 });
436 return failure(leftover.wasInterrupted());
437 }
438
439 void runOnOperation() override {
440 DeviceOp device = getOperation();
441 OpBuilder builder(device.getContext());
442
443 LoweringContext ctx{device, device.getTargetModel().hasProperty(
444 AIETargetModel::UsesSemaphoreLocks)};
445
446 annotateUnrollHints(device, ctx, builder);
447 for (auto coreOp : device.getOps<CoreOp>()) {
448 emitCounters(coreOp, ctx, builder);
449 }
450
451 RewritePatternSet patterns(device.getContext());
452 patterns.insert<LowerAcquire, LowerRelease>(device.getContext(), ctx);
453 walkAndApplyPatterns(device, std::move(patterns));
454 if (ctx.sawError) {
455 return signalPassFailure();
456 }
457
458 SmallVector<Operation *> toErase;
459 device.walk([&](Operation *op) {
460 if (isa<ObjectFifoAcquireOp, ObjectFifoReleaseOp>(op)) {
461 toErase.push_back(op);
462 }
463 });
464 for (Operation *op : toErase) {
465 op->dropAllUses();
466 op->erase();
467 }
468
469 for (auto endpoint : llvm::make_early_inc_range(
470 device.getOps<ObjectFifoCoreEndpointOp>())) {
471 endpoint.erase();
472 }
473
474 if (failed(promoteBookkeepingSlots(device))) {
475 return signalPassFailure();
476 }
477 }

36/// Tracks, per DMA channel, the pushes not yet known to have completed along
37/// one execution path. The sequence analysis below joins control-flow paths.
38class DmaQueueModel {
39public:
40 /// {col, row, direction, channel}. Queues are per channel and per direction,
41 /// and shim tiles are one per column, so the same channel number in two
42 /// columns is two independent queues.
43 using ChannelKey = std::array<int, 4>;
44
45 /// True when a push on `key` would land on a full queue. `depth` of 0 means
46 /// the target has no queued-task model and nothing can overflow.
47 bool wouldOverflow(const ChannelKey &key, uint32_t depth) const {
48 if (depth == 0)
49 return false;
50 auto it = queued.find(key);
51 return it != queued.end() && it->second.pushes.size() >= depth;
52 }
53
54 /// Record a push. Every push occupies a slot, not just issue_token ones, and
55 /// a repeat_count of N still occupies exactly one.
56 void push(const ChannelKey &key, bool issuesToken) {
57 auto &q = queued[key];
58 q.pushes.push_back(issuesToken && !q.unknownTokens);
59 }
60
61 /// Retire what a task-completion-token await proves has drained. The sync
62 /// pops whichever token the channel produces first -- the oldest outstanding
63 /// token-issuing push, not necessarily the task the await names -- and
64 /// in-order execution means everything queued ahead of it has drained too.
65 /// With no token-issuing push queued, nothing can be retired.
66 void awaitToken(const ChannelKey &key) {
67 auto it = queued.find(key);
68 if (it == queued.end())
69 return;
70 auto &q = it->second;
71 if (q.unknownTokens)
72 return;
73 auto *tok = llvm::find(q.pushes, true);
74 if (tok != q.pushes.end())
75 q.pushes.erase(q.pushes.begin(), std::next(tok));
76 }
77
78 /// Record that a queue-space poll has guaranteed room on `key`. Keeps the
79 /// newest depth-1 entries rather than clearing. Polling does not consume
80 /// completion tokens: if it drops a token-bearing entry, later syncs cannot
81 /// safely be credited to newer pushes.
82 void noteSpaceGuaranteed(const ChannelKey &key, uint32_t depth) {
83 auto &q = queued[key];
84 if (depth > 0 && q.pushes.size() > depth - 1) {
85 auto end = q.pushes.end() - (depth - 1);
86 q.unknownTokens |=
87 llvm::is_contained(llvm::make_range(q.pushes.begin(), end), true);
88 q.pushes.erase(q.pushes.begin(), end);
89 if (q.unknownTokens)
90 llvm::fill(q.pushes, false);
91 }
92 }
93
94 /// True the first time a channel is reported. An over-subscribed channel
95 /// usually stays that way for every later push, and repeating the same
96 /// diagnostic per push buries it.
97 bool shouldReport(const ChannelKey &key) {
98 return reported.insert(key).second;
99 }
100
101 /// Outstanding pushes on `key`, for diagnostics.
102 size_t outstanding(const ChannelKey &key) const {
103 auto it = queued.find(key);
104 return it == queued.end() ? 0 : it->second.pushes.size();
105 }
106
107 /// The whole per-channel queue, which is what the rolled-loop fixed point
108 /// below iterates on. `reported` is deliberately not part of it: it is
109 /// diagnostic bookkeeping, not queue state, and folding it in would stop
110 /// two otherwise identical states from comparing equal.
111 struct ChannelState {
112 llvm::SmallVector<bool, 8> pushes;
113 bool unknownTokens = false;
114 bool operator==(const ChannelState &other) const {
115 return pushes == other.pushes && unknownTokens == other.unknownTokens;
116 }
117 };
118 using State = std::map<ChannelKey, ChannelState>;
119 const State &state() const { return queued; }
120 void setState(State s) { queued = std::move(s); }
121
122private:
123 State queued;
124 std::set<ChannelKey> reported;
125};

127/// The channel an aiex.npu.sync retires on, or nullopt when it cannot be
128/// pinned down. Its six operands are exactly a ChannelKey, with direction
129/// numbered as AIE::DMAChannelDir because DmaWaitToSyncPattern builds that
130/// operand by casting the enum. Unresolvable retires nothing, which
131/// over-reports; guessing the other way would hide a real hang.
132inline std::optional<DmaQueueModel::ChannelKey> syncChannelKey(NpuSyncOp sync) {
133 std::optional<uint32_t> col = getConstantIntOperand(sync.getColumn());
134 std::optional<uint32_t> row = getConstantIntOperand(sync.getRow());
135 std::optional<uint32_t> dir = getConstantIntOperand(sync.getDirection());
136 std::optional<uint32_t> chan = getConstantIntOperand(sync.getChannel());
137 std::optional<uint32_t> colNum = getConstantIntOperand(sync.getColumnNum());
138 std::optional<uint32_t> rowNum = getConstantIntOperand(sync.getRowNum());
139 if (!col || !row || !dir || !chan || !colNum || !rowNum)
140 return std::nullopt;
141 // Nothing in tree settles whether a sync spanning a range of tiles waits for
142 // one token per tile or one for the range -- the cert lowering ignores both
143 // fields -- and crediting nothing is the only reading sound either way.
144 if (*colNum != 1 || *rowNum != 1)
145 return std::nullopt;
146 return DmaQueueModel::ChannelKey{
147 static_cast<int>(*col), static_cast<int>(*row), static_cast<int>(*dir),
148 static_cast<int>(*chan)};
149}
150
151/// Retire what an aiex.npu.sync proves has drained. This is the lowered form of
152/// npu.dma_wait and the same hardware event, so it retires by the same rule;
153/// sequences written or lowered down to raw syncs would otherwise look like
154/// they never drain a channel at all.
155inline void awaitSync(DmaQueueModel &queue, NpuSyncOp sync) {
156 if (std::optional<DmaQueueModel::ChannelKey> key = syncChannelKey(sync))
157 queue.awaitToken(*key);
158}

160/// Shared wording for the two lowering paths, so the explanation does not
161/// depend on whether the transfer came from dma_start_task or
162/// npu.dma_memcpy_nd.
163inline mlir::InFlightDiagnostic
164emitQueueOverflowWarning(mlir::Operation *op, int col, int row,
165 AIE::DMAChannelDir dir, uint32_t channel,
166 uint32_t depth, size_t outstanding) {
167 return op->emitWarning()
168 << "pushes a DMA task onto tile (" << col << "," << row << ") "
169 << AIE::stringifyDMAChannelDir(dir) << " channel " << channel
170 << ", whose task queue is only " << depth << " deep, with "
171 << outstanding
172 << " push(es) not yet known to have completed. A push onto a full "
173 "queue is dropped by hardware and its transfer never runs, so "
174 "whatever waits on it blocks forever. Whether it happens depends "
175 "on how fast the consumer drains, which is why this is a warning "
176 "and not an error. Drain the channel with an await on this tile, "
177 "direction and channel before pushing again";
178}

180/// Emit a poll that blocks until the channel's task queue has a free slot,
181/// immediately before `before`. Fails when the target reports no pollable
182/// occupancy register, or when `depth` is not a power of two -- a maskpoll
183/// tests masked equality, not "<", so "occupancy < depth" has to be the single
184/// "depth bit is clear" test (for depth 4, bit 22 of Task_Queue_Size).
185inline mlir::LogicalResult
186insertQueueSpaceWait(mlir::Operation *before, const AIE::AIETargetModel &tm,
187 int col, int row, uint32_t channel, AIE::DMAChannelDir dir,
188 uint32_t depth) {
189 uint32_t fieldMask = tm.getDmaTaskQueueSizeMask();
190 if (!fieldMask || depth == 0 || (depth & (depth - 1)) != 0)
191 return mlir::failure();
192 std::optional<uint32_t> statusAddr =
193 tm.getDmaStatusAddress(col, row, channel, dir);
194 if (!statusAddr)
195 return mlir::failure();
196 uint32_t depthBit = depth << llvm::countr_zero(fieldMask);
197 if ((depthBit & fieldMask) != depthBit)
198 return mlir::failure();
199
200 mlir::OpBuilder b(before);
201 mlir::Location loc = before->getLoc();
202 auto i32 = b.getI32Type();
203 auto cst = [&](uint32_t v) {
204 return mlir::arith::ConstantOp::create(b, loc, i32, b.getI32IntegerAttr(v))
205 .getResult();
206 };
207 NpuMaskPollOp::create(b, loc, cst(*statusAddr), cst(0), cst(depthBit),
208 /*buffer=*/nullptr, /*column=*/nullptr,
209 /*row=*/nullptr);
210 return mlir::success();
211}

213inline constexpr llvm::StringLiteral queueDiagnosedAttr =
214 "aiex.dma_queue_overflow_diagnosed";

216/// Handle a push on `key` that would land on a full queue: poll where one can
217/// be emitted, warn where it cannot (see insertQueueSpaceWait above for when
218/// that is). Warning rather than failing keeps a default from rejecting
219/// designs that build today on a target the user can do nothing about; the
220/// note is what keeps it from declining quietly.
221inline void guardQueueOverflow(DmaQueueModel &queue, mlir::Operation *push,
222 const AIE::AIETargetModel &tm,
223 const DmaQueueModel::ChannelKey &key,
224 uint32_t depth, bool enforce) {
225 int col = key[0], row = key[1];
226 auto dir = static_cast<AIE::DMAChannelDir>(key[2]);
227 auto chan = static_cast<uint32_t>(key[3]);
228
229 if (enforce && mlir::succeeded(insertQueueSpaceWait(push, tm, col, row, chan,
230 dir, depth))) {
231 queue.noteSpaceGuaranteed(key, depth);
232 return;
233 }
234 if (!queue.shouldReport(key))
235 return;
236 // Keep diagnostics per sequence/channel across task and combined lowering.
237 // This marker suppresses only warnings, never queue accounting or guards.
238 if (auto seq = push->getParentOfType<AIE::RuntimeSequenceOp>()) {
239 auto channel = mlir::DenseI32ArrayAttr::get(push->getContext(), key);
240 llvm::SmallVector<mlir::Attribute> diagnosed;
241 if (auto previous =
242 seq->getAttrOfType<mlir::ArrayAttr>(queueDiagnosedAttr)) {
243 if (llvm::is_contained(previous, channel))
244 return;
245 llvm::append_range(diagnosed, previous);
246 }
247 diagnosed.push_back(channel);
248 seq->setAttr(queueDiagnosedAttr,
249 mlir::ArrayAttr::get(push->getContext(), diagnosed));
250 }
251 mlir::InFlightDiagnostic diag = emitQueueOverflowWarning(
252 push, col, row, dir, chan, depth, queue.outstanding(key));
253 if (enforce)
254 diag.attachNote() << "the compiler would have waited for a free slot here, "
255 "but this target reports no pollable task-queue "
256 "occupancy register for this channel";
257}

259/// What one op in a loop body does to a channel queue. The two lowering paths
260/// spell pushes and awaits with different ops, so the fixed point below takes
261/// the classification from its caller and keeps the FIFO rule to itself.
262struct QueueEffect {
263 enum class Kind { Ignore, Push, Await };
264 Kind kind = Kind::Ignore;
265 DmaQueueModel::ChannelKey key{};
266 bool issuesToken = false;
267
268 static QueueEffect push(DmaQueueModel::ChannelKey key, bool issuesToken) {
269 return {Kind::Push, key, issuesToken};
270 }
271 static QueueEffect await(DmaQueueModel::ChannelKey key) {
272 return {Kind::Await, key, false};
273 }
274};

276/// Guard every push in a runtime sequence that can land on a full queue.
277///
278/// Retain all reachable states at branches and loop exits, including zero
279/// iterations. Nested loops are analyzed recursively, not walked once. Queue
280/// states are finite because overflowing pushes are capped at the target depth;
281/// targets without a queued-task model must be ignored rather than accumulated.
282/// Analyze each independent channel separately to avoid a Cartesian product
283/// of states across channels.
284template <typename EffectFn>
285inline void guardSequenceQueueDepth(mlir::Region &body, DmaQueueModel &queue,
286 const AIE::AIETargetModel &tm, bool enforce,
287 EffectFn effectOf) {
288 auto depthOf = [&](const DmaQueueModel::ChannelKey &k) {
289 return tm.getDmaTaskQueueDepth(k[0], k[1], k[3],
290 static_cast<AIE::DMAChannelDir>(k[2]));
291 };
292
293 using States = llvm::SmallVector<DmaQueueModel::State, 8>;
294 std::set<DmaQueueModel::ChannelKey> keys;
295 for (const auto &[key, channel] : queue.state())
296 if (depthOf(key) != 0)
297 keys.insert(key);
298 body.walk([&](mlir::Operation *op) {
299 QueueEffect e = effectOf(op);
300 if (e.kind != QueueEffect::Kind::Ignore && depthOf(e.key) != 0)
301 keys.insert(e.key);
302 });
303 DmaQueueModel::State entry = queue.state();
304 DmaQueueModel::State merged = entry;
305 for (const auto &key : keys) {
306 uint32_t depth = depthOf(key);
307 auto channelEffectOf = [&](mlir::Operation *op) {
308 QueueEffect e = effectOf(op);
309 return e.key == key ? e : QueueEffect{};
310 };
311 llvm::SetVector<mlir::Operation *> overflowing;
312 std::map<mlir::Operation *, DmaQueueModel::State> witnesses;
313 auto append = [](States &to, const States &from) {
314 for (const auto &state : from)
315 if (!llvm::is_contained(to, state))
316 to.push_back(state);
317 };
318
319 // Earlier passes may already have guarded a start. Credit only polls whose
320 // constant address and depth bit prove space on a channel tracked here.
321 auto pollsChannel = [&](NpuMaskPollOp poll) {
322 auto address = poll.getAbsoluteAddress();
323 auto mask = getConstantIntOperand(poll.getMask());
324 auto value = getConstantIntOperand(poll.getValue());
325 uint32_t fieldMask = tm.getDmaTaskQueueSizeMask();
326 if (!address || !mask || !value || !fieldMask ||
327 (depth & (depth - 1)) != 0)
328 return false;
329 uint32_t depthBit = depth << llvm::countr_zero(fieldMask);
330 auto status = tm.getDmaStatusAddress(
331 key[0], key[1], key[3], static_cast<AIE::DMAChannelDir>(key[2]));
332 return depthBit && status == address &&
333 (depthBit & fieldMask) == depthBit &&
334 (*mask & depthBit) == depthBit && (*value & depthBit) == 0;
335 };
336
337 // Unknown region semantics (including unstructured CFGs) must not credit
338 // conditional waits. Guard their pushes and conservatively forget tokens.
339 auto unknownRegion = [&](mlir::Region &region, States states) {
340 region.walk([&](mlir::Operation *op) {
341 QueueEffect e = channelEffectOf(op);
342 if (e.kind != QueueEffect::Kind::Push)
343 return;
344 for (auto &state : states) {
345 auto &channel = state[e.key];
346 channel.pushes.assign(depth, false);
347 channel.unknownTokens = true;
348 if (overflowing.insert(op))
349 witnesses[op] = state;
350 }
351 });
352 States unique;
353 append(unique, states);
354 return unique;
355 };
356
357 std::function<States(mlir::Region &, States)> analyze;
358 analyze = [&](mlir::Region &region, States states) -> States {
359 if (!region.hasOneBlock())
360 return unknownRegion(region, std::move(states));
361 for (mlir::Operation &op : region.front()) {
362 if (auto loop = mlir::dyn_cast<mlir::scf::ForOp>(&op)) {
363 auto lb = mlir::getConstantIntValue(loop.getLowerBound());
364 auto ub = mlir::getConstantIntValue(loop.getUpperBound());
365 if (lb && ub && *lb >= *ub &&
366 (!loop->hasAttr("unsignedCmp") || (*lb >= 0 && *ub >= 0)))
367 continue;
368 // Every loop header state is a possible exit for an unknown trip
369 // count. Include the entry: even an await-only loop might execute
370 // zero times.
371 States reachable = states;
372 for (size_t i = 0; i < reachable.size(); ++i) {
373 States next = analyze(loop.getRegion(), States{reachable[i]});
374 append(reachable, next);
375 }
376 states = std::move(reachable);
377 continue;
378 }
379 if (auto branch = mlir::dyn_cast<mlir::scf::IfOp>(&op)) {
380 States alternatives = analyze(branch.getThenRegion(), states);
381 append(alternatives, analyze(branch.getElseRegion(), states));
382 states = std::move(alternatives);
383 continue;
384 }
385 for (mlir::Region &nested : op.getRegions())
386 states = unknownRegion(nested, std::move(states));
387
388 if (auto poll = mlir::dyn_cast<NpuMaskPollOp>(&op))
389 if (pollsChannel(poll)) {
390 States polled;
391 for (const auto &state : states) {
392 DmaQueueModel path;
393 path.setState(state);
394 path.noteSpaceGuaranteed(key, depth);
395 append(polled, States{path.state()});
396 }
397 states = std::move(polled);
398 }
399
400 QueueEffect e = channelEffectOf(&op);
401 if (e.kind == QueueEffect::Kind::Ignore)
402 continue;
403 States next;
404 for (const auto &state : states) {
405 DmaQueueModel path;
406 path.setState(state);
407 if (e.kind == QueueEffect::Kind::Await) {
408 path.awaitToken(e.key);
409 } else {
410 if (path.wouldOverflow(e.key, depth)) {
411 if (overflowing.insert(&op))
412 witnesses[&op] = state;
413 path.noteSpaceGuaranteed(e.key, depth);
414 }
415 path.push(e.key, e.issuesToken);
416 }
417 append(next, States{path.state()});
418 }
419 states = std::move(next);
420 }
421 return states;
422 };
423
424 DmaQueueModel::State channelEntry;
425 if (auto it = entry.find(key); it != entry.end())
426 channelEntry.insert(*it);
427 States exits = analyze(body, States{channelEntry});
428 for (mlir::Operation *push : overflowing) {
429 queue.setState(witnesses[push]);
430 guardQueueOverflow(queue, push, tm, key, depth, enforce);
431 }
432
433 // Preserve exact straight-line state for callers. If control-flow paths
434 // disagree, retain the largest occupancy without crediting ambiguous
435 // tokens.
436 DmaQueueModel::State channelExit = exits.front();
437 for (const auto &state : exits)
438 for (const auto &[key, channel] : state) {
439 auto &dest = channelExit[key];
440 if (!(dest == channel)) {
441 dest.pushes.assign(
442 std::max(dest.pushes.size(), channel.pushes.size()), false);
443 dest.unknownTokens = true;
444 }
445 }
446 if (auto it = channelExit.find(key); it != channelExit.end())
447 merged[key] = it->second;
448 }
449 queue.setState(std::move(merged));
450}

40/// Returns true iff the parameter-sync preamble (use_lock acquire) should be
41/// emitted for this core. Reads the explicit attribute if present; otherwise
42/// defaults to true iff any 'aiex.read_scratchpad_parameter' op is present in
43/// the core body.
44static bool shouldEmitParameterSyncPreamble(CoreOp coreOp) {
45 if (auto attr = coreOp.getEmitParameterSyncPreambleAttr()) {
46 return attr.getValue();
47 }
48 bool found = false;
49 coreOp.getBody().walk([&](ReadScratchpadParameterOp) { found = true; });
50 return found;
51}

53// Warned here, not at NPU lowering, which inlines a sequence at every call
54// site and so would warn once per call.
55static void warnIfRuntimeOffsetMayRound(Operation *op, Type bufType) {
56 uint32_t elemBytes =
57 llvm::cast<BaseMemRefType>(bufType).getElementTypeBitWidth() / 8;
58 if (elemBytes == 0 || elemBytes % 4 == 0) {
59 return;
60 }
61 mlir::emitWarning(op->getLoc())
62 << "runtime offset parameter on a " << (elemBytes * 8)
63 << "-bit element type: the firmware masks the BD address register with "
64 "0xFFFFFFFC, so a value that is not a multiple of "
65 << (4 / std::gcd(4u, elemBytes))
66 << " elements is silently rounded down to the 4-byte boundary below "
67 "instead of being rejected";
68}

70/// A DMA with runtime offset `o` and length `l` touches elements
71/// [offset + o, last + o + lengthStep * l].
72struct TransferExtent {
73 int64_t bufferElems;
74 int64_t offset;
75 int64_t last;
76 int64_t lengthStep;
77};
78
79/// `sizes`/`strides` are innermost-first and empty for a linear BD.
80static std::optional<TransferExtent>
81getTransferExtent(Type bufType, int64_t offset, int64_t lenElems,
82 SmallVector<int64_t> sizes, SmallVector<int64_t> strides,
83 std::optional<int64_t> lengthUnit) {
84 auto buffer = llvm::cast<BaseMemRefType>(bufType);
85 if (!buffer.hasStaticShape())
86 return std::nullopt;
87 TransferExtent extent{buffer.getNumElements(), offset, offset, 0};
88 if (!lengthUnit) {
89 if (sizes.empty())
90 extent.last += lenElems - 1;
91 for (auto [size, stride] : llvm::zip(sizes, strides))
92 extent.last += (size - 1) * stride;
93 return extent;
94 }
95 bool linear = sizes.empty();
96 sizes.resize(std::max<size_t>(sizes.size(), 3), 1);
97 strides.resize(sizes.size(), 0);
98 for (size_t i = 3; i < sizes.size(); ++i)
99 extent.last += (sizes[i] - 1) * strides[i];
100 if (linear || AIEX::isContiguousTransfer(sizes, strides)) {
101 extent.last += lenElems - 1;
102 extent.lengthStep = *lengthUnit;
103 return extent;
104 }
105 AIE::placeRuntimeLengthDimension(sizes, strides);
106 int64_t rowElems = sizes[0] * sizes[1];
107 // A static length of 0 counts -1 steps, so that the first added unit's
108 // steps start at 0.
109 extent.last += (sizes[0] - 1) * strides[0] + (sizes[1] - 1) * strides[1] +
110 (llvm::divideCeilSigned(lenElems, rowElems) - 1) * strides[2];
111 extent.lengthStep = *lengthUnit / rowElems * strides[2];
112 return extent;
113}

115static std::optional<TransferExtent>
116getTransferExtent(NpuDmaMemcpyNdOp op, std::optional<int64_t> lengthUnit) {
117 std::optional<SmallVector<int64_t>> offsets =
118 getConstantIntValues(op.getMixedOffsets());
119 std::optional<SmallVector<int64_t>> sizes =
120 getConstantIntValues(op.getMixedSizes());
121 std::optional<SmallVector<int64_t>> strides =
122 getConstantIntValues(op.getMixedStrides());
123 int64_t elemBits = op.getElementTypeBitwidth();
124 if (!offsets || !sizes || !strides || elemBits % 8 != 0)
125 return std::nullopt;
126 std::reverse(sizes->begin(), sizes->end());
127 std::reverse(strides->begin(), strides->end());
128 return getTransferExtent(
129 op.getMemref().getType(), op.getOffsetInBytes() / (elemBits / 8),
130 (*sizes)[0] * (*sizes)[1] * (*sizes)[2], *sizes, *strides, lengthUnit);
131}

133/// The range of a BD's runtime offset, if it is known to stay within the
134/// buffer. An offset computed from a runtime-sequence loop's induction variable
135/// is only folded once the loop is unrolled, after this pass.
136static std::optional<std::pair<int64_t, int64_t>>
137getOffsetRange(Value offset, int64_t bufferElems, DataFlowSolver *solver) {
138 if (!solver)
139 return std::nullopt;
140 auto *state = solver->lookupState<dataflow::IntegerValueRangeLattice>(offset);
141 if (!state || state->getValue().isUninitialized())
142 return std::nullopt;
143 const ConstantIntRanges &range = state->getValue().getValue();
144 // The offset may be wider than 64 bits; an extremum that does not fit is
145 // outside the buffer anyway.
146 std::optional<int64_t> lo = range.smin().trySExtValue();
147 std::optional<int64_t> hi = range.smax().trySExtValue();
148 if (!lo || !hi || *lo < 0 || *hi >= bufferElems)
149 return std::nullopt;
150 return std::make_pair(*lo, *hi);
151}

153/// With a runtime offset, the extent starts at its smallest value and ends
154/// at its largest.
155static std::optional<TransferExtent>
156getTransferExtent(AIE::DMABDOp op, std::optional<int64_t> lengthUnit,
157 DataFlowSolver *solver) {
158 auto buffer = llvm::cast<BaseMemRefType>(op.getBuffer().getType());
159 std::optional<int64_t> offset = 0, offsetMin = 0;
160 if (op.hasOffset()) {
161 offset = offsetMin = op.getConstantOffset();
162 if (!offset && buffer.hasStaticShape()) {
163 if (auto range =
164 getOffsetRange(op.getOffset(), buffer.getNumElements(), solver)) {
165 offsetMin = range->first;
166 offset = range->second;
167 }
168 }
169 }
170 std::optional<int64_t> len;
171 if (op.hasLen())
172 len = op.getConstantLen();
173 else if (buffer.hasStaticShape())
174 len = buffer.getNumElements();
175 std::optional<SmallVector<int64_t>> sizes =
176 getConstantIntValues(op.getMixedSizes());
177 std::optional<SmallVector<int64_t>> strides =
178 getConstantIntValues(op.getMixedStrides());
179 if (!offset || !len || !sizes || !strides)
180 return std::nullopt;
181 std::reverse(sizes->begin(), sizes->end());
182 std::reverse(strides->begin(), strides->end());
183 std::optional<TransferExtent> extent =
184 getTransferExtent(buffer, *offset, *len, *sizes, *strides, lengthUnit);
185 if (extent)
186 extent->offset = *offsetMin;
187 return extent;
188}

190/// The largest length parameter value that keeps the shim BD's Buffer_Length,
191/// the static length plus that many units in 32-bit words, within its field.
192static int64_t getMaxLengthSteps(Operation *op, int64_t lenBytes,
193 int64_t unitBytes) {
194 int64_t maxWords =
195 getTargetModel(op).getDmaBdMaxLen(AIETileType::ShimNOCTile);
196 return (maxWords - lenBytes / 4) / (unitBytes / 4);
197}
198
199static std::optional<int64_t> getMaxLengthSteps(NpuDmaMemcpyNdOp op) {
200 std::optional<SmallVector<int64_t>> sizes =
201 getConstantIntValues(op.getMixedSizes());
202 std::optional<int64_t> lengthUnit = op.getLengthUnit();
203 if (!sizes || !lengthUnit)
204 return std::nullopt;
205 int64_t elemBytes = op.getElementTypeBitwidth() / 8;
206 return getMaxLengthSteps(op,
207 (*sizes)[1] * (*sizes)[2] * (*sizes)[3] * elemBytes,
208 *lengthUnit * elemBytes);
209}

211static std::optional<int64_t> getMaxLengthSteps(AIE::DMABDOp op) {
212 std::optional<int64_t> lengthUnit = op.getLengthUnit();
213 if (!lengthUnit)
214 return std::nullopt;
215 return getMaxLengthSteps(op, op.getLenInBytes(),
216 *lengthUnit * op.getBufferElementTypeWidthInBytes());
217}

219/// Returns true iff the parameter-sync preamble (create_scratchpad + set_lock)
220/// should be emitted into this sequence. Reads the explicit attribute if
221/// present; otherwise defaults to true iff the parent device uses any
222/// scratchpad parameter, from a core or a DMA BD, and the runtime sequence does
223/// not already contain a `aiex.sync_scratchpad_parameters_from_host` marker.
224static bool shouldEmitParameterSyncPreamble(RuntimeSequenceOp seqOp) {
225 if (auto attr = seqOp.getEmitParameterSyncPreambleAttr()) {
226 return attr.getValue();
227 }
228 bool hasManualSync = false;
229 seqOp.walk([&](SyncScratchpadParametersFromHostOp) { hasManualSync = true; });
230 if (hasManualSync) {
231 return false;
232 }
233 auto device = seqOp->getParentOfType<DeviceOp>();
234 if (!device) {
235 return false;
236 }
237 bool found = false;
238 device.walk([&](Operation *op) {
239 if (found) {
240 return;
241 }
242 if (llvm::isa<ReadScratchpadParameterOp>(op) ||
243 op->hasAttr("offset_parameter") ||
244 op->hasAttr("offset_state_table_idx") ||
245 op->hasAttr("length_parameter") ||
246 op->hasAttr("length_state_table_idx")) {
247 found = true;
248 }
249 });
250 return found;
251}

258 // For each read_scratchpad_parameter of a unique parameter, create a 2xi32
259 // buffer and store a reference to it on the ReadScratchpadParameterOp as
260 // the `buffer` attribute.
261 void allocateBuffers(DeviceOp device, OpBuilder &builder) {
262 MLIRContext *ctx = device.getContext();
263 unsigned uniquingCounter = 0;
264
265 DenseMap<std::pair<StringRef, Operation *>, BufferOp> seen;
266
267 device.walk([&](ReadScratchpadParameterOp readOp) {
268 auto coreOp = readOp->getParentOfType<CoreOp>();
269 TileOp tile = coreOp.getTileOp();
270 StringRef paramName = readOp.getParameter();
271 auto key = std::make_pair(paramName, tile.getOperation());
272
273 if (seen.count(key)) {
274 readOp.setBufferAttr(
275 FlatSymbolRefAttr::get(ctx, *seen[key].getSymName()));
276 return;
277 }
278
279 builder.setInsertionPointAfter(tile);
280 // Buffer must be 8 bytes: update_from_scratchpad always writes a 48-bit
281 // value across two 32-bit registers at [RegOff] and [RegOff+4]. The
282 // firmware masks Reg[0] with 0xFFFFFFFC (lower 2 bits forced to 0)
283 // because it was designed for 4-byte-aligned DMA BD addresses. The host
284 // library's `ParameterScratchpad::write` left-shifts by 2 before
285 // writing to the scratchpad, and the core right-shifts by 2 after
286 // loading. Note that this limits effective parameter values to 30
287 // bits.
288 auto bufType = MemRefType::get({2}, builder.getI32Type());
289 std::string prefix =
290 ("__param_" + paramName + "_" + std::to_string(tile.getCol()) + "_" +
291 std::to_string(tile.getRow()) + "_")
292 .str();
293 std::string bufName =
294 AIE::generateUniqueSymbolName(device, prefix, uniquingCounter);
295 auto buf =
296 BufferOp::create(builder, readOp.getLoc(), bufType, tile,
297 builder.getStringAttr(bufName), /*address=*/nullptr,
298 /*initial_value=*/nullptr, /*mem_bank=*/nullptr);
299 seen[key] = buf;
300
301 readOp.setBufferAttr(
302 FlatSymbolRefAttr::get(ctx, *seen[key].getSymName()));
303 });
304 }

306 // Lower each read_scratchpad_parameter to: load from buffer[0], shift right
307 // by 2, and cast to the result type. The buffer to use comes from the
308 // `buffer` attribute set by allocateBuffers().
309 void lowerReadParameters(DeviceOp device, OpBuilder &builder) {
310 SmallVector<ReadScratchpadParameterOp> readOps;
311 device.walk([&](ReadScratchpadParameterOp op) { readOps.push_back(op); });
312
313 for (auto readOp : readOps) {
314 FlatSymbolRefAttr bufRef = readOp.getBufferAttr();
315 auto buf = AIE::lookupNamedOpIn<BufferOp>(device, bufRef.getAttr());
316
317 builder.setInsertionPoint(readOp);
318 Value c0 = arith::ConstantIndexOp::create(builder, readOp.getLoc(), 0);
319 Value raw = memref::LoadOp::create(builder, readOp.getLoc(), buf, c0);
320 Value c2 = arith::ConstantOp::create(builder, readOp.getLoc(),
321 builder.getI32IntegerAttr(2));
322 Value decoded = arith::ShRUIOp::create(builder, readOp.getLoc(), raw, c2);
323
324 Type resultType = readOp.getResult().getType();
325 Value result = decoded;
326 if (resultType.isInteger() && resultType != builder.getI32Type()) {
327 result = arith::TruncIOp::create(builder, readOp.getLoc(), resultType,
328 decoded);
329 } else if (resultType.isBF16()) {
330 Value masked = arith::TruncIOp::create(builder, readOp.getLoc(),
331 builder.getI16Type(), decoded);
332 result = arith::BitcastOp::create(builder, readOp.getLoc(), resultType,
333 masked);
334 }
335
336 readOp.getResult().replaceAllUsesWith(result);
337 readOp.erase();
338 }
339 }

341 // For each core in `device` with shouldEmitParameterSyncPreamble()==true,
342 // create an aie.lock (no lockID; AIEAssignLockIDs will assign one later) and
343 // insert aie.use_lock(Acquire, 1) at the top of the core body. Returns the
344 // list of lock values created (one per qualifying core, in walk order).
345 SmallVector<Value> emitCorePreambles(DeviceOp device, OpBuilder &builder) {
346 SmallVector<Value> syncLocks;
347 device.walk([&](CoreOp coreOp) {
348 if (!shouldEmitParameterSyncPreamble(coreOp)) {
349 return;
350 }
351
352 TileOp tile = coreOp.getTileOp();
353 builder.setInsertionPointAfter(tile);
354 auto lockOp = LockOp::create(
355 builder, coreOp.getLoc(), builder.getIndexType(), tile.getResult(),
356 /*lockID=*/IntegerAttr{}, builder.getI32IntegerAttr(0),
357 /*sym_name=*/StringAttr{});
358 syncLocks.push_back(lockOp.getResult());
359
360 Block &bodyBlock = coreOp.getBody().front();
361 builder.setInsertionPointToStart(&bodyBlock);
362 UseLockOp::create(builder, coreOp.getLoc(), lockOp.getResult(),
363 LockAction::Acquire, 1);
364
365 // Mark as done so the pass is idempotent.
366 coreOp.setEmitParameterSyncPreambleAttr(builder.getBoolAttr(false));
367 });
368 return syncLocks;
369 }

371 // For each runtime sequence in `device` with shouldEmitParameterSyncPreamble
372 // ()==true, insert a marker SyncScratchpadParametersFromHostOp after the
373 // last top-level load_pdi of the sequence body, or at its start if there is
374 // none.
375 LogicalResult emitSequencePreambles(DeviceOp device, OpBuilder &builder) {
376 WalkResult result = device.walk([&](RuntimeSequenceOp seqOp) {
377 if (!shouldEmitParameterSyncPreamble(seqOp)) {
378 return WalkResult::advance();
379 }
380
381 Region &region = seqOp.getBody();
382 if (region.empty())
383 region.emplaceBlock();
384 Block &body = region.front();
385 builder.setInsertionPointToStart(&body);
386 // Loading a PDI resets the core buffers and locks the sync writes.
387 for (NpuLoadPdiOp loadPdi : body.getOps<NpuLoadPdiOp>())
388 builder.setInsertionPointAfter(loadPdi);
389 for (Operation &op :
390 llvm::make_range(body.begin(), builder.getInsertionPoint())) {
391 WalkResult early = op.walk([&](Operation *inner) {
392 auto dmaOp = dyn_cast<NpuDmaMemcpyNdOp>(inner);
393 auto bdOp = dyn_cast<AIE::DMABDOp>(inner);
394 if ((dmaOp && (dmaOp.getOffsetStateTableIdxAttr() ||
395 dmaOp.getLengthStateTableIdxAttr())) ||
396 (bdOp && (bdOp.getOffsetStateTableIdxAttr() ||
397 bdOp.getLengthStateTableIdxAttr()))) {
398 inner->emitOpError(
399 "reads a scratchpad parameter before the last "
400 "aiex.npu.load_pdi of its runtime sequence, which is where "
401 "the scratchpad is created. Load the PDIs first, or place "
402 "aiex.sync_scratchpad_parameters_from_host before this op.");
403 return WalkResult::interrupt();
404 }
405 return WalkResult::advance();
406 });
407 if (early.wasInterrupted())
408 return WalkResult::interrupt();
409 }
410 SyncScratchpadParametersFromHostOp::create(builder, seqOp.getLoc());
411
412 // Mark as done so the pass is idempotent.
413 seqOp.setEmitParameterSyncPreambleAttr(builder.getBoolAttr(false));
414 return WalkResult::advance();
415 });
416 return failure(result.wasInterrupted());
417 }

419 // Lower a sync_scratchpad_parameters_from_host marker op in place to its
420 // component ops, using the parameter information for the device that contains
421 // it:
422 // 1. npu.create_scratchpad(scratchpadSize)
423 // 2. For each (state_idx, buffer_ref) in paramEntries: npu.write32(0) +
424 // npu.update_from_scratchpad
425 // 3. set_lock(lock, 1) for each lock in syncLocks
426 void lowerSyncParametersOp(
427 SyncScratchpadParametersFromHostOp syncOp, uint32_t scratchpadSize,
428 const SmallVector<std::pair<uint8_t, FlatSymbolRefAttr>> &paramEntries,
429 const SmallVector<Value> &syncLocks) {
430 OpBuilder builder(syncOp);
431 Location loc = syncOp.getLoc();
432
433 // A module with no parameters has nothing to stage, and size 0 is invalid.
434 if (scratchpadSize > 0)
435 NpuCreateScratchpadOp::create(builder, loc, scratchpadSize);
436
437 for (auto &[stateIdx, bufRef] : paramEntries) {
438 // Zero the destination before the additive UpdateScratchpad.
439 NpuWrite32Op::create(builder, loc,
440 /*address=*/createConstantI32(builder, loc, 0),
441 /*value=*/createConstantI32(builder, loc, 0), bufRef,
442 /*column=*/nullptr, /*row=*/nullptr);
443 NpuUpdateFromScratchpadOp::create(
444 builder, loc, stateIdx, StateTableFunc::Incr,
445 /*func_arg=*/static_cast<uint32_t>(0),
446 /*address=*/static_cast<uint32_t>(0), bufRef,
447 /*column=*/nullptr, /*row=*/nullptr);
448 }
449
450 for (Value lock : syncLocks) {
451 SetLockOp::create(builder, loc, lock, createConstantI32(builder, loc, 1));
452 }
453
454 syncOp.erase();
455 }

457 // Lower every sync_scratchpad_parameters_from_host op in `device` to its
458 // component ops in place, using parameter info for this device only.
459 void lowerSyncParametersOps(
460 DeviceOp device, uint32_t scratchpadSize,
461 const SmallVector<std::pair<uint8_t, FlatSymbolRefAttr>> &paramEntries,
462 const SmallVector<Value> &syncLocks) {
463 SmallVector<SyncScratchpadParametersFromHostOp> syncOps;
464 device.walk(
465 [&](SyncScratchpadParametersFromHostOp op) { syncOps.push_back(op); });
466 for (SyncScratchpadParametersFromHostOp syncOp : syncOps) {
467 lowerSyncParametersOp(syncOp, scratchpadSize, paramEntries, syncLocks);
468 }
469 }

488 void runOnOperation() override {
489 ModuleOp moduleOp = getOperation();
490 OpBuilder builder(&getContext());
491
492 // Ranges are only needed for a parameterized BD with a runtime offset.
493 std::unique_ptr<DataFlowSolver> solver;
494 WalkResult needRanges = moduleOp.walk([](AIE::DMABDOp bd) {
495 bool param = bd.getLengthParameterAttr() || bd.getOffsetParameterAttr();
496 return param && bd.getOffset() && !bd.getConstantOffset()
497 ? WalkResult::interrupt()
498 : WalkResult::advance();
499 });
500 if (needRanges.wasInterrupted()) {
501 solver = std::make_unique<DataFlowSolver>();
502 solver->load<dataflow::DeadCodeAnalysis>();
503 solver->load<dataflow::SparseConstantPropagation>();
504 solver->load<dataflow::IntegerRangeAnalysis>();
505 if (failed(solver->initializeAndRun(moduleOp)))
506 solver.reset();
507 }
508
509 // Step 1: collect every parameter in the module.
510 SmallVector<ScratchpadParameterOp> allParams;
511 moduleOp.walk([&](ScratchpadParameterOp p) { allParams.push_back(p); });
512
513 if (allParams.size() > 32) {
514 InFlightDiagnostic diag =
515 moduleOp.emitError("Module declares ")
516 << allParams.size()
517 << " parameters but the scratchpad supports at most 32. The "
518 "scratchpad is a single hardware resource shared by all PDIs "
519 "loaded by a runtime sequence.";
520 for (auto p : allParams)
521 diag.attachNote(p.getLoc()) << "parameter '" << p.getSymName() << "'";
522 return signalPassFailure();
523 }
524
525 // Step 2: determine each parameter's kind from its usage, erroring on
526 // mixed use. A parameter is "core" if any aiex.read_scratchpad_parameter
527 // references it; "addr" if any DMA op references it via offset_parameter.
528 // If both, emit an error. A DMA length_parameter works with either kind,
529 // so it takes the kind of the parameter's other uses, and "addr" if it is
530 // the only use; the lowering compensates in the update's multiplier.
531 DenseMap<StringRef, bool> usedAsCore;
532 DenseMap<StringRef, bool> usedAsLength;
533 DenseMap<StringRef, bool> usedAsAddr;
534 moduleOp.walk([&](ReadScratchpadParameterOp op) {
535 usedAsCore[op.getParameter()] = true;
536 });
537 auto markDmaUses = [&](FlatSymbolRefAttr offsetRef,
538 FlatSymbolRefAttr lengthRef) {
539 if (offsetRef)
540 usedAsAddr[offsetRef.getValue()] = true;
541 if (lengthRef)
542 usedAsLength[lengthRef.getValue()] = true;
543 };
544 moduleOp.walk([&](NpuDmaMemcpyNdOp op) {
545 markDmaUses(op.getOffsetParameterAttr(), op.getLengthParameterAttr());
546 });
547 moduleOp.walk([&](AIE::DMABDOp op) {
548 markDmaUses(op.getOffsetParameterAttr(), op.getLengthParameterAttr());
549 });
550
551 for (auto p : allParams) {
552 StringRef name = p.getSymName();
553 bool core = usedAsCore.lookup(name);
554 bool length = usedAsLength.lookup(name);
555 bool addr = usedAsAddr.lookup(name);
556 if (core && addr) {
557 p.emitError("parameter '")
558 << name
559 << "' is used both as an aiex.read_scratchpad_parameter source "
560 "(core) and as a DMA offset_parameter (addr); a parameter "
561 "must have a single kind";
562 return signalPassFailure();
563 }
564 p.setKindAttr(ScratchpadParameterKindAttr::get(
565 &getContext(), addr || (length && !core)
566 ? ScratchpadParameterKind::Addr
567 : ScratchpadParameterKind::Core));
568 }
569
570 // Step 3: assign global state_table_idx in walk order, 0..N-1.
571 for (auto [i, p] : llvm::enumerate(allParams)) {
572 p.setStateTableIdxAttr(builder.getIntegerAttr(
573 builder.getIntegerType(8, /*isSigned=*/false), i));
574 }
575
576 // Step 3b: rewrite DMA `offset_parameter` / `length_parameter` symbol
577 // references to plain `offset_state_table_idx` / `length_state_table_idx`
578 // integer attributes, so downstream `aie`-dialect passes do not need to
579 // resolve `aiex.scratchpad_parameter` symbols.
580 auto rewriteParam =
581 [&](Operation *op, FlatSymbolRefAttr ref, StringRef attrName,
582 StringRef idxAttrName) -> FailureOr<ScratchpadParameterOp> {
583 auto paramOp =
584 moduleOp.lookupSymbol<ScratchpadParameterOp>(ref.getAttr());
585 if (!paramOp) {
586 op->emitOpError() << attrName << " '" << ref.getValue()
587 << "' not found. Declare it at module scope with "
588 "aiex.scratchpad_parameter.";
589 return failure();
590 }
591 if (!paramOp.getType().isInteger(32)) {
592 auto err = op->emitOpError()
593 << attrName << " '" << ref.getValue()
594 << "' must have type i32, got " << paramOp.getType() << ".";
595 err.attachNote(paramOp.getLoc()) << "Parameter declared here.";
596 return failure();
597 }
598 uint8_t stateIdx =
599 static_cast<uint8_t>(paramOp.getStateTableIdx().value());
600 op->setAttr(idxAttrName,
601 builder.getIntegerAttr(
602 builder.getIntegerType(8, /*isSigned=*/false), stateIdx));
603 op->removeAttr(attrName);
604 return paramOp;
605 };
606 // The values of each DMA offset or length parameter that keep every
607 // transfer using it within its buffer, judged with the transfer's other
608 // parameter at 0.
609 DenseMap<StringRef, std::pair<int64_t, int64_t>> bounds;
610 // (length, offset, length step) -> max of offset + length step * length.
611 llvm::MapVector<std::tuple<StringRef, StringRef, int64_t>, int64_t>
612 jointBounds;
613 auto rewriteParams =
614 [&](Operation *op, FlatSymbolRefAttr offsetRef,
615 FlatSymbolRefAttr lengthRef, Type bufType,
616 std::optional<TransferExtent> extent,
617 std::optional<int64_t> maxLengthSteps) -> LogicalResult {
618 bool joint = offsetRef == lengthRef;
619 int64_t room = extent ? extent->bufferElems - 1 - extent->last : 0;
620 if (offsetRef) {
621 if (failed(rewriteParam(op, offsetRef, "offset_parameter",
622 "offset_state_table_idx")))
623 return failure();
624 warnIfRuntimeOffsetMayRound(op, bufType);
625 auto &[lo, hi] =
626 bounds.try_emplace(offsetRef.getValue(), INT32_MIN, INT32_MAX)
627 .first->second;
628 if (extent) {
629 lo = std::max(lo, -extent->offset);
630 hi = std::min(hi, llvm::divideFloorSigned(
631 room, 1 + (joint ? extent->lengthStep : 0)));
632 }
633 }
634 if (lengthRef) {
635 FailureOr<ScratchpadParameterOp> lengthParam = rewriteParam(
636 op, lengthRef, "length_parameter", "length_state_table_idx");
637 if (failed(lengthParam))
638 return failure();
639 auto &[lo, hi] =
640 bounds.try_emplace(lengthRef.getValue(), INT32_MIN, INT32_MAX)
641 .first->second;
642 lo = std::max<int64_t>(lo, 0);
643 if (maxLengthSteps)
644 hi = std::min(hi, *maxLengthSteps);
645 if (lengthParam->getKind() == ScratchpadParameterKind::Core) {
646 op->setAttr("length_core_encoded", builder.getUnitAttr());
647 hi = std::min<int64_t>(hi, (1 << 30) - 1);
648 }
649 if (extent && !joint && extent->lengthStep > 0)
650 hi = std::min(hi, llvm::divideFloorSigned(room, extent->lengthStep));
651 if (extent && offsetRef && !joint && extent->lengthStep > 0) {
652 auto [it, inserted] = jointBounds.try_emplace(
653 {lengthRef.getValue(), offsetRef.getValue(), extent->lengthStep},
654 room);
655 it->second = std::min(it->second, room);
656 }
657 }
658 for (FlatSymbolRefAttr ref : {offsetRef, lengthRef}) {
659 if (!ref)
660 continue;
661 auto [lo, hi] = bounds.lookup(ref.getValue());
662 if (lo > hi)
663 return op->emitOpError("no value of parameter '")
664 << ref.getValue()
665 << "' keeps every transfer using it within its buffer";
666 }
667 return success();
668 };
669 WalkResult rewriteResult = moduleOp.walk([&](Operation *op) {
670 if (auto dmaOp = dyn_cast<NpuDmaMemcpyNdOp>(op)) {
671 FlatSymbolRefAttr offsetRef = dmaOp.getOffsetParameterAttr();
672 FlatSymbolRefAttr lengthRef = dmaOp.getLengthParameterAttr();
673 if (!offsetRef && !lengthRef)
674 return WalkResult::advance();
675 if (failed(rewriteParams(
676 op, offsetRef, lengthRef, dmaOp.getMemref().getType(),
677 getTransferExtent(dmaOp, lengthRef ? dmaOp.getLengthUnit()
678 : std::nullopt),
679 getMaxLengthSteps(dmaOp))))
680 return WalkResult::interrupt();
681 } else if (auto bdOp = dyn_cast<AIE::DMABDOp>(op)) {
682 FlatSymbolRefAttr offsetRef = bdOp.getOffsetParameterAttr();
683 FlatSymbolRefAttr lengthRef = bdOp.getLengthParameterAttr();
684 if (!offsetRef && !lengthRef)
685 return WalkResult::advance();
686 std::optional<int64_t> lengthUnit;
687 if (lengthRef)
688 lengthUnit = bdOp.getLengthUnit();
689 if (failed(rewriteParams(
690 op, offsetRef, lengthRef, bdOp.getBuffer().getType(),
691 getTransferExtent(bdOp, lengthUnit, solver.get()),
692 getMaxLengthSteps(bdOp))))
693 return WalkResult::interrupt();
694 }
695 return WalkResult::advance();
696 });
697 if (rewriteResult.wasInterrupted()) {
698 return signalPassFailure();
699 }
700 for (ScratchpadParameterOp p : allParams) {
701 auto it = bounds.find(p.getSymName());
702 if (it == bounds.end())
703 continue;
704 p.setMinValueAttr(builder.getI32IntegerAttr(it->second.first));
705 p.setMaxValueAttr(builder.getI32IntegerAttr(it->second.second));
706 }
707 llvm::MapVector<StringRef, SmallVector<Attribute>> jointBoundAttrs;
708 for (auto [key, max] : jointBounds) {
709 auto [length, offset, step] = key;
710 jointBoundAttrs[length].push_back(JointBoundAttr::get(
711 &getContext(), FlatSymbolRefAttr::get(&getContext(), offset), step,
712 max));
713 }
714 for (ScratchpadParameterOp p : allParams) {
715 auto *it = jointBoundAttrs.find(p.getSymName());
716 if (it != jointBoundAttrs.end())
717 p.setJointBoundsAttr(builder.getArrayAttr(it->second));
718 }
719
720 // Step 4: per-device lowering.
721 SmallVector<DeviceOp> devices;
722 moduleOp.walk([&](DeviceOp d) { devices.push_back(d); });
723 for (auto d : devices) {
724 allocateBuffers(d, builder);
725
726 // Collect unique (stateIdx, bufferRef) pairs for core-kind parameters in
727 // this device. Buffer attrs are set by allocateBuffers() above.
728 SmallVector<std::pair<uint8_t, FlatSymbolRefAttr>> paramEntries;
729 DenseSet<StringRef> seenBufs;
730 d.walk([&](ReadScratchpadParameterOp readOp) {
731 FlatSymbolRefAttr bufRef = readOp.getBufferAttr();
732 if (!bufRef || !seenBufs.insert(bufRef.getValue()).second) {
733 return;
734 }
735 auto paramOp =
736 moduleOp.lookupSymbol<ScratchpadParameterOp>(readOp.getParameter());
737 uint8_t stateIdx =
738 static_cast<uint8_t>(paramOp.getStateTableIdx().value());
739 paramEntries.push_back({stateIdx, bufRef});
740 });
741
742 // Emit lock + use_lock preambles in cores, then insert a marker
743 // SyncScratchpadParametersFromHostOp at the start of each qualifying
744 // runtime sequence. Immediately lower every
745 // sync_scratchpad_parameters_from_host op in the device (both the
746 // preamble-inserted ones and any user-written ones) using this device's
747 // parameter info.
748 SmallVector<Value> syncLocks = emitCorePreambles(d, builder);
749 if (failed(emitSequencePreambles(d, builder)))
750 return signalPassFailure();
751 uint32_t scratchpadSize = static_cast<uint32_t>(allParams.size() * 4);
752 lowerSyncParametersOps(d, scratchpadSize, paramEntries, syncLocks);
753
754 lowerReadParameters(d, builder);
755 }
756
757 // Step 5: emit the single params.txt for the module.
758 if (failed(emitParamsFile())) {
759 return signalPassFailure();
760 }
761 }
762};

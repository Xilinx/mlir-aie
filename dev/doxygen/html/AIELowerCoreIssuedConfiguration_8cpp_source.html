39/// The core reaches its own tile's registers through a window at this offset;
40/// see the write_tm intrinsic, whose own default says the same.
41constexpr uint64_t kTileMemoryMappedBase = 0x80000;
42
43/// The names of the store and load intrinsics for this architecture.
44struct TileAccessIntrinsics {
45 std::string write, read;
46};
47
48static std::optional<TileAccessIntrinsics> tileAccessIntrinsics(AIEArch arch) {
49 switch (arch) {
50 case AIEArch::AIE2:
51 return TileAccessIntrinsics{"llvm.aie2.write.tm", "llvm.aie2.read.tm"};
52 case AIEArch::AIE2p:
53 return TileAccessIntrinsics{"llvm.aie2p.write.tm", "llvm.aie2p.read.tm"};
54 default:
55 return std::nullopt;
56 }
57}

59/// The writes aie-rt makes to push `op`'s task, as (tile offset, value).
60static LogicalResult
61pushWrites(DmaPushQueueOp op, int col, int row,
62 const AIETargetModel &targetModel,
63 SmallVectorImpl<std::pair<uint64_t, uint32_t>> &writes) {
64 AIERTControl ctl(targetModel);
65 if (failed(ctl.setIOBackend(/*aieSim=*/false, /*xaieDebug=*/false)))
66 return op.emitOpError("could not start aie-rt to collect its writes");
67 ctl.startTransaction();
68 // Only the task queue. Rewriting the channel's control register while the
69 // channel runs loses the state it keeps between packets, and the channel
70 // this pushes onto is already configured and running.
71 if (failed(ctl.pushToBdQueueAndEnable(
72 *op.getOperation(), col, row, op.getChannel(), op.getDirection(),
73 op.getBdId(), op.getRepeatCount(), /*padValue=*/0, op.getOutOfOrder(),
74 /*configureChannel=*/false)))
75 return failure();
76
77 std::vector<uint8_t> txn = ctl.exportSerializedTransaction();
78 std::vector<TransactionBinaryOperation> ops;
79 if (!parseTransactionBinary(txn, ops))
80 return op.emitOpError("could not read back what aie-rt recorded");
81
82 // Only this tile's registers are the core's to write, and only a plain
83 // store is. Anything else means aie-rt pushes a task in a way this lowering
84 // does not carry, and silently dropping it would leave the channel without
85 // the task.
86 uint64_t tileBase =
87 (static_cast<uint64_t>(col) << targetModel.getColumnShift()) |
88 (static_cast<uint64_t>(row) << targetModel.getRowShift());
89 uint64_t tileSpan = 1ull << targetModel.getRowShift();
90 for (const TransactionBinaryOperation &txnOp : ops) {
91 if (txnOp.cmd.Opcode != XAie_TxnOpcode::XAIE_IO_WRITE)
92 return op.emitOpError("pushing this task takes a ")
93 << static_cast<int>(txnOp.cmd.Opcode)
94 << " transaction command, which a core cannot make; only plain "
95 "register writes lower here";
96 uint64_t addr = txnOp.cmd.RegOff;
97 if (addr < tileBase || addr >= tileBase + tileSpan)
98 return op.emitOpError("pushing this task writes ")
99 << addr << ", outside tile (" << col << ", " << row
100 << "); a core reaches only its own tile's registers";
101 writes.emplace_back(addr - tileBase, txnOp.cmd.Value);
102 }
103 if (writes.empty())
104 return op.emitOpError("aie-rt recorded no writes for this push");
105 return success();
106}

109 : public xilinx::AIE::impl::AIELowerCoreIssuedConfigurationBase<

112 /// A declaration of `name`, inside the device, which is a symbol table of
113 /// its own, so that the calls resolve here; outlining the cores later takes
114 /// the declaration out to the module along with them.
115 func::FuncOp declare(DeviceOp device, StringRef name, TypeRange argTypes,
116 TypeRange resultTypes) {
117 if (auto decl = device.lookupSymbol<func::FuncOp>(name))
118 return decl;
119 OpBuilder builder(device.getContext());
120 builder.setInsertionPointToStart(device.getBody());
121 auto decl = func::FuncOp::create(
122 builder, device.getLoc(), name,
123 FunctionType::get(device.getContext(), argTypes, resultTypes));
124 decl.setPrivate();
125 return decl;
126 }
127
128 /// A pointer to `offset` in this tile's register window.
129 Value tilePointer(OpBuilder &builder, Location loc, uint64_t offset) {
130 Type i32 = builder.getI32Type();
131 Type ptr = LLVM::LLVMPointerType::get(builder.getContext());
132 Value addr = arith::ConstantOp::create(
133 builder, loc, i32,
134 builder.getI32IntegerAttr(
135 static_cast<int32_t>(kTileMemoryMappedBase + offset)));
136 return LLVM::IntToPtrOp::create(builder, loc, ptr, addr);
137 }
138
139 LogicalResult lowerPush(DmaPushQueueOp op, const AIETargetModel &targetModel,
140 func::FuncOp write) {
141 CoreOp core = op->getParentOfType<CoreOp>();
142 TileOp tile = core.getTileOp();
143 SmallVector<std::pair<uint64_t, uint32_t>> writes;
144 if (failed(
145 pushWrites(op, tile.getCol(), tile.getRow(), targetModel, writes)))
146 return failure();
147 // The configuration enables the bus for this core before it runs.
148 core.setProcessorBus(true);
149
150 OpBuilder builder(op);
151 Type i32 = builder.getI32Type();
152 for (auto [offset, value] : writes) {
153 Value valueOp = arith::ConstantOp::create(
154 builder, op.getLoc(), i32,
155 builder.getI32IntegerAttr(static_cast<int32_t>(value)));
156 Value addrPtr = tilePointer(builder, op.getLoc(), offset);
157 // A call to a declaration, not llvm.call_intrinsic: the intrinsic is
158 // the AIE backend's, and translation to LLVM IR runs against an LLVM
159 // that does not have it. A named callee is what the other AIE
160 // intrinsics the cores use are reached through too.
161 func::CallOp::create(builder, op.getLoc(), write,
162 ValueRange{valueOp, addrPtr});
163 }
164 op.erase();
165 return success();
166 }
167
168 LogicalResult lowerPoll(DmaPollForQueueSpaceOp op,
169 const AIETargetModel &targetModel,
170 func::FuncOp read) {
171 CoreOp core = op->getParentOfType<CoreOp>();
172 TileOp tile = core.getTileOp();
173 std::optional<uint32_t> statusAddr = targetModel.getDmaStatusAddress(
174 tile.getCol(), tile.getRow(), op.getChannel(), op.getDirection());
175 if (!statusAddr)
176 return op.emitOpError("this tile reports no DMA status register to read");
177 uint32_t mask = targetModel.getDmaTaskQueueSizeMask();
178 if (mask == 0)
179 return op.emitOpError(
180 "this tile reports no task-queue field in its DMA status");
181 core.setProcessorBus(true);
182
183 OpBuilder builder(op);
184 Location loc = op.getLoc();
185 Type i32 = builder.getI32Type();
186 uint32_t localAddr = *statusAddr & ((1u << targetModel.getRowShift()) - 1);
187 Value addrPtr = tilePointer(builder, loc, localAddr);
188 Value maskOp = arith::ConstantOp::create(
189 builder, loc, i32,
190 builder.getI32IntegerAttr(static_cast<int32_t>(mask)));
191 // The field counts the tasks the queue holds, so it reads the depth
192 // exactly when the queue has no room left.
193 uint32_t full = targetModel.getDmaTaskQueueDepth()
194 << llvm::countr_zero(mask);
195 Value fullOp = arith::ConstantOp::create(
196 builder, loc, i32,
197 builder.getI32IntegerAttr(static_cast<int32_t>(full)));
198
199 // Read the field until it falls below the depth. The core does nothing
200 // else meanwhile.
201 auto loop = scf::WhileOp::create(builder, loc, TypeRange{}, ValueRange{});
202 OpBuilder::InsertionGuard guard(builder);
203 Block *before = builder.createBlock(&loop.getBefore());
204 builder.setInsertionPointToStart(before);
205 Value status = func::CallOp::create(builder, loc, read, ValueRange{addrPtr})
206 .getResult(0);
207 Value size = arith::AndIOp::create(builder, loc, status, maskOp);
208 Value isFull = arith::CmpIOp::create(builder, loc, arith::CmpIPredicate::eq,
209 size, fullOp);
210 scf::ConditionOp::create(builder, loc, isFull, ValueRange{});
211
212 Block *after = builder.createBlock(&loop.getAfter());
213 builder.setInsertionPointToStart(after);
214 scf::YieldOp::create(builder, loc, ValueRange{});
215
216 op.erase();
217 return success();
218 }

220 void runOnOperation() override {
221 DeviceOp device = getOperation();
222 const AIETargetModel &targetModel = device.getTargetModel();
223
224 SmallVector<DmaPushQueueOp> pushes;
225 SmallVector<DmaPollForQueueSpaceOp> polls;
226 device.walk([&](Operation *op) {
227 if (auto push = dyn_cast<DmaPushQueueOp>(op))
228 pushes.push_back(push);
229 else if (auto poll = dyn_cast<DmaPollForQueueSpaceOp>(op))
230 polls.push_back(poll);
231 });
232 if (pushes.empty() && polls.empty())
233 return;
234
235 std::optional<TileAccessIntrinsics> intrinsics =
236 tileAccessIntrinsics(targetModel.getTargetArch());
237 if (!intrinsics) {
238 Operation *first = pushes.empty() ? polls.front().getOperation()
239 : pushes.front().getOperation();
240 first->emitOpError("this device has no access a core can make to its "
241 "own tile's registers");
242 return signalPassFailure();
243 }
244
245 OpBuilder builder(device.getContext());
246 Type i32 = builder.getI32Type();
247 Type ptr = LLVM::LLVMPointerType::get(device.getContext());
248
249 if (!polls.empty()) {
250 func::FuncOp read = declare(device, intrinsics->read, {ptr}, {i32});
251 for (DmaPollForQueueSpaceOp op : polls)
252 if (failed(lowerPoll(op, targetModel, read)))
253 return signalPassFailure();
254 }
255 if (!pushes.empty()) {
256 func::FuncOp write =
257 declare(device, intrinsics->write, {i32, ptr}, TypeRange{});
258 for (DmaPushQueueOp op : pushes)
259 if (failed(lowerPush(op, targetModel, write)))
260 return signalPassFailure();
261 }
262 }
263};

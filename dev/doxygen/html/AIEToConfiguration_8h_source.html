71// Enum for specifying which tile types to reset
72enum class ResetTileType : unsigned {
73 None = 0,
74 ShimNOC = 1 << 0,
75 MemTile = 1 << 1,
76 CoreTile = 1 << 2,
77 All = ShimNOC | MemTile | CoreTile
78};
79
80inline bool hasFlag(ResetTileType value, ResetTileType flag) {
81 return (static_cast<unsigned>(value) & static_cast<unsigned>(flag)) != 0;
82}
83
84// Enum for specifying when to reset
85enum class ResetMode {
86 Never, // Never perform reset
87 IfUsed, // Reset only if the tile is used in the device
88 IfUsedFineGrained, // Reset only individual locks/connections that are used
89 IfChanged, // Reset only if the tile configuration changed from previous
90 IfChangedFineGrained, // Reset only individual locks/connections that changed
91 Always // Reset all tiles of the specified type
92};
93
94// Configuration for different reset operations
95struct ResetConfig {
96 ResetTileType tileType;
97 ResetMode mode;
98
99 ResetConfig(ResetTileType tt = ResetTileType::None,
100 ResetMode m = ResetMode::Never)
101 : tileType(tt), mode(m) {}
102};

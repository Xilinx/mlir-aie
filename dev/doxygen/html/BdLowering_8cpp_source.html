24// SsaStridePolicy: arith-emitting mirror of ConstStridePolicy.
25//
26// Arithmetic is i32, matching the BD word fields. This agrees with
27// ConstStridePolicy's int64 because encodeBdCommon bounds every runtime operand
28// before it reaches the policy, so no intermediate product overflows.
29//===----------------------------------------------------------------------===//

31Value SsaStridePolicy::cst(int64_t c) const {
32 auto i32ty = builder.getIntegerType(32);
33 return arith::ConstantOp::create(builder, loc, IntegerAttr::get(i32ty, c));
34}

36Value SsaStridePolicy::mul(Value v, int64_t c) const {
37 return arith::MulIOp::create(builder, loc, v, cst(c));
38}

40Value SsaStridePolicy::div(Value v, int64_t c) const {
41 // Hardware granularity scaling is exact (verifier enforces divisibility);
42 // unsigned division matches the constant policy on the non-negative values
43 // that reach here.
44 return arith::DivUIOp::create(builder, loc, v, cst(c));
45}

75Value getAsValue(OpBuilder &builder, Location loc, OpFoldResult ofr,
76 Type intType) {
77 if (auto constVal = getConstantIntValue(ofr))
78 return arith::ConstantOp::create(builder, loc,
79 IntegerAttr::get(intType, *constVal));
80 Value val = cast<Value>(ofr);
81 if (val.getType() != intType) {
82 unsigned valBits = val.getType().getIntOrFloatBitWidth();
83 unsigned tgtBits = intType.getIntOrFloatBitWidth();
84 if (valBits > tgtBits)
85 val = arith::TruncIOp::create(builder, loc, intType, val);
86 else
87 val = arith::ExtUIOp::create(builder, loc, intType, val);
88 }
89 return val;
90}

92Value buildArgPlusValue(OpBuilder &builder, Location loc,
93 ArrayRef<OpFoldResult> elementOffsets,
94 ArrayRef<OpFoldResult> strides, int64_t elemWidthBytes,
95 int64_t baseByteOffset) {
96 auto i32ty = builder.getIntegerType(32);
97
98 // Fast path: everything constant -> fold to one arith.constant, matching the
99 // static lowering byte-for-byte.
100 bool allConst = llvm::all_of(elementOffsets,
101 [](OpFoldResult o) {
102 return getConstantIntValue(o).has_value();
103 }) &&
104 llvm::all_of(strides, [](OpFoldResult s) {
105 return getConstantIntValue(s).has_value();
106 });
107 if (allConst) {
108 int64_t bytes = baseByteOffset;
109 for (auto [o, s] : llvm::zip(elementOffsets, strides)) {
110 auto oc = getConstantIntValue(o);
111 auto sc = getConstantIntValue(s);
112 assert(oc && sc && "allConst already verified these are constant");
113 bytes += (*oc) * (*sc) * elemWidthBytes;
114 }
115 // Reachable only for a constant offset; the runtime path below stays i32.
116 if (bytes > std::numeric_limits<uint32_t>::max() || bytes < 0)
117 return arith::ConstantOp::create(
118 builder, loc, IntegerAttr::get(builder.getIntegerType(64), bytes));
119 return arith::ConstantOp::create(builder, loc,
120 IntegerAttr::get(i32ty, bytes));
121 }
122
123 // Runtime path: sum(offset[i] * stride[i]) * elemWidthBytes + base, as arith.
124 Value acc =
125 arith::ConstantOp::create(builder, loc, IntegerAttr::get(i32ty, 0));
126 for (auto [o, s] : llvm::zip(elementOffsets, strides)) {
127 Value ov = getAsValue(builder, loc, o, i32ty);
128 Value sv = getAsValue(builder, loc, s, i32ty);
129 Value prod = arith::MulIOp::create(builder, loc, ov, sv);
130 acc = arith::AddIOp::create(builder, loc, acc, prod);
131 }
132 Value width = arith::ConstantOp::create(
133 builder, loc, IntegerAttr::get(i32ty, elemWidthBytes));
134 acc = arith::MulIOp::create(builder, loc, acc, width);
135 if (baseByteOffset != 0) {
136 Value base = arith::ConstantOp::create(
137 builder, loc, IntegerAttr::get(i32ty, baseByteOffset));
138 acc = arith::AddIOp::create(builder, loc, acc, base);
139 }
140 return acc;
141}

143Value buildBdWord(OpBuilder &builder, Location loc,
144 ArrayRef<std::tuple<Value, uint32_t, uint32_t>> fields) {
145 auto i32ty = builder.getIntegerType(32);
146 Value result =
147 arith::ConstantOp::create(builder, loc, IntegerAttr::get(i32ty, 0));
148 for (auto &[val, mask, shift] : fields) {
149 Value masked = val;
150 if (mask != 0xFFFFFFFF) {
151 auto maskConst = arith::ConstantOp::create(
152 builder, loc, IntegerAttr::get(i32ty, (int64_t)(int32_t)mask));
153 masked = arith::AndIOp::create(builder, loc, masked, maskConst);
154 }
155 if (shift > 0) {
156 auto shiftConst = arith::ConstantOp::create(
157 builder, loc, IntegerAttr::get(i32ty, shift));
158 masked = arith::ShLIOp::create(builder, loc, masked, shiftConst);
159 }
160 result = arith::OrIOp::create(builder, loc, result, masked);
161 }
162 return result;
163}

165Value getBdRegisterBase(OpBuilder &builder, Location loc,
166 const AIE::AIETargetModel &targetModel, int tileCol,
167 int tileRow, OpFoldResult bdId) {
168 auto i32ty = builder.getIntegerType(32);
169 if (auto c = getConstantIntValue(bdId))
170 return createConstantI32(builder, loc,
171 static_cast<uint32_t>(targetModel.getDmaBdAddress(
172 tileCol, tileRow, *c)));
173 // Runtime bd_id: base + bd_id * bdStride, with base/stride from the (linear)
174 // target-model address function.
175 uint64_t addrForId0 = targetModel.getDmaBdAddress(tileCol, tileRow, 0);
176 uint64_t bdStride =
177 targetModel.getDmaBdAddress(tileCol, tileRow, 1) - addrForId0;
178 Value bdIdVal = getAsValue(builder, loc, bdId, i32ty);
179 return arith::AddIOp::create(
180 builder, loc, createConstantI32(builder, loc, addrForId0),
181 arith::MulIOp::create(builder, loc, bdIdVal,
182 createConstantI32(builder, loc, bdStride)));
183}

187// Arrays are innermost-first: [d0, d1, d2, iter].
188struct EncodedBd {
189 Value inS[4], inT[4]; // element counts, as given
190 Value hwS[4], hwT[4]; // granule-scaled wraps and -1-biased steps
191 Value bufLen; // buffer_length, in address-gen granules
192 Value iterSizeField; // iteration_size, zeroed for a pure repeat
193 Value repeatCount; // outer-dim repeat for the caller's queue push
194 bool isLinear = false; // d0/d1/d2 folded into buffer_length
195};

197LogicalResult encodeBdCommon(OpBuilder &builder, Location loc,
198 const AIE::AIETargetModel &tm, int tileCol,
199 int tileRow, ArrayRef<OpFoldResult> mixedSizes,
200 ArrayRef<OpFoldResult> mixedStrides,
201 uint64_t elemWidth, Value bufLenOverride,
202 EncodedBd &out) {
203 auto i32ty = builder.getIntegerType(32);
204 uint32_t gran = tm.getAddressGenGranularity();
205 SmallVector<OpFoldResult, 4> sizesRev(llvm::reverse(mixedSizes));
206 SmallVector<OpFoldResult, 4> stridesRev(llvm::reverse(mixedStrides));
207 // The encoding below is i32 arithmetic, and a RUNTIME operand that does not
208 // fit it wraps: an i64 is truncated (2^32 + 1 becomes 1), and d0's size and
209 // every stride are multiplied by elemWidth before they are scaled down to
210 // granules. A wrapped value can then pass the field guards further down, so
211 // bound each operand at its original width first. Only the d1..d3 sizes are
212 // used unscaled, and an i32 one needs no bound. Every BD field is far
213 // narrower than these bounds, so no valid value is rejected here.
214 int64_t scaledMax =
215 std::numeric_limits<int32_t>::max() / std::max<uint64_t>(elemWidth, 1);
216 auto guardOperand = [&](OpFoldResult in, bool scaled) {
217 if (getConstantIntValue(in))
218 return;
219 Value v = cast<Value>(in);
220 if (!scaled && v.getType().getIntOrFloatBitWidth() <= 32)
221 return;
222 NpuAssertBdFieldOp::create(
223 builder, loc, v,
224 builder.getI32IntegerAttr(
225 scaled ? scaledMax : std::numeric_limits<int32_t>::max()));
226 };
227 for (int i = 0; i < 4; i++) {
228 guardOperand(sizesRev[i], /*scaled=*/i == 0);
229 guardOperand(stridesRev[i], /*scaled=*/true);
230 }
231 for (int i = 0; i < 4; i++) {
232 out.inS[i] = getAsValue(builder, loc, sizesRev[i], i32ty);
233 out.inT[i] = getAsValue(builder, loc, stridesRev[i], i32ty);
234 }
235 SsaStridePolicy policy(builder, loc);
236 encodeHardwareStridesWraps(policy, elemWidth, gran, out.inS, out.inT, out.hwS,
237 out.hwT);
238
239 // buffer_length: the caller's runtime len if supplied (dma_task), else the
240 // d0*d1*d2 hardware-unit size-product (dma_memcpy_nd). hwS[0] already carries
241 // the elemWidth/gran scaling; d1/d2 are element counts.
242 uint64_t lenMax = tm.getDmaBdMaxLen(tileCol, tileRow);
243 bool lenGuarded = false;
244 out.bufLen = bufLenOverride;
245 if (!out.bufLen) {
246 auto cst0 = [&](int i) { return getConstantIntValue(sizesRev[i]); };
247 bool unitOuter = [&] {
248 auto s1 = cst0(1), s2 = cst0(2);
249 return s1 && *s1 == 1 && s2 && *s2 == 1;
250 }();
251 if ((cst0(0) && cst0(1) && cst0(2)) || unitOuter) {
252 out.bufLen = arith::MulIOp::create(
253 builder, loc,
254 arith::MulIOp::create(builder, loc, out.hwS[0], out.inS[1]),
255 out.inS[2]);
256 } else {
257 // A runtime factor can make the i32 product wrap into a length that
258 // passes its guard, so multiply in i64 and bound each partial product.
259 // hwS[0] is below 2^31 once its operand guard holds, each d1/d2 size
260 // below 2^32 unsigned, and a partial product that passed its guard below
261 // 2^31, so no product wraps in i64 before the guard that rejects it.
262 auto i64ty = builder.getI64Type();
263 auto wide = [&](Value v) -> Value {
264 return arith::ExtUIOp::create(builder, loc, i64ty, v);
265 };
266 auto cap = (int64_t)std::min<uint64_t>(
267 lenMax, std::numeric_limits<int32_t>::max());
268 Value len = wide(out.hwS[0]);
269 for (int i = 1; i < 3; i++) {
270 if (auto c = cst0(i); c && *c == 1)
271 continue;
272 len = arith::MulIOp::create(builder, loc, len, wide(out.inS[i]));
273 NpuAssertBdFieldOp::create(builder, loc, len,
274 builder.getI32IntegerAttr(cap));
275 }
276 out.bufLen = arith::TruncIOp::create(builder, loc, i32ty, len);
277 lenGuarded = true;
278 }
279 }

281 auto cst = [&](OpFoldResult v) { return getConstantIntValue(v); };
282 auto constEq = [&](OpFoldResult v, int64_t c) {
283 auto k = cst(v);
284 return k && *k == c;
285 };
286 // A transfer with no data layout transformation at all (mirrors
287 // AIEX::isLinearTransfer). Canonicalization zeroes size-1 strides before this
288 // runs, hence the stride == 0 tests.
289 auto knownLinear = [&]() {
290 return constEq(sizesRev[1], 1) && constEq(sizesRev[2], 1) &&
291 constEq(stridesRev[0], 1) && constEq(stridesRev[1], 0) &&
292 constEq(stridesRev[2], 0);
293 };
294 // A contiguous row-major scan: each outer stride is the product of the inner
295 // sizes (mirrors AIEX::isContiguousTransfer). A runtime operand needed for
296 // the test falls back to ND mode.
297 auto knownContiguous = [&]() -> bool {
298 auto s0 = cst(stridesRev[0]);
299 if (!s0 || *s0 != 1)
300 return false;
301 auto sz0 = cst(sizesRev[0]);
302 auto d1sz = cst(sizesRev[1]);
303 auto d1st = cst(stridesRev[1]);
304 if (!d1sz || *d1sz != 1)
305 if (!sz0 || !d1st || *d1st != *sz0)
306 return false;
307 auto d2sz = cst(sizesRev[2]);
308 auto d2st = cst(stridesRev[2]);
309 if (!d2sz || *d2sz != 1) {
310 auto prod01 =
311 (sz0 && d1sz) ? std::optional<int64_t>(*sz0 * *d1sz) : std::nullopt;
312 if (!prod01 || !d2st || *d2st != *prod01)
313 return false;
314 }
315 return true;
316 };
317 // Folding d0/d1/d2 into buffer_length dodges the 10-bit d0 wrap, but only a
318 // shim NOC tile's 32-bit length field can absorb a merely-contiguous scan
319 // (17 bits on a mem tile, 14 on a core tile). The static path draws the line
320 // in the same place, in AIEDMATasksToNPU.cpp's treatAsLinear.
321 out.isLinear = knownLinear() ||
322 (tm.isShimNOCTile(tileCol, tileRow) && knownContiguous());
323
324 // Guard a RUNTIME value against a BD field too narrow to hold it; the
325 // packer's mask would otherwise truncate it silently on hardware. Constants
326 // are verifier-checked instead.
327 auto guardField = [&](OpFoldResult in, Value hwVal, int64_t fieldMax) {
328 if (getConstantIntValue(in))
329 return;
330 NpuAssertBdFieldOp::create(builder, loc, hwVal,
331 builder.getI32IntegerAttr(fieldMax));
332 };
333 int64_t wrapMax = (1ll << tm.getDmaBdWrapBits(tileCol, tileRow)) - 1;
334 int64_t iterMax = (1ll << tm.getDmaBdIterBits(tileCol, tileRow)) - 1;
335 int64_t stepMax = (1ll << tm.getDmaBdStepBits(tileCol, tileRow)) - 1;
336 if (!out.isLinear) {
337 guardField(sizesRev[0], out.hwS[0], wrapMax);
338 guardField(sizesRev[1], out.hwS[1], wrapMax);
339 // The step fields are narrower than the values that reach them (17 bits on
340 // a mem tile, 13 on a core tile), so a runtime stride needs the same guard
341 // a constant gets from verifyStridesWraps.
342 guardField(stridesRev[0], out.hwT[0], stepMax);
343 guardField(stridesRev[1], out.hwT[1], stepMax);
344 guardField(stridesRev[2], out.hwT[2], stepMax);
345 }
346 guardField(sizesRev[3], out.hwS[3], iterMax);
347 // hwT[3] collapses to 0 unless the iteration wrap is > 1, which is the same
348 // condition under which verifyStridesWraps exempts it -- so this guard is
349 // inert in the pure-repeat case rather than rejecting it.
350 guardField(stridesRev[3], out.hwT[3], stepMax);
351
352 // Neither check fires on a shim NOC tile, whose buffer_length owns the whole
353 // 32-bit word, nor on a size-product the i64 path above already bounded.
354 if (!lenGuarded && lenMax < std::numeric_limits<uint32_t>::max()) {
355 if (auto constLen = getConstantIntValue(out.bufLen)) {
356 if (*constLen < 0 || (uint64_t)*constLen > lenMax)
357 return emitError(loc)
358 << "buffer length of " << *constLen
359 << " address-generation granules exceeds the " << lenMax
360 << " this tile's BD buffer_length field can hold.";
361 } else {
362 NpuAssertBdFieldOp::create(builder, loc, out.bufLen,
363 builder.getI32IntegerAttr((int64_t)lenMax));
364 }
365 }
366
367 // Guard a RUNTIME size/stride whose byte extent must be a whole number of
368 // granules (mirrors verifyStridesWraps). Guard is on the input element count;
369 // constants are verifier-checked.
370 int64_t divisor = bdGranuleDivisor(elemWidth, gran);
371 auto guardDivisible = [&](OpFoldResult in, Value inVal, bool allowUnit) {
372 if (divisor <= 1 || getConstantIntValue(in))
373 return;
374 NpuAssertBdDivisibleOp::create(builder, loc, inVal, (uint32_t)divisor,
375 /*allow_unit=*/allowUnit);
376 };
377 guardDivisible(sizesRev[0], out.inS[0], /*allowUnit=*/false);
378 // The innermost stride collapses to hardware 0 when contiguous or
379 // granule-aligned; guard with the unit-stride exemption. Outer strides don't
380 // collapse.
381 guardDivisible(stridesRev[0], out.inT[0], /*allowUnit=*/true);
382 for (int i = 1; i < 4; i++)
383 guardDivisible(stridesRev[i], out.inT[i], /*allowUnit=*/false);
384
385 // iteration_size. A zero outer stride is a pure repeat (carried by
386 // repeat_count), so the field must be 0 like AIEDmaToNpu; gate it on the
387 // stride while leaving hwS[3] for repeatCount. hwT[3] already collapses to 0.
388 Value zeroI32 = createConstantI32(builder, loc, 0);
389 Value iterStridePos = arith::CmpIOp::create(
390 builder, loc, arith::CmpIPredicate::sgt, out.inT[3], zeroI32);
391 out.iterSizeField =
392 arith::SelectOp::create(builder, loc, iterStridePos, out.hwS[3], zeroI32);
393
394 // repeat_count for the queue push is the biased outer size (N > 1 ? N - 1 :
395 // 0), matching the static path. A constant folds so the static push_queue
396 // lowering can consume it; a runtime size yields the SSA hwS[3].
397 if (auto outerConst = getConstantIntValue(sizesRev[3])) {
398 int64_t r = *outerConst > 1 ? *outerConst - 1 : 0;
399 out.repeatCount =
400 arith::ConstantOp::create(builder, loc, IntegerAttr::get(i32ty, r));
401 } else {
402 out.repeatCount = out.hwS[3];
403 }
404 return success();
405}

407// dynamic-matches-static-words.mlir holds this to WriteBdToBlockWritePattern.
408void packBdWords(OpBuilder &builder, Location loc,
409 const AIE::AIETargetModel &tm, const AIE::DmaBdLayout &layout,
410 const BdTemplateFields &f, const EncodedBd &e,
411 uint32_t burstLength, uint32_t axcache,
412 SmallVectorImpl<Value> &wordsOut) {
413 using Field = std::tuple<Value, uint32_t, uint32_t>;
414 SmallVector<uint32_t, 8> constWords(layout.numWords, 0);
415 SmallVector<SmallVector<Field, 3>, 8> runtimeWords(layout.numWords);
416 auto set = [&](const AIE::DmaBdField &field, uint64_t value) {
417 if (field.exists())
418 constWords[field.word] |= field.place(value);
419 };
420 auto add = [&](const AIE::DmaBdField &field, Value value) {
421 if (field.exists())
422 runtimeWords[field.word].push_back({value, field.mask(), field.shift});
423 };
424
425 set(layout.enablePacket, f.enable_packet);
426 set(layout.packetType, f.packet_type);
427 set(layout.packetId, f.packet_id);
428 set(layout.outOfOrderId, f.out_of_order_id);
429 if (layout.burstLength.exists())
430 set(layout.burstLength, AIE::getShimBurstLengthEncoding(tm, burstLength));
431 set(layout.axcache, axcache);
432 set(layout.nextBd, f.next_bd_id);
433 set(layout.useNextBd, f.use_next_bd);
434 set(layout.validBd, 1);
435 set(layout.lockRelValue, f.lock_rel_val);
436 set(layout.lockRelId, f.lock_rel_id);
437 set(layout.lockAcqEnable, f.lock_acq_enable);
438 set(layout.lockAcqValue, f.lock_acq_val);
439 set(layout.lockAcqId, f.lock_acq_id);
440
441 add(layout.bufferLength, e.bufLen);
442 if (!e.isLinear) {
443 add(layout.d0Size, e.hwS[0]);
444 add(layout.d0Stride, e.hwT[0]);
445 add(layout.d1Size, e.hwS[1]);
446 add(layout.d1Stride, e.hwT[1]);
447 add(layout.d2Stride, e.hwT[2]);
448 }
449 add(layout.iterationSize, e.iterSizeField);
450 add(layout.iterationStride, e.hwT[3]);
451
452 wordsOut.clear();
453 for (auto [constWord, fields] : llvm::zip(constWords, runtimeWords)) {
454 if (fields.empty()) {
455 wordsOut.push_back(createConstantI32(builder, loc, constWord));
456 continue;
457 }
458 auto [value, mask, shift] = fields.front();
459 if (constWord == 0 && fields.size() == 1 && mask == 0xFFFFFFFF &&
460 shift == 0) {
461 wordsOut.push_back(value);
462 continue;
463 }
464 Value word = buildBdWord(builder, loc, fields);
465 if (constWord != 0)
466 word = arith::OrIOp::create(
467 builder, loc, createConstantI32(builder, loc, constWord), word);
468 wordsOut.push_back(word);
469 }
470}

474LogicalResult
475buildBdWords(OpBuilder &builder, Location loc,
476 const AIE::AIETargetModel &targetModel, int tileCol, int tileRow,
477 const BdTemplateFields &f, ArrayRef<OpFoldResult> mixedSizes,
478 ArrayRef<OpFoldResult> mixedStrides, uint64_t elemWidth,
479 uint32_t burstLength, uint32_t axcache, Value bufLenOverride,
480 Value &repeatCountOut, SmallVectorImpl<Value> &wordsOut) {
481 EncodedBd e;
482 if (failed(encodeBdCommon(builder, loc, targetModel, tileCol, tileRow,
483 mixedSizes, mixedStrides, elemWidth, bufLenOverride,
484 e)))
485 return failure();
486 repeatCountOut = e.repeatCount;
487
488 const AIE::DmaBdLayout *layout = targetModel.getDmaBdLayout(tileCol, tileRow);
489 assert(layout && "buildBdWords called for a tile with no DMA BD layout "
490 "(rejected by the caller)");
491 packBdWords(builder, loc, targetModel, *layout, f, e, burstLength, axcache,
492 wordsOut);
493 return success();
494}

27// One way to multiply an m x k lhs tile by a k x n rhs tile into an m x n
28// accumulator. Names give the operand types and m x k x n.
29enum class MatMulRecipeKind {
30 // AIE2: one `mac.conf` each.
31 AIE2_I8xI4_4x16x8,
32 AIE2_I8xI8_4x8x8,
33 AIE2_I16xI8_4x4x8,
34 AIE2_I16xI16_4x2x8,
35 AIE2_I16xI8_2x8x8_Acc64,
36 AIE2_I16xI8_4x8x4_Acc64,
37 AIE2_I16xI16_2x4x8_Acc64,
38 AIE2_I16xI16_4x4x4_Acc64,
39 AIE2_I32xI16_4x2x4_Acc64,
40 AIE2_BF16_4x8x4,
41 // AIE2P integer matrix modes.
42 AIE2P_I8_8x8x8,
43 AIE2P_I16_8x2x8,
44 // AIE2P bf16 through bfp16 (block floating point, one exponent per 8 values
45 // along k), the `8x8_8x8T` matrix mode.
46 AIE2P_BFP16_8x8x8,
47 AIE2P_BFP16_4x8x8,
48 // AIE2P bf16 through elementwise multiplies of broadcast rows and columns.
49 AIE2P_BF16Elem_8x8x4,
50 AIE2P_BF16Elem_4x8x4,
51 AIE2P_BF16Outer_8x1x8,
52};
53
54struct MatMulRecipe {
55 MatMulRecipeKind kind;
56 // The matmul ops take lhs as [m, k] and rhs as [k, n]. These say whether
57 // the recipe reads lhs as [k, m] and rhs as [n, k] instead, so that an
58 // operand stored that way needs no shuffle.
59 bool lhsKM;
60 bool rhsNK;
61};
62
63// The recipe for an m x k x n tile with these element types, if there is one.
64// Integer types are matched by width; their signedness is a separate MAC
65// setting.
66std::optional<MatMulRecipe> getMatMulRecipe(bool aie2p, mlir::Type lhsElemTy,
67 mlir::Type rhsElemTy,
68 mlir::Type accElemTy, int64_t m,
69 int64_t k, int64_t n);

71// The shuffle modes that transpose a rows x cols tile of `elemTy` held
72// row-major. Empty when a dim is 1, as the tile then needs no data movement.
73// One mode when the tile fills 512 bits, or 256 bits on AIE2P (the tile then
74// sits in the low half of a 512-bit register). Two modes, the low and high
75// halves of the result, for an 8x8 tile of 16-bit values on AIE2P. Nullopt
76// for any other tile.
77std::optional<llvm::SmallVector<ShuffleMode, 2>>
78getTileTransposeModes(bool aie2p, mlir::Type elemTy, int64_t rows,
79 int64_t cols);
80
81// Checks that lhs times rhs into acc is a tile some recipe multiplies. lhs is
82// stored [m, k], or [k, m] when `lhsTransposed`; rhs is stored [k, n], or
83// [n, k] when `rhsTransposed`; acc is [m, n]. Each stored layout must be the
84// one the recipe reads, or one that getTileTransposeModes turns into it.
85// Returns the recipe. On failure, reports why through `emitError` unless it
86// is null.
87std::optional<MatMulRecipe> checkMatMulTile(
88 bool aie2p, mlir::VectorType lhsTy, mlir::VectorType rhsTy,
89 mlir::VectorType accTy, bool lhsTransposed, bool rhsTransposed,
90 llvm::function_ref<mlir::InFlightDiagnostic()> emitError = nullptr);
91
92} // namespace xilinx::aievec

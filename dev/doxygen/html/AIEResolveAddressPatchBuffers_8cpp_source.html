29struct ResolveBufferOperand : OpRewritePattern<NpuAddressPatchOp> {
30 using OpRewritePattern::OpRewritePattern;
31
32 LogicalResult matchAndRewrite(NpuAddressPatchOp op,
33 PatternRewriter &rewriter) const override {
34 if (!op.getBuffer()) {
35 return failure();
36 }
37
38 auto seq = op->getParentOfType<AIE::RuntimeSequenceOp>();
39 if (!seq) {
40 return rewriter.notifyMatchFailure(
41 op, "must be inside a runtime sequence to resolve its host buffer");
42 }
43 std::optional<SubviewTraceResult> traced =
44 traceSubviewToBlockArgument(op.getBuffer());
45 if (!traced || traced->rootArg.getOwner() != &seq.getBody().front()) {
46 return rewriter.notifyMatchFailure(
47 op, "buffer must be an argument of the enclosing runtime sequence, "
48 "or a static contiguous view of one");
49 }
50 std::optional<unsigned> argIdx = getHostBufferArgIndex(traced->rootArg);
51 if (!argIdx) {
52 return rewriter.notifyMatchFailure(
53 op, "buffer must resolve to a memref argument");
54 }
55
56 Value argPlus = op.getArgPlus();
57 if (traced->offsetInBytes != 0) {
58 std::optional<uint32_t> base = getConstantIntOperand(argPlus);
59 if (base) {
60 argPlus = createConstantI32(rewriter, op.getLoc(),
61 *base + traced->offsetInBytes);
62 } else {
63 Value shift =
64 createConstantI32(rewriter, op.getLoc(), traced->offsetInBytes);
65 argPlus = arith::AddIOp::create(rewriter, op.getLoc(), argPlus, shift);
66 }
67 }
68
69 rewriter.replaceOpWithNewOp<NpuAddressPatchOp>(
70 op, op.getAddr(), op.getAddrVal(), static_cast<int32_t>(*argIdx),
71 argPlus);
72 return success();
73 }
74};

76struct AIEResolveAddressPatchBuffersPass
77 : xilinx::AIEX::impl::AIEResolveAddressPatchBuffersBase<
78 AIEResolveAddressPatchBuffersPass> {
79
80 void runOnOperation() override {
81 RewritePatternSet patterns(&getContext());
82 patterns.add<ResolveBufferOperand>(&getContext());
83 walkAndApplyPatterns(getOperation(), std::move(patterns));
84
85 // A pattern that declines to match leaves the operand in place. The targets
86 // cannot emit that, so report it here instead.
87 WalkResult unresolved = getOperation().walk([&](NpuAddressPatchOp op) {
88 if (op.getBuffer()) {
89 op.emitOpError("could not resolve its buffer operand to a host buffer "
90 "index of the enclosing runtime sequence");
91 return WalkResult::interrupt();
92 }
93 return WalkResult::advance();
94 });
95 if (unresolved.wasInterrupted()) {
96 signalPassFailure();
97 }
98 }
99};

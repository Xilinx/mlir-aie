41void xilinx::AIE::addObjectFifoLoweringPasses(OpPassManager &pm,
42 bool packetSwitched,
43 bool skipVerify,
44 bool reserveControllerPacketIds,
45 bool checkQueueSpace) {
46 pm.addPass(createAIEObjectFifoSplitPass());
47 if (!skipVerify) {
48 pm.addPass(createAIEObjectFifoVerifyPass());
49 }
50 pm.addPass(
51 createAIEAssignPacketIdsPass(packetSwitched, reserveControllerPacketIds));
52 pm.addPass(createAIEObjectFifoAllocatePass());
53 AIEObjectFifoInsertCoreQueuePushesOptions pushOptions;
54 pushOptions.checkQueueSpace = checkQueueSpace;
55 pm.addPass(createAIEObjectFifoInsertCoreQueuePushesPass(pushOptions));
56 pm.addPass(createAIEObjectFifoLowerDMAsPass());
57 pm.addPass(createAIEObjectFifoLowerCoresPass());
58 pm.addPass(createAIEObjectFifoErasePoolsPass());
59}

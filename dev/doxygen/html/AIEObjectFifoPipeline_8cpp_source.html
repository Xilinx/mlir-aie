32void xilinx::AIE::addObjectFifoLoweringPasses(OpPassManager &pm,
33 bool packetSwitched,
34 bool skipVerify) {
35 pm.addPass(createAIEObjectFifoSplitPass());
36 if (!skipVerify) {
37 pm.addPass(createAIEObjectFifoVerifyPass());
38 }
39 pm.addPass(createAIEAssignPacketIdsPass(packetSwitched));
40 pm.addPass(createAIEObjectFifoAllocatePass());
41 pm.addPass(createAIEObjectFifoLowerDMAsPass());
42 pm.addPass(createAIEObjectFifoLowerCoresPass());
43 pm.addPass(createAIEObjectFifoErasePoolsPass());
44}

44std::optional<DmaChannelProgram> makeProgram(Operation *op, DeviceOp device) {
45 auto parentTile = [](Operation *op) -> std::optional<TileID> {
46 if (auto element = op->getParentOfType<TileElement>())
47 return tileOf(element.getTile());
48 return std::nullopt;
49 };
50 if (auto start = dyn_cast<DMAStartOp>(op)) {
51 std::optional<TileID> tile = parentTile(op);
52 if (!tile)
53 return std::nullopt;
54 DmaChannelProgram p{
55 op, {*tile, start.getChannelDir(), start.getChannelIndex()}};
56 p.passes += start.getRepeatCount();
57 llvm::SmallPtrSet<Block *, 8> seen;
58 for (Block *b = start.getDest(); b;) {
59 if (!seen.insert(b).second) {
60 p.loops = true;
61 p.loopStart = llvm::find(p.bds, b) - p.bds.begin();
62 break;
63 }
64 p.bds.push_back(b);
65 auto next = dyn_cast<NextBDOp>(b->getTerminator());
66 b = next ? next.getDest() : nullptr;
67 }
68 return p;
69 }
70 if (auto dma = dyn_cast<DMAOp>(op)) {
71 std::optional<TileID> tile = parentTile(op);
72 if (!tile)
73 return std::nullopt;
74 DmaChannelProgram p{op,
75 {*tile, dma.getChannelDir(), dma.getChannelIndex()}};
76 p.passes += dma.getRepeatCount();
77 for (Region &r : dma.getBds())
78 p.regions.push_back(&r);
79 p.loops = dma.getLoop();
80 return p;
81 }
82 if (auto task = dyn_cast<AIEX::DMAConfigureTaskOp>(op)) {
83 std::optional<TileID> tile = tileOf(task.getTile());
84 if (!tile)
85 return std::nullopt;
86 DmaChannelProgram p{
87 op, {*tile, task.getDirection(), static_cast<int>(task.getChannel())}};
88 p.regions.push_back(&task.getBody());
89 p.passes += task.getRepeatCount();
90 return p;
91 }
92 if (auto task = dyn_cast<AIEX::DMAConfigureTaskForOp>(op)) {
93 auto alloc = ShimDMAAllocationOp::getForSymbol(
94 device, task.getAlloc().getRootReference());
95 if (!alloc)
96 return std::nullopt;
97 std::optional<TileID> tile = tileOf(alloc.getTile());
98 if (!tile)
99 return std::nullopt;
100 DmaChannelProgram p{op,
101 {*tile, alloc.getChannelDir(),
102 static_cast<int>(alloc.getChannelIndex())}};
103 p.regions.push_back(&task.getBody());
104 p.passes += task.getRepeatCount();
105 return p;
106 }
107 return std::nullopt;
108}

110// Every DMA channel program: aie.dma_start and aie.dma inside the tile DMA

112SmallVector<DmaChannelProgram> collectDmaPrograms(DeviceOp device) {
113 SmallVector<DmaChannelProgram> programs;
114 device.walk([&](Operation *op) {
115 if (isa<DMAStartOp, DMAOp, AIEX::DMAConfigureTaskOp,
116 AIEX::DMAConfigureTaskForOp>(op))
117 if (std::optional<DmaChannelProgram> p = makeProgram(op, device))
118 programs.push_back(std::move(*p));
119 });
120 return programs;

123template <typename OpT>
124void forEachInProgram(const DmaChannelProgram &p,
125 llvm::function_ref<void(OpT)> fn) {
126 for (Block *b : p.bds)
127 for (auto op : b->getOps<OpT>())
128 fn(op);
129 for (Region *r : p.regions)
130 r->walk([&](OpT op) { fn(op); });

145// The ops that can create `task`, following it back through the results and
146// block arguments of runtime control flow (scf.for iter_args, scf.if results).
147// Empty when some value it can come from has no creator here.
148SmallVector<Operation *> taskCreators(Value task) {
149 SmallVector<Operation *> creators;
150 llvm::SmallPtrSet<Value, 8> seen;
151 SmallVector<Value> worklist{task};
152 while (!worklist.empty()) {
153 Value v = worklist.pop_back_val();
154 if (!seen.insert(v).second)
155 continue;
156 Operation *op = v.getDefiningOp();
157 if (isa_and_nonnull<AIEX::DMAConfigureTaskOp, AIEX::DMAConfigureTaskForOp,
158 AIEX::DMAStartBdChainOp, AIEX::DMAStartBdChainForOp>(
159 op)) {
160 creators.push_back(op);
161 continue;
162 }
163 Operation *owner =
164 op ? op : cast<BlockArgument>(v).getOwner()->getParentOp();
165 auto branch = dyn_cast_or_null<RegionBranchOpInterface>(owner);
166 if (!branch)
167 return {};
168 RegionBranchInverseSuccessorMapping mapping;
169 branch.getSuccessorInputOperandMapping(mapping);
170 auto it = mapping.find(v);
171 if (it == mapping.end())
172 return {};
173 for (OpOperand *operand : it->second)
174 worklist.push_back(operand->get());
175 }
176 return creators;
177}

179// The channels `task` can run on; empty when that is not known here.
180SmallVector<TileDMAChannel> channelsOfTask(DeviceOp device, Value task) {
181 SmallVector<TileDMAChannel> keys;
182 for (Operation *op : taskCreators(task)) {
183 std::optional<TileDMAChannel> key;
184 auto onTile = [&](Value tileValue, DMAChannelDir dir, int channel) {
185 if (std::optional<TileID> tile = tileOf(tileValue))
186 key = TileDMAChannel{*tile, dir, channel};
187 };
188 if (auto configure = dyn_cast<AIEX::DMAConfigureTaskOp>(op))
189 onTile(configure.getTile(), configure.getDirection(),
190 configure.getChannel());
191 else if (auto chain = dyn_cast<AIEX::DMAStartBdChainOp>(op))
192 onTile(chain.getTile(), chain.getDirection(), chain.getChannel());
193 else if (auto configure = dyn_cast<AIEX::DMAConfigureTaskForOp>(op))
194 key = channelOfSymbol(device, configure.getAlloc().getRootReference());
195 else if (auto chain = dyn_cast<AIEX::DMAStartBdChainForOp>(op))
196 key = channelOfSymbol(device, chain.getAlloc());
197 if (!key)
198 return {};
199 if (!llvm::is_contained(keys, *key))
200 keys.push_back(*key);
201 }
202 return keys;
203}

205// Visits the ops in `region` in program order, going through the body of
206// each loop twice, so that what one iteration does follows all the previous
207// iteration did.
208void walkLoopsTwice(Region &region, function_ref<void(Operation *)> fn) {
209 for (Operation &op : region.getOps()) {
210 fn(&op);
211 auto branch = dyn_cast<RegionBranchOpInterface>(op);
212 int passes = branch && branch.hasLoop() ? 2 : 1;
213 for (int pass = 0; pass < passes; pass++)
214 for (Region &r : op.getRegions())
215 walkLoopsTwice(r, fn);
216 }
217}

219// Packet ids each sending channel is programmed with.
220std::map<TileDMAChannel, std::set<int>> collectSentPacketIDs(DeviceOp device) {
221 std::map<TileDMAChannel, std::set<int>> ids;
222 for (const DmaChannelProgram &p : collectDmaPrograms(device)) {
223 if (p.dma.dir != DMAChannelDir::MM2S)
224 continue;
225 forEachInProgram<DMABDOp>(p, [&](DMABDOp bd) {
226 if (std::optional<PacketInfoAttr> packet = bd.getPacket())
227 ids[p.dma].insert(packet->getPktId());
228 });
229 }
230 for (auto alloc : device.getOps<ShimDMAAllocationOp>()) {
231 std::optional<PacketInfoAttr> packet = alloc.getPacket();
232 std::optional<TileID> tile = tileOf(alloc.getTile());
233 if (packet && tile)
234 ids[{*tile, alloc.getChannelDir(),
235 static_cast<int>(alloc.getChannelIndex())}]
236 .insert(packet->getPktId());
237 }
238 device.walk([&](AIEX::NpuDmaMemcpyNdOp memcpy) {
239 std::optional<PacketInfoAttr> packet = memcpy.getPacket();
240 if (!packet)
241 return;
242 if (std::optional<TileDMAChannel> key =
243 channelOfSymbol(device, memcpy.getMetadata().getRootReference()))
244 ids[*key].insert(packet->getPktId());
245 });
246 return ids;
247}

251 explicit StreamTracer(DeviceOp device)
252 : maxPacketID(getTargetModel(device).getMaxPacketId()) {
253 for (auto sb : device.getOps<SwitchboxOp>())
254 switchboxes[sb.getTileOp().getTileID()] = sb;

260 std::vector<RoutedStream> trace() {
261 for (auto &[tile, mux] : shimMuxes)
262 for (Port p : inputPorts(mux.getConnections()))
263 if (p.bundle != WireBundle::North)
264 traceFrom({tile, p}, mux, p);
265 for (auto &[tile, sb] : switchboxes)
266 for (Port p : inputPorts(sb.getConnections()))
267 if (!isDirectional(p.bundle))
268 traceFrom({tile, p}, sb, p);
269 return std::move(streams);
270 }

278 static SmallVector<Port> inputPorts(Region &connections) {
279 SmallVector<Port> ports;
280 for (Operation &op : connections.front()) {
281 std::optional<Port> p;
282 if (auto connect = dyn_cast<ConnectOp>(op))
283 p = connect.sourcePort();
284 else if (auto rules = dyn_cast<PacketRulesOp>(op))
285 p = rules.sourcePort();
286 if (p && !llvm::is_contained(ports, *p))
287 ports.push_back(*p);
288 }

292 void traceFrom(StreamEndpoint src, Interconnect interconnect, Port input) {
293 std::optional<std::set<int>> ids;
294 // A tile sends from its MM2S channel on the port of the same number.
295 if (src.port.bundle == WireBundle::DMA) {
296 auto it = sentIDs.find({src.tile, DMAChannelDir::MM2S, src.port.channel});
297 if (it != sentIDs.end())
298 ids = it->second;
299 }
300 if (!ids) {
301 step(src, {interconnect, input}, std::nullopt);
302 return;
303 }
304 for (int id : *ids)
305 step(src, {interconnect, input}, id);
306 }

308 // The switchbox input a master port of `from` drives, or the tile port it
309 // leaves the fabric at.
310 std::variant<Hop, StreamEndpoint> follow(Interconnect from, Port out) {
311 TileID here{from.colIndex(), from.rowIndex()};
312 if (isa<ShimMuxOp>(from)) {
313 if (out.bundle == WireBundle::North)
314 if (auto sb = switchboxes.find(here); sb != switchboxes.end())
315 return Hop{sb->second, {WireBundle::South, out.channel}};
316 return StreamEndpoint{here, out};
317 }
318 if (!isDirectional(out.bundle))
319 return StreamEndpoint{here, out};
320 if (out.bundle == WireBundle::South)
321 if (auto mux = shimMuxes.find(here); mux != shimMuxes.end())
322 return Hop{mux->second, {WireBundle::North, out.channel}};
323 TileID next = here;
324 WireBundle in = WireBundle::South;
325 switch (out.bundle) {
326 case WireBundle::North:
327 next.row++;
328 in = WireBundle::South;
329 break;
330 case WireBundle::South:
331 next.row--;
332 in = WireBundle::North;
333 break;
334 case WireBundle::East:
335 next.col++;
336 in = WireBundle::West;
337 break;
338 default:
339 next.col--;
340 in = WireBundle::East;
341 break;
342 }
343 if (auto sb = switchboxes.find(next); sb != switchboxes.end())
344 return Hop{sb->second, {in, out.channel}};
345 return StreamEndpoint{here, out};
346 }

348 // Follows the stream `src` sends with `id` on from `hop`, along every
349 // branch that does not come back to an input it already took.
350 void step(StreamEndpoint src, Hop hop, std::optional<int> id) {
351 std::pair<Operation *, Port> input{hop.interconnect, hop.input};
352 if (!onPath.insert(input).second)
353 return;
354 llvm::scope_exit leave([&] { onPath.erase(input); });
355 auto sb = dyn_cast<SwitchboxOp>(hop.interconnect.getOperation());
356 Block &b = hop.interconnect.getConnections().front();
357
358 auto next = [&](Port out, std::optional<int> nextID,
359 std::optional<int> arbiter, bool keepsPktHeader = false) {
360 if (sb)
361 path.push_back({sb.getTileOp().getTileID(), hop.input, arbiter});
362 llvm::scope_exit backtrack([&] {
363 if (sb)
364 path.pop_back();
365 });
366 std::variant<Hop, StreamEndpoint> to = follow(hop.interconnect, out);
367 if (auto *endpoint = std::get_if<StreamEndpoint>(&to))
368 streams.push_back({src, *endpoint, nextID, keepsPktHeader, path});
369 else
370 step(src, std::get<Hop>(to), nextID);
371 };
372
373 for (auto connect : b.getOps<ConnectOp>())
374 if (connect.sourcePort() == hop.input)
375 next(connect.destPort(), id, std::nullopt);
376
377 for (auto rules : b.getOps<PacketRulesOp>()) {
378 if (rules.sourcePort() != hop.input)
379 continue;
380 auto route = [&](PacketRuleOp rule, int ruleID) {
381 std::optional<int> arbiter;
382 if (auto amsel = rule.getAmsel().getDefiningOp<AMSelOp>())
383 arbiter = amsel.arbiterIndex();
384 for (auto masterSet : b.getOps<MasterSetOp>())
385 if (llvm::is_contained(masterSet.getAmsels(), rule.getAmsel()))
386 next(masterSet.destPort(), ruleID, arbiter,
387 masterSet.keepsPktHeader());
388 };
389 for (int packetID = id.value_or(0); packetID <= id.value_or(maxPacketID);
390 ++packetID)
391 for (auto rule : rules.getRules().front().getOps<PacketRuleOp>())
392 if ((packetID & rule.maskInt()) ==
393 (rule.valueInt() & rule.maskInt())) {
394 route(rule, packetID);
395 break;
396 }
397 }
398 }
399
400 std::map<TileID, SwitchboxOp> switchboxes;
401 std::map<TileID, ShimMuxOp> shimMuxes;
402 std::map<TileDMAChannel, std::set<int>> sentIDs;
403 std::vector<RoutedStream> streams;
404 // The interconnect inputs the stream `step` follows has taken, and the
405 // switchboxes it has passed.
406 llvm::DenseSet<std::pair<Operation *, Port>> onPath;
407 SmallVector<StreamHop, 8> path;
408 int maxPacketID;
409};

417std::vector<RoutedStream> AIE::requestedStreams(DeviceOp device) {
418 std::vector<RoutedStream> streams;
419 for (auto flow : device.getOps<FlowOp>()) {
420 std::optional<TileID> srcTile = tileOf(flow.getSource());
421 std::optional<TileID> dstTile = tileOf(flow.getDest());
422 if (srcTile && dstTile)
423 streams.push_back(
424 {{*srcTile, {flow.getSourceBundle(), flow.sourceIndex()}},
425 {*dstTile, {flow.getDestBundle(), flow.destIndex()}},
426 std::nullopt});
427 }
428 // The router writes one keep_pkt_header per destination port, the last
429 // packet flow's into it.
430 DenseMap<std::pair<TileID, Port>, std::optional<bool>> keepAt;
431 for (auto flow : device.getOps<PacketFlowOp>())
432 for (auto dst : flow.getPorts().front().getOps<PacketDestOp>())
433 if (std::optional<TileID> dstTile = tileOf(dst.getTile()))
434 keepAt[{*dstTile, dst.port()}] = flow.getKeepPktHeader();
435 for (auto flow : device.getOps<PacketFlowOp>()) {
436 Block &b = flow.getPorts().front();
437 for (auto src : b.getOps<PacketSourceOp>()) {
438 std::optional<TileID> srcTile = tileOf(src.getTile());
439 if (!srcTile)
440 continue;
441 for (auto dst : b.getOps<PacketDestOp>())
442 if (std::optional<TileID> dstTile = tileOf(dst.getTile()))
443 streams.push_back(
444 {{*srcTile, src.port()},
445 {*dstTile, dst.port()},
446 flow.IDInt(),
447 keepsPktHeader(*dstTile, dst.port(),
448 keepAt.lookup({*dstTile, dst.port()})),
449 {},
450 static_cast<int>(flow.getMask().value_or(~0))});
451 }
452 }
453 return streams;
454}

472// The BD blocks of a program in the order its channel runs them.
473static SmallVector<Block *> chainBlocks(const DmaChannelProgram &p) {
474 SmallVector<Block *> sequence(p.bds);
475 for (Region *r : p.regions)
476 sequence.push_back(&r->front());
477 return sequence;
478}

480// Index into `sequence` of the block a program runs at `step`: the blocks
481// before the cycle once, then the cycle over and over. A looping chain cycles
482// from where it returns to, a repeated one over the whole chain.
483static size_t blockAt(const DmaChannelProgram &p, size_t size, uint64_t step) {
484 size_t start = p.loops ? p.loopStart : 0;
485 if (step < start)
486 return step;
487 return start + (step - start) % (size - start);
488}

490// The value `use` acquires or releases, if it is a constant.
491static std::optional<int64_t> lockAmount(UseLockOp use) {
492 APInt amount;

494 !matchPattern(use.getValue(), m_ConstantInt(&amount)))
495 return std::nullopt;

509std::optional<uint64_t>
510StreamVolumeAnalysis::sendVolume(const RoutedStream &stream) const {
511 if (stream.src.port.bundle != WireBundle::DMA)
512 return std::nullopt;
513 TileDMAChannel dma{stream.src.tile, DMAChannelDir::MM2S,
514 stream.src.port.channel};
515 constexpr uint64_t pktHeaderBytes = 4;
516 auto headerBytes = [&](std::optional<PacketInfoAttr> packet) -> uint64_t {
517 return packet && stream.keepsPktHeader ? pktHeaderBytes : 0;
518 };
519 auto bytesOf = [&](DMABDOp bd) -> uint64_t {
520 if (!stream.carries(bd.getPacket()))
521 return 0;
522 return bd.getLenInBytes() + headerBytes(bd.getPacket());
523 };
524 bool known = false;
525 uint64_t total = 0;
526 if (auto it = programs.find(dma); it != programs.end()) {
527 for (const DmaChannelProgram &p : it->second) {
528 bool carried = false;
529 uint64_t bytes = 0;
530 forEachInProgram<DMABDOp>(p, [&](DMABDOp bd) {
531 if (!stream.carries(bd.getPacket()))
532 return;
533 carried = true;
534 bytes += bytesOf(bd);
535 });
536 known = true;
537 if (!carried)
538 continue;
539 if (p.loops) {
540 std::optional<uint64_t> looped = loopedVolume(p, bytesOf);
541 if (!looped)
542 return std::nullopt;
543 total += *looped;
544 continue;
545 }
546 uint64_t runs = p.passes;
547 if (isa<AIEX::DMAConfigureTaskOp, AIEX::DMAConfigureTaskForOp>(p.op)) {
548 auto task = dyn_cast<AIEX::DMAConfigureTaskOp>(p.op);
549 auto taskFor = dyn_cast<AIEX::DMAConfigureTaskForOp>(p.op);
550 if ((task && task.getRepeatCountVal()) ||
551 (taskFor && taskFor.getRepeatCountVal()))
552 return std::nullopt;
553 runs = 0;
554 for (Operation *user : p.op->getUsers()) {
555 if (isa<AIEX::DMAAwaitTaskOp, AIEX::DMAFreeTaskOp>(user))
556 continue;
557 // Any other user, such as an scf.yield or an scf.for init, can
558 // forward the task to starts this does not count.
559 if (!isa<AIEX::DMAStartTaskOp>(user))
560 return std::nullopt;
561 if (user->getParentOfType<LoopLikeOpInterface>())
562 return std::nullopt;
563 runs += p.passes;
564 }
565 }
566 total += bytes * runs;
567 }
568 }
569 if (auto it = memcpys.find(dma); it != memcpys.end()) {
570 for (Operation *op : it->second) {
571 auto memcpy = cast<AIEX::NpuDmaMemcpyNdOp>(op);
572 std::optional<PacketInfoAttr> packet = memcpy.getPacket();
573 if (!packet)
574 packet = ShimDMAAllocationOp::getForSymbol(
575 device, memcpy.getMetadata().getRootReference())
576 .getPacket();
577 if (!stream.carries(packet))
578 continue;
579 auto type = dyn_cast<BaseMemRefType>(memcpy.getMemref().getType());
580 if (memcpy->getParentOfType<LoopLikeOpInterface>() || !type ||
581 !type.getElementType().isIntOrFloat())
582 return std::nullopt;
583 std::optional<SmallVector<int64_t>> sizes =
584 getConstantIntValues(memcpy.getMixedSizes());
585 if (!sizes)
586 return std::nullopt;
587 uint64_t elements = 1;
588 for (int64_t n : *sizes)
589 elements *= n;
590 // The outermost dimension re-runs the BD (as iterations, or as repeats
591 // when its stride is 0), and each run sends its own header.
592 uint64_t runs = sizes->front();
593 known = true;
594 total += elements * type.getElementTypeBitWidth() / 8 +
595 runs * headerBytes(packet);
596 }
597 }
598 if (!known)
599 return std::nullopt;
600 return total;
601}

612bool StreamVolumeAnalysis::walkLoop(const DmaChannelProgram &p,
613 function_ref<void(DMABDOp)> visit) const {
614 // AIE1 locks hold a state rather than count tokens.
615 if (getTargetModel(device).getTargetArch() == AIEArch::AIE1 ||
616 !visiting.insert(&p).second)
617 return false;
618 llvm::scope_exit done([&] { visiting.erase(&p); });
619 SmallVector<Block *> sequence = chainBlocks(p);
620 if (sequence.empty())
621 return false;
622 DenseMap<Operation *, uint64_t> tokens;
623 for (int step = 0; step < maxBDSteps; step++) {
624 for (Operation &bdOp : *sequence[blockAt(p, sequence.size(), step)]) {
625 if (auto use = dyn_cast<UseLockOp>(bdOp)) {
626 auto lock = use.getLock().getDefiningOp<LockOp>();
627 std::optional<uint64_t> n = lockTokens(use);
628 if (!n || (!use.release() && !use.acquireGE()))
629 return false;
630 auto it = tokens.find(lock);
631 if (it == tokens.end()) {
632 std::optional<uint64_t> others = tokensFromOthers(lock, &p);
633 if (!others)
634 return false;
635 it = tokens.try_emplace(lock, lock.getInit().value_or(0) + *others)
636 .first;
637 }
638 if (use.release())
639 it->second += *n;
640 else if (it->second < *n)
641 return true;
642 else
643 it->second -= *n;
644 } else if (auto bd = dyn_cast<DMABDOp>(bdOp)) {
645 visit(bd);
646 }
647 }
648 }
649 return false;
650}
651
652// Whether `op` runs at most once each time its core runs its body.
653static bool onceInCore(Operation *op) {
654 for (Operation *parent = op->getParentOp(); parent;
655 op = parent, parent = parent->getParentOp()) {
656 if (!op->getParentRegion()->hasOneBlock() ||
657 isa<LoopLikeOpInterface>(parent))
658 return false;
659 if (isa<CoreOp>(parent))
660 return true;
661 }
662 return false;
663}
664
665std::optional<uint64_t>
666StreamVolumeAnalysis::tokensFromOthers(LockOp lock,
667 const DmaChannelProgram *self) const {
668 llvm::SetVector<const DmaChannelProgram *> releasers;
669 uint64_t tokens = 0;
670 for (Operation *user : lock->getUsers()) {
671 auto use = dyn_cast<UseLockOp>(user);
672 if (!use)
673 return std::nullopt;
674 if (!use.release())
675 continue;
676 auto owner = lockUseProgram.find(use);
677 if (owner == lockUseProgram.end()) {
678 std::optional<uint64_t> n = lockTokens(use);
679 if (!onceInCore(use) || !n)
680 return std::nullopt;
681 tokens += *n;
682 continue;
683 }
684 if (owner->second != self)
685 releasers.insert(owner->second);
686 }
687 for (const DmaChannelProgram *program : releasers) {
688 std::optional<uint64_t> n = releasesOver(*program, lock);
689 if (!n)
690 return std::nullopt;
691 tokens += *n;
692 }
693 return tokens;
694}

696std::optional<bool>
697StreamVolumeAnalysis::maySendAfter(const RoutedStream &first,
698 const RoutedStream &then) const {
699 if (first.src != then.src)
700 return std::nullopt;
701 if (!first.packetID || !then.packetID ||
702 ((*first.packetID ^ *then.packetID) & first.packetMask &
703 then.packetMask) == 0)
704 return true;
705 if (first.src.port.bundle != WireBundle::DMA)
706 return std::nullopt;
707 TileDMAChannel dma{first.src.tile, DMAChannelDir::MM2S,
708 first.src.port.channel};
709 auto it = programs.find(dma);
710 if (it == programs.end() || it->second.size() != 1 ||
711 !isa<DMAStartOp, DMAOp>(it->second.front().op) || memcpys.count(dma))
712 return std::nullopt;
713 const DmaChannelProgram &p = it->second.front();
714 bool sent = false, after = false;
715 auto visit = [&](DMABDOp bd) {
716 after |= sent && then.carries(bd.getPacket());
717 sent |= first.carries(bd.getPacket());
718 };
719 if (p.loops) {
720 if (!walkLoop(p, visit))
721 return std::nullopt;
722 return after;
723 }
724 SmallVector<Block *> sequence = chainBlocks(p);
725 for (uint64_t pass = 0; pass < p.passes && !after; pass++)
726 for (Block *b : sequence)
727 for (DMABDOp bd : b->getOps<DMABDOp>())
728 visit(bd);
729 return after;
730}
731
732std::optional<uint64_t>
733StreamVolumeAnalysis::releasesOver(const DmaChannelProgram &p,
734 LockOp lock) const {
735 if (!isa<DMAStartOp, DMAOp>(p.op))
736 return std::nullopt;
737 SmallVector<Block *> sequence = chainBlocks(p);
738 if (!p.loops) {
739 uint64_t perPass = 0;
740 for (Block *b : sequence)
741 for (auto use : b->getOps<UseLockOp>())
742 if (use.release() && use.getLock().getDefiningOp() == lock) {
743 std::optional<uint64_t> n = lockTokens(use);
744 if (!n)
745 return std::nullopt;
746 perPass += *n;
747 }
748 return perPass * p.passes;
749 }
750 // A looping receiver finishes only the BDs what it is sent fills.
751 if (p.dma.dir != DMAChannelDir::S2MM || sequence.empty())
752 return std::nullopt;
753 StreamEndpoint endpoint{p.dma.tile, {WireBundle::DMA, p.dma.channel}};
754 uint64_t received = 0;
755 for (const RoutedStream &s : streams) {
756 if (s.dst != endpoint)
757 continue;
758 std::optional<uint64_t> bytes = sendVolume(s);
759 if (!bytes)
760 return std::nullopt;
761 received += *bytes;
762 }
763 uint64_t filled = 0, tokens = 0;
764 for (int step = 0; step < maxBDSteps; step++) {
765 for (Operation &bdOp : *sequence[blockAt(p, sequence.size(), step)]) {
766 if (auto bd = dyn_cast<DMABDOp>(bdOp)) {
767 if (filled + bd.getLenInBytes() > received)
768 return tokens;
769 filled += bd.getLenInBytes();
770 } else if (auto use = dyn_cast<UseLockOp>(bdOp)) {
771 if (!use.release() || use.getLock().getDefiningOp() != lock)
772 continue;
773 std::optional<uint64_t> n = lockTokens(use);
774 if (!n)
775 return std::nullopt;
776 tokens += *n;
777 }
778 }
779 }
780 return std::nullopt;
781}

783static std::optional<uint64_t> programCapacity(const DmaChannelProgram &p,
784 DeviceOp device) {
785 if (!isa<DMAStartOp, DMAOp>(p.op))
786 return 0;
787
788 // Run the BD chain from its initial lock values until an acquire blocks.
789 SmallVector<Block *> sequence = chainBlocks(p);
790 if (sequence.empty())
791 return 0;
792 // AIE1 locks hold a state an acquire waits to equal and a release sets; later
793 // locks count, and an acquire waiting for an exact count is taken to block.
794 bool stateLocks = getTargetModel(device).getTargetArch() == AIEArch::AIE1;
795 DenseMap<Operation *, int64_t> lockValues;
796 uint64_t bytes = 0;
797 // Lock values and bytes when the cycle last began. A cycle that leaves every
798 // lock where it found it, or with more tokens, runs again the same way.
799 std::optional<DenseMap<Operation *, int64_t>> cycleLocks;
800 uint64_t cycleBytes = 0;
801 size_t start = p.loops ? p.loopStart : 0;
802 for (uint64_t step = 0;; step++) {
803 if (!p.loops && step >= p.passes * sequence.size())
804 return bytes;
805 size_t i = blockAt(p, sequence.size(), step);
806 if (i == start) {
807 if (cycleLocks && cycleLocks->size() == lockValues.size() &&
808 llvm::all_of(lockValues, [&](const auto &lv) {
809 int64_t before = cycleLocks->at(lv.first);
810 return stateLocks ? lv.second == before : lv.second >= before;
811 })) {
812 if (p.loops)
813 return std::nullopt;
814 return bytes +
815 (p.passes - step / sequence.size()) * (bytes - cycleBytes);
816 }
817 cycleLocks = lockValues;
818 cycleBytes = bytes;
819 }
820 // Past the analysis limit, what it took in so far is a safe capacity.
821 if (step >= static_cast<uint64_t>(maxBDSteps))
822 return bytes;
823 for (Operation &bdOp : *sequence[i]) {
824 if (auto use = dyn_cast<UseLockOp>(bdOp)) {
825 std::optional<int64_t> n = lockAmount(use);
826 if (!n)
827 return 0;
828 auto lock = use.getLock().getDefiningOp<LockOp>();
829 int64_t &value =
830 lockValues.try_emplace(lock, lock.getInit().value_or(0))
831 .first->second;
832 if (stateLocks) {
833 if (use.release())
834 value = *n;
835 else if (value != *n)
836 return bytes;
837 } else if (use.release()) {
838 value += *n;
839 } else if (!use.acquireGE() || value < *n) {
840 return bytes;
841 } else {
842 value -= *n;
843 }
844 } else if (auto bd = dyn_cast<DMABDOp>(bdOp)) {
845 bytes += bd.getLenInBytes();
846 }
847 }
848 }
849}

851std::optional<uint64_t>
852StreamVolumeAnalysis::receiveCapacity(const StreamEndpoint &endpoint) const {
853 if (endpoint.port.bundle != WireBundle::DMA)
854 return 0;
855 auto it = programs.find(
856 {endpoint.tile, DMAChannelDir::S2MM, endpoint.port.channel});
857 if (it == programs.end())
858 return 0;
859 std::optional<uint64_t> capacity;
860 for (const DmaChannelProgram &p : it->second)
861 if (std::optional<uint64_t> c = programCapacity(p, device))
862 capacity = std::min(*c, capacity.value_or(*c));
863 return capacity;
864}
865
866bool StreamVolumeAnalysis::canFill(const StreamEndpoint &endpoint,
867 ArrayRef<RoutedStream> streams) const {
868 std::optional<uint64_t> capacity = receiveCapacity(endpoint);
869 if (!capacity)
870 return false;
871 uint64_t sent = 0;
872 for (const RoutedStream &s : streams) {
873 if (s.dst != endpoint)
874 continue;
875 std::optional<uint64_t> bytes = sendVolume(s);
876 if (!bytes)
877 return true;
878 sent += *bytes;
879 }
880 return sent > *capacity;
881}
882
883StreamWaitGraph::StreamWaitGraph(DeviceOp device,
884 ArrayRef<RoutedStream> streams,
885 const StreamVolumeAnalysis &volumes) {
886 std::set<TileDMAChannel> neverFull;
887 for (const RoutedStream &s : streams)
888 if (s.dst.port.bundle == WireBundle::DMA &&
889 !volumes.canFill(s.dst, streams))
890 neverFull.insert({s.dst.tile, DMAChannelDir::S2MM, s.dst.port.channel});
891 auto waitsOnLocks = [&](unsigned agent) {
892 const Agent &a = agents[agent];
893 return a.kind == Agent::Kind::Core || !neverFull.count(a.dma);
894 };
895
896 // Who acquires and who releases each lock.
897 DenseMap<Operation *, llvm::SetVector<unsigned>> acquirers, releasers;
898 auto noteLock = [&](UseLockOp use, unsigned agent) {
899 Operation *lock = use.getLock().getDefiningOp();
900 if (!lock)
901 return;
902 if (use.getAction() == LockAction::Release)
903 releasers[lock].insert(agent);
904 else
905 acquirers[lock].insert(agent);
906 };
907
908 // Agents whose waits the design spells out; see the end of this constructor
909 // for the rest.
910 for (auto core : device.getOps<CoreOp>()) {
911 unsigned agent = getOrCreate(Agent::core(core.getTileOp().getTileID()));
912 modeled.insert(agent);
913 core.walk([&](UseLockOp use) { noteLock(use, agent); });
914 }
915 for (const DmaChannelProgram &p : collectDmaPrograms(device)) {
916 unsigned agent = getOrCreate(Agent::channel(p.dma));
917 modeled.insert(agent);
918 forEachInProgram<UseLockOp>(p,
919 [&](UseLockOp use) { noteLock(use, agent); });
920 }
921 // Document order, not pointer order: edge order picks the chain a
922 // diagnostic reports.
923 for (auto lock : device.getOps<LockOp>()) {
924 auto waiting = acquirers.find(lock), it = releasers.find(lock);
925 if (waiting == acquirers.end() || it == releasers.end())
926 continue;
927 for (unsigned p : waiting->second)
928 for (unsigned q : it->second)
929 if (p != q && waitsOnLocks(p))
930 addEdge(p, q, EdgeKind::Lock);
931 }
932
933 for (const RoutedStream &s : streams) {
934 std::optional<unsigned> from, to;
935 if (std::optional<Agent> agent = Agent::at(s.src, true))
936 from = getOrCreate(*agent);
937 if (std::optional<Agent> agent = Agent::at(s.dst, false))
938 to = getOrCreate(*agent);
939 if (!from || !to || *from == *to)
940 continue;
941 addEdge(*from, *to, EdgeKind::Stream);
942 addEdge(*to, *from, EdgeKind::Stream);
943 }
944
945 // The host issues a channel's transfer only after every wait before it in
946 // the runtime sequence completes. Each issue is modeled: a later one can
947 // follow waits the first did not. A wait on a channel not known here may be
948 // on any the sequence issues, and an issue on one, or a raw register write,
949 // may start any channel.
950 unsigned numStreamAgents = agents.size();
951 for (auto sequence : device.getOps<RuntimeSequenceOp>()) {
952 auto bySymbol = [&](SymbolRefAttr symbol) -> SmallVector<TileDMAChannel> {
953 if (std::optional<TileDMAChannel> key =
954 channelOfSymbol(device, symbol.getRootReference()))
955 return {*key};
956 return {};
957 };
958 auto issued =
959 [&](Operation *op) -> std::optional<SmallVector<TileDMAChannel>> {
960 if (auto memcpy = dyn_cast<AIEX::NpuDmaMemcpyNdOp>(op))
961 return bySymbol(memcpy.getMetadata());
962 if (auto start = dyn_cast<AIEX::DMAStartTaskOp>(op))
963 return channelsOfTask(device, start.getTask());
964 if (isa<AIEX::DMAStartBdChainOp, AIEX::DMAStartBdChainForOp>(op))
965 return channelsOfTask(device, op->getResult(0));
966 if (auto push = dyn_cast<AIEX::NpuPushQueueOp>(op))
967 return SmallVector<TileDMAChannel>{
968 {{static_cast<int>(push.getColumn()),
969 static_cast<int>(push.getRow())},
970 push.getDirection(),
971 static_cast<int>(push.getChannel())}};
972 if (isa<AIEX::NpuWrite32Op, AIEX::NpuMaskWrite32Op,
973 AIEX::NpuBlockWriteOp>(op))
974 return SmallVector<TileDMAChannel>{};
975 return std::nullopt;
976 };
977 auto awaited =
978 [&](Operation *op) -> std::optional<SmallVector<TileDMAChannel>> {
979 if (auto dmaWait = dyn_cast<AIEX::NpuDmaWaitOp>(op))
980 return bySymbol(dmaWait.getSymbolAttr());
981 if (auto await = dyn_cast<AIEX::DMAAwaitTaskOp>(op))
982 return channelsOfTask(device, await.getTask());
983 if (auto sync = dyn_cast<AIEX::NpuSyncOp>(op)) {
984 std::optional<int64_t> col = getConstantIntValue(sync.getColumn()),
985 row = getConstantIntValue(sync.getRow()),
986 dir = getConstantIntValue(sync.getDirection()),
987 channel = getConstantIntValue(sync.getChannel()),
988 cols = getConstantIntValue(sync.getColumnNum()),
989 rows = getConstantIntValue(sync.getRowNum());
990 if (!col || !row || !dir || !channel || !cols || !rows)
991 return SmallVector<TileDMAChannel>{};
992 SmallVector<TileDMAChannel> keys;
993 for (int64_t c = *col; c < *col + *cols; c++)
994 for (int64_t r = *row; r < *row + *rows; r++)
995 keys.push_back({{static_cast<int>(c), static_cast<int>(r)},
996 *dir ? DMAChannelDir::MM2S : DMAChannelDir::S2MM,
997 static_cast<int>(*channel)});
998 return keys;
999 }
1000 if (isa<AIEX::NpuMaskPollOp>(op))
1001 return SmallVector<TileDMAChannel>{};
1002 return std::nullopt;
1003 };
1004 llvm::SetVector<TileDMAChannel, SmallVector<TileDMAChannel>,
1005 std::set<TileDMAChannel>>
1006 issuedKeys;
1007 sequence.walk([&](Operation *op) {
1008 if (std::optional<SmallVector<TileDMAChannel>> keys = issued(op))
1009 issuedKeys.insert(keys->begin(), keys->end());
1010 });
1011
1012 llvm::SetVector<unsigned> waited;
1013 walkLoopsTwice(sequence.getBody(), [&](Operation *op) {
1014 if (std::optional<SmallVector<TileDMAChannel>> keys = issued(op)) {
1015 for (const TileDMAChannel &key : *keys) {
1016 unsigned agent = getOrCreate(Agent::channel(key));
1017 modeled.insert(agent);
1018 for (unsigned w : waited)
1019 if (w != agent)
1020 addEdge(agent, w, EdgeKind::Host);
1021 }
1022 if (keys->empty())
1023 for (unsigned a = 0; a < numStreamAgents; a++)
1024 if (agents[a].kind == Agent::Kind::Channel)
1025 for (unsigned w : waited)
1026 if (w != a)
1027 addEdge(a, w, EdgeKind::Host);
1028 } else if (std::optional<SmallVector<TileDMAChannel>> keys =
1029 awaited(op)) {
1030 for (const TileDMAChannel &key :
1031 keys->empty() ? issuedKeys.getArrayRef() : ArrayRef(*keys)) {
1032 waited.insert(getOrCreate(Agent::channel(key)));
1033 for (unsigned a = 0; a < numStreamAgents; a++)
1034 if (agents[a].kind == Agent::Kind::Controller &&
1035 agents[a].dma.tile.col == key.tile.col)
1036 waited.insert(a);
1037 }
1038 }
1039 });
1040 }
1041
1042 // A channel nothing programs here is programmed elsewhere, in ways this
1043 // cannot see, so it may wait on anything else on its tile. One on a shim
1044 // tile is driven by the host, which may wait on any other shim tile first.
1045 const AIETargetModel &targetModel = getTargetModel(device);
1046 for (Agent &a : agents)
1047 a.onShim = a.kind == Agent::Kind::Channel &&
1048 targetModel.isShimNOCorPLTile(a.dma.tile.col, a.dma.tile.row);
1049 for (unsigned a = 0; a < agents.size(); a++) {
1050 if (modeled.contains(a) || agents[a].kind == Agent::Kind::Controller ||
1051 !waitsOnLocks(a))
1052 continue;
1053 for (unsigned b = 0; b < agents.size(); b++)
1054 if (b != a && agents[b].kind != Agent::Kind::Controller &&
1055 (agents[b].dma.tile == agents[a].dma.tile ||
1056 (agents[a].onShim && agents[b].onShim)))
1057 addEdge(a, b, EdgeKind::Lock);
1058 }
1059
1060 LLVM_DEBUG({
1061 for (unsigned a = 0; a < agents.size(); a++)
1062 for (const Edge &e : edges[a])
1063 llvm::dbgs() << "Wait: " << describe(a) << " on " << describe(e.to)
1064 << (e.kind == EdgeKind::Lock ? " (lock)"
1065 : e.kind == EdgeKind::Stream ? " (stream)"
1066 : " (host)")
1067 << (e.kind == EdgeKind::Lock && !modeled.contains(a)
1068 ? ", assumed"
1069 : "")
1070 << "\n";
1071 });
1072}

1130SmallVector<unsigned>
1131StreamWaitGraph::waitChain(ArrayRef<unsigned> from, ArrayRef<unsigned> targets,
1132 ArrayRef<unsigned> avoid) const {
1133 llvm::DenseMap<unsigned, std::optional<unsigned>> parent;
1134 for (unsigned a : avoid)
1135 parent.try_emplace(a, std::nullopt);
1136 std::deque<unsigned> worklist;
1137 for (unsigned a : from)
1138 if (parent.try_emplace(a, std::nullopt).second)
1139 worklist.push_back(a);
1140 while (!worklist.empty()) {
1141 unsigned a = worklist.front();
1142 worklist.pop_front();
1143 if (llvm::is_contained(targets, a)) {
1144 SmallVector<unsigned> chain;
1145 for (std::optional<unsigned> at = a; at; at = parent.lookup(*at))
1146 chain.push_back(*at);
1147 std::reverse(chain.begin(), chain.end());
1148 return chain;
1149 }
1150 for (const Edge &e : edges[a])
1151 if (parent.try_emplace(e.to, a).second)
1152 worklist.push_back(e.to);
1153 }
1154 return {};
1155}

1185 size_t g) const {
1186 const RoutedStream &fs = streams[f], &gs = streams[g];
1187 std::optional<unsigned> fDst = graph.agentAt(fs.dst, false);
1188 if (!fDst)
1189 return {};
1190 SmallVector<unsigned> own;
1191 if (std::optional<unsigned> fSrc = graph.agentAt(fs.src, true))
1192 own.push_back(*fSrc);
1193 own.push_back(*fDst);
1194 SmallVector<unsigned> drainers = graph.drainersOf(*fDst);
1195 SmallVector<unsigned> avoid, targets;
1196 for (unsigned a : own)
1197 if (!llvm::is_contained(drainers, a))
1198 avoid.push_back(a);
1199 for (std::optional<unsigned> a :
1200 {graph.agentAt(gs.src, true), graph.agentAt(gs.dst, false)})
1201 if (a && !llvm::is_contained(own, *a))
1202 targets.push_back(*a);
1203 if (targets.empty())
1204 return {};
1205 return graph.waitChain(drainers, targets, avoid);
1206}

1224std::string StreamDeadlockAnalysis::explainBlock(size_t f, size_t g) const {
1225 const RoutedStream &fs = streams[f], &gs = streams[g];
1226 SmallVector<unsigned> chain = blockingChain(f, g);
1227 std::string s;
1228 llvm::raw_string_ostream os(s);
1229 os << describeStream(fs) << " can fill its receiver, and draining that "
1230 << "waits on ";
1231 for (auto [i, a] : llvm::enumerate(chain))
1232 os << (i ? ", then " : "") << graph.describe(a);
1233 os << (graph.agentAt(gs.dst, false) == chain.back() ? ", which receives "
1234 : ", which sends ")
1235 << describeStream(gs) << '.';
1236 for (const std::string &a : assumptions(f, g))
1237 os << ' ' << a;
1238 return s;
1239}
1240
1241SmallVector<std::string> StreamDeadlockAnalysis::assumptions(size_t f,
1242 size_t g) const {
1243 const RoutedStream &fs = streams[f], &gs = streams[g];
1244 SmallVector<std::string> assumed;
1245 for (const RoutedStream &other : streams)
1246 if (other.dst == fs.dst && !volumes.sendVolume(other)) {
1247 assumed.push_back("The volume " + describeStream(other) +
1248 " carries is unknown, so it is assumed to overrun its "
1249 "receiver.");
1250 break;
1251 }
1252 if (fs.src == gs.src && !volumes.maySendAfter(fs, gs))
1253 assumed.push_back("Both come from " +
1254 describeTilePort(fs.src.tile, fs.src.port) +
1255 ", and the order it sends in is not modeled.");
1256 SmallVector<unsigned> chain = blockingChain(f, g);
1257 std::optional<unsigned> receiver = graph.agentAt(fs.dst, false);
1258 assert(receiver && "a stream that can block has a receiver");
1259 SmallVector<unsigned> waiters{*receiver};
1260 waiters.append(chain.begin(), std::prev(chain.end()));
1261 for (auto [i, a] : llvm::enumerate(waiters))
1262 if (!graph.isModeled(a) &&
1263 !llvm::is_contained(ArrayRef(waiters).take_front(i), a))
1264 assumed.push_back(
1265 "Nothing in the design programs " + graph.describe(a) +
1266 ", so it is assumed to wait on anything on its tile" +
1267 (graph.getAgent(a).onShim ? " or on another shim tile." : "."));
1268 return assumed;
1269}

1338// The chains holdCycle follows from a tree stuck at a receiver to the trees
1339// it waits on, which hold for any routing.
1340const DenseMap<size_t, std::pair<size_t, size_t>> &
1341StreamConflicts::waitsFrom(size_t a) const {
1342 std::optional<DenseMap<size_t, std::pair<size_t, size_t>>> &cached = waits[a];
1343 if (cached)
1344 return *cached;
1345 DenseMap<size_t, std::pair<size_t, size_t>> &reached = cached.emplace();
1346 std::deque<size_t> work{a};
1347 while (!work.empty()) {
1348 size_t u = work.front();
1349 work.pop_front();
1350 for (size_t g = 0; g < treeMembers.size(); g++) {
1351 if (g == a || reached.contains(g) ||
1352 !streams[treeMembers[g].front()].packetID)
1353 continue;
1354 auto by = [&]() -> std::optional<std::pair<size_t, size_t>> {
1355 for (size_t x : treeMembers[u])
1356 for (size_t m : treeMembers[g])
1357 if (blocks(x, m))
1358 return std::pair{x, m};
1359 return std::nullopt;
1360 }();
1361 if (!by)
1362 continue;
1363 reached[g] = *by;
1364 work.push_back(g);
1365 }
1366 }
1367 return reached;
1368}

1378std::string StreamConflicts::explain(size_t s, size_t t) const {
1379 const StreamDeadlockAnalysis &a = getAnalysis();
1380 if (a.canBlock(s, t))
1381 return a.explainBlock(s, t);
1382 if (a.canBlock(t, s))
1383 return a.explainBlock(t, s);
1384 if (!waitsFrom(treeOf[s]).contains(treeOf[t]))
1385 std::swap(s, t);
1386 SmallVector<std::string> steps;
1387 for (size_t g = treeOf[t]; g != treeOf[s];) {
1388 auto [x, m] = waitsFrom(treeOf[s]).at(g);
1389 steps.push_back(a.explainBlock(x, m));
1390 g = treeOf[x];
1391 }
1392 std::reverse(steps.begin(), steps.end());
1393 return llvm::join(steps, " ");
1394}

1396SmallVector<std::pair<size_t, size_t>> StreamConflicts::unavoidable() const {
1397 SmallVector<std::pair<size_t, size_t>> pairs;
1398 const StreamDeadlockAnalysis &a = getAnalysis();
1399 for (size_t s = 0; s < numRequested; s++)
1400 for (size_t t = 0; t < numRequested; t++) {
1401 if (s == t || !(streams[s].packetID || streams[t].packetID) ||
1402 !related(s, t) || !a.canBlock(s, t))

1409 }

1413namespace {
1414// A wait in the graph holdCycle searches: a head stuck at one node waits on
1415// one at `to`, for the reason `step` gives, if any.
1416struct WaitEdge {
1417 size_t to;
1418 std::optional<HoldCycle::Step> step;
1419 // Where the trees a link wait is between merged above the link: the
1420 // switchbox and the arbiter that granted them the link in turn.
1421 std::optional<std::pair<TileID, int>> merged = std::nullopt;
1422};
1423// The graph holdCycle searches, built as it is walked. Node `entry` leads to
1424// every root.
1425struct WaitGraph {
1426 llvm::function_ref<ArrayRef<WaitEdge>(size_t)> successors;
1427 size_t entry;
1428};
1429using WaitNode = std::pair<const WaitGraph *, size_t>;
1430} // namespace
1431
1432namespace llvm {
1433template <>
1434struct GraphTraits<const WaitGraph *> {
1435 using NodeRef = WaitNode;
1436 struct ToNode {
1437 const WaitGraph *graph;
1438 NodeRef operator()(const WaitEdge &e) const { return {graph, e.to}; }
1439 };
1440 using ChildIteratorType = mapped_iterator<const WaitEdge *, ToNode>;
1441 static NodeRef getEntryNode(const WaitGraph *g) { return {g, g->entry}; }
1442 static ChildIteratorType child_begin(NodeRef n) {
1443 return map_iterator(n.first->successors(n.second).begin(), ToNode{n.first});
1444 }
1445 static ChildIteratorType child_end(NodeRef n) {
1446 return map_iterator(n.first->successors(n.second).end(), ToNode{n.first});
1447 }
1448};
1449} // namespace llvm
1450
1451std::optional<HoldCycle>
1452StreamConflicts::holdCycle(ArrayRef<SmallVector<StreamHop, 8>> routes,
1453 bool forcedWaits, bool definite) const {
1454 // The packets one source sends with one id move down every branch as one.
1455 struct Tree {
1456 SmallVector<size_t, 2> members;
1457 SmallVector<std::pair<TileID, Port>, 8> hops;
1458 SmallVector<int, 8> parent;
1459 SmallVector<std::optional<int>, 8> arbiter;
1460 size_t base = 0;
1461 };
1462 std::vector<Tree> trees;
1463 std::map<TreeKey, size_t> treeIDs;
1464 for (size_t i = 0; i < streams.size(); i++) {
1465 std::optional<TreeKey> key = treeKey(i);
1466 if (!key || getAnalysis().silent(i))
1467 continue;
1468 auto [it, inserted] = treeIDs.try_emplace(*key, trees.size());
1469 if (inserted)
1470 trees.emplace_back();
1471 Tree &tree = trees[it->second];
1472 tree.members.push_back(i);
1473 int prev = -1;
1474 for (const StreamHop &hop : routes[i]) {
1475 std::pair<TileID, Port> key{hop.tile, hop.input};
1476 int h = llvm::find(tree.hops, key) - tree.hops.begin();
1477 if (h == static_cast<int>(tree.hops.size())) {
1478 tree.hops.push_back(key);
1479 tree.parent.push_back(prev);
1480 tree.arbiter.push_back(hop.arbiter);
1481 }
1482 prev = h;
1483 }
1484 }
1485
1486 // One source sends one tree's packet at a time. Trees into one receiver are
1487 // both in flight at once, and wait on each other at its master port however
1488 // they are routed, where both enter its switchbox for it; and trees with one
1489 // id wait on each other where they meet and below, going on as one. Those
1490 // waits count only with `forcedWaits`.
1491 auto sameSource = [&](size_t a, size_t b) {
1492 return streams[trees[a].members.front()].src ==
1493 streams[trees[b].members.front()].src;
1494 };
1495 auto forced = [&](size_t a, int ha, size_t b, int hb, TileID tile) {
1496 Port ina = trees[a].hops[ha].second, inb = trees[b].hops[hb].second;
1497 auto into = [&](size_t m, Port in) {
1498 return streams[m].dst.tile == tile && !routes[m].empty() &&
1499 routes[m].back().tile == tile && routes[m].back().input == in;
1500 };
1501 for (size_t m : trees[a].members)
1502 for (size_t n : trees[b].members)
1503 if (streams[m].dst == streams[n].dst && into(m, ina) && into(n, inb))
1504 return true;
1505 if (streams[trees[a].members.front()].packetID !=
1506 streams[trees[b].members.front()].packetID)
1507 return false;
1508 for (auto above : llvm::enumerate(trees[a].parent))
1509 if (above.value() == ha &&
1510 llvm::any_of(llvm::enumerate(trees[b].parent), [&](auto below) {
1511 return below.value() == hb &&
1512 trees[b].hops[below.index()] == trees[a].hops[above.index()];
1513 }))
1514 return true;
1515 return false;
1516 };
1517
1518 // Per tree, a node for a head of it stuck entering each hop or at each
1519 // receiver, one for it stuck anywhere, and one per hop for it stuck
1520 // anywhere past that hop, so still holding what it took there.
1521 enum class Kind { Hop, Receiver, Anywhere, Past };
1522 struct Node {
1523 size_t tree;
1524 Kind kind;
1525 size_t index;
1526 };
1527 std::vector<Node> nodes;
1528 for (size_t t = 0; t < trees.size(); t++) {
1529 Tree &tree = trees[t];
1530 tree.base = nodes.size();
1531 for (size_t h = 0; h < tree.hops.size(); h++)
1532 nodes.push_back({t, Kind::Hop, h});
1533 for (size_t m = 0; m < tree.members.size(); m++)
1534 nodes.push_back({t, Kind::Receiver, m});
1535 nodes.push_back({t, Kind::Anywhere, 0});
1536 for (size_t h = 0; h < tree.hops.size(); h++)
1537 nodes.push_back({t, Kind::Past, h});
1538 }
1539 auto receiverNode = [&](size_t t, size_t m) {
1540 return trees[t].base + trees[t].hops.size() + m;
1541 };
1542 auto anywhereNode = [&](size_t t) {
1543 return receiverNode(t, trees[t].members.size());
1544 };
1545 auto pastNode = [&](size_t t, size_t h) { return anywhereNode(t) + 1 + h; };
1546
1547 std::map<std::pair<TileID, Port>, SmallVector<std::pair<size_t, size_t>, 4>>
1548 entering;
1549 std::map<TileID, SmallVector<std::pair<size_t, size_t>, 8>> passing;
1550 for (size_t t = 0; t < trees.size(); t++)
1551 for (size_t h = 0; h < trees[t].hops.size(); h++) {
1552 entering[trees[t].hops[h]].push_back({t, h});
1553 passing[trees[t].hops[h].first].push_back({t, h});
1554 }
1555
1556 // The arbiter both trees took to leave the switchbox above a hop they share
1557 // where they came into it apart.
1558 using Grant = std::pair<TileID, int>;
1559 auto mergedAt = [&](size_t a, int ha, size_t b,
1560 int hb) -> std::optional<Grant> {
1561 for (;;) {
1562 int pa = trees[a].parent[ha], pb = trees[b].parent[hb];
1563 if (pa < 0 || pb < 0)
1564 return std::nullopt;
1565 if (trees[a].hops[pa] != trees[b].hops[pb]) {
1566 std::optional<int> arbiter = trees[a].arbiter[pa];
1567 if (trees[a].hops[pa].first != trees[b].hops[pb].first || !arbiter ||
1568 arbiter != trees[b].arbiter[pb])
1569 return std::nullopt;
1570 return Grant{trees[a].hops[pa].first, *arbiter};
1571 }
1572 ha = pa;
1573 hb = pb;
1574 }
1575 };
1576
1577 using Edge = WaitEdge;
1578 SmallVector<size_t> roots;
1579 std::vector<std::optional<SmallVector<Edge, 4>>> edges(nodes.size() + 1);
1580 auto successors = [&](size_t n) -> ArrayRef<Edge> {
1581 if (edges[n])
1582 return *edges[n];
1583 SmallVector<Edge, 4> out;
1584 if (n == nodes.size()) {
1585 for (size_t root : roots)
1586 out.push_back({root, std::nullopt});
1587 return *(edges[n] = std::move(out));
1588 }
1589 const Node &node = nodes[n];
1590 const Tree &u = trees[node.tree];
1591 switch (node.kind) {
1592 case Kind::Hop: {
1593 auto [tile, input] = u.hops[node.index];
1594 size_t waiting = u.members.front();
1595 for (auto [t, ht] : entering.at({tile, input})) {
1596 if (t != node.tree && sameSource(node.tree, t))
1597 continue;
1598 size_t sharer = trees[t].members.front();
1599 bool sameId = streams[waiting].packetID == streams[sharer].packetID;
1600 if (t != node.tree && (!sameId || forcedWaits))
1601 out.push_back(
1602 {pastNode(t, ht),
1603 HoldCycle::Step{HoldCycle::Wait::Link, waiting, sharer, sharer,
1604 tile, input, input, -1, sameId},
1605 mergedAt(node.tree, node.index, t, ht)});
1606 std::optional<int> arbiter = trees[t].arbiter[ht];
1607 if (!arbiter)
1608 continue;
1609 // A tree's branches move as one, so where two come into a switchbox
1610 // apart onto one arbiter, the one holding it waits on the other.
1611 for (auto [v, hv] : passing.at(tile)) {
1612 Port holderInput = trees[v].hops[hv].second;
1613 bool self = v == node.tree && t == node.tree;
1614 if ((!self && (v == node.tree || v == t || sameSource(t, v))) ||
1615 holderInput == input || trees[v].arbiter[hv] != arbiter)
1616 continue;
1617 bool isForced = !self && forced(t, ht, v, hv, tile);
1618 if (isForced && !forcedWaits)
1619 continue;
1620 out.push_back(
1621 {pastNode(v, hv),
1622 HoldCycle::Step{HoldCycle::Wait::Arbiter, waiting, sharer,
1623 trees[v].members.front(), tile, input,
1624 holderInput, *arbiter, isForced}});
1625 }
1626 }
1627 break;
1628 }
1629 case Kind::Receiver: {
1630 // A tree into the full receiver is stuck there however it is routed.
1631 size_t s = u.members[node.index];
1632 for (size_t g = 0; g < trees.size(); g++) {
1633 if (g == node.tree || llvm::any_of(trees[g].members, [&](size_t m) {
1634 return streams[m].dst == streams[s].dst;
1635 }))
1636 continue;
1637 for (size_t m : trees[g].members)
1638 if (blocks(s, m) &&
1639 (!definite || getAnalysis().assumptions(s, m).empty())) {
1640 out.push_back({anywhereNode(g),
1641 HoldCycle::Step{HoldCycle::Wait::Drain, s, s, m,
1642 TileID{}, Port{}, Port{}, -1}});
1643 break;
1644 }
1645 }
1646 break;
1647 }
1648 case Kind::Anywhere:
1649 case Kind::Past: {
1650 llvm::SmallDenseSet<size_t, 8> behind;
1651 if (node.kind == Kind::Past)
1652 for (int h = node.index; h >= 0; h = u.parent[h])
1653 behind.insert(h);
1654 for (size_t h = 0; h < u.hops.size(); h++)
1655 if (!behind.contains(h))
1656 out.push_back({u.base + h, std::nullopt});
1657 for (size_t m = 0; m < u.members.size(); m++)
1658 out.push_back({receiverNode(node.tree, m), std::nullopt});
1659 break;
1660 }
1661 }
1662 return *(edges[n] = std::move(out));
1663 };
1664
1665 // A cycle of receivers alone waiting on each other is the design's own, so
1666 // only cycles through a wait some requested routing adds count.
1667 auto counts = [&](const Edge &e) {
1668 if (!e.step || e.step->wait == HoldCycle::Wait::Drain)
1669 return false;
1670 return e.step->waiting < numRequested || e.step->sharer < numRequested ||
1671 e.step->holding < numRequested;
1672 };
1673 for (size_t n = 0; n < nodes.size(); n++)
1674 if (nodes[n].kind == Kind::Hop && llvm::any_of(successors(n), counts))
1675 roots.push_back(n);
1676
1677 const WaitGraph graph{successors, nodes.size()};
1678 std::vector<int> component(nodes.size() + 1, -1);
1679 int components = 0;
1680 for (auto scc = llvm::scc_begin(&graph); !scc.isAtEnd(); ++scc, ++components)
1681 for (WaitNode n : *scc)
1682 component[n.second] = components;
1683
1684 // A counted wait within one component lies on a closed walk. No state has two
1685 // trees holding one arbiter (see HoldCycle), so a walk that needs that is no
1686 // deadlock. Nor is one where each packet is behind the next on links below
1687 // one arbiter they merged at, which granted them those links in one order, so
1688 // a walk from a link wait below one must also wait otherwise. The search
1689 // fixes or rules out a holder per arbiter until the shortest walk left
1690 // agrees. Each clash splits it in two, so past maxWalkSearches it keeps the
1691 // first walk, which at worst steers the router off a routing that cannot
1692 // deadlock.
1693 struct Holders {
1694 std::optional<size_t> fixed;
1695 SmallVector<size_t, 2> excluded;
1696 };
1697 using Constraints = std::map<Grant, Holders>;
1698 auto allowed = [](const Edge &edge, const Constraints &c) {
1699 if (!edge.step || edge.step->wait != HoldCycle::Wait::Arbiter)
1700 return true;
1701 auto it = c.find({edge.step->tile, edge.step->arbiter});
1702 if (it == c.end())
1703 return true;
1704 size_t holder = edge.step->holding;
1705 return it->second.fixed ? *it->second.fixed == holder
1706 : !llvm::is_contained(it->second.excluded, holder);
1707 };
1708 auto closeWalk =
1709 [&](size_t x, const Edge &e,
1710 const Constraints &c) -> std::optional<SmallVector<const Edge *>> {
1711 if (!allowed(e, c))
1712 return std::nullopt;
1713 // A state is a node and whether the walk there has waited other than on
1714 // a link below e's merge.
1715 auto ordered = [&](const Edge &edge) {
1716 return !edge.step || edge.merged == e.merged;
1717 };
1718 auto state = [](size_t node, bool waited) { return 2 * node + waited; };
1719 size_t start = state(e.to, !e.merged), end = state(x, true);
1720 DenseMap<size_t, std::pair<size_t, const Edge *>> via;
1721 via[start] = {start, nullptr};
1722 std::deque<size_t> worklist{start};
1723 while (!via.contains(end)) {
1724 if (worklist.empty())
1725 return std::nullopt;
1726 size_t n = worklist.front();
1727 worklist.pop_front();
1728 for (const Edge &next : successors(n / 2)) {
1729 size_t to = state(next.to, n % 2 || !ordered(next));
1730 if (component[next.to] == component[x] && allowed(next, c) &&
1731 via.try_emplace(to, n, &next).second)
1732 worklist.push_back(to);
1733 }
1734 }
1735 SmallVector<const Edge *> path;
1736 for (size_t n = end; n != start; n = via.at(n).first)
1737 path.push_back(via.at(n).second);
1738 path.push_back(&e);
1739 std::reverse(path.begin(), path.end());
1740 return path;
1741 };
1742 auto toCycle = [](ArrayRef<const Edge *> path) {
1743 HoldCycle cycle;
1744 for (const Edge *edge : path)
1745 if (edge->step)
1746 cycle.steps.push_back(*edge->step);
1747 return cycle;
1748 };
1749 constexpr int maxWalkSearches = 1024;
1750 for (size_t x : roots)
1751 for (const Edge &e : successors(x)) {
1752 if (!counts(e) || component[e.to] != component[x])
1753 continue;
1754 std::optional<SmallVector<const Edge *>> first;
1755 SmallVector<Constraints> pending{{}};
1756 int search = 0;
1757 for (; !pending.empty(); search++) {
1758 if (search == maxWalkSearches) {
1759 assert(first && "only a walk found adds pending constraints");
1760 LLVM_DEBUG(llvm::dbgs() << "Hold cycle search gave up after "
1761 << search << " walks\n");
1762 return toCycle(*first);
1763 }
1764 Constraints c = pending.pop_back_val();
1765 std::optional<SmallVector<const Edge *>> path = closeWalk(x, e, c);
1766 if (!path)
1767 continue;
1768 if (!first)
1769 first = path;
1770 std::map<Grant, size_t> held;
1771 std::optional<std::pair<Grant, size_t>> clash;
1772 for (const Edge *edge : *path) {
1773 if (!edge->step || edge->step->wait != HoldCycle::Wait::Arbiter)
1774 continue;
1775 auto [it, inserted] = held.try_emplace(
1776 {edge->step->tile, edge->step->arbiter}, edge->step->holding);
1777 if (!inserted && it->second != edge->step->holding) {
1778 clash = *it;
1779 break;
1780 }
1781 }
1782 if (!clash) {

8// This pass expands `npu.load_pdi` operations that reference a device. There
9// are two output modes, controlled by the `ctrl-pkt` option:
10//
11// 1. Default (ctrl-pkt=false): replaces each `load_pdi @device` with
12// a. an empty device PDI load (`load_pdi @empty_N`), which causes the
13// firmware to reset the device, and
14// b. explicit `aiex.npu.write32`/`aiex.npu.blockwrite` configuration ops.
15// 2. With ctrl-pkt=true: replaces each `load_pdi @device` with
16// a. a `load_pdi @ctrl_pkt_overlay`, which configures the NPU to stream
17// further configuration as control packets, and
18// b. a sequence of `aiex.npu.control_packet` ops carrying the device's
19// configuration.
20//
21//===----------------------------------------------------------------------===//

53// Name of the overlay device loaded ahead of streaming configuration as
54// control packets. The device is expected to exist in the module (e.g.,
55// emitted by the `aie-generate-column-control-overlay` pass).
56static constexpr llvm::StringLiteral kCtrlPktOverlayName = "ctrl_pkt_overlay";
57// Name of an alternating copy of the overlay device, used to avoid PDI
58// address caching when the same overlay would otherwise be loaded twice in a
59// row (the firmware would treat the second load as a no-op).
60static constexpr llvm::StringLiteral kCtrlPktOverlayCopyName =
61 "ctrl_pkt_overlay_copy";
62
63// Look up `kCtrlPktOverlayName` and return a clone of it named
64// `kCtrlPktOverlayCopyName`, creating it once if needed. Returns nullptr on
65// error.
66static AIE::DeviceOp getOrCreateCtrlPktOverlayCopy(ModuleOp moduleOp,
67 Operation *errLocOp) {
68 if (auto existing =
69 moduleOp.lookupSymbol<AIE::DeviceOp>(kCtrlPktOverlayCopyName))
70 return existing;
71
72 auto orig = moduleOp.lookupSymbol<AIE::DeviceOp>(kCtrlPktOverlayName);
73 if (!orig) {
74 errLocOp->emitError("ctrl-pkt mode requires a `@")
75 << kCtrlPktOverlayName << "` device in the module";
76 return nullptr;
77 }
78
79 OpBuilder builder(moduleOp.getContext());
80 builder.setInsertionPointAfter(orig);
81 auto *cloned = builder.clone(*orig.getOperation());
82 auto clonedDev = cast<AIE::DeviceOp>(cloned);
83 clonedDev.setSymName(kCtrlPktOverlayCopyName);
84 return clonedDev;
85}

87// A stream-switch setting: a port, or with `slot` >= 0 one of its packet rule
88// slots.
89struct SwitchPort {
90 AIE::TileID tile;
91 bool master;
92 AIE::Port port;
93 int slot;
94
95 bool operator<(const SwitchPort &other) const {
96 return std::tie(tile, master, port, slot) <
97 std::tie(other.tile, other.master, other.port, other.slot);
98 }
99
100 std::string describe() const {
101 std::string s =
102 llvm::formatv("tile ({0}, {1}) {2} {3} : {4}", tile.col, tile.row,
103 master ? "master" : "slave",
104 AIE::stringifyWireBundle(port.bundle), port.channel);
105 if (slot >= 0)
106 s += llvm::formatv(" slot {0}", slot).str();
107 return s;
108 }
109};

111struct PortSetting {
112 SwitchPort port;
113 std::string value;
114 Operation *op;
115 bool skipped;
116};
117
118// The settings `op` writes, and whether a reload skips each. Each packet rule
119// fills one slot, and a reload skips the rules marked
120// is_ctrl_pkt_overlay. It skips enabling the port if it skips any of them.
121static SmallVector<PortSetting> portSettings(Operation &op,
122 AIE::SwitchboxOp sb) {
123 auto key = [&](bool master, AIE::Port port, int slot = -1) {
124 return SwitchPort{{sb.colIndex(), sb.rowIndex()}, master, port, slot};
125 };
126 auto amsel = [](Value v) {
127 auto a = v.getDefiningOp<AIE::AMSelOp>();
128 return llvm::formatv("{0}.{1}", a.arbiterIndex(), a.getMselValue()).str();
129 };
130 AIE::AIEDialect::IsCtrlPktOverlayAttrHelper overlay(op.getContext());
131 bool skipped = overlay.isAttrPresent(&op);
132 SmallVector<PortSetting> settings;
133 if (auto connect = dyn_cast<AIE::ConnectOp>(op)) {
134 AIE::Port src = connect.sourcePort();
135 settings.push_back(
136 {key(true, connect.destPort()),
137 llvm::formatv("circuit {0} : {1}",
138 AIE::stringifyWireBundle(src.bundle), src.channel),
139 &op, skipped});
140 settings.push_back({key(false, src), "circuit", &op, skipped});
141 } else if (auto masterSet = dyn_cast<AIE::MasterSetOp>(op)) {
142 int arbiter = -1;
143 int mask = 0;
144 for (Value v : masterSet.getAmsels()) {
145 auto a = cast<AIE::AMSelOp>(v.getDefiningOp());
146 arbiter = a.arbiterIndex();
147 mask |= 1 << a.getMselValue();
148 }
149 std::string value =
150 llvm::formatv("packet {0}/{1} {2}", arbiter, mask,
151 masterSet.keepsPktHeader() ? "keep" : "drop")
152 .str();
153 settings.push_back({key(true, masterSet.destPort()), value, &op, skipped});
154 } else if (auto rules = dyn_cast<AIE::PacketRulesOp>(op)) {
155 // An empty packet_rules op writes nothing to the hardware.
156 auto ruleOps = rules.getRules().front().getOps<AIE::PacketRuleOp>();
157 if (ruleOps.empty())
158 return settings;
159 bool anySkipped = skipped;
160 for (auto [slot, rule] : llvm::enumerate(ruleOps)) {
161 bool ruleSkipped = skipped || overlay.isAttrPresent(rule);
162 anySkipped |= ruleSkipped;
163 settings.push_back({key(false, rules.sourcePort(), slot),
164 llvm::formatv("{0}/{1}>{2}", rule.valueInt(),
165 rule.maskInt(), amsel(rule.getAmsel())),
166 rule, ruleSkipped});
167 }
168 settings.insert(settings.begin(), {key(false, rules.sourcePort()), "packet",
169 &op, anySkipped});
170 }
171 return settings;
172}

174// A control-packet reload streams `device`'s configuration over the routes
175// `overlay` set up, skipping what is marked is_ctrl_pkt_overlay. So the
176// overlay must already write every setting the reload skips, the same way,
177// and the reload may write nothing the overlay writes.
178static LogicalResult verifyReloadKeepsOverlay(AIE::DeviceOp device,
179 AIE::DeviceOp overlay) {
180 StringRef overlayName = overlay.getSymName();
181 std::map<SwitchPort, std::string> overlaySettings;
182 overlay.walk([&](AIE::SwitchboxOp sb) {
183 for (Operation &op : sb.getConnections().front())
184 for (PortSetting &setting : portSettings(op, sb))
185 overlaySettings[setting.port] = setting.value;
186 });
187 WalkResult result = device.walk([&](AIE::SwitchboxOp sb) {
188 for (Operation &op : sb.getConnections().front()) {
189 for (auto &[port, value, setter, skipped] : portSettings(op, sb)) {
190 auto it = overlaySettings.find(port);
191 if (skipped && it == overlaySettings.end()) {
192 setter->emitError()
193 << "a control-packet reload skips this op, but @" << overlayName
194 << " does not set " << port.describe();
195 return WalkResult::interrupt();
196 }
197 if (skipped && it->second != value) {
198 setter->emitError()
199 << "a control-packet reload skips this op, but @" << overlayName
200 << " sets " << port.describe() << " differently";
201 return WalkResult::interrupt();
202 }
203 if (!skipped && it != overlaySettings.end()) {
204 setter->emitError()
205 << "a control-packet reload rewrites " << port.describe()
206 << ", which the control packets of @" << overlayName
207 << " route through";
208 return WalkResult::interrupt();
209 }
210 }
211 }
212 return WalkResult::advance();
213 });
214 return failure(result.wasInterrupted());
215}

217// A control-packet reload reaches a tile only if `overlay` routes control
218// packets to its TileControl port. `packets` are the reload's.
219static LogicalResult
220verifyOverlayReachesTiles(NpuLoadPdiOp loadPdiOp, AIE::DeviceOp overlay,
221 iterator_range<Block::iterator> packets) {
222 std::set<AIE::TileID> reached;
223 overlay.walk([&](AIE::SwitchboxOp sb) {
224 for (Operation &op : sb.getConnections().front()) {
225 std::optional<AIE::Port> dest;
226 if (auto master = dyn_cast<AIE::MasterSetOp>(op))
227 dest = master.destPort();
228 else if (auto connect = dyn_cast<AIE::ConnectOp>(op))
229 dest = connect.destPort();
230 if (dest == AIE::Port{AIE::WireBundle::TileControl, 0})
231 reached.insert({sb.colIndex(), sb.rowIndex()});
232 }
233 });
234 for (Operation &op : packets) {
235 auto packet = dyn_cast<NpuControlPacketOp>(op);
236 if (!packet)
237 continue;
238 AIE::TileID tile{static_cast<int>(packet.getColumnFromAddr()),
239 static_cast<int>(packet.getRowFromAddr())};
240 if (reached.count(tile))
241 continue;
242 return loadPdiOp.emitError()
243 << "a control-packet reload configures tile (" << tile.col << ", "
244 << tile.row << "), but @" << overlay.getSymName()
245 << " routes no control packets to it";
246 }
247 return success();
248}

250// The empty device whose PDI load makes the firmware reset the array. `parity`
251// alternates so that two consecutive loads never name the same PDI (see
252// transformLoadPdi).
253static AIE::DeviceOp getOrCreateEmptyDevice(ModuleOp moduleOp,
254 AIE::AIEDevice deviceType,
255 unsigned parity, Location loc) {
256 std::string emptyName = "empty_" + std::to_string(parity);
257 if (auto existing = moduleOp.lookupSymbol<AIE::DeviceOp>(emptyName)) {
258 // Shared reset PDIs retain every load and trailing sequence's provenance.
259 loc = FusedLoc::get(moduleOp.getContext(), {existing.getLoc(), loc});
260 existing->setLoc(loc);
261 existing.getRegion().front().getTerminator()->setLoc(loc);
262 return existing;
263 }
264
265 OpBuilder builder(moduleOp.getContext());
266 builder.setInsertionPointToStart(moduleOp.getBody());
267 auto emptyDevice = AIE::DeviceOp::create(builder, loc, deviceType,
268 builder.getStringAttr(emptyName));
269 emptyDevice.getRegion().emplaceBlock();
270 builder.setInsertionPointToEnd(&emptyDevice.getRegion().front());
271 AIE::EndOp::create(builder, loc);
272 return emptyDevice;
273}

275// Helper to transform a single load_pdi operation
276static LogicalResult transformLoadPdi(NpuLoadPdiOp loadPdiOp, ModuleOp moduleOp,
277 unsigned index,
278 AIEX::ExpandMode defaultMode) {
279 static unsigned long i = 0;
280 OpBuilder builder(loadPdiOp);
281
282 // Only process load_pdi ops that reference a device
283 auto deviceRefAttr = loadPdiOp.getDeviceRefAttr();
284 if (!deviceRefAttr) {
285 return success();
286 }
287
288 // Per-op annotation takes precedence; fall back to pass default
289 AIEX::ExpandMode mode = loadPdiOp.getExpandMode().value_or(defaultMode);
290 if (mode == AIEX::ExpandMode::none)
291 return success();
292 bool ctrlPkt = (mode == AIEX::ExpandMode::ctrlpkt);
293
294 auto referencedDevice = moduleOp.lookupSymbol<AIE::DeviceOp>(deviceRefAttr);
295 if (!referencedDevice) {
296 loadPdiOp.emitError("Referenced symbol '")
297 << deviceRefAttr.getValue() << "' is not a device";
298 return failure();
299 }
300
301 FlatSymbolRefAttr preloadRef;
302 AIE::DeviceOp overlay;
303 if (ctrlPkt) {
304 // Overlay device PDI
305 // Alternate between the original overlay and a clone of it on every
306 // other load. Loading the same PDI twice in a row gets cached by the
307 // firmware (the second load becomes a no-op), so we need two distinct
308 // PDI addresses that carry the same overlay configuration.
309 StringRef overlayName =
310 (index % 2 == 0) ? kCtrlPktOverlayName : kCtrlPktOverlayCopyName;
311 if (index % 2 != 0) {
312 overlay =
313 getOrCreateCtrlPktOverlayCopy(moduleOp, loadPdiOp.getOperation());
314 if (!overlay)
315 return failure();
316 } else {
317 overlay = moduleOp.lookupSymbol<AIE::DeviceOp>(kCtrlPktOverlayName);
318 if (!overlay) {
319 loadPdiOp.emitError("ctrl-pkt mode requires a `@")
320 << kCtrlPktOverlayName << "` device in the module";
321 return failure();
322 }
323 }
324 if (failed(verifyReloadKeepsOverlay(referencedDevice, overlay)))
325 return failure();
326 preloadRef = FlatSymbolRefAttr::get(builder.getContext(), overlayName);
327 } else {
328 // Empty device PDI (triggers firmware reset)
329 AIE::DeviceOp emptyDevice = getOrCreateEmptyDevice(
330 moduleOp, referencedDevice.getDevice(), index % 2, loadPdiOp.getLoc());
331 preloadRef = FlatSymbolRefAttr::get(emptyDevice.getSymNameAttr());
332 }
333
334 builder.setInsertionPoint(loadPdiOp);
335
336 // Emit the preload load_pdi (either empty-device reset or ctrl_pkt_overlay).
337 NpuLoadPdiOp preload;
338 if (ctrlPkt) {
339 preload =
340 NpuLoadPdiOp::create(builder, loadPdiOp.getLoc(), preloadRef,
341 /*id=*/nullptr, /*size=*/nullptr,
342 /*address=*/nullptr,
343 /*expand_mode=*/
344 AIEX::ExpandModeAttr::get(builder.getContext(),
345 AIEX::ExpandMode::none));
346 } else {
347 NpuLoadPdiOp::create(builder, loadPdiOp.getLoc(), preloadRef,
348 loadPdiOp.getIdAttr(), loadPdiOp.getSizeAttr(),
349 loadPdiOp.getAddressAttr(),
350 /*expand_mode=*/
351 AIEX::ExpandModeAttr::get(builder.getContext(),
352 AIEX::ExpandMode::none));
353 }
354
355 // Step 2: generate and insert configuration ops.
356 auto outputType = ctrlPkt ? AIEToConfigurationOutputType::ControlPacket
357 : AIEToConfigurationOutputType::Transaction;
358 std::string prefix = ctrlPkt ? ("loadpdi_ctrlpkt_" + std::to_string(i) + "_")
359 : ("loadpdi_" + std::to_string(i));
360 if (failed(xilinx::AIE::generateAndInsertConfigOps(
361 builder, referencedDevice, /*clElfDir=*/"", outputType, prefix,
362 /*skipCtrlPktOverlay=*/ctrlPkt))) {
363 loadPdiOp.emitError("Failed to generate configuration operations");
364 return failure();
365 }
366 if (ctrlPkt &&
367 failed(verifyOverlayReachesTiles(
368 loadPdiOp, overlay,
369 {std::next(preload->getIterator()), loadPdiOp->getIterator()})))
370 return failure();
371
372 // Erase the original load_pdi operation
373 loadPdiOp.erase();
374
375 i++;
376
377 return success();
378}

380struct AIEExpandLoadPdiPass
381 : public xilinx::AIEX::impl::AIEExpandLoadPdiBase<AIEExpandLoadPdiPass> {
382 using AIEExpandLoadPdiBase::AIEExpandLoadPdiBase;
383
384 void getDependentDialects(DialectRegistry &registry) const override {
385 registry
386 .insert<memref::MemRefDialect, AIE::AIEDialect, AIEX::AIEXDialect>();
387 }
388
389 void runOnOperation() override {
390 auto module = getOperation();
391
392 // Collect all load_pdi operations in program order;
393 // need to collect once, then transform all collected ops;
394 // since the transform inserts a new preload load_pdi, we can't transform
395 // as we walk or it'd infinitely recurse.
396 SmallVector<NpuLoadPdiOp> loadPdiOps;
397
398 module.walk(
399 [&](NpuLoadPdiOp loadPdiOp) { loadPdiOps.push_back(loadPdiOp); });
400
401 // Per enclosing runtime sequence: the parity of its FIRST and LAST reset,
402 // plus the device type to reset. `transformLoadPdi` picks the empty device
403 // as `index % 2` over the module-wide op order, and that order includes
404 // ops this pass skips, so a sequence's first reset is not necessarily
405 // `@empty_0` and its parity cannot be recovered from a count. Captured
406 // BEFORE transforming, which erases the ops.
407 struct ResetParity {
408 unsigned firstParity = 0;
409 unsigned lastParity = 0;
410 bool seen = false;
411 AIE::AIEDevice device = {};
412 };
413 llvm::MapVector<AIE::RuntimeSequenceOp, ResetParity> resetsPerSequence;
414 for (auto [index, loadPdiOp] : llvm::enumerate(loadPdiOps)) {
415 auto deviceRefAttr = loadPdiOp.getDeviceRefAttr();
416 if (!deviceRefAttr)
417 continue;
418 if (loadPdiOp.getExpandMode().value_or(clCtrlPkt
419 ? AIEX::ExpandMode::ctrlpkt
420 : AIEX::ExpandMode::write32) !=
421 AIEX::ExpandMode::write32)
422 continue;
423 auto seq = loadPdiOp->getParentOfType<AIE::RuntimeSequenceOp>();
424 auto dev = module.lookupSymbol<AIE::DeviceOp>(deviceRefAttr);
425 if (!seq || !dev)
426 continue;
427 auto &entry = resetsPerSequence[seq];
428 if (!entry.seen) {
429 entry.firstParity = index % 2;
430 entry.seen = true;
431 }
432 entry.lastParity = index % 2;
433 entry.device = dev.getDevice();
434 }
435
436 // Map the legacy bool option to the new ExpandMode enum.
437 AIEX::ExpandMode defaultMode =
438 clCtrlPkt ? AIEX::ExpandMode::ctrlpkt : AIEX::ExpandMode::write32;
439
440 // Transform load_pdi ops
441 unsigned idx = 0;
442 for (auto loadPdiOp : loadPdiOps) {
443 if (failed(transformLoadPdi(loadPdiOp, module, idx, defaultMode))) {
444 signalPassFailure();
445 return;
446 }
447 idx++;
448 }
449
450 // The `index % 2` alternation above keeps two consecutive resets on
451 // different PDIs WITHIN a sequence. But the host re-runs the whole sequence
452 // on every dispatch, so the alternation has to hold across that boundary
453 // too: when a sequence's last reset lands on the same empty PDI as its
454 // first, the firmware caches the address and no-ops the next dispatch's
455 // first load, and the configuration that follows lands on a device that was
456 // never reset. That is silent -- the first dispatch is correct and later
457 // ones return non-deterministic garbage.
458 //
459 // Append one more reset, of the parity opposite the last one, so the
460 // sequence ends somewhere its own start will not repeat. The empty PDI is a
461 // few hundred bytes, so the cost is negligible next to the configuration it
462 // guards.
463 for (auto &[seq, info] : resetsPerSequence) {
464 if (!info.seen || info.firstParity != info.lastParity)
465 continue;
466 AIE::DeviceOp emptyDevice = getOrCreateEmptyDevice(
467 module, info.device, 1 - info.lastParity, seq.getLoc());
468 OpBuilder builder(seq.getContext());
469 Block &body = seq.getBody().front();
470 if (!body.empty() && body.back().hasTrait<OpTrait::IsTerminator>())
471 builder.setInsertionPoint(&body.back());
472 else
473 builder.setInsertionPointToEnd(&body);
474 NpuLoadPdiOp::create(
475 builder, seq.getLoc(),
476 FlatSymbolRefAttr::get(emptyDevice.getSymNameAttr()),
477 /*id=*/nullptr, /*size=*/nullptr, /*address=*/nullptr,
478 /*expand_mode=*/
479 AIEX::ExpandModeAttr::get(seq.getContext(), AIEX::ExpandMode::none));
480 }
481 }
482};

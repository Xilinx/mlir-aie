8// This pass expands `npu.load_pdi` operations that reference a device. There
9// are two output modes, controlled by the `ctrl-pkt` option:
10//
11// 1. Default (ctrl-pkt=false): replaces each `load_pdi @device` with
12// a. an empty device PDI load (`load_pdi @empty_N`), which causes the
13// firmware to reset the device, and
14// b. explicit `aiex.npu.write32`/`aiex.npu.blockwrite` configuration ops.
15// 2. With ctrl-pkt=true: replaces each `load_pdi @device` with
16// a. a `load_pdi @ctrl_pkt_overlay`, which configures the NPU to stream
17// further configuration as control packets, and
18// b. a sequence of `aiex.npu.control_packet` ops carrying the device's
19// configuration.
20//
21//===----------------------------------------------------------------------===//

49// Name of the overlay device loaded ahead of streaming configuration as
50// control packets. The device is expected to exist in the module (e.g.,
51// emitted by the `aie-generate-column-control-overlay` pass).
52static constexpr llvm::StringLiteral kCtrlPktOverlayName = "ctrl_pkt_overlay";
53// Name of an alternating copy of the overlay device, used to avoid PDI
54// address caching when the same overlay would otherwise be loaded twice in a
55// row (the firmware would treat the second load as a no-op).
56static constexpr llvm::StringLiteral kCtrlPktOverlayCopyName =
57 "ctrl_pkt_overlay_copy";
58
59// Look up `kCtrlPktOverlayName` and return a clone of it named
60// `kCtrlPktOverlayCopyName`, creating it once if needed. Returns nullptr on
61// error.
62static AIE::DeviceOp getOrCreateCtrlPktOverlayCopy(ModuleOp moduleOp,
63 Operation *errLocOp) {
64 if (auto existing =
65 moduleOp.lookupSymbol<AIE::DeviceOp>(kCtrlPktOverlayCopyName))
66 return existing;
67
68 auto orig = moduleOp.lookupSymbol<AIE::DeviceOp>(kCtrlPktOverlayName);
69 if (!orig) {
70 errLocOp->emitError("ctrl-pkt mode requires a `@")
71 << kCtrlPktOverlayName << "` device in the module";
72 return nullptr;
73 }
74
75 OpBuilder builder(moduleOp.getContext());
76 builder.setInsertionPointAfter(orig);
77 auto *cloned = builder.clone(*orig.getOperation());
78 auto clonedDev = cast<AIE::DeviceOp>(cloned);
79 clonedDev.setSymName(kCtrlPktOverlayCopyName);
80 return clonedDev;
81}

83// The empty device whose PDI load makes the firmware reset the array. `parity`
84// alternates so that two consecutive loads never name the same PDI (see
85// transformLoadPdi).
86static AIE::DeviceOp getOrCreateEmptyDevice(ModuleOp moduleOp,
87 AIE::AIEDevice deviceType,
88 unsigned parity) {
89 std::string emptyName = "empty_" + std::to_string(parity);
90 if (auto existing = moduleOp.lookupSymbol<AIE::DeviceOp>(emptyName))
91 return existing;
92
93 OpBuilder builder(moduleOp.getContext());
94 builder.setInsertionPointToStart(moduleOp.getBody());
95 auto loc = builder.getUnknownLoc();
96 auto emptyDevice = AIE::DeviceOp::create(builder, loc, deviceType,
97 builder.getStringAttr(emptyName));
98 emptyDevice.getRegion().emplaceBlock();
99 builder.setInsertionPointToEnd(&emptyDevice.getRegion().front());
100 AIE::EndOp::create(builder, loc);
101 return emptyDevice;
102}

104// Helper to transform a single load_pdi operation
105static LogicalResult transformLoadPdi(NpuLoadPdiOp loadPdiOp, ModuleOp moduleOp,
106 unsigned index,
107 AIEX::ExpandMode defaultMode) {
108 static unsigned long i = 0;
109 OpBuilder builder(loadPdiOp);
110
111 // Only process load_pdi ops that reference a device
112 auto deviceRefAttr = loadPdiOp.getDeviceRefAttr();
113 if (!deviceRefAttr) {
114 return success();
115 }
116
117 // Per-op annotation takes precedence; fall back to pass default
118 AIEX::ExpandMode mode = loadPdiOp.getExpandMode().value_or(defaultMode);
119 if (mode == AIEX::ExpandMode::none)
120 return success();
121 bool ctrlPkt = (mode == AIEX::ExpandMode::ctrlpkt);
122
123 auto referencedDevice = moduleOp.lookupSymbol<AIE::DeviceOp>(deviceRefAttr);
124 if (!referencedDevice) {
125 loadPdiOp.emitError("Referenced symbol '")
126 << deviceRefAttr.getValue() << "' is not a device";
127 return failure();
128 }
129
130 FlatSymbolRefAttr preloadRef;
131 if (ctrlPkt) {
132 // Overlay device PDI
133 // Alternate between the original overlay and a clone of it on every
134 // other load. Loading the same PDI twice in a row gets cached by the
135 // firmware (the second load becomes a no-op), so we need two distinct
136 // PDI addresses that carry the same overlay configuration.
137 StringRef overlayName =
138 (index % 2 == 0) ? kCtrlPktOverlayName : kCtrlPktOverlayCopyName;
139 if (index % 2 != 0) {
140 AIE::DeviceOp copy =
141 getOrCreateCtrlPktOverlayCopy(moduleOp, loadPdiOp.getOperation());
142 if (!copy)
143 return failure();
144 } else if (!moduleOp.lookupSymbol<AIE::DeviceOp>(kCtrlPktOverlayName)) {
145 loadPdiOp.emitError("ctrl-pkt mode requires a `@")
146 << kCtrlPktOverlayName << "` device in the module";
147 return failure();
148 }
149 preloadRef = FlatSymbolRefAttr::get(builder.getContext(), overlayName);
150 } else {
151 // Empty device PDI (triggers firmware reset)
152 AIE::DeviceOp emptyDevice = getOrCreateEmptyDevice(
153 moduleOp, referencedDevice.getDevice(), index % 2);
154 preloadRef = FlatSymbolRefAttr::get(emptyDevice.getSymNameAttr());
155 }
156
157 builder.setInsertionPoint(loadPdiOp);
158
159 // Emit the preload load_pdi (either empty-device reset or ctrl_pkt_overlay).
160 if (ctrlPkt) {
161 NpuLoadPdiOp::create(builder, loadPdiOp.getLoc(), preloadRef,
162 /*id=*/nullptr, /*size=*/nullptr,
163 /*address=*/nullptr,
164 /*expand_mode=*/
165 AIEX::ExpandModeAttr::get(builder.getContext(),
166 AIEX::ExpandMode::none));
167 } else {
168 NpuLoadPdiOp::create(builder, loadPdiOp.getLoc(), preloadRef,
169 loadPdiOp.getIdAttr(), loadPdiOp.getSizeAttr(),
170 loadPdiOp.getAddressAttr(),
171 /*expand_mode=*/
172 AIEX::ExpandModeAttr::get(builder.getContext(),
173 AIEX::ExpandMode::none));
174 }
175
176 // Step 2: generate and insert configuration ops.
177 auto outputType = ctrlPkt ? AIEToConfigurationOutputType::ControlPacket
178 : AIEToConfigurationOutputType::Transaction;
179 std::string prefix = ctrlPkt ? ("loadpdi_ctrlpkt_" + std::to_string(i) + "_")
180 : ("loadpdi_" + std::to_string(i));
181 if (failed(xilinx::AIE::generateAndInsertConfigOps(
182 builder, referencedDevice, /*clElfDir=*/"", outputType, prefix,
183 /*skipCtrlPktOverlay=*/ctrlPkt))) {
184 loadPdiOp.emitError("Failed to generate configuration operations");
185 return failure();
186 }
187
188 // Erase the original load_pdi operation
189 loadPdiOp.erase();
190
191 i++;
192
193 return success();
194}

196struct AIEExpandLoadPdiPass
197 : public xilinx::AIEX::impl::AIEExpandLoadPdiBase<AIEExpandLoadPdiPass> {
198 using AIEExpandLoadPdiBase::AIEExpandLoadPdiBase;
199
200 void getDependentDialects(DialectRegistry &registry) const override {
201 registry
202 .insert<memref::MemRefDialect, AIE::AIEDialect, AIEX::AIEXDialect>();
203 }
204
205 void runOnOperation() override {
206 auto module = getOperation();
207
208 // Collect all load_pdi operations in program order;
209 // need to collect once, then transform all collected ops;
210 // since the transform inserts a new preload load_pdi, we can't transform
211 // as we walk or it'd infinitely recurse.
212 SmallVector<NpuLoadPdiOp> loadPdiOps;
213
214 module.walk(
215 [&](NpuLoadPdiOp loadPdiOp) { loadPdiOps.push_back(loadPdiOp); });
216
217 // Per enclosing runtime sequence: the parity of its FIRST and LAST reset,
218 // plus the device type to reset. `transformLoadPdi` picks the empty device
219 // as `index % 2` over the module-wide op order, and that order includes
220 // ops this pass skips, so a sequence's first reset is not necessarily
221 // `@empty_0` and its parity cannot be recovered from a count. Captured
222 // BEFORE transforming, which erases the ops.
223 struct ResetParity {
224 unsigned firstParity = 0;
225 unsigned lastParity = 0;
226 bool seen = false;
227 AIE::AIEDevice device = {};
228 };
229 llvm::MapVector<AIE::RuntimeSequenceOp, ResetParity> resetsPerSequence;
230 for (auto [index, loadPdiOp] : llvm::enumerate(loadPdiOps)) {
231 auto deviceRefAttr = loadPdiOp.getDeviceRefAttr();
232 if (!deviceRefAttr)
233 continue;
234 if (loadPdiOp.getExpandMode().value_or(clCtrlPkt
235 ? AIEX::ExpandMode::ctrlpkt
236 : AIEX::ExpandMode::write32) !=
237 AIEX::ExpandMode::write32)
238 continue;
239 auto seq = loadPdiOp->getParentOfType<AIE::RuntimeSequenceOp>();
240 auto dev = module.lookupSymbol<AIE::DeviceOp>(deviceRefAttr);
241 if (!seq || !dev)
242 continue;
243 auto &entry = resetsPerSequence[seq];
244 if (!entry.seen) {
245 entry.firstParity = index % 2;
246 entry.seen = true;
247 }
248 entry.lastParity = index % 2;
249 entry.device = dev.getDevice();
250 }
251
252 // Map the legacy bool option to the new ExpandMode enum.
253 AIEX::ExpandMode defaultMode =
254 clCtrlPkt ? AIEX::ExpandMode::ctrlpkt : AIEX::ExpandMode::write32;
255
256 // Transform load_pdi ops
257 unsigned idx = 0;
258 for (auto loadPdiOp : loadPdiOps) {
259 if (failed(transformLoadPdi(loadPdiOp, module, idx, defaultMode))) {
260 signalPassFailure();
261 return;
262 }
263 idx++;
264 }
265
266 // The `index % 2` alternation above keeps two consecutive resets on
267 // different PDIs WITHIN a sequence. But the host re-runs the whole sequence
268 // on every dispatch, so the alternation has to hold across that boundary
269 // too: when a sequence's last reset lands on the same empty PDI as its
270 // first, the firmware caches the address and no-ops the next dispatch's
271 // first load, and the configuration that follows lands on a device that was
272 // never reset. That is silent -- the first dispatch is correct and later
273 // ones return non-deterministic garbage.
274 //
275 // Append one more reset, of the parity opposite the last one, so the
276 // sequence ends somewhere its own start will not repeat. The empty PDI is a
277 // few hundred bytes, so the cost is negligible next to the configuration it
278 // guards.
279 for (auto &[seq, info] : resetsPerSequence) {
280 if (!info.seen || info.firstParity != info.lastParity)
281 continue;
282 AIE::DeviceOp emptyDevice =
283 getOrCreateEmptyDevice(module, info.device, 1 - info.lastParity);
284 OpBuilder builder(seq.getContext());
285 Block &body = seq.getBody().front();
286 if (!body.empty() && body.back().hasTrait<OpTrait::IsTerminator>())
287 builder.setInsertionPoint(&body.back());
288 else
289 builder.setInsertionPointToEnd(&body);
290 NpuLoadPdiOp::create(
291 builder, seq.getLoc(),
292 FlatSymbolRefAttr::get(emptyDevice.getSymNameAttr()),
293 /*id=*/nullptr, /*size=*/nullptr, /*address=*/nullptr,
294 /*expand_mode=*/
295 AIEX::ExpandModeAttr::get(seq.getContext(), AIEX::ExpandMode::none));
296 }
297 }
298};

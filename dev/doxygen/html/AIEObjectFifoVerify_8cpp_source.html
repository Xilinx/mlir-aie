25struct SegmentActors {
26 SmallVector<Operation *> fillers;
27 SmallVector<Operation *> drainers;
28};
29
30struct AIEObjectFifoVerifyPass
31 : public xilinx::AIE::impl::AIEObjectFifoVerifyBase<
32 AIEObjectFifoVerifyPass> {
33
34 /// Every segment needs an actor at each end, so that what one writes another
35 /// reads. A segment may go unfilled when the pool's objects start full.
36 LogicalResult verifyActors(ObjectFifoPoolOp pool,
37 ArrayRef<SegmentActors> actors) {
38 bool external = llvm::any_of(pool.getBufferOps(), [](BufferLike buffer) {
39 return isa<ExternalBufferOp>(buffer.getOperation());
40 });
41 for (auto [index, segment] : llvm::enumerate(actors)) {
42 if (segment.fillers.size() > 1) {
43 return pool.emitOpError("segment ")
44 << index << " is filled by more than one endpoint";
45 }
46 if (segment.drainers.size() > 1) {
47 return pool.emitOpError("segment ")
48 << index << " is drained by more than one endpoint";
49 }
50 // The host works an external pool's buffers, so it stands in for the end
51 // this device does not program.
52 bool hostDrains = external && !segment.fillers.empty();
53 bool hostFills = external && !segment.drainers.empty();
54 if (segment.drainers.empty() && !hostDrains) {
55 return pool.emitOpError("segment ") << index << " has no drainer";
56 }
57 if (segment.fillers.empty() && !pool.getInitValues() && !hostFills) {
58 return pool.emitOpError("segment ") << index << " has no filler";
59 }
60 }
61 return success();
62 }
63
64 /// An endpoint that is not connected carries data nowhere.
65 LogicalResult verifyFlows(DeviceOp device) {
66 DenseMap<StringRef, int> appearances;
67 for (auto flow : device.getOps<RouteOp>()) {
68 appearances[flow.getSource()]++;
69 for (auto dest : flow.getDestinations().getAsRange<FlatSymbolRefAttr>()) {
70 appearances[dest.getValue()]++;
71 }
72 }
73
74 for (auto endpoint : device.getOps<RouteEndpoint>()) {
75 int count =
76 appearances.lookup(cast<SymbolOpInterface>(*endpoint).getName());
77 if (count == 0) {
78 return endpoint->emitOpError("is not connected by any flow");
79 }
80 if (count > 1) {
81 return endpoint->emitOpError(
82 "drives one channel, so at most one flow may name it, but "
83 "it is named ")
84 << count << " times";
85 }
86 }
87 return success();
88 }
89
90 /// A loop body that releases more than it acquires underflows the held count
91 /// as it repeats, whatever the trip count. Accesses split across nesting
92 /// levels are excluded: balancing them needs the inner trip counts. This runs
93 /// before lowering because afterwards the imbalance survives only as
94 /// loop-carried lock values, which no scan can decide.
95 LogicalResult verifyOverRelease(DeviceOp device) {
96 for (auto coreOp : device.getOps<CoreOp>()) {
97 WalkResult result = coreOp.walk([&](scf::ForOp forOp) {
98 auto directlyIn = [&](Operation *op) {
99 return op->getParentOfType<scf::ForOp>() == forOp;
100 };
101 DenseMap<StringRef, int64_t> acquired;
102 DenseMap<StringRef, int64_t> released;
103 DenseMap<StringRef, Operation *> blame;
104 DenseSet<StringRef> spansNestedLoop;
105
106 forOp.getBody()->walk([&](ObjectFifoAcquireOp a) {
107 StringRef key = a.getObjFifoName();
108 if (!directlyIn(a)) {
109 spansNestedLoop.insert(key);
110 } else {
111 acquired[key] += a.acqNumber();
112 blame.try_emplace(key, a);
113 }
114 });
115 forOp.getBody()->walk([&](ObjectFifoReleaseOp r) {
116 StringRef key = r.getObjFifoName();
117 if (!directlyIn(r)) {
118 spansNestedLoop.insert(key);
119 } else {
120 released[key] += r.relNumber();
121 blame.try_emplace(key, r);
122 }
123 });
124
125 for (auto &[key, count] : released) {
126 if (spansNestedLoop.contains(key) || count <= acquired.lookup(key)) {
127 continue;
128 }
129 blame.lookup(key)->emitOpError(
130 "cannot release more elements than are already acquired");
131 return WalkResult::interrupt();
132 }
133 return WalkResult::advance();
134 });
135 if (result.wasInterrupted()) {
136 return failure();
137 }
138 }
139 return success();
140 }
141
142 void runOnOperation() override {
143 DeviceOp device = getOperation();
144
145 DenseMap<Operation *, SmallVector<SegmentActors>> actorsPerPool;
146 for (auto pool : device.getOps<ObjectFifoPoolOp>()) {
147 actorsPerPool[pool].resize(pool.getSegmentOps().size());
148 }
149
150 auto record = [&](auto endpoint) {
151 ObjectFifoPoolOp pool = endpoint.getPoolOp();
152 if (!pool) {
153 return;
154 }
155 auto &actors = actorsPerPool[pool];
156 std::vector<ObjectFifoSegmentOp> all = pool.getSegmentOps();
157 for (ObjectFifoSegmentOp segment : endpoint.getSelectedSegments()) {
158 size_t index = llvm::find(all, segment) - all.begin();
159 (endpoint.drains() ? actors[index].drainers : actors[index].fillers)
160 .push_back(endpoint);
161 }
162 };
163
164 for (auto endpoint : device.getOps<ObjectFifoCoreEndpointOp>()) {
165 record(endpoint);
166 }
167 for (auto endpoint : device.getOps<ObjectFifoDmaEndpointOp>()) {
168 record(endpoint);
169 }
170
171 for (auto pool : device.getOps<ObjectFifoPoolOp>()) {
172 if (failed(verifyActors(pool, actorsPerPool[pool]))) {
173 return signalPassFailure();
174 }
175 }
176
177 if (failed(verifyFlows(device)) || failed(verifyOverRelease(device))) {
178 return signalPassFailure();
179 }
180 }
181};

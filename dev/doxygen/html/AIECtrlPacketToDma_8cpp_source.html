32static ShimDMAAllocationOp getCtrlPktShimAlloc(DeviceOp device, int col) {
33 AIEDialect::IsCtrlPktOverlayAttrHelper overlay(device.getContext());
34 for (auto alloc : device.getOps<AIE::ShimDMAAllocationOp>())
35 if (overlay.isAttrPresent(alloc) &&
36 alloc.getChannelDir() == AIE::DMAChannelDir::MM2S &&
37 alloc.getTileOp().colIndex() == col)
38 return alloc;
39 return {};
40}

42struct AIECtrlPacketInferTilesPass
43 : xilinx::AIEX::impl::AIECtrlPacketInferTilesBase<
44 AIECtrlPacketInferTilesPass> {
45 void runOnOperation() override {
46 DeviceOp device = getOperation();
47 const auto &targetModel = device.getTargetModel();
48 OpBuilder devBuilder = OpBuilder::atBlockBegin(device.getBody());
49
50 auto sequenceOps = device.getOps<AIE::RuntimeSequenceOp>();
51 for (auto f : sequenceOps) {
52 for (auto ctrlPktOp : f.getOps<NpuControlPacketOp>()) {
53 auto tOp = TileOp::getOrCreate(devBuilder, device,
54 (int)ctrlPktOp.getColumnFromAddr(),
55 (int)ctrlPktOp.getRowFromAddr());
56 // Assign controller id
57 auto tileIDMap = targetModel.getTileToControllerIdMap(true);
58 if (tOp->hasAttr("controller_id"))
59 continue;
60 auto pktInfoAttr = AIE::PacketInfoAttr::get(
61 tOp->getContext(), /*pkt_type*/ 0,
62 /*pkt_id*/ tileIDMap[{tOp.colIndex(), tOp.rowIndex()}]);
63 tOp->setAttr("controller_id", pktInfoAttr);
64 }
65 }
66 }

70 : xilinx::AIEX::impl::AIECtrlPacketToDmaBase<AIECtrlPacketToDmaPass> {

88 static DMAConfigureTaskForOp emitTask(OpBuilder &builder, Location loc,
89 Value buffer, const BatchInfo &batch) {
90 auto task = DMAConfigureTaskForOp::create(
91 builder, loc, builder.getIndexType(),
92 SymbolRefAttr::get(builder.getContext(), batch.shimDmaAllocName),
93 /*issue_token=*/true, /*repeat_count=*/0,
94 /*repeat_count_val=*/Value());
95 OpBuilder::InsertionGuard guard(builder);
96 builder.createBlock(&task.getBody());
97 AIE::DMABDOp::create(builder, loc, buffer,
98 static_cast<int>(batch.startOffset),
99 static_cast<int>(batch.totalSize));
100 AIE::EndOp::create(builder, loc);
101 return task;
102 }

104 void runOnOperation() override {
105 DeviceOp device = getOperation();
106 const auto &targetModel = device.getTargetModel();
107 auto *ctx = device->getContext();
108 auto loc = device->getLoc();
109 SymbolTable symbols(device);
110
111 if (targetModel.getTargetArch() == AIEArch::AIE1)
112 return; // Disable this pass for AIE1; AIE1 support NYI.
113
114 device.walk([](DMAAwaitTaskOp await) {
115 if (isa_and_nonnull<NpuControlPacketOp, ControlPacketBarrierOp>(
116 await->getNextNode()))
117 await.removeDeferCompletionAttr();
118 });
119
120 auto sequenceOps = device.getOps<AIE::RuntimeSequenceOp>();
121 for (auto f : sequenceOps) {
122 if (f.getBody().empty())
123 continue;
124 Block &entry = f.getBody().front();
125 auto controlPacketOps = f.getOps<AIEX::NpuControlPacketOp>();
126 if (controlPacketOps.empty())
127 continue;
128
129 auto ctrlPktMemrefType = MemRefType::get(
130 ShapedType::kDynamic, IntegerType::get(ctx, 32), nullptr, nullptr);
131 auto packetData = entry.addArgument(ctrlPktMemrefType, loc);
132 std::vector<BatchInfo> batches;
133
134 int64_t ddrOffset = 0;
135
136 NpuControlPacketOp previous;
137 bool previousIntervalMulticast = false;
138 for (auto [index, interval] : llvm::enumerate(packetIntervals(entry))) {
139 llvm::DenseSet<uint32_t> multicastWords;
140 llvm::DenseMap<TileID, uint32_t> multicastRoutes;
141 bool disjointMulticastRoutes = true;
142 bool disjointMulticast = interval.ordinary;
143 auto groups = controlPacketMulticasts(
144 device->getParentOfType<ModuleOp>(), interval.device);
145 if (failed(groups))
146 return signalPassFailure();
147 for (NpuControlPacketOp packet : interval.packets)
148 if (auto id = packet.getMulticastId()) {
149 if (*id >= groups->size()) {
150 packet.emitOpError("multicast_id ")
151 << *id << " has no declaration in the active overlay";
152 return signalPassFailure();
153 }
154 for (auto tile : (*groups)[*id].tiles.getAsRange<AIE::TileAttr>()) {
155 auto destination = tile.getTileID();
156 auto [found, inserted] =
157 multicastRoutes.try_emplace(destination, *id);
158 disjointMulticastRoutes &= inserted || found->second == *id;
159 uint32_t address =
160 (destination.col << targetModel.getColumnShift()) |
161 (destination.row << targetModel.getRowShift()) |
162 packet.getLocalAddress();
163 unsigned words = packet.getData() ? packet.getData()->size()
164 : *packet.getLength();
165 for (unsigned word = 0; word < words; ++word)
166 multicastWords.insert(address + 4 * word);
167 }
168 }
169 for (NpuControlPacketOp packet : interval.packets)
170 if (!packet.getMulticastId() && packet.getData())
171 for (size_t word = 0; word < packet.getData()->size(); ++word)
172 disjointMulticast &=
173 !multicastWords.contains(packet.getAddress() + 4 * word);
174 bool new_batch = !previous || previous.getOpcode() != 0 ||
175 batches.back().delivery != interval.delivery ||
176 previousIntervalMulticast;
177 for (NpuControlPacketOp ctrlPktOp : interval.packets) {
178 int col = ctrlPktOp.getColumnFromAddr();
179 int row = ctrlPktOp.getRowFromAddr();
180
181 // A read's length describes its response, not an outgoing payload.
182 int64_t ctrlPktSize = 2; // ctrl info word + packet header
183 if (auto data = ctrlPktOp.getData())
184 ctrlPktSize += data->size();
185
186 // Read requests must end their DMA transfer. A batch is one stream
187 // packet, routed by its first header, so its packets share their
188 // destination.
189 bool batchable = ctrlPktOp.getOpcode() == 0;
190 std::optional<int32_t> multicast;
191 if (auto id = ctrlPktOp.getMulticastId())
192 multicast = *id;
193 if (targetModel.getTargetArch() == AIEArch::AIE2p && batchable &&
194 !new_batch && batches.back().tileId == TileID{col, row} &&
195 batches.back().multicast == multicast) {
196 batches.back().totalSize += ctrlPktSize;
197 batches.back().interval = index;
198 batches.back().ordinary &= interval.ordinary;
199 batches.back().independent &= interval.independent;
200 batches.back().disjointMulticast &= disjointMulticast;
201 batches.back().disjointMulticastRoutes &= disjointMulticastRoutes;
202 } else {
203 ShimDMAAllocationOp alloc;
204 if (multicast) {
205 auto source = (*groups)[*multicast].allocation;
206 alloc = symbols.lookup<ShimDMAAllocationOp>(source.getSymName());
207 if (!alloc || alloc.getChannelDir() != DMAChannelDir::MM2S ||
208 alloc.getTileOp().getTileID() !=
209 source.getTileOp().getTileID() ||
210 alloc.getChannelIndex() != source.getChannelIndex()) {
211 ctrlPktOp.emitOpError("multicast allocation @")
212 << source.getSymName()
213 << " requires the matching imported shim DMA allocation";
214 return signalPassFailure();
215 }
216 } else {
217 alloc = getCtrlPktShimAlloc(device, col);
218 }
219 if (!alloc) {
220 ctrlPktOp.emitOpError("column ")
221 << col
222 << " has no control packet shim DMA allocation; run "
223 "-aie-generate-column-control-overlay with "
224 "configuration=true, then -aie-import-overlay first";
225 return signalPassFailure();
226 }
227 batches.push_back({TileID{col, row}, ddrOffset, ctrlPktSize,
228 alloc.getSymName().str(), ctrlPktOp,
229 interval.delivery, static_cast<unsigned>(index),
230 interval.ordinary, interval.independent,
231 disjointMulticast, disjointMulticastRoutes,
232 multicast});
233 }
234 new_batch = !batchable;
235 ddrOffset += ctrlPktSize;
236 previous = ctrlPktOp;
237 }
238 previousIntervalMulticast =
239 llvm::any_of(interval.packets, [](NpuControlPacketOp packet) {
240 return packet.getMulticastId().has_value();
241 });
242 }
243
244 auto batchIt = batches.begin();
245 bool pendingMulticast = false;
246 for (; batchIt != batches.end(); ++batchIt) {
247 OpBuilder builder(batchIt->first);
248 DMAConfigureTaskForOp task =
249 emitTask(builder, batchIt->first->getLoc(), packetData, *batchIt);
250 DMAStartTaskOp::create(builder, loc, task);
251 auto next = std::next(batchIt);
252 bool independent = next != batches.end() &&
253 ((batchIt->independent && next->independent) ||
254 (batchIt->ordinary && next->ordinary &&
255 next->interval == batchIt->interval));
256 pendingMulticast |= batchIt->multicast.has_value();
257 bool defer =
258 next != batches.end() && next->delivery == batchIt->delivery &&
259 batchIt->independent == next->independent &&
260 (batchIt->independent || next->interval == batchIt->interval) &&
261 (!pendingMulticast || next->interval == batchIt->interval) &&
262 ((!batchIt->multicast && !next->multicast) ||
263 (((batchIt->independent && next->independent) ||
264 (batchIt->ordinary && next->ordinary)) &&
265 next->interval == batchIt->interval &&
266 ((batchIt->multicast && next->multicast &&
267 batchIt->disjointMulticastRoutes) ||
268 (!next->multicast && batchIt->disjointMulticast)))) &&
269 ((chainBds && next->shimDmaAllocName == batchIt->shimDmaAllocName &&
270 (independent || next->tileId == batchIt->tileId)) ||
271 (parallelShims && independent));
272 DMAAwaitTaskOp::create(builder, loc, task,
273 defer ? builder.getUnitAttr() : UnitAttr{});
274 pendingMulticast &= defer;
275 }
276 for (NpuControlPacketOp packet :
277 llvm::make_early_inc_range(controlPacketOps))
278 packet.erase();
279 }
280 device.walk([](ControlPacketBarrierOp barrier) { barrier.erase(); });
281 }
282};

35void AIEX::addLoweringStage(OpPassManager &pm, bool dynamicObjFifos,
36 bool packetSwObjFifos, bool ctrlPktOverlay,
37 bool loadPdiToCtrlPkt, bool skipObjectFifoVerify,
38 llvm::ArrayRef<int> coverColumns,
39 bool adaptiveOverlay, bool checkQueueSpace,
40 unsigned multicastSpan) {
41 pm.addPass(createLowerAffinePass());
42 pm.addPass(createAIECanonicalizeDevicePass());
43 // The verifier checks a parameter-waiting Worker's fifo volume per dispatch
44 // from the aiex.wait_scratchpad_parameters and parameter syncs that the
45 // scratchpad lowering below replaces. Split is idempotent, so the lowering
46 // after it splits again and does not verify twice.
47 if (!skipObjectFifoVerify) {
48 OpPassManager &verify = pm.nest<DeviceOp>();
49 verify.addPass(createAIEObjectFifoSplitPass());
50 verify.addPass(createAIEObjectFifoVerifyPass());
51 }
52 // Before lock IDs and buffer addresses, which its new locks and buffers need.
53 pm.addPass(createAIELowerScratchpadParametersPass());
54
55 pm.nest<DeviceOp>().addPass(createAIEObjectFifoSplitPass());
56 AIEGenerateColumnControlOverlayOptions overlay;
57 overlay.configuration = ctrlPktOverlay || loadPdiToCtrlPkt;
58 overlay.adaptive = adaptiveOverlay;
59 overlay.multicastSpan = loadPdiToCtrlPkt ? multicastSpan : 0;
60 overlay.coverColumns = llvm::to_vector(coverColumns);
61 overlay.packetSwitched = packetSwObjFifos;
62 pm.addPass(createAIEGenerateColumnControlOverlayPass(overlay));
63 pm.addPass(createAIEImportOverlayPass());
64 pm.addPass(createAIEAnnotateShimSharingPass());
65 OpPassManager &dpm = pm.nest<DeviceOp>();
66 dpm.addPass(createAIEAssignPacketIdsPass(packetSwObjFifos, ctrlPktOverlay));
67 dpm.addPass(createAIEObjectFifoAllocatePass());
68 AIEObjectFifoInsertCoreQueuePushesOptions pushOptions;
69 pushOptions.checkQueueSpace = checkQueueSpace;
70 dpm.addPass(createAIEObjectFifoInsertCoreQueuePushesPass(pushOptions));
71 dpm.addPass(createAIEObjectFifoLowerDMAsPass());
72 dpm.addPass(createAIEObjectFifoLowerCoresPass());
73 dpm.addPass(createAIEObjectFifoErasePoolsPass());
74 AIEObjectFifoUnrollOptions unroll;
75 unroll.clDefaultDynamic = dynamicObjFifos;
76 dpm.addPass(createAIEObjectFifoUnrollPass(unroll));
77 dpm.addPass(createAIENormalizeDmaBdDimsPass());
78 dpm.addPass(createAIEAssignLockIDsPass());
79 dpm.addPass(createAIEReserveRuntimeBDIDsPass());
80 dpm.addPass(createAIEAssignBufferDescriptorIDsPass());
81 dpm.addPass(createAIELowerCoreIssuedConfigurationPass());
82 dpm.addPass(createAIELowerCascadeFlowsPass());
83 dpm.addPass(createAIEBroadcastPacketPass());
84 dpm.addPass(createAIELowerMulticastPass());
85 pm.nest<DeviceOp>().addPass(createAIEAssignTileCtrlIDsPass());
86
87 // A buffer's name becomes a symbol in its core's object.
88 pm.nest<DeviceOp>().addPass(createAIEPrepareBuffersPass());
89}

97void AIEX::addRoutingStage(OpPassManager &pm, bool allowDeadlockProne) {
98 pm.nest<DeviceOp>().addPass(createAIESplitFlowViasPass());
99 OpPassManager &dpm = pm.nest<DeviceOp>();
100 AIERoutePathfinderFlowsOptions options;
101 options.clAllowDeadlockProne = allowDeadlockProne;
102 dpm.addPass(createAIEPathfinderPass(options));

105void AIEX::registerAIEResourceStagesPipeline() {
106 PassPipelineRegistration<AIEResourceStagesOptions>(
107 "aie-resource-stages",
108 "aiecc's placement, trace, lowering, buffer address and routing stages",
109 [](OpPassManager &pm, const AIEResourceStagesOptions &options) {
110 AIEPlaceTilesOptions place;
111 place.clPlacerType = options.placer;
112 place.clCoresPerCol = options.coresPerCol;
113 place.clSASeed = options.saSeed;
114 place.clSAEffort = options.saEffort;
115 place.clPlacementBudget = options.placementBudget;
116 place.clCtrlPktOverlay =
117 options.ctrlPktOverlay || options.loadPdiToCtrlPkt;
118 place.clCtrlPktOverlayAdaptive = options.adaptiveOverlay;
119 place.clPacketSwObjectFifos = options.packetSwObjFifos;
120 addPlacementStage(pm, place);
121 addTraceStage(pm);
122 addLoweringStage(pm, options.dynamicObjFifos, options.packetSwObjFifos,
123 options.ctrlPktOverlay || options.loadPdiToCtrlPkt,
124 options.loadPdiToCtrlPkt, options.skipObjectFifoVerify,
125 {}, options.adaptiveOverlay, /*checkQueueSpace=*/true,
126 options.multicastSpan);
127 addBufferAddressStage(pm, options.placementBudget);
128 addRoutingStage(pm, options.allowDeadlockProneRouting);
129 });
130}

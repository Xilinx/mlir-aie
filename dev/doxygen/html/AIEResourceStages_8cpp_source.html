24void AIEX::addTraceStage(OpPassManager &pm) {

34void AIEX::addLoweringStage(OpPassManager &pm, bool dynamicObjFifos,
35 bool packetSwObjFifos, bool ctrlPktOverlay,
36 bool loadPdiToCtrlPkt, bool skipObjectFifoVerify) {
37 pm.addPass(createLowerAffinePass());
38 pm.addPass(createAIECanonicalizeDevicePass());
39 // Before lock IDs and buffer addresses, which its new locks and buffers need.
40 pm.addPass(createAIELowerScratchpadParametersPass());
41
42 // With the overlay requested, it claims its shim DMA channels before
43 // objectFIFO lowering works around them; otherwise it runs after tile
44 // control IDs are assigned.
45 AIEGenerateColumnControlOverlayOptions overlay;
46 overlay.clRouteShimDmaToTileCTRL = ctrlPktOverlay;
47 overlay.clEmitStandaloneOverlay = ctrlPktOverlay && loadPdiToCtrlPkt;
48 if (ctrlPktOverlay)
49 pm.addPass(createAIEGenerateColumnControlOverlayPass(overlay));
50
51 OpPassManager &dpm = pm.nest<DeviceOp>();
52 addObjectFifoLoweringPasses(dpm, packetSwObjFifos, skipObjectFifoVerify);
53 AIEObjectFifoUnrollOptions unroll;
54 unroll.clDefaultDynamic = dynamicObjFifos;
55 dpm.addPass(createAIEObjectFifoUnrollPass(unroll));
56 dpm.addPass(createAIENormalizeDmaBdDimsPass());
57 dpm.addPass(createAIEAssignLockIDsPass());
58 dpm.addPass(createAIEReserveRuntimeBDIDsPass());
59 dpm.addPass(createAIEAssignBufferDescriptorIDsPass());
60 dpm.addPass(createAIELowerCascadeFlowsPass());
61 dpm.addPass(createAIEBroadcastPacketPass());
62 dpm.addPass(createAIELowerMulticastPass());
63 dpm.addPass(createAIEAssignTileCtrlIDsPass());
64
65 if (!ctrlPktOverlay)
66 pm.addPass(createAIEGenerateColumnControlOverlayPass(overlay));
67
68 // A buffer's name becomes a symbol in its core's object.
69 pm.nest<DeviceOp>().addPass(createAIEPrepareBuffersPass());
70}

86void AIEX::registerAIEResourceStagesPipeline() {
87 PassPipelineRegistration<AIEResourceStagesOptions>(
88 "aie-resource-stages",
89 "aiecc's placement, trace, lowering, buffer address and routing stages",
90 [](OpPassManager &pm, const AIEResourceStagesOptions &options) {
91 AIEPlaceTilesOptions place;
92 place.clPlacerType = options.placer;
93 place.clCoresPerCol = options.coresPerCol;
94 place.clSASeed = options.saSeed;
95 place.clSAEffort = options.saEffort;
96 addPlacementStage(pm, place);
97 addTraceStage(pm);
98 addLoweringStage(pm, options.dynamicObjFifos, options.packetSwObjFifos,
99 options.ctrlPktOverlay || options.loadPdiToCtrlPkt,
100 options.loadPdiToCtrlPkt,
101 options.skipObjectFifoVerify);
102 addBufferAddressStage(pm, options.placementBudget);
103 addRoutingStage(pm, options.allowDeadlockProneRouting);
104 });
105}

24std::string AIE::joinNames(llvm::ArrayRef<std::string> names, size_t shown) {
25 size_t more = names.size() > shown ? names.size() - shown : 0;
26 llvm::ArrayRef<std::string> listed = names.drop_back(more);
27 std::string list;
28 for (auto [i, name] : llvm::enumerate(listed))
29 list += (i == 0 ? ""
30 : i + 1 == listed.size() && !more ? " and "
31 : ", ") +
32 name;
33 if (more)
34 list += " and " + std::to_string(more) + " more";
35 return list;
36}

564// Invoke `fn(peer, thisIsFirst)` for each edge in `adjacency` that
565// mentions `op`. `peer` is the OTHER endpoint of the edge; `thisIsFirst`
566// is true when `op` is `edge.first`.
567template <typename F>
568inline void forEachPeer(mlir::Operation *op, const Placer::Adjacency &adjacency,
569 F &&fn) {
570 auto it = adjacency.tileToEdges.find(op);
571 if (it == adjacency.tileToEdges.end())
572 return;
573 for (unsigned idx : it->second) {
574 auto [first, second] = adjacency.edges[idx];
575 bool thisIsFirst = first.getOperation() == op;
576 TileLike peer = thisIsFirst ? second : first;
577 fn(peer, thisIsFirst);
578 }
579}

581// Whether two compute tiles can share one memory module: unlike buffer
582// ownership (where only the consumer must reach the owner's module),
583// sharing has no owner/consumer asymmetry, so either tile reaching the
584// other's module is enough. isLegalMemAffinity itself is directional (see
585// AIETargetModel::getSharedMemory's None/First/Second/Either), so compute-
586// peer neighbor-ness -- what `--aie-objectfifo-split`'s auto transport
587// actually tests -- has to check both.
588inline bool isSharedComputeMemory(const AIETargetModel &targetModel, TileID a,
589 TileID b) {
590 return targetModel.isLegalMemAffinity(a.col, a.row, b.col, b.row) ||
591 targetModel.isLegalMemAffinity(b.col, b.row, a.col, a.row);
592}

594// Invoke `fn(TileID)` on every other CoreTile that can share compute-peer
595// memory with `at` per the target model.
596template <typename F>
597inline void forEachMemAffinityNeighbor(const AIETargetModel &targetModel,
598 TileID at, F &&fn) {
599 for (int col = 0; col < targetModel.columns(); ++col)
600 for (int row = 0; row < targetModel.rows(); ++row)
601 if (TileID other{col, row};
602 other != at &&
603 targetModel.getTileType(col, row) == AIETileType::CoreTile &&
604 isSharedComputeMemory(targetModel, at, other))
605 fn(other);
606}

608// Sequential placement algorithm
609//
610// Greedy, single-pass placer that maps each `aie.logical_tile` (LTO) to a
611// physical (col, row). Four phases:
612//
613// 1. Collect LogicalTileOp / ObjectFifo* / Cascade* / Flow* / PacketFlow* /
614// Route from the device.
615// 2. Build per-LTO constraints: buffer/cascade/compute-peer adjacencies,
616// channel requirements, and needNeighborIn/Out (the minimum number of
617// compute peers per LTO that MUST land on a shared-L1 neighbor to fit
618// the per-tile DMA budget). Group the compute LTOs by the non-core
619// tile each shares the most flows with (buildCoreGroups).
620// 3. Place compute tiles. Make each core group contiguous, then sort by
621// constraint level (pinned -> partial -> unpinned) and by descending
622// placementPriority. For each LTO: validate buffer/cascade/compute-peer
623// constraints and channel budget, then claim the tile, trying first
624// the columns that keep its group together. Unpinned candidates are
625// tried in two passes: first the slots with column stream capacity and
626// not soft-reserved for another demand-bearing LTO's peers, then a
627// fallback accepting overloaded columns and reserved slots. Placing a
628// demand-bearing LTO soft-reserves its mem-affinity neighbor slots for
629// that LTO's compute peers.
630// 4. Place each remaining non-core (mem/shim) LTO near the column
631// centroid of its placed-core peers. With mergeLogicalTiles == true
632// (default) several non-core LTOs may share one physical tile when
633// DMA capacity allows, preferring tiles whose North/South stream ports
634// and runtime-sequence BDs also fit; with mergeLogicalTiles == false
635// each non-core LTO claims its own physical tile.
636//
637// Greedy: once a tile is chosen the decision is final, with no backtracking.
638// If a later LTO's constraints become unsatisfiable given prior placements,
639// the placer emits an error pointing at the LTOs the user can manually pin
640// to break the deadlock.
641//
642// Compute-to-compute ObjectFifos consume no DMA channel when their endpoints
643// land on shared-L1 neighbor tiles, so they enter the placer as a separate
644// compute-peer adjacency rather than as channel demand.

993// across all nets.

996// overflow, core tile overflow, DMA channel overuse, BD count

998// locks LockPlan cannot place at the current positions.
999//
1000// 3. Cascade penalty (hard, blocks legality): penalizes cascade put/get

1136/// A DenseMap that notes what each write through operator[] overwrote, so
1137/// rollback() can restore it as of the last commit().
1138template <typename Key, typename Value>
1139class JournaledMap {
1140public:
1141 using Map = llvm::DenseMap<Key, Value>;
1142 Value &operator[](const Key &key) {
1143 auto [it, added] = map.try_emplace(key);
1144 saved.emplace_back(key,
1145 added ? std::nullopt : std::optional<Value>(it->second));
1146 return it->second;
1147 }
1148 Value lookup(const Key &key) const { return map.lookup(key); }
1149 typename Map::const_iterator find(const Key &key) const {
1150 return map.find(key);
1151 }
1152 bool contains(const Key &key) const { return map.contains(key); }
1153 typename Map::const_iterator begin() const { return map.begin(); }
1154 typename Map::const_iterator end() const { return map.end(); }
1155 void clear() {
1156 map.clear();
1157 saved.clear();
1158 }
1159 void commit() { saved.clear(); }
1160 void rollback() {
1161 for (auto &[key, old] : llvm::reverse(saved)) {
1162 if (old)
1163 map[key] = std::move(*old);
1164 else
1165 map.erase(key);
1166 }
1167 saved.clear();
1168 }
1169
1170private:
1171 Map map;
1172 llvm::SmallVector<std::pair<Key, std::optional<Value>>> saved;
1173};

26struct AIEReorderCtrlPacketsByTilePass
27 : xilinx::AIEX::impl::AIEReorderCtrlPacketsByTileBase<
28 AIEReorderCtrlPacketsByTilePass> {
29 using Base::Base;

31 void normalize(SmallVectorImpl<NpuControlPacketOp> &packets) {
32 if (packets.size() < 2 || (!addressOrder && !eliminateDeadWrites)) {
33 packets.clear();
34 return;
35 }
36 Block *block = packets.back()->getBlock();
37 auto end = std::next(packets.back()->getIterator());
38 llvm::DenseSet<uint32_t> written;
39 SmallVector<NpuControlPacketOp> surviving;
40 for (NpuControlPacketOp packet : llvm::reverse(packets)) {
41 if (!eliminateDeadWrites) {
42 surviving.push_back(packet);
43 continue;
44 }
45 auto data = packet.getData().value();
46 SmallVector<bool, 4> live;
47 for (size_t index = 0; index < data.size(); ++index)
48 live.push_back(written.insert(packet.getAddress() + 4 * index).second);
49 if (llvm::all_of(live, [](bool value) { return value; })) {
50 surviving.push_back(packet);
51 continue;
52 }
53 OpBuilder builder(packet);
54 for (size_t begin = 0; begin < data.size();) {
55 if (!live[begin]) {
56 ++begin;
57 continue;
58 }
59 size_t limit = begin + 1;
60 while (limit < data.size() && live[limit])
61 ++limit;
62 surviving.push_back(NpuControlPacketOp::create(
63 builder, packet.getLoc(),
64 builder.getUI32IntegerAttr(packet.getAddress() + 4 * begin),
65 IntegerAttr{}, packet.getOpcodeAttr(), packet.getStreamIdAttr(),
66 builder.getDenseI32ArrayAttr(data.slice(begin, limit - begin))));
67 begin = limit;
68 }
69 packet.erase();
70 }
71 if (addressOrder) {
72 llvm::sort(surviving,
73 [](NpuControlPacketOp left, NpuControlPacketOp right) {
74 return left.getAddress() < right.getAddress();
75 });
76 for (NpuControlPacketOp packet : surviving)
77 packet->moveBefore(block, end);
78 }
79 packets.clear();
80 }
81
82 void runOnOperation() override {
83 getOperation().walk([&](AIE::RuntimeSequenceOp sequence) {
84 sequence.walk([&](Block *block) {
85 for (PacketInterval interval : packetIntervals(*block)) {
86 if (addressOrder && interval.independent &&
87 llvm::none_of(interval.packets, [](NpuControlPacketOp packet) {
88 return packet.getMulticastId().has_value();
89 })) {
90 auto end = std::next(interval.packets.back()->getIterator());
91 llvm::stable_sort(interval.packets, [](NpuControlPacketOp left,
92 NpuControlPacketOp right) {
93 return left.getTileFromAddr() < right.getTileFromAddr();
94 });
95 for (NpuControlPacketOp packet : interval.packets)
96 packet->moveBefore(block, end);
97 }
98 if (!interval.ordinary)
99 continue;
100 SmallVector<NpuControlPacketOp> packets;
101 llvm::DenseSet<uint32_t> written;
102 auto finish = [&] {
103 normalize(packets);
104 written.clear();
105 };
106 for (NpuControlPacketOp packet : interval.packets) {
107 if (packet.getMulticastId()) {
108 finish();
109 continue;
110 }
111 if (!packets.empty() &&
112 packets.back().getStreamId() != packet.getStreamId())
113 finish();
114 if (addressOrder && !eliminateDeadWrites) {
115 for (size_t index = 0; index < packet.getData()->size(); ++index)
116 if (written.contains(packet.getAddress() + 4 * index)) {
117 finish();
118 break;
119 }
120 for (size_t index = 0; index < packet.getData()->size(); ++index)
121 written.insert(packet.getAddress() + 4 * index);
122 }
123 packets.push_back(packet);
124 }
125 finish();
126 }
127 });
128 });
129 }
130};

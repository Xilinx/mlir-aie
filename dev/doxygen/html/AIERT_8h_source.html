26// RAII helper. Constructs while AIERTControl is recording an aie-rt
27// transaction, captures the current XAie_TxnInst command count at entry, and
28// at scope end attributes every aie-rt command produced during the scope to
29// `loc`. exportSerializedTransaction() projects the captured ranges onto the
30// serialized transaction's operations, and the AIEToConfiguration round-trip
31// reads them back out so re-emitted aiex.npu.* ops carry the source op's MLIR
32// Location instead of the device's fallback location.
33class TxnLocBracket {
34public:
35 TxnLocBracket(AIERTControl &ctl, mlir::Location loc);
36 ~TxnLocBracket();
37 TxnLocBracket(const TxnLocBracket &) = delete;
38 TxnLocBracket &operator=(const TxnLocBracket &) = delete;
39
40private:
41 AIERTControl &ctl;
42 mlir::Location loc;
43 uint32_t startCmds;
44};

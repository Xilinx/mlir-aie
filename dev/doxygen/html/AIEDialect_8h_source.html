52// Supplies `Symbol`'s visibility accessors for ops that keep visibility in a
53// plain `sym_visibility` attribute rather than in a tablegen-declared argument
54// (which is what upstream's `SymbolVisibility` trait requires).
55template <typename ConcreteType>
56struct AttrBasedSymbolVisibility
57 : mlir::OpTrait::TraitBase<ConcreteType, AttrBasedSymbolVisibility> {
58 static constexpr llvm::StringRef getVisibilityAttrName() {
59 return "sym_visibility";
60 }
61
62 // `mlir::detail::verifySymbol` only checks the inherent attribute, so it
63 // never sees ours.
64 static mlir::LogicalResult verifyTrait(mlir::Operation *op) {
65 mlir::Attribute vis = op->getAttr(getVisibilityAttrName());
66 if (!vis)
67 return mlir::success();
68 auto visStrAttr = llvm::dyn_cast<mlir::StringAttr>(vis);
69 if (!visStrAttr)
70 return op->emitOpError()
71 << "requires visibility attribute '" << getVisibilityAttrName()
72 << "' to be a string attribute, but got " << vis;
73 if (!llvm::is_contained(
74 llvm::ArrayRef<llvm::StringRef>{"public", "private", "nested"},
75 visStrAttr.getValue()))
76 return op->emitOpError()
77 << "visibility expected to be one of [\"public\", \"private\", "
78 "\"nested\"], but got "
79 << visStrAttr;
80 return mlir::success();
81 }
82
83 mlir::SymbolTable::Visibility getVisibility() {
84 mlir::StringAttr vis =
85 this->getOperation()->template getAttrOfType<mlir::StringAttr>(
86 getVisibilityAttrName());
87 if (!vis)
88 return mlir::SymbolTable::Visibility::Public;
89 return llvm::StringSwitch<mlir::SymbolTable::Visibility>(vis.getValue())
90 .Case("private", mlir::SymbolTable::Visibility::Private)
91 .Case("nested", mlir::SymbolTable::Visibility::Nested)
92 .Default(mlir::SymbolTable::Visibility::Public);
93 }
94
95 void setVisibility(mlir::SymbolTable::Visibility vis) {
96 mlir::Operation *op = this->getOperation();
97 if (vis == mlir::SymbolTable::Visibility::Public) {
98 op->removeAttr(getVisibilityAttrName());
99 return;
100 }
101 assert((vis == mlir::SymbolTable::Visibility::Private ||
102 vis == mlir::SymbolTable::Visibility::Nested) &&
103 "unknown symbol visibility kind");
104 llvm::StringRef visName =
105 vis == mlir::SymbolTable::Visibility::Private ? "private" : "nested";
106 op->setAttr(getVisibilityAttrName(),
107 mlir::StringAttr::get(op->getContext(), visName));
108 }
109};

396// A draining walk over a whole stream of objects rather than one of them:
397// output `j` walks `inner` from object `advance * j - top` on, reading `span`
398// consecutive objects through the inner dimension `rowDim` steps by one
399// object (none when `span` is 1); the `top` objects before the first and
400// `bottom` past the last of `objects` read as zeros.
401struct StreamWindow {
402 int64_t objectSize;
403 int64_t outputs;
404 int64_t advance;
405 int64_t span;
406 int64_t top;
407 int64_t bottom;
408 int64_t objects;
409 int64_t rowDim;
410 llvm::SmallVector<BDDimLayoutAttr> inner;
411
412 // Elements each output moves.
413 int64_t transfer() const;
414};
415
416// One output of a StreamWindow: the first object it reads, counting the `top`
417// zero objects, and the lock units it takes before and gives back after.
418struct StreamWindowStep {
419 int64_t first;
420 int64_t acquire;
421 int64_t release;
422};

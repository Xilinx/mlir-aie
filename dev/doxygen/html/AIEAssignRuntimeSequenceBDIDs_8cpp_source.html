7//
8// Assigns buffer-descriptor (BD) IDs to the DMA tasks configured in a runtime
9// sequence.
10//
11// This pass runs on straight-line IR: --aie-unroll-runtime-sequence-loops has
12// unrolled every constant-trip scf.for, and canonicalization has folded every
13// constant-predicate scf.if, before this pass runs. So in the static path no
14// scf op survives to reach the allocator -- a rolled ping-pong over a constant
15// loop is just N straight-line configures whose ids ordinary liveness reuse
16// recycles. Any scf.for/scf.if still present is therefore runtime-valued, which
17// the static path cannot lower (the runtime sequence becomes a flat, branchless
18// NPU instruction stream); such forms are rejected here for the dynamic EmitC
19// path (Phase 2).
20//
21// aie-split-long-repeats has already made every start exactly one queue push.
22//
23// Before allocating, the pass also checks the per-channel hardware resources a
24// sequence can exhaust (see verifyChannelUsage): task-completion-token (TCT)
25// imbalance -- an await with no matching issue_token push on its channel would
26// deadlock the host -- and DMA task-queue overflow, where more transfers are
27// pushed onto a channel than its queue holds. On the straight-line IR the
28// allocator sees, both are single-pass per-channel counts.
29//
30// With reclaim-bds, when a tile runs out of ids, the pass takes them back from
31// a started task that was never released (see reclaimFor) rather than failing:
32// from one a status poll already proves finished, or else by inserting a poll
33// that does. Only allocations that would fail change, so a sequence that fits
34// compiles to the same instructions either way. It is opt-in because the poll
35// never returns if the task's completion depends on a later push.
36//
37//===----------------------------------------------------------------------===//

78 static std::optional<DmaQueueModel::ChannelKey>
79 otherTokenChannel(Operation *op) {
80 if (auto push = dyn_cast<NpuPushQueueOp>(op)) {
81 if (push.getIssueToken())
82 return DmaQueueModel::ChannelKey{static_cast<int>(push.getColumn()),
83 static_cast<int>(push.getRow()),
84 static_cast<int>(push.getDirection()),
85 static_cast<int>(push.getChannel())};
86 } else if (auto copy = dyn_cast<NpuDmaMemcpyNdOp>(op)) {
87 auto allocation = AIE::ShimDMAAllocationOp::getForSymbol(
88 op->getParentOfType<AIE::DeviceOp>(),
89 copy.getMetadata().getRootReference());
90 if (allocation &&
91 (copy.getIssueToken() ||
92 allocation.getChannelDir() == AIE::DMAChannelDir::S2MM)) {
93 if (AIE::TileOp tile = allocation.getTileOp())
94 return DmaQueueModel::keyOf(tile, allocation.getChannelDir(),
95 allocation.getChannelIndex());
96 }
97 }
98 return std::nullopt;
99 }

139 // Reject control flow the static path cannot lower. Constant-trip scf.for is
140 // unrolled and constant-predicate scf.if is folded before this pass, so any
141 // scf op here is runtime-valued and belongs to the dynamic EmitC path.
142 LogicalResult rejectRuntimeControlFlow(AIE::RuntimeSequenceOp seq) {
143 WalkResult wr = seq.walk([&](Operation *op) -> WalkResult {
144 if (isa<scf::ForOp, scf::IfOp, scf::WhileOp>(op)) {
145 op->emitOpError(
146 "Runtime-valued control flow in a runtime sequence is not "
147 "supported by this static BD-ID allocation pass. Either pass "
148 "only constant-valued predicates to scf.for and scf.if so "
149 "`aie-unroll-runtime-sequence-loops` can unroll/fold them, or "
150 "use the dynamic EmitC path instead.");
151 return WalkResult::interrupt();
152 }
153 return WalkResult::advance();
154 });
155 return failure(wr.wasInterrupted());
156 }

158 // Track the two per-channel resources a sequence can exhaust. Their failure
159 // modes are opposites: over-production is harmless for the TCT FIFO (leftover
160 // tokens cost nothing) but not for the task queue, where a push onto a full
161 // queue is dropped and its transfer never runs.
162 //
163 // Queue counting differs from token counting in three ways: every push takes
164 // a slot, not just issue_token ones; a non-token push is still retired
165 // implicitly, since in-order execution means awaiting a token drains
166 // everything queued ahead of it too; and a repeat_count does not multiply
167 // slots. rejectRuntimeControlFlow has already run, so one program-order pass
168 // over straight-line IR is exact.
169 LogicalResult verifyChannelUsage(AIE::RuntimeSequenceOp seq) {
170 using ChannelKey = DmaQueueModel::ChannelKey;
171
172 const AIETargetModel &tm =
173 seq->getParentOfType<AIE::DeviceOp>().getTargetModel();
174
175 std::map<ChannelKey, int> avail;
176 DmaQueueModel queue;
177
178 WalkResult wr = seq.walk([&](Operation *op) -> WalkResult {
179 if (auto start = dyn_cast<DMAStartTaskOp>(op)) {
180 DMAConfigureTaskOp cfg = start.getTaskOp();
181 if (!cfg)
182 return WalkResult::advance();
183 ChannelKey key = channelOf(cfg);
184 AIE::TileOp tile = cfg.getTileOp();
185 uint32_t depth = tm.getDmaTaskQueueDepth(
186 tile.getCol(), tile.getRow(), cfg.getChannel(), cfg.getDirection());
187 if (queue.wouldOverflow(key, depth))
188 guardQueueOverflow(queue, start, tm, key, depth, enforceQueueDepth);
189 bool issuesToken = start.getPushIssueToken(cfg);
190 queue.push(key, issuesToken);
191 if (issuesToken)
192 avail[key]++;
193 } else if (auto await = dyn_cast<DMAAwaitTaskOp>(op)) {
194 DMAConfigureTaskOp cfg = await.getTaskOp();
195 // A non-issue_token await is diagnosed later by aie-dma-tasks-to-npu;
196 // an unresolved task is diagnosed by the recycle path. Skip both here.
197 if (!cfg || !cfg.getIssueToken())
198 return WalkResult::advance();
199 ChannelKey key = channelOf(cfg);
200 int &tokens = avail[key];
201 if (tokens < 1) {
202 await.emitOpError(
203 "awaits a task-completion token on a channel where no "
204 "outstanding token is guaranteed to have been produced; the "
205 "runtime sequence would block here forever. Ensure a prior "
206 "issue_token dma_start_task on the same tile, direction and "
207 "channel reaches this await");
208 return WalkResult::interrupt();
209 }
210 tokens--;
211 queue.awaitToken(key);
212 } else if (auto key = otherAwaitChannel(op)) {
213 // Raw waits may also consume tokens from outside this sequence. Do not
214 // diagnose those here, but never reuse a token already consumed by one.
215 avail[*key] = std::max(0, avail[*key] - 1);
216 queue.awaitToken(*key);
217 } else if (auto key = otherTokenChannel(op)) {
218 avail[*key]++;
219 queue.push(*key, true);
220 }
221 return WalkResult::advance();
222 });
223 return failure(wr.wasInterrupted());
224 }

226 LogicalResult validate(AIE::RuntimeSequenceOp seq) {
227 // Reject runtime control flow first, so the token-balance pass below runs
228 // on straight-line IR and needs no control-flow reasoning.
229 if (failed(rejectRuntimeControlFlow(seq)))
230 return failure();
231 if (failed(verifyChannelUsage(seq)))
232 return failure();
233 return success();
234 }

236 LogicalResult allocateConfigure(DMAConfigureTaskOp op) {
237 AIE::TileOp tile = op.getTileOp();
238 BdIdGenerator &gen = getGeneratorForTile(tile);
239
240 const AIETargetModel &targetModel =
241 tile->getParentOfType<AIE::DeviceOp>().getTargetModel();
242
243 // First, honor all the user-specified BD IDs.
244 WalkResult result = op.walk<WalkOrder::PreOrder>([&](AIE::DMABDOp bd_op) {
245 if (bd_op.getBdId().has_value()) {
246 if (!targetModel.isBdChannelAccessible(tile.getCol(), tile.getRow(),
247 bd_op.getBdId().value(),
248 op.getChannel())) {
249 bd_op.emitOpError("Buffer descriptor ID ")
250 << bd_op.getBdId().value() << " cannot be submitted on channel "
251 << op.getChannel() << " of tile (" << tile.getCol() << ","
252 << tile.getRow()
253 << "), which partitions its buffer descriptors by channel "
254 "parity: an even channel reaches only the low half of the "
255 "ids and an odd channel only the high half.";
256 return WalkResult::interrupt();
257 }
258 if (gen.bdIdAlreadyAssigned(bd_op.getBdId().value())) {
259 op.emitOpError("Specified buffer descriptor ID ")
260 << bd_op.getBdId().value()
261 << " is already in use. Release the earlier task first: "
262 "aiex.dma_await_task waits for hardware completion before its "
263 "BDs are reusable. aiex.dma_free_task also releases them but "
264 "does NOT wait, so it is only safe when some other "
265 "synchronization already guarantees that task has finished "
266 "(see programming_guide/section-2/section-2d/DMATasks.md).";
267 return WalkResult::interrupt();
268 }
269 checkReallocation(tile, bd_op.getBdId().value(), bd_op);
270 gen.assignBdId(bd_op.getBdId().value());
271 }
272 return WalkResult::advance();
273 });
274 if (result.wasInterrupted())
275 return failure();
276
277 // Now allocate BD IDs for all unspecified BDs.
278 result = op.walk<WalkOrder::PreOrder>([&](AIE::DMABDOp bd_op) {
279 if (bd_op.getBdId().has_value())
280 return WalkResult::advance();
281 // channelIndex matters on a MemTile, where the AIE2 model partitions
282 // BDs by channel parity (isBdChannelAccessible: an even channel can
283 // only submit ids below 24, an odd channel only 24 and above).
284 std::optional<int32_t> next_id = gen.nextBdId(op.getChannel());
285 while (!next_id && reclaimBds && succeeded(reclaimFor(op)))
286 next_id = gen.nextBdId(op.getChannel());
287 if (!next_id) {
288 auto diag =
289 op.emitOpError()
290 << "Too many simultaneously active buffer descriptors on tile ("
291 << tile.getCol() << "," << tile.getRow()
292 << "), which supports up to "
293 << targetModel.getNumBDsForChannel(tile.getCol(), tile.getRow(),
294 op.getChannel())
295 << ". Emit an aiex.dma_await_task to free BDs for reuse; it "
296 "waits for hardware completion, so the recycled ids are no "
297 "longer in flight. aiex.dma_free_task also recycles ids but "
298 "does NOT wait for completion -- using it before the task "
299 "has finished is a race -- so reach for it only when some "
300 "other synchronization already guarantees completion (see "
301 "programming_guide/section-2/section-2d/DMATasks.md).";
302 if (reclaimBds)
303 diag << " The compiler could not take any back either: every "
304 "task holding one is not started yet, is started again "
305 "later, or runs on a channel whose status it cannot "
306 "poll.";
307 else
308 diag << " Or let the compiler poll for finished tasks and take "
309 "their ids (reclaim-bds, aiecc --reclaim-runtime-bds), "
310 "provided no task depends on a push issued after it.";
311 return WalkResult::interrupt();
312 }
313 checkReallocation(tile, *next_id, bd_op);
314 bd_op.setBdId(next_id);
315 return WalkResult::advance();
316 });
317 if (result.wasInterrupted())
318 return failure();
319
320 liveByTile[tile].push_back(op);
321 return success();
322 }

324 // One push onto a channel: the task it started (null for a raw push or
325 // memcpy transfer) and whether that push issues a token. A task started more
326 // than once may issue a token on only some of its pushes.
327 struct StartedTask {
328 DMAConfigureTaskOp task;
329 bool issuesToken;
330 // Program order of the push, for picking the oldest task to reclaim.
331 uint64_t order;
332 // A status poll proved it finished. Its token, if any, is still waiting
333 // for an await, so it stays here until one consumes it.
334 bool finished = false;
335 };
336 // Pushes on each channel, in program order, whose tokens no await has
337 // consumed yet.
338 std::map<DmaQueueModel::ChannelKey, SmallVector<StartedTask, 8>>
339 startedOnChannel;
340 uint64_t pushCount = 0;
341
342 void notePush(const DmaQueueModel::ChannelKey &key, DMAConfigureTaskOp task,
343 bool issuesToken) {
344 startedOnChannel[key].push_back({task, issuesToken, pushCount++});
345 }

347 static bool isStarted(ArrayRef<StartedTask> started,
348 DMAConfigureTaskOp task) {
349 return llvm::any_of(started,
350 [&](const StartedTask &s) { return s.task == task; });
351 }
352 static bool isRunning(ArrayRef<StartedTask> started,
353 DMAConfigureTaskOp task) {
354 return llvm::any_of(started, [&](const StartedTask &s) {
355 return s.task == task && !s.finished;
356 });
357 }

359 llvm::SmallPtrSet<Operation *, 16> knownComplete;

383 // An await consumes the oldest outstanding token on the channel, regardless
384 // of the configure named by its SSA operand. Only that token-issuing task and
385 // the tasks queued ahead of it are known to have finished.
386 void noteAwaited(const DmaQueueModel::ChannelKey &key) {
387 auto &started = startedOnChannel[key];
388 auto *it = llvm::find_if(
389 started, [](const StartedTask &s) { return s.issuesToken; });
390 if (it == started.end())
391 return;
392 SmallVector<StartedTask, 8> retired(started.begin(), std::next(it));
393 started.erase(started.begin(), std::next(it));
394 for (const StartedTask &s : retired)
395 if (s.task && !isStarted(started, s.task))
396 knownComplete.insert(s.task);
397 }

399 // A status poll proved at most `unfinished` pushes on `key` are still queued
400 // or running. The channel runs its queue in order, so those are the newest.
401 void noteDrained(const DmaQueueModel::ChannelKey &key, size_t unfinished) {
402 auto &started = startedOnChannel[key];
403 if (started.size() <= unfinished)
404 return;
405 auto done = MutableArrayRef<StartedTask>(started).drop_back(unfinished);
406 for (StartedTask &s : done)
407 s.finished = true;
408 for (StartedTask &s : done)
409 if (s.task && !isRunning(started, s.task))
410 pollProven.insert(s.task);
411 }

413 // Credit a maskpoll on a channel's status register, whether the queue-depth
414 // guard, reclaimFor or the design emitted it. A poll bounding the queue size
415 // by b leaves at most b + 1 unfinished (see getDmaTaskQueueSizeMask); one
416 // that also clears the running and stall bits leaves none.
417 void notePoll(NpuMaskPollOp poll) {
418 std::optional<uint32_t> address = poll.getAbsoluteAddress();
419 std::optional<uint32_t> mask = getConstantIntOperand(poll.getMask());
420 std::optional<uint32_t> value = getConstantIntOperand(poll.getValue());
421 const AIETargetModel &tm =
422 poll->getParentOfType<AIE::DeviceOp>().getTargetModel();
423 uint32_t field = tm.getDmaTaskQueueSizeMask();
424 uint32_t idle = tm.getDmaChannelIdleMask();
425 if (!address || !mask || !value || !field)
426 return;
427 for (auto &[key, started] : startedOnChannel) {
428 if (tm.getDmaStatusAddress(key[0], key[1], key[3],
429 static_cast<AIE::DMAChannelDir>(key[2])) !=
430 address)
431 continue;
432 if (idle && (*mask & idle) == idle && (*value & idle) == 0)
433 return noteDrained(key, 0);
434 unsigned shift = llvm::countr_zero(field);
435 uint32_t bound = 0;
436 for (uint32_t size = 0; size <= field >> shift; ++size)
437 if (((size << shift) & *mask & field) == (*value & *mask & field))
438 bound = size;
439 return noteDrained(key, bound + 1);
440 }
441 }

443 // Record ids released without any completion guarantee. nextBdId scans upward
444 // from 0, so a just-freed low id is the first one handed out again -- the
445 // worst case for aliasing a BD that is still running.
446 void noteFreedInFlight(DMAConfigureTaskOp cfg, Operation *freeOp) {
447 if (!isStarted(startedOnChannel[channelOf(cfg)], cfg))
448 return;
449 AIE::TileOp tile = cfg.getTileOp();
450 auto &ids = freedInFlight[{tile.getCol(), tile.getRow()}];
451 cfg.walk([&](AIE::DMABDOp bd) {
452 if (bd.getBdId().has_value())
453 ids[bd.getBdId().value()] = {cfg, freeOp};
454 });
455 }

457 // Warn where the hazard actually bites: reusing the id, not releasing it.
458 void checkReallocation(AIE::TileOp tile, uint32_t id, AIE::DMABDOp bd) {
459 if (!warnUnsafeBdReuse)
460 return;
461 auto tileIt = freedInFlight.find({tile.getCol(), tile.getRow()});
462 if (tileIt == freedInFlight.end())
463 return;
464 auto idIt = tileIt->second.find(id);
465 if (idIt == tileIt->second.end())
466 return;
467 ReleasedTask released = idIt->second;
468 tileIt->second.erase(idIt);
469 if (knownComplete.contains(released.configure) ||
470 pollProven.contains(released.configure))
471 return;
472 auto diag =
473 bd->emitWarning()
474 << "reuses buffer descriptor ID " << id << " on tile (" << tile.getCol()
475 << "," << tile.getRow()
476 << ") after it was released by an aiex.dma_free_task that had no "
477 "completion guarantee, so the DMA it belonged to may still be "
478 "running and this reprograms it underneath. Consume the outstanding "
479 "tokens on the same tile, direction and channel through a task "
480 "queued at or after this transfer -- each await consumes the oldest "
481 "token, regardless of the task it names";
482 diag.attachNote(released.freeOp->getLoc()) << "released here";
483 }

485 // Configures already completed by an aiex.dma_await_task. Awaiting a task
486 // returns its BD ids to the pool (like a free), but a subsequent
487 // aiex.dma_free_task of the same task is the common "wait, then release"
488 // idiom, not a double free -- so freeing an awaited task's already-returned
489 // ids is tolerated.
490 llvm::SmallPtrSet<Operation *, 8> awaitedConfigures;
491 // Awaited configures whose IDs remain live until their FIFO completion.
492 llvm::SmallPtrSet<Operation *, 8> pendingAwaitReleases;
493
494 // Return the ids of the configure's chain to the pool. `isAwait` records the
495 // configure so a later free of it is treated as a redundant release rather
496 // than a double free. Otherwise an already-freed id is a real double free (or
497 // a free of a task that was never started) and is an error.
498 LogicalResult recycle(DMAConfigureTaskOp task_op, Operation *freeOp,
499 bool isAwait) {
500 pendingAwaitReleases.erase(task_op);
501 // Those IDs may now belong to another configure. A redundant release must
502 // not inspect or change the generator's current ownership.
503 if (awaitedConfigures.contains(task_op) || reclaimed.contains(task_op))
504 return success();
505 BdIdGenerator &gen = getGeneratorForTile(task_op.getTileOp());
506 WalkResult result = task_op.walk<WalkOrder::PreOrder>([&](AIE::DMABDOp bd) {
507 if (!bd.getBdId().has_value()) {
508 bd.emitOpError("Free called on BD chain with unassigned IDs.");
509 return WalkResult::interrupt();
510 }
511 if (gen.bdIdAlreadyAssigned(bd.getBdId().value())) {
512 gen.freeBdId(bd.getBdId().value());
513 } else {
514 freeOp->emitOpError("frees buffer descriptor ID ")
515 << bd.getBdId().value()
516 << ", which is not currently in use; it was already completed by "
517 "an "
518 "earlier aiex.dma_free_task or aiex.dma_await_task";
519 return WalkResult::interrupt();
520 }
521 return WalkResult::advance();
522 });
523 if (result.wasInterrupted())
524 return failure();
525 releasedTasks.insert(task_op);
526 if (isAwait)
527 awaitedConfigures.insert(task_op);
528 else
529 noteFreedInFlight(task_op, freeOp);
530 return success();
531 }

533 // Resolve a free/await to its configure. In straight-line IR the task value
534 // is defined directly by the configure op; anything else is an unlowered task
535 // op.
536 LogicalResult recycleTask(Value task, Operation *op, bool isAwait) {
537 auto cfg = dyn_cast_or_null<DMAConfigureTaskOp>(task.getDefiningOp());
538 if (!cfg) {
539 auto err = op->emitOpError(
540 "does not reference a valid configure_task operation.");
541 if (Operation *def = task.getDefiningOp()) {
542 if (isa<DMAStartBdChainOp>(def))
543 err.attachNote(def->getLoc())
544 << "Lower this operation first using the "
545 "--aie-materialize-bd-chains pass.";
546 if (isa<DMAConfigureTaskForOp>(def))
547 err.attachNote(def->getLoc())
548 << "Lower this operation first using the "
549 "--aie-substitute-shim-dma-allocations pass.";
550 }
551 return err;
552 }
553 if (!isAwait)
554 return recycle(cfg, op, /*isAwait=*/false);
555 pendingAwaitReleases.insert(cfg);
556 return recycleCompletedTasks(op);
557 }

559 LogicalResult recycleCompletedTasks(Operation *op) {
560 SmallVector<Operation *, 8> pending(pendingAwaitReleases.begin(),
561 pendingAwaitReleases.end());
562 for (Operation *pendingOp : pending) {
563 auto pendingCfg = cast<DMAConfigureTaskOp>(pendingOp);
564 // An earlier await may have named this task while consuming an older
565 // token. Revisit its release when a later await proves it complete.
566 if (!knownComplete.contains(pendingCfg))
567 continue;
568 // Retain ownership across any remaining starts of the same configure.
569 if (llvm::any_of(pendingCfg.getResult().getUsers(), [&](Operation *user) {
570 return isa<DMAStartTaskOp>(user) &&
571 user->getBlock() == op->getBlock() &&
572 op->isBeforeInBlock(user);
573 }))
574 continue;
575 if (failed(recycle(pendingCfg, op, /*isAwait=*/true)))
576 return failure();
577 }
578 return success();
579 }

581 static bool isStartedAfter(DMAConfigureTaskOp task, Operation *op) {
582 return llvm::any_of(task.getResult().getUsers(), [&](Operation *user) {
583 return isa<DMAStartTaskOp>(user) && user->getBlock() == op->getBlock() &&
584 op->isBeforeInBlock(user);
585 });
586 }

588 // Take back the ids of a started task that is not started again after `op`
589 // and holds an id `op`'s channel can use (on a mem tile, its parity's half).
590 // One a poll already proved finished is taken as is. Otherwise a poll before
591 // `op` proves one finished: the oldest with j >= 1 pushes queued behind it,
592 // for which Task_Queue_Size <= j - 1 is proof (as a masked equality, the
593 // largest 2^k - 1 <= j - 1), so it waits on no push not yet issued; failing
594 // that, the oldest, until its channel is idle. The poll never returns if that
595 // task's inputs come from a later push through a core, which the compiler
596 // cannot see; a design with such a dependence releases its own BDs.
597 LogicalResult reclaimFor(DMAConfigureTaskOp op) {
598 AIE::TileOp tile = op.getTileOp();
599 const AIETargetModel &tm =
600 tile->getParentOfType<AIE::DeviceOp>().getTargetModel();
601 auto isCandidate = [&](DMAConfigureTaskOp task) {
602 if (releasedTasks.contains(task) || isStartedAfter(task, op))
603 return false;
604 bool usable = false;
605 task.walk([&](AIE::DMABDOp bd) {
606 usable |= bd.getBdId() &&
607 tm.isBdChannelAccessible(tile.getCol(), tile.getRow(),
608 *bd.getBdId(), op.getChannel());
609 });
610 return usable;
611 };
612 auto take = [&](DMAConfigureTaskOp task) {
613 BdIdGenerator &gen = getGeneratorForTile(tile);
614 task.walk([&](AIE::DMABDOp bd) { gen.freeBdId(*bd.getBdId()); });
615 reclaimed.insert(task);
616 releasedTasks.insert(task);
617 pendingAwaitReleases.erase(task);
618 };
619
620 ArrayRef<DMAConfigureTaskOp> live = liveByTile[tile];
621 for (DMAConfigureTaskOp task : live)
622 if ((knownComplete.contains(task) || pollProven.contains(task)) &&
623 isCandidate(task)) {
624 take(task);
625 return success();
626 }
627
628 uint32_t field = tm.getDmaTaskQueueSizeMask();
629 uint32_t idle = tm.getDmaChannelIdleMask();
630 DMAConfigureTaskOp victim;
631 size_t victimQueuedBehind = 0;
632 uint64_t victimOrder = 0;
633 for (DMAConfigureTaskOp task : live) {
634 if (!isCandidate(task))
635 continue;
636 DmaQueueModel::ChannelKey key = channelOf(task);
637 if (!tm.getDmaStatusAddress(key[0], key[1], key[3],
638 static_cast<AIE::DMAChannelDir>(key[2])))
639 continue;
640 ArrayRef<StartedTask> started = startedOnChannel[key];
641 auto last =
642 llvm::find_if(llvm::reverse(started),
643 [&](const StartedTask &s) { return s.task == task; });
644 // Never started, or retired by an await and caught above.
645 if (last == started.rend() || last->finished)
646 continue;
647 size_t queuedBehind = std::distance(started.rbegin(), last);
648 if (queuedBehind == 0 ? !idle : !field)
649 continue;
650 // Rule 1 before rule 2, then oldest.
651 if (victim && std::make_pair(queuedBehind == 0, last->order) >=
652 std::make_pair(victimQueuedBehind == 0, victimOrder))
653 continue;
654 victim = task;
655 victimQueuedBehind = queuedBehind;
656 victimOrder = last->order;
657 }
658 if (!victim)
659 return failure();
660
661 DmaQueueModel::ChannelKey key = channelOf(victim);
662 std::optional<uint32_t> status = tm.getDmaStatusAddress(
663 key[0], key[1], key[3], static_cast<AIE::DMAChannelDir>(key[2]));
664 if (!status)
665 return failure();
666 uint32_t mask = idle;
667 size_t unfinished = 0;
668 if (victimQueuedBehind > 0) {
669 unsigned shift = llvm::countr_zero(field);
670 uint32_t bound =
671 std::min<uint32_t>(llvm::bit_floor(victimQueuedBehind),
672 llvm::bit_floor((field >> shift) + 1) / 2) -
673 1;
674 mask = field & ~(bound << shift);
675 unfinished = bound + 1;
676 }
677 OpBuilder b(op);
678 auto cst = [&](uint32_t v) {
679 return arith::ConstantOp::create(b, op.getLoc(), b.getI32Type(),
680 b.getI32IntegerAttr(v))
681 .getResult();
682 };
683 Value maskValue = cst(mask);
684 Value compareValue = cst(0);
685 Value statusValue = cst(*status);
686 auto poll = NpuMaskPollOp::create(b, op.getLoc(), statusValue, compareValue,
687 maskValue,
688 /*buffer=*/nullptr, /*column=*/nullptr,
689 /*row=*/nullptr);
690 SmallVector<int32_t> ids;
691 victim.walk([&](AIE::DMABDOp bd) { ids.push_back(*bd.getBdId()); });
692 auto remark = poll.emitRemark()
693 << "tile (" << tile.getCol() << "," << tile.getRow()
694 << ") has no free buffer descriptor id, so this poll waits "
695 << (victimQueuedBehind > 0
696 ? "until the queue behind an earlier task is short "
697 "enough to prove it finished"
698 : "until the channel of an earlier task is idle")
699 << " and takes back its ids ";
700 llvm::interleaveComma(ids, remark);
701 remark.attachNote(victim.getLoc()) << "the task it waits for";
702 noteDrained(key, unfinished);
703 take(victim);
704 return success();
705 }

707 // All of this is scoped to one runtime sequence. `gens` restarts BD id
708 // allocation per sequence, so hazard state left over from an earlier one
709 // describes ids that no longer name the same tasks. freedInFlight is the
710 // one that bites rather than merely misleads: it holds aiex.dma_free_task
711 // pointers, and those ops are erased once the sequence that owns them is
712 // done, so carrying an entry forward leaves a dangling note location.
713 void resetPerSequenceState() {
714 gens.clear();
715 awaitedConfigures.clear();
716 pendingAwaitReleases.clear();
717 startedOnChannel.clear();
718 pushCount = 0;
719 knownComplete.clear();
720 pollProven.clear();
721 reclaimed.clear();
722 releasedTasks.clear();
723 liveByTile.clear();
724 freedInFlight.clear();
725 }

727 void runOnOperation() override {
728 AIE::DeviceOp device = getOperation();
729
730 WalkResult wr = device.walk([&](AIE::RuntimeSequenceOp seq) -> WalkResult {
731 // Skip sequences already handled by the dynamic free-list pool path
732 // (aie-lower-dynamic-bd-pool): they draw BD ids at runtime via
733 // dma_bd_pool_pop and keep their scf.for rolled, which the static
734 // straight-line allocator neither needs to touch nor can validate.
735 bool dynamicPool = false;
736 bool hasTasks = false;
737 seq.walk([&](Operation *op) {
738 dynamicPool |= isa<DMABdPoolPopOp>(op);
739 hasTasks |=
740 isa<DMAConfigureTaskOp, DMAConfigureTaskForOp, DMAStartBdChainOp,
741 DMAStartTaskOp, DMAAwaitTaskOp, DMAFreeTaskOp>(op);
742 });
743 // Already-lowered instruction-only control flow needs no BD allocation.
744 if (dynamicPool || !hasTasks)
745 return WalkResult::advance();
746
747 if (failed(validate(seq)))
748 return WalkResult::interrupt();
749 resetPerSequenceState();
750
751 // Straight-line walk. Collect frees to erase after (recycling reads the
752 // configure the free points at, so erase only once the walk is done).
753 // This allocation strategy works only for straight-line IR without
754 // branching or conditionals; the verifier of this pass ensures this is
755 // the case on the input IR.
756 SmallVector<DMAFreeTaskOp> frees;
757 WalkResult r = seq.walk([&](Operation *op) -> WalkResult {
758 if (auto cfg = dyn_cast<DMAConfigureTaskOp>(op)) {
759 if (failed(allocateConfigure(cfg)))
760 return WalkResult::interrupt();
761 } else if (auto start = dyn_cast<DMAStartTaskOp>(op)) {
762 if (DMAConfigureTaskOp cfg = start.getTaskOp()) {
763 // Only an explicit aiex.dma_free_task releases a task that is
764 // started again (an await or a reclaim keeps the ids of one that
765 // is), so this start would push ids that may already describe
766 // another task's transfer.
767 if (releasedTasks.contains(cfg)) {
768 start.emitOpError(
769 "starts a task whose buffer descriptors an earlier "
770 "aiex.dma_free_task released; their ids may already be "
771 "reprogrammed for another task. Free the task after its "
772 "last start instead.");
773 return WalkResult::interrupt();
774 }
775 knownComplete.erase(cfg);
776 pollProven.erase(cfg);
777 notePush(channelOf(cfg), cfg, start.getPushIssueToken(cfg));
778 }
779 } else if (auto poll = dyn_cast<NpuMaskPollOp>(op)) {
780 notePoll(poll);
781 } else if (auto await = dyn_cast<DMAAwaitTaskOp>(op)) {
782 if (DMAConfigureTaskOp cfg = await.getTaskOp())
783 if (cfg.getIssueToken())
784 noteAwaited(channelOf(cfg));
785 if (failed(recycleTask(await.getTask(), await, /*isAwait=*/true)))
786 return WalkResult::interrupt();
787 } else if (auto key = otherAwaitChannel(op)) {
788 noteAwaited(*key);
789 if (failed(recycleCompletedTasks(op)))
790 return WalkResult::interrupt();
791 } else if (auto freeOp = dyn_cast<DMAFreeTaskOp>(op)) {
792 if (failed(recycleTask(freeOp.getTask(), freeOp, /*isAwait=*/false)))
793 return WalkResult::interrupt();
794 frees.push_back(freeOp);
795 } else if (auto key = otherTokenChannel(op)) {
796 notePush(*key, DMAConfigureTaskOp{}, true);
797 }
798 return WalkResult::advance();
799 });
800 if (r.wasInterrupted())
801 return WalkResult::interrupt();
802 for (DMAFreeTaskOp freeOp : frees)
803 freeOp.erase();
804 return WalkResult::advance();
805 });
806 if (wr.wasInterrupted())
807 return signalPassFailure();
808 }
809};

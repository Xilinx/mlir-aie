7//
8// Assigns buffer-descriptor (BD) IDs to the DMA tasks configured in a runtime
9// sequence.
10//
11// This pass runs on straight-line IR: --aie-unroll-runtime-sequence-loops has
12// unrolled every constant-trip scf.for, and canonicalization has folded every
13// constant-predicate scf.if, before this pass runs. So in the static path no
14// scf op survives to reach the allocator -- a rolled ping-pong over a constant
15// loop is just N straight-line configures whose ids ordinary liveness reuse
16// recycles. Any scf.for/scf.if still present is therefore runtime-valued, which
17// the static path cannot lower (the runtime sequence becomes a flat, branchless
18// NPU instruction stream); such forms are rejected here for the dynamic EmitC
19// path (Phase 2).
20//
21// Before allocating, the pass also checks the per-channel hardware resources a
22// sequence can exhaust (see verifyChannelUsage): task-completion-token (TCT)
23// imbalance -- an await with no matching issue_token push on its channel would
24// deadlock the host -- and DMA task-queue overflow, where more transfers are
25// pushed onto a channel than its queue holds. On the straight-line IR the
26// allocator sees, both are single-pass per-channel counts.
27//
28//===----------------------------------------------------------------------===//

70 otherTokenChannel(Operation *op) {
71 if (auto push = dyn_cast<NpuPushQueueOp>(op)) {
72 if (push.getIssueToken())

123 auto it = gens.find(tile);

134 // Reject control flow the static path cannot lower. Constant-trip scf.for is
135 // unrolled and constant-predicate scf.if is folded before this pass, so any
136 // scf op here is runtime-valued and belongs to the dynamic EmitC path.
137 LogicalResult rejectRuntimeControlFlow(AIE::RuntimeSequenceOp seq) {
138 WalkResult wr = seq.walk([&](Operation *op) -> WalkResult {
139 if (isa<scf::ForOp, scf::IfOp, scf::WhileOp>(op)) {
140 op->emitOpError(
141 "Runtime-valued control flow in a runtime sequence is not "
142 "supported by this static BD-ID allocation pass. Either pass "
143 "only constant-valued predicates to scf.for and scf.if so "
144 "`aie-unroll-runtime-sequence-loops` can unroll/fold them, or "
145 "use the dynamic EmitC path instead.");
146 return WalkResult::interrupt();
147 }
148 return WalkResult::advance();
149 });
150 return failure(wr.wasInterrupted());
151 }

153 // Track the two per-channel resources a sequence can exhaust. Their failure
154 // modes are opposites: over-production is harmless for the TCT FIFO (leftover
155 // tokens cost nothing) but not for the task queue, where a push onto a full
156 // queue is dropped and its transfer never runs.
157 //
158 // Queue counting differs from token counting in three ways: every push takes
159 // a slot, not just issue_token ones; a non-token push is still retired
160 // implicitly, since in-order execution means awaiting a token drains
161 // everything queued ahead of it too; and repeat_count does not multiply
162 // slots. rejectRuntimeControlFlow has already run, so one program-order pass
163 // over straight-line IR is exact.
164 LogicalResult verifyChannelUsage(AIE::RuntimeSequenceOp seq) {
165 using ChannelKey = DmaQueueModel::ChannelKey;
166 auto keyOf = [](DMAConfigureTaskOp cfg) -> ChannelKey {
167 AIE::TileOp tile = cfg.getTileOp();
168 return {tile.getCol(), tile.getRow(),
169 static_cast<int>(cfg.getDirection()),
170 static_cast<int>(cfg.getChannel())};
171 };
172
173 const AIETargetModel &tm =
174 seq->getParentOfType<AIE::DeviceOp>().getTargetModel();
175
176 std::map<ChannelKey, int> avail;
177 DmaQueueModel queue;
178
179 WalkResult wr = seq.walk([&](Operation *op) -> WalkResult {
180 if (auto start = dyn_cast<DMAStartTaskOp>(op)) {
181 DMAConfigureTaskOp cfg = start.getTaskOp();
182 if (!cfg)
183 return WalkResult::advance();
184 ChannelKey key = keyOf(cfg);
185 AIE::TileOp tile = cfg.getTileOp();
186 uint32_t depth = tm.getDmaTaskQueueDepth(
187 tile.getCol(), tile.getRow(), cfg.getChannel(), cfg.getDirection());
188 if (queue.wouldOverflow(key, depth))
189 guardQueueOverflow(queue, start, tm, key, depth, enforceQueueDepth);
190 queue.push(key, cfg.getIssueToken());
191 if (cfg.getIssueToken())
192 avail[key]++;
193 } else if (auto await = dyn_cast<DMAAwaitTaskOp>(op)) {
194 DMAConfigureTaskOp cfg = await.getTaskOp();
195 // A non-issue_token await is diagnosed later by aie-dma-tasks-to-npu;
196 // an unresolved task is diagnosed by the recycle path. Skip both here.
197 if (!cfg || !cfg.getIssueToken())
198 return WalkResult::advance();
199 ChannelKey key = keyOf(cfg);
200 int &tokens = avail[key];
201 if (tokens < 1) {
202 await.emitOpError(
203 "awaits a task-completion token on a channel where no "
204 "outstanding token is guaranteed to have been produced; the "
205 "runtime sequence would block here forever. Ensure a prior "
206 "issue_token dma_start_task on the same tile, direction and "
207 "channel reaches this await");
208 return WalkResult::interrupt();
209 }
210 tokens--;
211 queue.awaitToken(key);
212 } else if (auto key = otherAwaitChannel(op)) {
213 // Raw waits may also consume tokens from outside this sequence. Do not
214 // diagnose those here, but never reuse a token already consumed by one.
215 avail[*key] = std::max(0, avail[*key] - 1);
216 queue.awaitToken(*key);
217 } else if (auto key = otherTokenChannel(op)) {
218 avail[*key]++;
219 queue.push(*key, true);
220 }
221 return WalkResult::advance();
222 });
223 return failure(wr.wasInterrupted());
224 }

226 LogicalResult validate(AIE::RuntimeSequenceOp seq) {
227 // Reject runtime control flow first, so the token-balance pass below runs
228 // on straight-line IR and needs no control-flow reasoning.
229 if (failed(rejectRuntimeControlFlow(seq)))
230 return failure();
231 if (failed(verifyChannelUsage(seq)))
232 return failure();
233 return success();
234 }

236 LogicalResult allocateConfigure(DMAConfigureTaskOp op) {
237 AIE::TileOp tile = op.getTileOp();
238 BdIdGenerator &gen = getGeneratorForTile(tile);
239
240 // First, honor all the user-specified BD IDs.
241 WalkResult result = op.walk<WalkOrder::PreOrder>([&](AIE::DMABDOp bd_op) {
242 if (bd_op.getBdId().has_value()) {
243 if (gen.bdIdAlreadyAssigned(bd_op.getBdId().value())) {
244 op.emitOpError("Specified buffer descriptor ID ")
245 << bd_op.getBdId().value()
246 << " is already in use. Release the earlier task first: "
247 "aiex.dma_await_task waits for hardware completion before its "
248 "BDs are reusable. aiex.dma_free_task also releases them but "
249 "does NOT wait, so it is only safe when some other "
250 "synchronization already guarantees that task has finished "
251 "(see programming_guide/section-2/section-2d/DMATasks.md).";
252 return WalkResult::interrupt();
253 }
254 checkReallocation(tile, bd_op.getBdId().value(), bd_op);
255 gen.assignBdId(bd_op.getBdId().value());
256 }
257 return WalkResult::advance();
258 });
259 if (result.wasInterrupted())
260 return failure();
261
262 // Now allocate BD IDs for all unspecified BDs.
263 result =
264 op.walk<WalkOrder::PreOrder>([&](AIE::DMABDOp bd_op) {
265 if (bd_op.getBdId().has_value())
266 return WalkResult::advance();
267 // channelIndex only affects allocation on MemTiles, where the AIE2
268 // model partitions BDs by channel parity (isBdChannelAccessible).
269 // Runtime sequences configure BDs on shim (and compute) tiles only,
270 // which are channel-agnostic (always accessible), so passing 0 is
271 // correct here.
272 std::optional<int32_t> next_id = gen.nextBdId(/*channelIndex=*/0);
273 if (!next_id) {
274 const AIETargetModel &tm =
275 tile->getParentOfType<AIE::DeviceOp>().getTargetModel();
276 op.emitOpError()
277 << "Too many simultaneously active buffer descriptors on tile ("
278 << tile.getCol() << "," << tile.getRow()
279 << "), which supports up to "
280 << tm.getNumBDs(tile.getCol(), tile.getRow())
281 << ". Emit an aiex.dma_await_task to free BDs for reuse; it "
282 "waits for hardware completion, so the recycled ids are no "
283 "longer in flight. aiex.dma_free_task also recycles ids but "
284 "does NOT wait for completion -- using it before the task "
285 "has finished is a race -- so reach for it only when some "
286 "other synchronization already guarantees completion (see "
287 "programming_guide/section-2/section-2d/DMATasks.md).";
288 return WalkResult::interrupt();
289 }
290 checkReallocation(tile, *next_id, bd_op);
291 bd_op.setBdId(next_id);
292 return WalkResult::advance();
293 });
294 if (result.wasInterrupted())
295 return failure();
296
297 return success();
298 }

300 // Tasks started on each channel, in program order, that are not yet known to
301 // have completed.
302 std::map<DmaQueueModel::ChannelKey, SmallVector<DMAConfigureTaskOp, 8>>
303 startedOnChannel;
304 // Configures whose completion an await has established.
305 llvm::SmallPtrSet<Operation *, 16> knownComplete;
306 // BD ids released by aiex.dma_free_task while the task could still have been
307 // in flight, keyed by tile, with the task and the free that released them.
308 struct ReleasedTask {
309 DMAConfigureTaskOp configure;
310 Operation *freeOp;
311 };
312 std::map<std::pair<int, int>, std::map<uint32_t, ReleasedTask>> freedInFlight;

314 static DmaQueueModel::ChannelKey channelOf(DMAConfigureTaskOp cfg) {
315 AIE::TileOp tile = cfg.getTileOp();
316 return {tile.getCol(), tile.getRow(), static_cast<int>(cfg.getDirection()),
317 static_cast<int>(cfg.getChannel())};
318 }

320 // An await consumes the oldest outstanding token on the channel, regardless
321 // of the configure named by its SSA operand. Only that token-issuing task and
322 // the tasks queued ahead of it are known to have finished.
323 void noteAwaited(const DmaQueueModel::ChannelKey &key) {
324 auto &started = startedOnChannel[key];
325 // A null configure represents a token from a raw push or memcpy transfer.
326 auto *it = llvm::find_if(started, [](DMAConfigureTaskOp task) {
327 return !task || task.getIssueToken();
328 });
329 if (it == started.end())
330 return;
331 SmallVector<DMAConfigureTaskOp, 8> retired(started.begin(), std::next(it));
332 started.erase(started.begin(), std::next(it));
333 for (DMAConfigureTaskOp task : retired)
334 if (task && !llvm::is_contained(started, task))
335 knownComplete.insert(task);
336 }

338 // Record ids released without any completion guarantee. nextBdId scans upward
339 // from 0, so a just-freed low id is the first one handed out again -- the
340 // worst case for aliasing a BD that is still running.
341 void noteFreedInFlight(DMAConfigureTaskOp cfg, Operation *freeOp) {
342 if (!llvm::is_contained(startedOnChannel[channelOf(cfg)], cfg))
343 return;
344 AIE::TileOp tile = cfg.getTileOp();
345 auto &ids = freedInFlight[{tile.getCol(), tile.getRow()}];
346 cfg.walk([&](AIE::DMABDOp bd) {
347 if (bd.getBdId().has_value())
348 ids[bd.getBdId().value()] = {cfg, freeOp};
349 });
350 }

352 // Warn where the hazard actually bites: reusing the id, not releasing it.
353 void checkReallocation(AIE::TileOp tile, uint32_t id, AIE::DMABDOp bd) {
354 if (!warnUnsafeBdReuse)
355 return;
356 auto tileIt = freedInFlight.find({tile.getCol(), tile.getRow()});
357 if (tileIt == freedInFlight.end())
358 return;
359 auto idIt = tileIt->second.find(id);
360 if (idIt == tileIt->second.end())
361 return;
362 ReleasedTask released = idIt->second;
363 tileIt->second.erase(idIt);
364 if (knownComplete.contains(released.configure))
365 return;
366 auto diag =
367 bd->emitWarning()
368 << "reuses buffer descriptor ID " << id << " on tile (" << tile.getCol()
369 << "," << tile.getRow()
370 << ") after it was released by an aiex.dma_free_task that had no "
371 "completion guarantee, so the DMA it belonged to may still be "
372 "running and this reprograms it underneath. Consume the outstanding "
373 "tokens on the same tile, direction and channel through a task "
374 "queued at or after this transfer -- each await consumes the oldest "
375 "token, regardless of the task it names";
376 diag.attachNote(released.freeOp->getLoc()) << "released here";
377 }

379 // Configures already completed by an aiex.dma_await_task. Awaiting a task
380 // returns its BD ids to the pool (like a free), but a subsequent
381 // aiex.dma_free_task of the same task is the common "wait, then release"
382 // idiom, not a double free -- so freeing an awaited task's already-returned
383 // ids is tolerated.
384 llvm::SmallPtrSet<Operation *, 8> awaitedConfigures;
385 // Awaited configures whose IDs remain live until their FIFO completion.
386 llvm::SmallPtrSet<Operation *, 8> pendingAwaitReleases;
387
388 // Return the ids of the configure's chain to the pool. `isAwait` records the
389 // configure so a later free of it is treated as a redundant release rather
390 // than a double free. Otherwise an already-freed id is a real double free (or
391 // a free of a task that was never started) and is an error.
392 LogicalResult recycle(DMAConfigureTaskOp task_op, Operation *freeOp,
393 bool isAwait) {
394 pendingAwaitReleases.erase(task_op);
395 // Those IDs may now belong to another configure. A redundant release must
396 // not inspect or change the generator's current ownership.
397 if (awaitedConfigures.contains(task_op))
398 return success();
399 BdIdGenerator &gen = getGeneratorForTile(task_op.getTileOp());
400 WalkResult result = task_op.walk<WalkOrder::PreOrder>([&](AIE::DMABDOp bd) {
401 if (!bd.getBdId().has_value()) {
402 bd.emitOpError("Free called on BD chain with unassigned IDs.");
403 return WalkResult::interrupt();
404 }
405 if (gen.bdIdAlreadyAssigned(bd.getBdId().value())) {
406 gen.freeBdId(bd.getBdId().value());
407 } else {
408 freeOp->emitOpError("frees buffer descriptor ID ")
409 << bd.getBdId().value()
410 << ", which is not currently in use; it was already completed by "
411 "an "
412 "earlier aiex.dma_free_task or aiex.dma_await_task";
413 return WalkResult::interrupt();
414 }
415 return WalkResult::advance();
416 });
417 if (result.wasInterrupted())
418 return failure();
419 if (isAwait)
420 awaitedConfigures.insert(task_op);
421 else
422 noteFreedInFlight(task_op, freeOp);
423 return success();
424 }

426 // Resolve a free/await to its configure. In straight-line IR the task value
427 // is defined directly by the configure op; anything else is an unlowered task
428 // op.
429 LogicalResult recycleTask(Value task, Operation *op, bool isAwait) {
430 auto cfg = dyn_cast_or_null<DMAConfigureTaskOp>(task.getDefiningOp());
431 if (!cfg) {
432 auto err = op->emitOpError(
433 "does not reference a valid configure_task operation.");
434 if (Operation *def = task.getDefiningOp()) {
435 if (isa<DMAStartBdChainOp>(def))
436 err.attachNote(def->getLoc())
437 << "Lower this operation first using the "
438 "--aie-materialize-bd-chains pass.";
439 if (isa<DMAConfigureTaskForOp>(def))
440 err.attachNote(def->getLoc())
441 << "Lower this operation first using the "
442 "--aie-substitute-shim-dma-allocations pass.";
443 }
444 return err;
445 }
446 if (!isAwait)
447 return recycle(cfg, op, /*isAwait=*/false);
448 pendingAwaitReleases.insert(cfg);
449 return recycleCompletedTasks(op);
450 }

452 LogicalResult recycleCompletedTasks(Operation *op) {
453 SmallVector<Operation *, 8> pending(pendingAwaitReleases.begin(),
454 pendingAwaitReleases.end());
455 for (Operation *pendingOp : pending) {
456 auto pendingCfg = cast<DMAConfigureTaskOp>(pendingOp);
457 // An earlier await may have named this task while consuming an older
458 // token. Revisit its release when a later await proves it complete.
459 if (!knownComplete.contains(pendingCfg))
460 continue;
461 // Retain ownership across any remaining starts of the same configure.
462 if (llvm::any_of(pendingCfg.getResult().getUsers(), [&](Operation *user) {
463 return isa<DMAStartTaskOp>(user) &&
464 user->getBlock() == op->getBlock() &&
465 op->isBeforeInBlock(user);
466 }))
467 continue;
468 if (failed(recycle(pendingCfg, op, /*isAwait=*/true)))
469 return failure();
470 }
471 return success();
472 }

474 // All of this is scoped to one runtime sequence. `gens` restarts BD id
475 // allocation per sequence, so hazard state left over from an earlier one
476 // describes ids that no longer name the same tasks. freedInFlight is the
477 // one that bites rather than merely misleads: it holds aiex.dma_free_task
478 // pointers, and those ops are erased once the sequence that owns them is
479 // done, so carrying an entry forward leaves a dangling note location.
480 void resetPerSequenceState() {
481 gens.clear();
482 awaitedConfigures.clear();
483 pendingAwaitReleases.clear();
484 startedOnChannel.clear();
485 knownComplete.clear();
486 freedInFlight.clear();
487 }

489 void runOnOperation() override {
490 AIE::DeviceOp device = getOperation();
491
492 WalkResult wr = device.walk([&](AIE::RuntimeSequenceOp seq) -> WalkResult {
493 // Skip sequences already handled by the dynamic free-list pool path
494 // (aie-lower-dynamic-bd-pool): they draw BD ids at runtime via
495 // dma_bd_pool_pop and keep their scf.for rolled, which the static
496 // straight-line allocator neither needs to touch nor can validate.
497 bool dynamicPool = false;
498 bool hasTasks = false;
499 seq.walk([&](Operation *op) {
500 dynamicPool |= isa<DMABdPoolPopOp>(op);
501 hasTasks |=
502 isa<DMAConfigureTaskOp, DMAConfigureTaskForOp, DMAStartBdChainOp,
503 DMAStartTaskOp, DMAAwaitTaskOp, DMAFreeTaskOp>(op);
504 });
505 // Already-lowered instruction-only control flow needs no BD allocation.
506 if (dynamicPool || !hasTasks)
507 return WalkResult::advance();
508
509 if (failed(validate(seq)))
510 return WalkResult::interrupt();
511 resetPerSequenceState();
512
513 // Straight-line walk. Collect frees to erase after (recycling reads the
514 // configure the free points at, so erase only once the walk is done).
515 // This allocation strategy works only for straight-line IR without
516 // branching or conditionals; the verifier of this pass ensures this is
517 // the case on the input IR.
518 SmallVector<DMAFreeTaskOp> frees;
519 WalkResult r = seq.walk([&](Operation *op) -> WalkResult {
520 if (auto cfg = dyn_cast<DMAConfigureTaskOp>(op)) {
521 if (failed(allocateConfigure(cfg)))
522 return WalkResult::interrupt();
523 } else if (auto start = dyn_cast<DMAStartTaskOp>(op)) {
524 if (DMAConfigureTaskOp cfg = start.getTaskOp()) {
525 knownComplete.erase(cfg);
526 startedOnChannel[channelOf(cfg)].push_back(cfg);
527 }
528 } else if (auto await = dyn_cast<DMAAwaitTaskOp>(op)) {
529 if (DMAConfigureTaskOp cfg = await.getTaskOp())
530 if (cfg.getIssueToken())
531 noteAwaited(channelOf(cfg));
532 if (failed(recycleTask(await.getTask(), await, /*isAwait=*/true)))
533 return WalkResult::interrupt();
534 } else if (auto key = otherAwaitChannel(op)) {
535 noteAwaited(*key);
536 if (failed(recycleCompletedTasks(op)))
537 return WalkResult::interrupt();
538 } else if (auto freeOp = dyn_cast<DMAFreeTaskOp>(op)) {
539 if (failed(recycleTask(freeOp.getTask(), freeOp, /*isAwait=*/false)))
540 return WalkResult::interrupt();
541 frees.push_back(freeOp);
542 } else if (auto key = otherTokenChannel(op)) {
543 startedOnChannel[*key].push_back(DMAConfigureTaskOp{});
544 }
545 return WalkResult::advance();
546 });
547 if (r.wasInterrupted())
548 return WalkResult::interrupt();
549 for (DMAFreeTaskOp freeOp : frees)
550 freeOp.erase();
551 return WalkResult::advance();
552 });
553 if (wr.wasInterrupted())
554 return signalPassFailure();
555 }

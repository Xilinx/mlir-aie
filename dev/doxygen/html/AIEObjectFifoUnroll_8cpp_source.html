42bool xilinx::AIE::isForeverLoop(scf::WhileOp whileOp) {
43 scf::ConditionOp condition = whileOp.getConditionOp();
44 return matchPattern(condition.getCondition(), m_One()) &&
45 llvm::equal(condition.getArgs(), whileOp.getBeforeArguments()) &&
46 llvm::all_of(whileOp.getBeforeBody()->without_terminator(),
47 [](Operation &op) { return isPure(&op); });
48}

50bool xilinx::AIE::isObjectFifoLoop(Operation *op) {
51 if (auto whileOp = dyn_cast<scf::WhileOp>(op)) {
52 return isForeverLoop(whileOp);
53 }
54 return isa<scf::ForOp>(op);
55}

57Operation *xilinx::AIE::getObjectFifoLoopParent(Operation *op) {

69// Statically known trip count of a loop, or nullopt if it cannot be computed.
70static std::optional<int64_t> getStaticTripCount(scf::ForOp forOp) {
71 if (forOp.getSingleLowerBound() && forOp.getSingleUpperBound() &&
72 forOp.getSingleStep()) {
73 if (std::optional<llvm::APInt> tc = forOp.getStaticTripCount()) {
74 return tc->getSExtValue();
75 }
76 }
77 return std::nullopt;
78}

80// Unroll a forever loop by `factor`: its body runs `factor` times per trip
81// around the loop, each copy taking the values the previous one yields. The
82// loop never exits, so there is no remainder to handle.
83static void unrollForeverLoop(scf::WhileOp whileOp, int64_t factor) {
84 Block *body = whileOp.getAfterBody();
85 auto yield = cast<scf::YieldOp>(body->getTerminator());
86 SmallVector<Operation *> bodyOps;
87 for (Operation &op : body->without_terminator()) {
88 bodyOps.push_back(&op);
89 }
90 OpBuilder builder(yield);
91 SmallVector<Value> yielded(yield.getOperands());
92 SmallVector<Value> carried = yielded;
93 for (int64_t i = 1; i < factor; ++i) {
94 IRMapping mapping;
95 mapping.map(body->getArguments(), carried);
96 for (Operation *op : bodyOps) {
97 builder.clone(*op, mapping);
98 }
99 for (auto [next, value] : llvm::zip_equal(carried, yielded)) {
100 next = mapping.lookupOrDefault(value);
101 }
102 }
103 yield->setOperands(carried);
104}

106// Remove redundant AIE1 binary-lock acquires.
107//
108// The dynamic lowering acquires the whole sliding window ([counter, counter +
109// acqNumber)) on every loop iteration; after unrolling and constant folding
110// that expands to an explicit `Acquire` of each element's lock.
111// This drops the needless re-acquire of locks already held.
112//
113// The analysis is block-local and conservatively assumes nothing is held on
114// block entry. That keeps the very first acquire of each unrolled loop body
115// (which legitimately re-establishes the window across the back-edge) while
116// dropping the intra-body re-acquires of locks whose element is still held.
117static bool removeRedundantBinaryAcquires(DeviceOp device) {
118 bool erased = false;
119 device.walk([&](Block *block) {
120 llvm::DenseSet<Value> held;
121 SmallVector<UseLockOp> toErase;
122 for (Operation &op : *block) {
123 auto useLock = dyn_cast<UseLockOp>(&op);
124 if (!useLock) {
125 continue;
126 }
127 Value lock = useLock.getLock();
128 if (useLock.acquire()) {
129 // A binary `Acquire` of a lock already in the held set is redundant.
130 if (!held.insert(lock).second) {
131 toErase.push_back(useLock);
132 }
133 } else if (useLock.release()) {
134 held.erase(lock);
135 }
136 }
137 for (UseLockOp op : toErase) {
138 op.erase();
139 }
140 erased |= !toErase.empty();
141 });
142 return erased;
143}

145// Marks a loop already offered to the peeling trial below, so that neither the
146// peeled loop nor the reverted copy is considered again.
147static constexpr llvm::StringLiteral kPeelTriedAttrName = "aie.peel_tried";
148
149// Number of `use_lock` ops in `op` whose lock value is not a compile-time
150// constant. UseLockOp::getConstantValue() reports an error when it fails, so
151// this probes the value directly.
152static int64_t countRuntimeLockValues(Operation *op) {
153 int64_t count = 0;
154 op->walk([&](UseLockOp useLockOp) {
155 if (!getConstantIntValue(useLockOp.getValue())) {
156 count++;
157 }
158 });
159 return count;
160}

162// Number of memrefs in `op` that are picked at run time: block arguments and
163// the results of region ops and selects.
164static int64_t countRuntimeMemrefs(Operation *op) {
165 int64_t count = 0;
166 auto isMemref = llvm::IsaPred<BaseMemRefType>;
167 op->walk([&](Block *block) {
168 count += llvm::count_if(block->getArgumentTypes(), isMemref);
169 });
170 op->walk([&](Operation *nested) {
171 if (nested != op &&
172 (nested->getNumRegions() > 0 || isa<arith::SelectOp>(nested))) {
173 count += llvm::count_if(nested->getResultTypes(), isMemref);
174 }
175 });
176 return count;
177}

179// Canonicalize, SCCP, canonicalize. The fold needs the folding patterns only:
180// region simplification walks every region of the device, runtime sequences
181// included, and the canonicalizes after this pass still run it.
182static OpPassManager buildFoldPipeline() {
183 GreedyRewriteConfig config;
184 config.setRegionSimplificationLevel(GreedySimplifyRegionLevel::Disabled);
185 OpPassManager pipeline(DeviceOp::getOperationName());
186 pipeline.addPass(mlir::createCanonicalizerPass(config));
187 pipeline.addPass(mlir::createSCCPPass());
188 pipeline.addPass(mlir::createCanonicalizerPass(config));
189 return pipeline;
190}

193 : xilinx::AIE::impl::AIEObjectFifoUnrollBase<AIEObjectFifoUnrollPass> {

199 /// Peel the first iteration off each loop whose lock values are still
200 /// computed at run time, keeping a peel only where it makes them constant.
201 ///
202 /// Whether a peel pays off depends on foldings that cannot be predicted from
203 /// the objectFifo access pattern alone (a conditional acquire may or may not
204 /// collapse, for instance), so each candidate is peeled for real and the
205 /// result measured. While the fold pipeline runs, the unpeeled core is held
206 /// aside as a detached clone, so exactly one version of the loop is ever
207 /// live; that clone is swapped back in when the peel bought nothing, and
208 /// destroyed otherwise.
209 ///
210 /// The measurement is a per-core count, so attributing a change to a peel
211 /// needs one peel per core per fold; the loop below drains the candidates one
212 /// round at a time, peeling every core in parallel so a round costs a single
213 /// run of the fold pipeline. Candidates are marked before they are tried and
214 /// never unmarked while draining, so the number of rounds is bounded by the
215 /// most objectFifo loops in any one core.
216 LogicalResult peelObjectFifoLoops(DeviceOp device) {
217 OpPassManager foldPipeline = buildFoldPipeline();
218
219 IRRewriter rewriter(device.getContext());
220 struct Trial {
221 CoreOp core;
222 Operation *unpeeled;
223 int64_t runtimeLocksBefore;
224 };
225
226 while (true) {
227 SmallVector<Trial> trials;
228 for (CoreOp core : SmallVector<CoreOp>(device.getOps<CoreOp>())) {
229 scf::ForOp target;
230 // Post-order, so the innermost untried loop is taken first.
231 core.walk([&](scf::ForOp forOp) {
232 if (forOp->hasAttr(kPeelTriedAttrName) ||
233 countRuntimeLockValues(forOp) == 0) {
234 return WalkResult::advance();
235 }
236 // A single-iteration loop has nothing to peel off.
237 std::optional<int64_t> trip = getStaticTripCount(forOp);
238 if (trip && *trip <= 1) {
239 return WalkResult::advance();
240 }
241 target = forOp;
242 return WalkResult::interrupt();
243 });
244 if (!target) {
245 continue;
246 }
247 // Marked before the core is cloned, so neither the peeled loop nor the
248 // fallback copy is offered as a candidate again.
249 target->setAttr(kPeelTriedAttrName, rewriter.getUnitAttr());
250
251 Trial trial{core, core->clone(), countRuntimeLockValues(core)};
252 scf::ForOp firstIteration;
253 if (failed(scf::peelForLoopFirstIteration(rewriter, target,
254 firstIteration))) {
255 trial.unpeeled->erase();
256 continue;
257 }
258 trials.push_back(trial);
259 }
260 if (trials.empty()) {
261 break;
262 }
263
264 if (failed(runPipeline(foldPipeline, device))) {
265 for (Trial &trial : trials) {
266 trial.unpeeled->erase();
267 }
268 return failure();
269 }
270
271 for (Trial &trial : trials) {
272 if (countRuntimeLockValues(trial.core) < trial.runtimeLocksBefore) {
273 trial.unpeeled->erase();
274 continue;
275 }
276 rewriter.setInsertionPoint(trial.core);
277 Operation *restored = rewriter.insert(trial.unpeeled);
278 rewriter.replaceOp(trial.core, restored->getResults());
279 }
280 }
281
282 device.walk(
283 [&](scf::ForOp forOp) { forOp->removeAttr(kPeelTriedAttrName); });
284 return success();
285 }

287 /// Run `foldPipeline`, first promoting the object slots collected above to
288 /// SSA on every core where that folds each object to its buffer; elsewhere
289 /// they are left in memory.
290 ///
291 /// A promoted object still picked at run time is a memref descriptor carried
292 /// through control flow, which LLVM's alias analysis cannot tell apart from
293 /// the others; left in memory, SROA reduces it to a pointer. As with peeling,
294 /// each core is promoted for real against a detached clone. Unless the
295 /// promoted core picks no memref at run time, the clone is swapped in and
296 /// folded too, and the promoted core returns only if it picks no more: with
297 /// both sides folded, unrelated folding cannot tip the comparison.
298 LogicalResult foldPromotingObjectSlots(DeviceOp device,
299 ArrayRef<memref::AllocaOp> slots,
300 OpPassManager &foldPipeline) {
301 if (slots.empty()) {
302 return runPipeline(foldPipeline, device);
303 }
304 struct Trial {
305 CoreOp core;
306 Operation *unpromoted;
307 };
308 SmallVector<Trial> trials;
309 llvm::SmallPtrSet<Operation *, 8> seen;
310 SmallVector<PromotableAllocationOpInterface> allocators;
311 for (memref::AllocaOp slot : slots) {
312 auto core = slot->getParentOfType<CoreOp>();
313 if (seen.insert(core).second) {
314 trials.push_back({core, core->clone()});
315 }
316 allocators.push_back(
317 cast<PromotableAllocationOpInterface>(slot.getOperation()));
318 }
319 DataLayout dataLayout = DataLayout::closest(device);
320 DominanceInfo dominance(device);
321 OpBuilder builder(device.getContext());
322 (void)tryToPromoteMemorySlots(allocators, builder, dataLayout, dominance);
323
324 if (failed(runPipeline(foldPipeline, device))) {
325 for (Trial &trial : trials) {
326 trial.unpromoted->erase();
327 }
328 return failure();
329 }
330
331 IRRewriter rewriter(device.getContext());
332 // Puts `in` in place of `out` and detaches `out`.
333 auto swap = [&](Operation *out, Operation *in) {
334 rewriter.setInsertionPoint(out);
335 rewriter.insert(in);
336 rewriter.replaceAllOpUsesWith(out, in->getResults());
337 out->remove();
338 };
339 struct Contest {
340 Operation *promoted;
341 Operation *unpromoted;
342 int64_t promotedRuntimeMemrefs;
343 };
344 SmallVector<Contest> contests;
345 bool restoredAny = false;
346 for (Trial &trial : trials) {
347 // mem2reg may also have left a slot in place.
348 WalkResult leftover = trial.core.walk([&](memref::AllocaOp allocaOp) {
349 return allocaOp->hasAttr(kObjectFifoObjectSlotAttrName)
350 ? WalkResult::interrupt()
351 : WalkResult::advance();
352 });
353 int64_t runtimeMemrefs = countRuntimeMemrefs(trial.core);
354 if (!leftover.wasInterrupted() && runtimeMemrefs == 0) {
355 trial.unpromoted->erase();
356 continue;
357 }
358 swap(trial.core, trial.unpromoted);
359 trial.unpromoted->walk([&](memref::AllocaOp allocaOp) {
360 allocaOp->removeAttr(kObjectFifoObjectSlotAttrName);
361 });
362 if (leftover.wasInterrupted()) {
363 trial.core->erase();
364 } else {
365 contests.push_back({trial.core, trial.unpromoted, runtimeMemrefs});
366 }
367 restoredAny = true;
368 }
369 if (!restoredAny) {
370 return success();
371 }
372
373 // The restored cores have not been folded yet.
374 if (failed(runPipeline(foldPipeline, device))) {
375 for (Contest &contest : contests) {
376 contest.promoted->erase();
377 }
378 return failure();
379 }
380 for (Contest &contest : contests) {
381 if (contest.promotedRuntimeMemrefs <=
382 countRuntimeMemrefs(contest.unpromoted)) {
383 swap(contest.unpromoted, contest.promoted);
384 contest.unpromoted->erase();
385 } else {
386 contest.promoted->erase();
387 }
388 }
389 return success();
390 }

392 void runOnOperation() override {
393 DeviceOp device = getOperation();
394
395 bool peelFirstIteration = false;
396 if (clPeelFirstIteration == "auto") {
397 peelFirstIteration = !clDefaultDynamic;
398 } else if (clPeelFirstIteration == "true") {
399 peelFirstIteration = true;
400 } else if (clPeelFirstIteration != "false") {
401 device.emitOpError("invalid peel-first-iteration value '")
402 << clPeelFirstIteration << R"('; expected "true", "false" or "auto")";
403 return signalPassFailure();
404 }
405
406 SmallVector<memref::AllocaOp> objectSlots;
407 for (auto coreOp : device.getOps<CoreOp>()) {
408 // `default-dynamic` picks the lowering for cores that do not pin their
409 // own choice; a core's `dynamic_objfifo_lowering` attribute overrides it
410 // in either direction. Dynamic cores keep their loops rolled (drop the
411 // hints, skip unrolling) and keep in memory the objects they reach from
412 // anywhere but straight-line code; their runtime bookkeeping is tidied by
413 // the shared fold/cleanup pipeline below.
414 if (coreOp.getDynamicObjfifoLowering().value_or(clDefaultDynamic)) {
415 coreOp.walk([&](Operation *op) {
416 op->removeAttr(kObjectFifoUnrollHintAttrName);
417 });
418 bool straightLine = coreOp.getBody().hasOneBlock();
419 coreOp.walk([&](memref::AllocaOp allocaOp) {
420 if (!allocaOp->hasAttr(kObjectFifoObjectSlotAttrName)) {
421 return;
422 }
423 if (straightLine &&
424 llvm::all_of(allocaOp->getUsers(), [&](Operation *user) {
425 return user->getParentOp() == coreOp.getOperation();
426 })) {
427 objectSlots.push_back(allocaOp);
428 } else {
429 allocaOp->removeAttr(kObjectFifoObjectSlotAttrName);
430 }
431 });
432 continue;
433 }
434 coreOp.walk([&](memref::AllocaOp allocaOp) {
435 if (allocaOp->hasAttr(kObjectFifoObjectSlotAttrName)) {
436 objectSlots.push_back(allocaOp);
437 }
438 });
439 SmallVector<Operation *> loops;
440 coreOp.walk([&](Operation *op) {
441 if (isObjectFifoLoop(op)) {
442 loops.push_back(op);
443 }
444 });
445
446 // Operation::walk uses post-order traversal by default, so a nested loop
447 // is visited before its enclosing loop; iterating the list in order thus
448 // processes the innermost loops first. Unrolling innermost loops first
449 // avoids invalidating references to inner loops when an outer loop (which
450 // duplicates its nested loops) is unrolled.
451 for (Operation *loop : loops) {
452 // Loops without the objectFifo unroll hint (the LCM of the accessed
453 // fifo depths) carry no objectFifo access; factor 1 leaves them rolled.
454 auto hint =
455 loop->getAttrOfType<IntegerAttr>(kObjectFifoUnrollHintAttrName);
456 int64_t unrollFactor = hint ? hint.getInt() : 1;
457 if (unrollFactor <= 1) {
458 continue;
459 }
460
461 // A forever loop has no trip count to cover, so repeating its body
462 // once per rotation is enough.
463 auto forOp = dyn_cast<scf::ForOp>(loop);
464 if (!forOp) {
465 unrollForeverLoop(cast<scf::WhileOp>(loop), unrollFactor);
466 continue;
467 }
468
469 std::optional<int64_t> trip = getStaticTripCount(forOp);
470 // When the loop performs fewer iterations than a full rotation of the
471 // objectFifos, unroll it completely: every iteration must map to an
472 // explicit buffer/lock slot.
473 if (trip && *trip <= unrollFactor) {
474 if (failed(mlir::loopUnrollFull(forOp))) {
475 forOp.emitOpError()
476 << "failed to fully unroll objectFifo loop (trip count "
477 << *trip << ")";
478 return signalPassFailure();
479 }
480 continue;
481 }
482
483 // Otherwise unroll by the rotation period. loopUnrollByFactor peels a
484 // cleanup/epilogue loop for the remaining iterations when the trip
485 // count is not an exact multiple of the factor.
486 FailureOr<mlir::UnrolledLoopInfo> info = mlir::loopUnrollByFactor(
487 forOp, static_cast<uint64_t>(unrollFactor));
488 if (failed(info)) {
489 forOp.emitOpError()
490 << "failed to unroll objectFifo loop by factor " << unrollFactor;
491 return signalPassFailure();
492 }
493
494 // The epilogue runs the remaining (< factor) iterations. Fully unroll
495 // it as well so that each of those iterations maps to an explicit
496 // buffer/lock rotation slot. This is best-effort: an epilogue with a
497 // non-constant trip count cannot be fully unrolled and is left rolled.
498 std::optional<scf::ForOp> epilogue =
499 info->epilogueLoopOp; // NOLINT(bugprone-unchecked-optional-access)
500 if (epilogue) {
501 (void)mlir::loopUnrollFull(*epilogue);
502 }
503 }
504
505 // Drop any lingering unroll hints so they do not leak into the output.
506 coreOp.walk([&](Operation *op) {
507 op->removeAttr(kObjectFifoUnrollHintAttrName);
508 });
509 }
510
511 // --aie-objectfifo-lower-cores promotes the lock bookkeeping counters to
512 // loop-carried SSA values, and the objects are promoted here where they
513 // fold. Once the loops have been unrolled by their rotation period, the
514 // objects and counters become loop-invariant, so every object and lock
515 // value collapses to a constant. Run that fold here as a scoped
516 // sub-pipeline: canonicalize exposes the constants, SCCP propagates them
517 // across any remainder loop that survives a partial unroll, and a final
518 // canonicalize deletes the now-dead counter arithmetic and iter_args.
519 OpPassManager foldPipeline = buildFoldPipeline();
520 if (failed(foldPromotingObjectSlots(device, objectSlots, foldPipeline))) {
521 return signalPassFailure();
522 }
523
524 if (peelFirstIteration && failed(peelObjectFifoLoops(device))) {
525 return signalPassFailure();
526 }
527
528 // With the window acquires now folded to concrete per-lock `Acquire` ops,
529 // drop the ones that re-acquire an already-held AIE1 binary lock. Only a
530 // drop leaves dead constants for a final canonicalize; the fold left the
531 // device canonical otherwise.
532 if (!removeRedundantBinaryAcquires(device)) {
533 return;
534 }
535 OpPassManager cleanupPipeline(DeviceOp::getOperationName());
536 cleanupPipeline.addPass(mlir::createCanonicalizerPass());
537 if (failed(runPipeline(cleanupPipeline, device))) {
538 return signalPassFailure();
539 }
540 }

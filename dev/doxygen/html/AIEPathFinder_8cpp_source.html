29// A connection's demand, the cost Dijkstra weighs it by, is
30// (demandBase + overCapacityCoeff * iterations it was over capacity) *
31// (demandBase + usedCapacityCoeff * streams using it).
32constexpr double overCapacityCoeff = 0.1;
33constexpr double usedCapacityCoeff = 0.02;
34constexpr double demandBase = 1.0;
35// A full connection's demand grows by this factor, or a prioritized flow's
36// by priorityDemandCoeff, each time another stream takes it.
37constexpr double demandCoeff = 1.1;
38constexpr double priorityDemandCoeff =
39 static_cast<double>(std::numeric_limits<int>::max());
40constexpr int maxCircuitStreamCapacity = 1;
41constexpr int maxPacketStreamCapacity = 32;
42// History added to a connection each time the routing check rejects it.
43constexpr int routingCheckPenalty = 5;
44// Iterations spent steering away from a usable routing the check rejects.
45constexpr int maxSteers = 16;
46// Routings in a row the check rejects only for onlyOverlayMasters before
47// routing gives up: the penalties move such a flow between the ports of the
48// switchbox it has to leave by.
49constexpr int maxOverlayFaults = 16;
50// Times the routing check may fault where two trees meet before they stop
51// meeting; each fault moves the meeting elsewhere.
52constexpr int maxMeetFaults = 4;
53// Cost added to a hop that shares an arbiter unit with a flow to avoid.
54constexpr double conflictSharePenalty = 4;
55// Channels per direction packet streams leave a capped tile by (see
56// Pathfinder::relax). A heuristic: fewer channels mean fewer master
57// sets per tile, but more flows on each.
58constexpr int packetFanoutCap = 2;
59// A multicast's next destination may branch off any hop its tree already
60// takes, starting at this cost per hop back to the source: enough of a
61// discount to share hops, while still preferring the shortest path to each
62// destination.
63constexpr double treeSeedFactor = 0.9;
64// A destination's branch is rerouted only when that saves more than this.
65constexpr double rerouteMinSaving = 1e-6;
66} // namespace

89llvm::Error DynamicTileAnalysis::runAnalysis(DeviceOp &device) {
90 LLVM_DEBUG(llvm::dbgs() << "\t---Begin DynamicTileAnalysis Constructor---\n");
91 // find the maxCol and maxRow
92 maxCol = device.getTargetModel().columns();
93 maxRow = device.getTargetModel().rows();
94
95 pathfinder.initialize(maxCol, maxRow, device.getTargetModel());
96
97 // For each flow (circuit + packet) in the device, add it to pathfinder. Each
98 // source can map to multiple different destinations (fanout). Control packet
99 // flows to be routed (as prioritized routings). Then followed by normal
100 // packet flows.
101 for (PacketFlowOp pktFlowOp : device.getOps<PacketFlowOp>()) {
102 Region &r = pktFlowOp.getPorts();
103 Block &b = r.front();
104 SmallVector<std::pair<TileID, Port>, 4> sources;
105 // Pass 1: collect all sources (order-independent; supports fan-in).
106 for (Operation &Op : b.getOperations()) {
107 if (auto pktSource = dyn_cast<PacketSourceOp>(Op)) {
108 auto srcTile = cast<TileOp>(pktSource.getTile().getDefiningOp());
109 sources.push_back(
110 {{srcTile.colIndex(), srcTile.rowIndex()}, pktSource.port()});
111 }
112 }
113
114 bool priorityFlow = pktFlowOp.getPriorityRoute().value_or(false);
115 // Pass 2: add a flow from every source to every destination so
116 // fan-in topologies are routed (not just the last source).
117 for (Operation &Op : b.getOperations()) {
118 if (auto pktDest = dyn_cast<PacketDestOp>(Op)) {
119 auto dstTile = cast<TileOp>(pktDest.getTile().getDefiningOp());
120 Port dstPort = pktDest.port();
121 TileID dstCoords = {dstTile.colIndex(), dstTile.rowIndex()};
122 for (auto &[srcCoords, srcPort] : sources) {
123 LLVM_DEBUG(llvm::dbgs()
124 << "\tAdding Packet Flow: (" << srcCoords.col << ", "
125 << srcCoords.row << ")"
126 << stringifyWireBundle(srcPort.bundle) << srcPort.channel
127 << " -> (" << dstCoords.col << ", " << dstCoords.row << ")"
128 << stringifyWireBundle(dstPort.bundle) << dstPort.channel
129 << "\n");
130 pathfinder.addFlow(srcCoords, srcPort, dstCoords, dstPort,
131 pktFlowOp.IDInt(), priorityFlow,
132 pktFlowOp.getLoc());
133 }
134 }
135 }
136 }
137
138 // Add circuit flows.
139 for (FlowOp flowOp : device.getOps<FlowOp>()) {
140 TileOp srcTile = cast<TileOp>(flowOp.getSource().getDefiningOp());
141 TileOp dstTile = cast<TileOp>(flowOp.getDest().getDefiningOp());
142 TileID srcCoords = {srcTile.colIndex(), srcTile.rowIndex()};
143 TileID dstCoords = {dstTile.colIndex(), dstTile.rowIndex()};
144 Port srcPort = {flowOp.getSourceBundle(), flowOp.getSourceChannel()};
145 Port dstPort = {flowOp.getDestBundle(), flowOp.getDestChannel()};
146 LLVM_DEBUG(llvm::dbgs()
147 << "\tAdding Flow: (" << srcCoords.col << ", " << srcCoords.row
148 << ")" << stringifyWireBundle(srcPort.bundle) << srcPort.channel
149 << " -> (" << dstCoords.col << ", " << dstCoords.row << ")"
150 << stringifyWireBundle(dstPort.bundle) << dstPort.channel
151 << "\n");
152 pathfinder.addFlow(srcCoords, srcPort, dstCoords, dstPort,
153 /*packetId=*/std::nullopt, /*isPriorityFlow=*/false,
154 flowOp.getLoc());
155 }
156
157 // Canonicalize all flows after both packet and circuit flows are collected.
158 pathfinder.sortFlows();
159
160 // A reload configures only the switchboxes the overlay reaches.
161 if (device.getHasCtrlPktOverlay()) {
162 llvm::DenseSet<TileID> reached;
163 for (PacketFlowOp pktFlowOp : device.getOps<PacketFlowOp>()) {
164 if (!pktFlowOp.getPriorityRoute().value_or(false))
165 continue;
166 for (auto pktDest : pktFlowOp.getPorts().getOps<PacketDestOp>())
167 if (pktDest.getBundle() == WireBundle::TileControl)
168 reached.insert(
169 cast<TileOp>(pktDest.getTile().getDefiningOp()).getTileID());
170 }
171 if (!reached.empty())
172 for (int row = 0; row <= maxRow; row++)
173 for (int col = 0; col <= maxCol; col++)
174 if (!reached.contains({col, row}))
175 pathfinder.excludeTile({col, row});
176 }
177
178 // add existing connections so Pathfinder knows which resources are
179 // available search all existing SwitchBoxOps for exising connections
180 for (SwitchboxOp switchboxOp : device.getOps<SwitchboxOp>()) {
181 if (failed(pathfinder.addFixedConnection(switchboxOp)))
182 return llvm::make_error<RoutingFailure>(
183 llvm::formatv("cannot add the fixed connections of the switchbox "
184 "at tile ({0}, {1})",
185 switchboxOp.colIndex(), switchboxOp.rowIndex()),
186 RoutingFaults{}, switchboxOp.getLoc());
187 }
188
189 // all flows are now populated, call the congestion-aware pathfinder
190 // algorithm
191 // check whether the pathfinder algorithm creates a legal routing
192 llvm::Expected<Routing> found = pathfinder.findPaths(maxIterations);
193 if (!found)
194 return found.takeError();
195 routing = std::move(*found);
196
197 // fill in coords to TileOps, SwitchboxOps, and ShimMuxOps
198 coordToTile.clear();
199 coordToSwitchbox.clear();
200 coordToShimMux.clear();
201 for (auto tileOp : device.getOps<TileOp>()) {
202 [[maybe_unused]] bool fresh =
203 coordToTile.try_emplace(tileOp.getTileID(), tileOp).second;
204 assert(fresh);
205 }
206 for (auto switchboxOp : device.getOps<SwitchboxOp>()) {
207 [[maybe_unused]] bool fresh =
208 coordToSwitchbox
209 .try_emplace({switchboxOp.colIndex(), switchboxOp.rowIndex()},
210 switchboxOp)
211 .second;
212 assert(fresh);
213 }
214 for (auto shimmuxOp : device.getOps<ShimMuxOp>()) {
215 [[maybe_unused]] bool fresh =
216 coordToShimMux
217 .try_emplace({shimmuxOp.colIndex(), shimmuxOp.rowIndex()},
218 shimmuxOp)
219 .second;
220 assert(fresh);
221 }
222
223 LLVM_DEBUG(llvm::dbgs() << "\t---End DynamicTileAnalysis Constructor---\n");
224 return llvm::Error::success();
225}

268void Pathfinder::initialize(int maxCol, int maxRow,
269 const AIETargetModel &targetModel) {
270 // Reset the graph and flows so a Pathfinder can be reused across analyses
271 // and devices; the dense-graph cache must be rebuilt for the new topology.
272 // The relax state (shareChannels, idsApart, relaxStep, cappedTiles,
273 // crowdedTiles) survives, so the ladder advances across attempts.
274 graph.clear();
275 flows.clear();
276 packetIdsTo.clear();
277 priorityIds.clear();
278 flowLocs.clear();
279 graphBuilt = false;
280 nodeIds.clear();
281 nodes.clear();
282 adjacency.clear();
283 distance.clear();
284 indexInHeap.clear();
285 colors.clear();
286 preds.clear();
287 predEdge.clear();
288
289 std::map<WireBundle, int> maxChannels;
290 auto intraconnect = [&](int col, int row) {
291 TileID coords = {col, row};
292 SwitchboxConnect &sb =
293 graph.try_emplace({coords, coords}, coords).first->second;
294
295 for (int i = 0, e = getMaxEnumValForWireBundle() + 1; i < e; ++i) {
296 WireBundle bundle = symbolizeWireBundle(i).value();
297 // get all ports into current switchbox
298 int channels =
299 targetModel.getNumSourceSwitchboxConnections(col, row, bundle);
300 if (channels == 0 && targetModel.isShimNOCorPLTile(col, row)) {
301 // wordaround for shimMux
302 channels = targetModel.getNumSourceShimMuxConnections(col, row, bundle);
303 }
304 for (int channel = 0; channel < channels; channel++) {
305 sb.srcPorts.push_back(Port{bundle, channel});
306 }
307 // get all ports out of current switchbox
308 channels = targetModel.getNumDestSwitchboxConnections(col, row, bundle);
309 if (channels == 0 && targetModel.isShimNOCorPLTile(col, row)) {
310 // wordaround for shimMux
311 channels = targetModel.getNumDestShimMuxConnections(col, row, bundle);
312 }
313 for (int channel = 0; channel < channels; channel++) {
314 sb.dstPorts.push_back(Port{bundle, channel});
315 }
316 maxChannels[bundle] = channels;
317 }
318 sb.resize();
319 // A shim's mux reaches its DMA, NOC and PLIO ports from any port.
320 auto isMuxed = [&](Port p) {
321 return targetModel.isShimNOCorPLTile(col, row) &&
322 llvm::is_contained(
323 {WireBundle::DMA, WireBundle::NOC, WireBundle::PLIO},
324 p.bundle);
325 };
326 for (auto [i, pIn] : llvm::enumerate(sb.srcPorts))
327 for (auto [j, pOut] : llvm::enumerate(sb.dstPorts))
328 sb.at(i, j).available =
329 targetModel.isLegalTileConnection(col, row, pIn.bundle, pIn.channel,
330 pOut.bundle, pOut.channel) ||
331 isMuxed(pIn) || isMuxed(pOut);
332 };
333
334 auto interconnect = [&](int col, int row, int targetCol, int targetRow,
335 WireBundle srcBundle, WireBundle dstBundle) {
336 TileID src = {col, row}, dst = {targetCol, targetRow};
337 SwitchboxConnect &sb =
338 graph.try_emplace({src, dst}, src, dst).first->second;
339 for (int channel = 0; channel < maxChannels[srcBundle]; channel++) {
340 sb.srcPorts.push_back(Port{srcBundle, channel});
341 sb.dstPorts.push_back(Port{dstBundle, channel});
342 }
343 sb.resize();
344 for (size_t i = 0; i < sb.srcPorts.size(); i++)
345 sb.at(i, i).available = true;
346 };
347
348 for (int row = 0; row <= maxRow; row++) {
349 for (int col = 0; col <= maxCol; col++) {
350 maxChannels.clear();
351 // connections within the same switchbox
352 intraconnect(col, row);
353
354 // connections between switchboxes
355 if (row > 0) {
356 // from south to north
357 interconnect(col, row, col, row - 1, WireBundle::South,
358 WireBundle::North);
359 }
360 if (row < maxRow) {
361 // from north to south
362 interconnect(col, row, col, row + 1, WireBundle::North,
363 WireBundle::South);
364 }
365 if (col > 0) {
366 // from east to west
367 interconnect(col, row, col - 1, row, WireBundle::West,
368 WireBundle::East);
369 }
370 if (col < maxCol) {
371 // from west to east
372 interconnect(col, row, col + 1, row, WireBundle::East,
373 WireBundle::West);
374 }
375 }
376 }
377}

379// Add a flow from src to dst can have an arbitrary number of dst locations
380// due to fanout.
381void Pathfinder::addFlow(TileID srcCoords, Port srcPort, TileID dstCoords,
382 Port dstPort, std::optional<int> packetId,
383 bool isPriorityFlow,
384 std::optional<mlir::Location> loc) {
385 isPriorityFlow &= constraints.prioritize ||
386 constraints.pinned.count({srcCoords, srcPort}) > 0;
387 if (loc)
388 flowLocs[{srcCoords, srcPort}].try_emplace({dstCoords, dstPort}, *loc);
389 if (packetId) {
390 auto &ids = packetIdsTo[{{srcCoords, srcPort}, {dstCoords, dstPort}}];
391 if (!llvm::is_contained(ids, *packetId))
392 ids.push_back(*packetId);
393 if (isPriorityFlow)
394 priorityIds[{srcCoords, srcPort}].insert(*packetId);
395 }
396 // A source has one flow, with all its destinations.
397 PathEndPoint src{srcCoords, srcPort}, dst{dstCoords, dstPort};
398 auto flow = llvm::find_if(flows, [&](const Flow &f) { return f.src == src; });
399 if (flow == flows.end()) {
400 // sortFlows assigns the packet groups.
401 flows.push_back(
402 Flow{packetId ? 0 : -1, isPriorityFlow, src, {dst}, packetId});
403 } else if (isPriorityFlow) {
404 flow->isPriorityFlow = true;
405 flow->dsts.insert(flow->dsts.begin(), dst);
406 } else {
407 flow->dsts.push_back(dst);
408 }
409}

452bool Pathfinder::splitSharedIds() {
453 if (splitShared)
454 return false;
455 splitShared = true;
456 shareChannels = false;
457 cappedTiles.clear();
458 std::map<PathEndPoint, std::set<int>> ids = idsBySource();
459 for (const auto &[src, srcIds] : ids)
460 for (const auto &[other, otherIds] : ids) {
461 const std::set<int> &theirs = otherIds;
462 if (!(src == other) &&
463 llvm::any_of(srcIds, [&](int id) { return theirs.count(id); }) &&
464 llvm::any_of(srcIds, [&](int id) { return !theirs.count(id); }))
465 return true;
466 }
467 return false;
468}

477constexpr RelaxStep relaxLadder[] = {
478 RelaxStep::ShareChannels, RelaxStep::CapCrowdedTiles,
479 RelaxStep::RouteIdsApart, RelaxStep::ShareChannels,

484bool Pathfinder::relax() {
485 if (!packetsFailed) {
486 LLVM_DEBUG(llvm::dbgs() << "No packet stream crosses an overused link\n");
487 return false;
488 }
489 while (relaxStep < std::size(relaxLadder)) {
490 switch (relaxLadder[relaxStep++]) {
491 case RelaxStep::ShareChannels:
492 if (shareAllChannels()) {
493 LLVM_DEBUG(llvm::dbgs() << "Relax: share channels\n");
494 return true;
495 }
496 break;
497 case RelaxStep::CapCrowdedTiles:
498 if (capCrowdedFanOut()) {
499 LLVM_DEBUG(llvm::dbgs() << "Relax: cap crowded tiles\n");
500 return true;
501 }
502 break;
503 case RelaxStep::RouteIdsApart:
504 if (routeIdsApart()) {
505 LLVM_DEBUG(llvm::dbgs() << "Relax: route ids apart\n");
506 return true;
507 }
508 // With one id per source, the steps after it would only repeat.
509 relaxStep = std::size(relaxLadder);
510 return false;
511 case RelaxStep::SplitSharedIds:
512 if (splitSharedIds()) {
513 LLVM_DEBUG(llvm::dbgs() << "Relax: split shared ids\n");
514 return true;
515 }
516 relaxStep = std::size(relaxLadder);
517 return false;
518 }
519 }
520 return false;
521}

523// Sort flows to (1) get deterministic routing, and (2) perform routings on
524// prioritized flows before others, for routing consistency on those flows.
525void Pathfinder::sortFlows() {
526 for (auto &flow : flows)
527 llvm::sort(flow.dsts);
528
529 // The groups shareAllChannels merges, whatever order the flows were
530 // added in: packet flows to a common destination are in one.
531 llvm::IntEqClasses groups(flows.size());
532 std::map<PathEndPoint, unsigned> firstTo;
533 for (auto [k, flow] : llvm::enumerate(flows)) {
534 if (flow.packetGroupId < 0)
535 continue;
536 for (const PathEndPoint &dst : flow.dsts) {
537 auto [it, fresh] = firstTo.try_emplace(dst, k);
538 if (!fresh)
539 groups.join(k, it->second);
540 }
541 }
542 llvm::DenseMap<unsigned, int> groupOf;
543 for (auto [k, flow] : llvm::enumerate(flows))
544 if (flow.packetGroupId >= 0)
545 flow.packetGroupId =
546 shareChannels
547 ? 0
548 : groupOf.try_emplace(groups.findLeader(k), groupOf.size())
549 .first->second;
550
551 auto flowRank = [](const Flow &flow) {
552 if (flow.isPriorityFlow)
553 return 0;
554 if (flow.packetGroupId >= 0)
555 return 1;
556 return 2;
557 };
558 llvm::sort(flows, [&](const Flow &lhs, const Flow &rhs) {
559 return std::make_pair(flowRank(lhs), lhs.src) <
560 std::make_pair(flowRank(rhs), rhs.src);
561 });
562}

582LogicalResult Pathfinder::addFixedConnection(SwitchboxOp switchboxOp) {
583 int col = switchboxOp.colIndex();
584 int row = switchboxOp.rowIndex();
585 TileID coords = {col, row};
586 auto &sb = graph[std::make_pair(coords, coords)];
587 // Validate every connect against the original connectivity before reserving
588 // any resource, so a broadcast (one source fanning out to several dests) does
589 // not invalidate its own sibling connects mid-scan. A destination port may be
590 // driven only once; a source port may legally recur across a broadcast.
591 llvm::SmallVector<std::pair<int, int>, 8> reserved;
592 llvm::SmallDenseSet<int, 8> claimedDsts;
593 for (ConnectOp connectOp : switchboxOp.getOps<ConnectOp>()) {
594 int srcIdx = sb.srcIndex(connectOp.sourcePort());
595 int dstIdx = sb.dstIndex(connectOp.destPort());
596 // Reject an illegal pair (absent from the switchbox model) or a second
597 // driver on the same output port; a repeated source port is a broadcast.
598 if (srcIdx < 0 || dstIdx < 0 || !sb.at(srcIdx, dstIdx).available ||
599 !claimedDsts.insert(dstIdx).second) {
600 return failure();
601 }
602 reserved.emplace_back(srcIdx, dstIdx);
603 }
604 // A pre-placed packet-switched output (an aie.masterset) also monopolizes its
605 // destination port. That stream-switch output is already configured for
606 // packet switching, so no circuit stream may drive it, and the router cannot
607 // emit a second masterset on the same port for another packet flow. The
608 // circuit pathfinder does not read packet ops, so without this reservation it
609 // may route a flow onto an output the masterset owns and produce two ops
610 // driving one destination.
611 llvm::SmallVector<int, 8> reservedMasterDsts;
612 for (MasterSetOp masterSetOp : switchboxOp.getOps<MasterSetOp>()) {
613 int dstIdx = sb.dstIndex(masterSetOp.destPort());
614 // Reject an output port absent from the switchbox model or already driven
615 // by a circuit connect or another masterset.
616 if (dstIdx < 0 || !claimedDsts.insert(dstIdx).second) {
617 return failure();
618 }
619 reservedMasterDsts.push_back(dstIdx);
620 }
621 // A circuit-switched ConnectOp monopolizes both its source port (the stream
622 // switch input) and its destination port (the output): no other stream may
623 // inject on that input, and the output can carry only this one stream.
624 // Reserving the whole column also reserves the outgoing wire, since that wire
625 // is reachable only by driving this output port.
626 for (auto [srcIdx, dstIdx] : reserved) {
627 for (size_t j = 0; j < sb.dstPorts.size(); j++)
628 sb.at(srcIdx, j).available = false;
629 for (size_t i = 0; i < sb.srcPorts.size(); i++)
630 sb.at(i, dstIdx).available = false;
631 }
632 // A masterset fixes only its output port. Its inputs arrive through arbiters
633 // that packet flows share, so the source rows stay free.
634 for (int dstIdx : reservedMasterDsts)
635 for (size_t i = 0; i < sb.srcPorts.size(); i++)
636 sb.at(i, dstIdx).available = false;
637 // A fixed op on a shim's South channel claims the port the shim mux carries
638 // on it (see shimMuxChannelFrom).
639 if (switchboxOp.getTileOp().isShimNOCorPLTile()) {
640 auto isMuxed = [](Port p) {
641 return llvm::is_contained(
642 {WireBundle::DMA, WireBundle::NOC, WireBundle::PLIO}, p.bundle);
643 };
644 llvm::SmallDenseSet<int, 8> southDsts, southSrcs;
645 for (int dstIdx : claimedDsts)
646 if (sb.dstPorts[dstIdx].bundle == WireBundle::South)
647 southDsts.insert(sb.dstPorts[dstIdx].channel);
648 for (auto [srcIdx, dstIdx] : reserved)
649 if (sb.srcPorts[srcIdx].bundle == WireBundle::South)
650 southSrcs.insert(sb.srcPorts[srcIdx].channel);
651 for (size_t j = 0; j < sb.dstPorts.size(); j++)
652 if (isMuxed(sb.dstPorts[j]) &&
653 southDsts.count(shimMuxChannelTo(sb.dstPorts[j])))
654 for (size_t i = 0; i < sb.srcPorts.size(); i++)
655 sb.at(i, j).available = false;
656 for (size_t i = 0; i < sb.srcPorts.size(); i++)
657 if (isMuxed(sb.srcPorts[i]) &&
658 southSrcs.count(shimMuxChannelFrom(sb.srcPorts[i])))
659 for (size_t j = 0; j < sb.dstPorts.size(); j++)
660 sb.at(i, j).available = false;
661 }
662 for (PacketRulesOp rulesOp : switchboxOp.getOps<PacketRulesOp>()) {
663 int srcIdx = sb.srcIndex(rulesOp.sourcePort());
664 if (srcIdx < 0)
665 return failure();
666 sb.packetOnlySrc.set(srcIdx);
667 }
668 return success();
669}

697// Build the dense integer node numbering and per-node adjacency once. The graph
698// topology is fixed across congestion iterations (only the demand weights
699// change), so this is computed a single time and the edges carry live pointers
700// into `graph` for demand lookups. A node's edges are sorted by the
701// PathEndPoint they reach, so Dijkstra breaks ties the same way every run.
702void Pathfinder::buildRoutingGraph() {
703 // Seed the dense node set with all flow endpoints (the only nodes Dijkstra is
704 // ever started from or traced back to). Remaining nodes are discovered as
705 // edge destinations below.
706 for (auto &f : flows) {
707 getOrAddNodeId(f.src);
708 for (auto &d : f.dsts)
709 getOrAddNodeId(d);
710 }
711
712 // Process nodes by growing index; getOrAddNodeId() may append new nodes as
713 // edge destinations are discovered, so re-read nodes.size() each iteration.
714 for (size_t id = 0; id < nodes.size(); id++) {
715 PathEndPoint src = nodes[id];
716 // The ports the crossbar connects this one to, and the neighbours' ports
717 // its wire reaches.
718 std::vector<PathEndPoint> dests;
719 auto intraIt = graph.find({src.coords, src.coords});
720 if (intraIt != graph.end()) {
721 auto &sb = intraIt->second;
722 for (auto [i, pIn] : llvm::enumerate(sb.srcPorts))
723 for (auto [j, pOut] : llvm::enumerate(sb.dstPorts))
724 if (pIn == src.port && sb.at(i, j).available)
725 dests.emplace_back(src.coords, pOut);
726 }
727 std::pair<TileID, Port> neighbors[] = {
728 {{src.coords.col, src.coords.row - 1},
729 {WireBundle::North, src.port.channel}},
730 {{src.coords.col - 1, src.coords.row},
731 {WireBundle::East, src.port.channel}},
732 {{src.coords.col, src.coords.row + 1},
733 {WireBundle::South, src.port.channel}},
734 {{src.coords.col + 1, src.coords.row},
735 {WireBundle::West, src.port.channel}}};
736 for (const auto &[neighborCoords, neighborPort] : neighbors) {
737 auto nIt = graph.find({src.coords, neighborCoords});
738 if (nIt != graph.end() &&
739 src.port.bundle == getConnectingBundle(neighborPort.bundle) &&
740 llvm::is_contained(nIt->second.dstPorts, neighborPort))
741 dests.emplace_back(neighborCoords, neighborPort);
742 }
743 llvm::sort(dests);
744
745 std::vector<Edge> edges;
746 edges.reserve(dests.size());
747 for (auto &dest : dests) {
748 auto &sb = graph.at({src.coords, dest.coords});
749 int i = sb.srcIndex(src.port);
750 int j = sb.dstIndex(dest.port);
751 assert(i >= 0 && j >= 0);
752 int destId = getOrAddNodeId(dest);
753 edges.push_back(Edge{destId, &sb, i, j});
754 }
755 // getOrAddNodeId above may have reallocated `adjacency` via index growth in
756 // later iterations, but we only assign this node's edges now.
757 if (adjacency.size() < nodes.size())
758 adjacency.resize(nodes.size());
759 adjacency[id] = std::move(edges);
760 }
761 size_t n = nodes.size();
762 if (adjacency.size() < n)
763 adjacency.resize(n);
764
765 // Size the reusable Dijkstra scratch buffers. These are indexed by state id,
766 // i.e. two entries per node -- one per side of the port.
767 distance.assign(2 * n, INF);
768 indexInHeap.assign(2 * n, 0);
769 colors.assign(2 * n, WHITE);
770 preds.assign(2 * n, -1);
771 predEdge.assign(2 * n, Edge{-1, nullptr, 0, 0});
772 graphBuilt = true;
773}

775double Pathfinder::edgeWeight(const Edge &e, std::optional<int> packetId,
776 const llvm::BitVector *avoid,
777 const llvm::BitVector *avoidBranch) const {
778 const SwitchboxConnect::Cell &cell = e.sb->at(e.i, e.j);
779 double w = cell.demand;
780 // Sharing it would take a second stream's capacity, which the demand
781 // only shows once the group is routed.
782 if (packetId && cell.packetFlowCount > 0 && cell.packetIds.count(*packetId))
783 w *= demandCoeff;
784 if (e.sb->srcCoords == e.sb->dstCoords)
785 for (const llvm::BitVector *flows : {avoid, avoidBranch})
786 if (flows && llvm::any_of(e.sb->unitFlows(e.j),
787 [&](int f) { return flows->test(f); }))
788 w += conflictSharePenalty;
789 return w;
790}

792// Dijkstra over the dense graph from the states in `seeds`, searching states
793// (node, PortSide) rather than bare nodes. Fills the `preds` and `predEdge`
794// scratch buffers, both indexed by state id.
795void Pathfinder::dijkstraShortestPaths(
796 ArrayRef<int> seeds, ArrayRef<double> seedCosts,
797 std::optional<int> packetId, const llvm::BitVector *avoid,
798 const llvm::DenseMap<int, llvm::BitVector> *branchAvoid,
799 const llvm::DenseSet<int> *stops, ArrayRef<int> targets,
800 llvm::function_ref<bool(int, int)> mayCross) {
801 llvm::fill(distance, INF);
802 llvm::fill(colors, static_cast<int8_t>(WHITE));
803 llvm::fill(preds, -1);
804 llvm::fill(indexInHeap, uint64_t{0});
805
806 using MutableQueue = d_ary_heap_indirect<
807 /*Value=*/int, /*Arity=*/4,
808 /*IndexInHeapPropertyMap=*/std::vector<uint64_t> &,
809 /*DistanceMap=*/std::vector<double> &,
810 /*Compare=*/std::less<>>;
811 MutableQueue Q(distance, indexInHeap);
812
813 for (auto [seed, cost] : llvm::zip_equal(seeds, seedCosts)) {
814 distance[seed] = cost;
815 colors[seed] = GRAY;
816 Q.push(seed);
817 }
818 // A settled state's distance and predecessor never change again.
819 SmallVector<int, 8> unsettled(targets);
820 while (!Q.empty()) {
821 int s = Q.top();
822 Q.pop();
823 if (llvm::is_contained(unsettled, s)) {
824 llvm::erase(unsettled, s);
825 if (unsettled.empty())
826 break;
827 }
828 if (stops && stops->count(s)) {
829 colors[s] = BLACK;
830 continue;
831 }
832 // In takes crossbar edges and lands on the Out side of the port it picks;
833 // Out takes the wire to the neighbour and lands on that tile's In side. Any
834 // other pairing would either turn the stream around inside a switchbox or
835 // ride a wire the crossbar was never set to drive.
836 const bool sIsOut = (s & 1) == Out;
837 const llvm::BitVector *avoidBranch = nullptr;
838 if (branchAvoid)
839 if (auto it = branchAvoid->find(s); it != branchAvoid->end())
840 avoidBranch = &it->second;
841 for (Edge &e : adjacency[stateNode(s)]) {
842 const bool isIntra = e.sb->srcCoords == e.sb->dstCoords;
843 if (sIsOut == isIntra ||
844 (isIntra && !packetId && e.sb->packetOnlySrc[e.i]) ||
845 (isIntra && packetId && e.sb->circuitOnlyDst[e.j]))
846 continue;
847 int dst = stateId(e.dst, isIntra ? Out : In);
848 if (isIntra && mayCross && !mayCross(s, dst))
849 continue;
850 double w = edgeWeight(e, packetId, avoid, avoidBranch);
851 if (colors[dst] == BLACK || distance[s] + w >= distance[dst])
852 continue;
853 distance[dst] = distance[s] + w;
854 preds[dst] = s;
855 predEdge[dst] = e;
856 if (colors[dst] == WHITE) {
857 colors[dst] = GRAY;
858 Q.push(dst);
859 } else {
860 Q.update(dst);
861 }
862 }
863 colors[s] = BLACK;
864 }
865}

867// What findPaths keeps from one flow and iteration to the next.
868struct Pathfinder::RouteState {
869 explicit RouteState(const Pathfinder &pf);
870
871 int groupOf(const Flow &f) const;
872 SmallVector<int, 4> idsTo(int flow, int dstState) const;
873 void relateParts();
874 int splitPart(int flow, const std::set<int> &ids);
875 bool isPinned(const Flow &f) const;
876
877 const Pathfinder &pf;
878 Routing routing;
879 // Stamp-based "processed" set (avoids O(n) clears per flow).
880 std::vector<uint32_t> processedStamp;
881 uint32_t curStamp = 0;
882 // The flows to route: one per source, less the packets split off into parts
883 // of their own (see below), which route as flows from the same source.
884 std::vector<Flow> parts;
885 std::map<PathEndPoint, SmallVector<int, 2>> partsOf;
886 // group flows based on packetGroupId; pinned trees go in first, so the
887 // flows routed around them see them.
888 llvm::MapVector<int, SmallVector<int, 8>> groupedFlows;
889 // Packet flows that conflict, by the ids each carries or by
890 // RoutingFaults::apart.
891 std::vector<llvm::BitVector> conflicting;
892 std::set<std::pair<PathEndPoint, PathEndPoint>> apartSources;
893 std::set<std::pair<PathEndPoint, PathEndPoint>> togetherSources;
894 // The packet ids each flow carries, in all and to each destination.
895 std::vector<std::set<int>> flowIds;
896 std::vector<SmallVector<int, 4>> sameId;
897 // Each packet flow's tree as routed so far this iteration: the state each
898 // hop reaches, from which state and by which edge; and its destinations.
899 std::vector<llvm::DenseMap<int, std::pair<int, Edge>>> treeOf;
900 std::vector<SmallVector<int, 4>> treeDsts;
901 // The hops each flow takes only by joining another flow's tree, by the flow
902 // it joined. A join the routing check faults is not made again.
903 std::vector<SmallVector<std::pair<int, Edge>, 8>> joinedHops;
904 std::set<std::pair<int, int>> noJoin;
905 // Where the routing check split a flow's tree: the tile it branches at, the
906 // ids to branch apart there, and the state of the destination that reaches
907 // the tile apart, if one does.
908 struct IdSplit {
909 TileID at;
910 int a, b;
911 std::optional<int> apart;
912 bool operator<(const IdSplit &other) const {
913 return std::tie(at, a, b, apart) <
914 std::tie(other.at, other.a, other.b, other.apart);
915 }
916 };
917 std::vector<std::set<IdSplit>> splitsOf;
918 // A split one tree cannot make, since a destination takes both ids, moves
919 // the second id to a part of its own. Parts of a source split apart at a
920 // tile leave it by different master ports: the part routed later keeps off
921 // those the other's tree takes there.
922 std::vector<std::set<std::pair<TileID, int>>> partSplits;
923 // Flows that found no one switchbox to meet a tree routed before them at,
924 // by that tree's flow; each such pair routes the other way round once.
925 std::set<std::pair<int, int>> unmet, reordered;
926 // A tree branching for two destinations at its source switchbox may leave
927 // no slave port there to meet it by. Each of an unmet pair then reaches the
928 // destinations it shares with the other by one master port out of it.
929 std::set<std::pair<int, int>> trunked;
930 // The flows that met a tree this iteration, by its flow, and how often the
931 // routing check faulted a meeting of each pair.
932 std::set<std::pair<int, int>> met;
933 std::map<std::pair<int, int>, int> meetFaults;
934 // Joined pairs the routing check faulted along with hops neither joined,
935 // which were left to the penalties on those hops once.
936 std::set<std::pair<int, int>> spared;
937 // A control-packet reload keeps the master sets the prioritized flows' trees
938 // leave each switchbox by, so another flow leaving by one of those master
939 // ports leaves by all of one such set and no others.
940 llvm::DenseSet<int> overlayMasters;
941 std::vector<SmallVector<int, 4>> overlaySets;
942 bool mayLeave(ArrayRef<int> masters, int next) const;
943 bool reorder(int later, int earlier, bool again = false);
944};

946Pathfinder::RouteState::RouteState(const Pathfinder &pf)
947 : pf(pf), processedStamp(2 * pf.nodes.size(), 0), parts(pf.flows),
948 conflicting(parts.size(), llvm::BitVector(parts.size())),
949 flowIds(parts.size()), treeOf(parts.size()), treeDsts(parts.size()),
950 joinedHops(parts.size()), splitsOf(parts.size()),
951 partSplits(parts.size()) {
952 for (auto [k, f] : llvm::enumerate(parts))
953 partsOf[f.src].push_back(k);
954 for (auto [k, f] : llvm::enumerate(parts))
955 groupedFlows[groupOf(f)].push_back(k);
956 for (const auto &[ends, ids] : pf.packetIdsTo)
957 flowIds[partsOf.at(ends.first).front()].insert(ids.begin(), ids.end());
958 relateParts();
959 // A pinned source's other packets follow the pinned tree where they go
960 // everywhere it goes, so they leave each switchbox by its master sets, and
961 // otherwise route as a part of their own.
962 for (size_t k = 0, n = parts.size(); k < n; k++) {
963 if (!isPinned(parts[k]))
964 continue;
965 const PathEndPoint &src = parts[k].src;
966 const std::set<int> &prioritized = pf.priorityIds.at(src);
967 std::set<PathEndPoint> treeDsts;
968 std::map<int, std::set<PathEndPoint>> dstsOf;
969 for (const auto &[ends, ids] : pf.packetIdsTo)
970 if (ends.first == src)
971 for (int id : ids)
972 (prioritized.count(id) ? treeDsts : dstsOf[id]).insert(ends.second);
973 std::set<int> others;
974 for (const auto &[id, dsts] : dstsOf)
975 if (!prioritized.count(id) && dsts != treeDsts)
976 others.insert(id);
977 if (!others.empty())
978 splitPart(k, others);
979 }
980 for (const auto &[src, tree] : pf.constraints.pinned) {
981 std::map<PathEndPoint, SmallVector<int, 4>> sets;
982 for (const TreeHop &h : tree) {
983 auto to = pf.nodeIds.find(h.to);
984 if (h.from.coords == h.to.coords && to != pf.nodeIds.end())
985 sets[h.from].push_back(stateId(to->second, Out));
986 }
987 for (auto &[_, set] : sets) {
988 llvm::sort(set);
989 overlayMasters.insert(set.begin(), set.end());
990 overlaySets.push_back(std::move(set));
991 }
992 }
993}

995// Route `later` before `earlier` from the next iteration on, unless the two
996// were reordered before and this is not `again`.
997bool Pathfinder::RouteState::reorder(int later, int earlier, bool again) {
998 if (!again && (reordered.count({earlier, later}) ||
999 !reordered.insert({later, earlier}).second))
1000 return false;
1001 SmallVector<int, 8> &group = groupedFlows[groupOf(parts[later])];
1002 if (!llvm::is_contained(group, earlier))
1003 return false;
1004 group.erase(llvm::find(group, later));
1005 group.insert(llvm::find(group, earlier), later);
1006 return true;
1007}

1010// port may leave it by `next` too; see overlayMasters.

1042void Pathfinder::RouteState::relateParts() {
1043 sameId.assign(parts.size(), {});
1044 conflicting.assign(parts.size(), llvm::BitVector(parts.size()));
1045 for (size_t a = 0; a < parts.size(); a++)
1046 for (size_t b = 0; b < parts.size(); b++) {
1047 // Trees with an id in common join where they meet, so are never apart.
1048 bool same = llvm::any_of(flowIds[a],
1049 [&](int id) { return flowIds[b].count(id); });
1050 if (a != b && same)
1051 sameId[a].push_back(b);
1052 if (a < b && parts[a].packetId && parts[b].packetId &&
1053 ((!same &&
1054 apartSources.count(std::minmax(parts[a].src, parts[b].src))) ||
1055 (pf.constraints.conflict &&
1056 pf.constraints.conflict(parts[a].src, flowIds[a], parts[b].src,
1057 flowIds[b])))) {
1058 conflicting[a].set(b);
1059 conflicting[b].set(a);
1060 }
1061 }

1064int Pathfinder::RouteState::splitPart(int flow, const std::set<int> &ids) {
1065 int part = parts.size();
1066 Flow f = parts[flow];
1067 LLVM_DEBUG({
1068 llvm::dbgs() << "\t\tRouting ids";
1069 for (int id : ids)
1070 llvm::dbgs() << ' ' << id;
1071 llvm::dbgs() << " from " << describeTilePort(f.src.coords, f.src.port)
1072 << " apart\n";
1073 });
1074 for (int id : ids)
1075 flowIds[flow].erase(id);
1076 flowIds.push_back(ids);
1077 auto carries = [&](int k, const PathEndPoint &p) {
1078 auto it = pf.packetIdsTo.find({f.src, p});
1079 return it != pf.packetIdsTo.end() &&
1080 llvm::any_of(it->second, [&](int i) { return flowIds[k].count(i); });
1081 };
1082 llvm::erase_if(f.dsts,
1083 [&](const PathEndPoint &p) { return !carries(part, p); });
1084 llvm::erase_if(parts[flow].dsts,
1085 [&](const PathEndPoint &p) { return !carries(flow, p); });
1086 f.packetId = *ids.begin();
1087 std::optional<int> &flowId = parts[flow].packetId;
1088 if (flowId && ids.count(*flowId))
1089 flowId = *flowIds[flow].begin();
1090 auto prioritized = [&](const std::set<int> &carried) {
1091 auto prio = pf.priorityIds.find(f.src);
1092 return prio != pf.priorityIds.end() && llvm::any_of(carried, [&](int id) {
1093 return prio->second.count(id);
1094 });
1095 };
1096 f.isPriorityFlow = prioritized(ids);
1097 parts[flow].isPriorityFlow = prioritized(flowIds[flow]);
1098 parts.push_back(f);
1099 partsOf[f.src].push_back(part);
1100 groupedFlows[groupOf(f)].push_back(part);
1101 relateParts();
1102 treeOf.emplace_back();
1103 treeDsts.emplace_back();
1104 joinedHops.emplace_back();
1105 for (auto [a, b] : std::set<std::pair<int, int>>(noJoin))
1106 if (a == flow || b == flow)
1107 noJoin.insert({a == flow ? part : a, b == flow ? part : b});
1108 splitsOf.push_back(splitsOf[flow]);
1109 partSplits.push_back(partSplits[flow]);
1110 for (auto [at, other] : partSplits[flow])
1111 partSplits[other].insert({at, part});
1112 for (auto [at, a, b, apart] : splitsOf[flow])
1113 if (ids.count(a) != ids.count(b) &&
1114 flowIds[flow].count(ids.count(a) ? b : a)) {
1115 partSplits[flow].insert({at, part});
1116 partSplits[part].insert({at, flow});
1117 }
1118 return part;
1119}

1121// A link with a channel no flow takes and a free master port driving it
1122// was overused only by the way the flows went; its flows could spread.
1123bool Pathfinder::hasRoom(const SwitchboxConnect &sb) const {
1124 if (sb.srcCoords == sb.dstCoords)
1125 return true;
1126 auto xbar = graph.find({sb.srcCoords, sb.srcCoords});
1127 for (size_t i = 0; i < sb.srcPorts.size(); i++)
1128 for (size_t j = 0; j < sb.dstPorts.size(); j++) {
1129 const SwitchboxConnect::Cell &c = sb.at(i, j);
1130 if (!c.available || c.usedCapacity > 0)
1131 continue;
1132 if (xbar == graph.end())
1133 return true;
1134 const SwitchboxConnect &x = xbar->second;
1135 int k = x.dstIndex(sb.srcPorts[i]);
1136 if (k < 0)
1137 return true;
1138 for (size_t row = 0; row < x.srcPorts.size(); row++)
1139 if (x.at(row, k).available)
1140 return true;
1141 }
1142 return false;
1143}

1145llvm::Error Pathfinder::explainNoRouting(const RouteState &st) const {
1146 // A prioritized flow keeps the route it takes alone, so the others
1147 // may have had to fit around it.
1148 auto overused = [](const SwitchboxConnect &sb) {
1149 return sb.srcCoords != sb.dstCoords &&
1150 llvm::any_of(sb.cells, [](const SwitchboxConnect::Cell &c) {
1151 return c.usedCapacity > maxCircuitStreamCapacity;
1152 });
1153 };
1154 auto link = [](const SwitchboxConnect &sb) -> std::string {
1155 return llvm::formatv("from tile ({0}, {1}) to ({2}, {3})", sb.srcCoords.col,
1156 sb.srcCoords.row, sb.dstCoords.col, sb.dstCoords.row);
1157 };
1158 const Flow *prioritized = nullptr;
1159 llvm::SetVector<const SwitchboxConnect *> held;
1160 for (auto [k, f] : llvm::enumerate(st.parts)) {
1161 if (!st.isPinned(f))
1162 continue;
1163 prioritized = prioritized ? prioritized : &f;
1164 for (const auto &[_, hop] : st.treeOf[k]) {
1165 if (overused(*hop.second.sb)) {
1166 return llvm::make_error<PinnedRoutingFailure>(
1167 describePrioritized(describeTilePort(f.src.coords, f.src.port)) +
1168 ", and it holds a channel " + link(*hop.second.sb) +
1169 " the other flows need.");
1170 }
1171 if (hop.second.sb->srcCoords != hop.second.sb->dstCoords)
1172 held.insert(hop.second.sb);
1173 }
1174 }
1175 if (prioritized && llvm::any_of(graph, [&](const auto &entry) {
1176 return overused(entry.second);
1177 })) {
1178 return llvm::make_error<PinnedRoutingFailure>(
1179 describePrioritized(
1180 describeTilePort(prioritized->src.coords, prioritized->src.port)) +
1181 ", and the router found no routing for the other flows around the "
1182 "channels it holds " +
1183 joinNames(llvm::to_vector(
1184 llvm::map_range(llvm::make_pointee_range(held), link))) +
1185 ".");
1186 }
1187 // The routing check says more than the overuse it led to.
1188 if (!checkReason.empty())
1189 return llvm::make_error<RoutingFailure>(checkReason);
1190 // Name the channel the last iteration overused that was overused in the
1191 // most iterations, on a link with no room if there is one, and the flows
1192 // the last one routed through it.
1193 const SwitchboxConnect *worst = nullptr;
1194 int worstI = 0, worstJ = 0;
1195 std::pair<bool, int> worstRank{false, 0};
1196 for (const auto &[_, sb] : graph) {
1197 std::optional<bool> roomless;
1198 for (size_t i = 0; i < sb.srcPorts.size(); i++)
1199 for (size_t j = 0; j < sb.dstPorts.size(); j++) {
1200 if (sb.at(i, j).usedCapacity <= maxCircuitStreamCapacity)
1201 continue;
1202 if (!roomless)
1203 roomless = !hasRoom(sb);
1204 std::pair<bool, int> rank{*roomless, sb.at(i, j).overCapacity};
1205 if (rank > worstRank) {
1206 worst = &sb;
1207 worstI = i;
1208 worstJ = j;
1209 worstRank = rank;
1210 }
1211 }
1212 }
1213 if (!worst)
1214 return llvm::make_error<RoutingFailure>("");
1215 bool crossbar = worst->srcCoords == worst->dstCoords;
1216 std::vector<std::string> users;
1217 for (const auto &[src, settings] : st.routing.settings) {
1218 auto it = settings.find(worst->srcCoords);
1219 if (it == settings.end())
1220 continue;
1221 const SwitchSetting &s = it->second;
1222 bool uses = false;
1223 for (size_t k = 0; k < s.dsts.size() && !uses; k++)
1224 uses = crossbar
1225 ? k < s.srcs.size() && s.srcs[k] == worst->srcPorts[worstI] &&
1226 s.dsts[k] == worst->dstPorts[worstJ]
1227 : llvm::is_contained(worst->srcPorts, s.dsts[k]);
1228 if (uses)
1229 users.push_back(describeTilePort(src.coords, src.port));
1230 }
1231 std::string where =
1232 crossbar ? llvm::formatv("the connection from {0} to {1} at tile ({2}, "
1233 "{3})",
1234 describePort(worst->srcPorts[worstI]),
1235 describePort(worst->dstPorts[worstJ]),
1236 worst->srcCoords.col, worst->srcCoords.row)
1237 .str()
1238 : "the links " + link(*worst);
1239 if (users.empty())
1240 return llvm::make_error<RoutingFailure>(
1241 "the router found no routing that fits " + where + ".");
1242 return llvm::make_error<RoutingFailure>(
1243 "the flows from " + joinNames(users) + " need " + where +
1244 ", and the router found no routing that fits them.");
1245}

1326Pathfinder::TreeBuilder::TreeBuilder(Pathfinder &pf, RouteState &st, int flow)
1327 : pf(pf), st(st), flow(flow), part(st.parts[flow]),
1328 avoid(part.packetId ? &st.conflicting[flow] : nullptr) {
1329 if (st.isPinned(part))
1330 pinned = &pf.constraints.pinned.at(part.src);
1331 // The flow source port feeds into its switchbox, so the tree starts
1332 // on its In side and the first edge taken is necessarily a crossbar
1333 // hop.
1334 int srcId = pf.nodeIds.at(part.src);
1335 tree.push_back(stateId(srcId, In));
1336 treeHops.push_back(0);
1337 st.processedStamp[tree.front()] = ++st.curStamp;
1338 for (const PathEndPoint &endPoint : part.dsts) {
1339 // Route to self: the port is both ends. Where its switchbox cannot
1340 // connect it to itself (Core to Core), the stream has to leave and
1341 // come back, which Dijkstra finds from the In side to the Out side.
1342 // A tree the flow joins may reach the port for it instead. A port in
1343 // overlayMasters is reached like any other destination, so mayLeave
1344 // still holds.
1345 int self = stateId(srcId, Out);
1346 if (endPoint == part.src && !st.overlayMasters.count(self) &&
1347 llvm::any_of(pf.adjacency[srcId],
1348 [&](const Edge &e) { return e.dst == srcId; })) {
1349 toSelf = true;
1350 continue;
1351 }
1352 pending.push_back(endPoint);
1353 }
1354 findJoins();
1355 for (auto it = st.trunked.lower_bound({flow, 0});
1356 it != st.trunked.end() && it->first == flow; ++it) {
1357 SmallVector<int, 4> shared;
1358 for (const PathEndPoint &dst : st.parts[it->second].dsts)
1359 if (llvm::is_contained(part.dsts, dst))
1360 shared.push_back(dstState(dst));
1361 if (shared.size() > 1)
1362 trunkDsts.insert(shared.begin(), shared.end());
1363 }
1364 for (auto [at, other] : st.partSplits[flow])
1365 for (const auto &[state, _] : st.treeOf[other])
1366 if ((state & 1) == Out && pf.nodes[stateNode(state)].coords == at)
1367 partOff.insert(state);
1368}

1390void Pathfinder::TreeBuilder::findJoins() {
1391 auto shared = [&](ArrayRef<int> ids, int other) {
1392 std::set<int> common;
1393 for (int id : ids)
1394 if (st.flowIds[other].count(id))
1395 common.insert(id);
1396 return common;
1397 };
1398 for (int other : st.sameId[flow]) {
1399 // A tree reaching its own source port takes no hop for it.
1400 const Flow &o = st.parts[other];
1401 int self = dstState(o.src);
1402 if (!st.treeDsts[other].empty() && llvm::is_contained(o.dsts, o.src) &&
1403 isPending(self) && !shared(st.idsTo(other, self), flow).empty())
1404 sharedDsts[other].push_back(self);
1405 for (int dst : st.treeDsts[other]) {
1406 std::set<int> common = shared(st.idsTo(other, dst), flow);
1407 if (common.empty())
1408 continue;
1409 // Joining shares the other flow's arbiters below the join with the
1410 // packets it takes there. A prioritized flow's tree is pinned, so it
1411 // joins only trees pinned with it. A destination of the other flow alone
1412 // keeps the flow off the hops toward it, as no join there is joinable.
1413 bool agree =
1414 (!isPending(dst) || common == shared(st.idsTo(flow, dst), other)) &&
1415 !(pf.constraints.conflict &&
1416 pf.constraints.conflict(part.src, common, o.src,
1417 st.flowIds[other])) &&
1418 !st.noJoin.count({flow, other}) &&
1419 (!part.isPriorityFlow || st.parts[other].isPriorityFlow);
1420 if (!agree)
1421 disagree.insert(other);
1422 else if (isPending(dst))
1423 sharedDsts[other].push_back(dst);
1424 for (auto it = st.treeOf[other].find(dst); it != st.treeOf[other].end();
1425 it = st.treeOf[other].find(it->second.first)) {
1426 int up = it->second.first;
1427 if ((up & 1) != Out)
1428 continue;
1429 stops.insert(up);
1430 if (!agree)
1431 unjoinable.insert(up);
1432 if (!llvm::is_contained(joins[up], dst))
1433 joins[up].push_back(dst);
1434 if (!llvm::is_contained(joinOwners[up], other))
1435 joinOwners[up].push_back(other);
1436 }
1437 }
1438 }
1439 // A tree with other ids reaching two of the flow's destinations is met the
1440 // same way, the flow taking its hops below the meeting with its own ids.
1441 // Not for a destination a tree with one of its ids also reaches, which the
1442 // flow has to meet on the way there instead.
1443 if (!part.packetId || part.isPriorityFlow)
1444 return;
1445 llvm::DenseSet<int> sameIdDsts;
1446 for (int other : st.sameId[flow])
1447 for (const PathEndPoint &dst : st.parts[other].dsts)
1448 if (!shared(st.idsTo(other, dstState(dst)), flow).empty())
1449 sameIdDsts.insert(dstState(dst));
1450 for (auto [other, o] : llvm::enumerate(st.parts)) {
1451 int k = other;
1452 if (k == flow || o.src == part.src || !o.packetId || o.isPriorityFlow ||
1453 st.treeDsts[k].empty() || llvm::is_contained(st.sameId[flow], k) ||
1454 st.conflicting[flow].test(k) || st.noJoin.count({flow, k}))
1455 continue;
1456 SmallVector<int, 4> dsts;
1457 for (int dst : st.treeDsts[k])
1458 if (isPending(dst) && dst != dstState(part.src) &&
1459 !sameIdDsts.count(dst) && st.treeOf[k].count(dst))
1460 dsts.push_back(dst);
1461 if (dsts.size() > 1) {
1462 sharedDsts[k] = std::move(dsts);
1463 foreign.insert(k);
1464 }
1465 }
1466}
1467
1468// Trees from two sources wait on each other wherever they meet (see HoldCycle),
1469// so two trees meeting at two switchboxes can each hold one arbiter and wait at
1470// the other. The flow meets each tree it shares destinations with at one
1471// switchbox, those it shares the most with first, on that tree's way from its
1472// source to where it branches for them, by a slave port that takes each master
1473// port the tree takes there toward them. Only a packet going to two of them can
1474// hold the arbiter where the trees meet for one and wait where they meet for
1475// the other, so two trees meet only when each sends some id to two shared
1476// destinations.
1477void Pathfinder::TreeBuilder::meet() {
1478 auto spans = [&](int k, ArrayRef<int> dsts) {
1479 return llvm::any_of(st.flowIds[k], [&](int id) {
1480 return llvm::count_if(dsts, [&](int dst) {
1481 return llvm::is_contained(st.idsTo(k, dst), id);
1482 }) > 1;
1483 });
1484 };
1485 auto mustMeet = [&](int other) {
1486 return spans(flow, sharedDsts[other]) && spans(other, sharedDsts[other]);
1487 };
1488 SmallVector<int, 4> owners;
1489 for (int other : llvm::make_first_range(sharedDsts))
1490 if (!disagree.count(other) && mustMeet(other))
1491 owners.push_back(other);
1492 llvm::stable_sort(owners, [&](int a, int b) {
1493 return sharedDsts[a].size() > sharedDsts[b].size();
1494 });
1495 for (int owner : owners) {
1496 llvm::erase_if(sharedDsts[owner], [&](int dst) { return !isPending(dst); });
1497 if (mustMeet(owner))
1498 meet(owner);
1499 }
1500}

1502void Pathfinder::TreeBuilder::meet(int owner) {
1503 st.unmet.insert({flow, owner});
1504 bool other = foreign.count(owner);
1505 const auto &t = st.treeOf[owner];
1506 const Flow &o = st.parts[owner];
1507 int root = stateId(pf.nodeIds.at(o.src), In);
1508 // The owner's states from its source down to each shared destination.
1509 SmallVector<SmallVector<int, 16>, 4> paths;
1510 for (int dst : sharedDsts[owner]) {
1511 SmallVector<int, 16> &path = paths.emplace_back(1, dst);
1512 for (auto it = t.find(dst); it != t.end(); it = t.find(it->second.first))
1513 path.push_back(it->second.first);
1514 if (path.size() == 1)
1515 path.push_back(root);
1516 if (path.back() != root)
1517 return;
1518 std::reverse(path.begin(), path.end());
1519 }
1520 size_t trunk = 0;
1521 while (llvm::all_of(paths, [&](const auto &path) {
1522 return trunk + 1 < path.size() && path[trunk] == paths[0][trunk];
1523 }))
1524 trunk++;
1525 struct Meeting {
1526 int in;
1527 SmallVector<std::pair<int, Edge>, 4> hops;
1528 double cost;
1529 size_t at;
1530 };
1531 SmallVector<Meeting, 8> meetings;
1532 SmallVector<int, 8> targets;
1533 // Below a meeting with other ids the flow takes the owner's hops itself.
1534 auto below = [&](size_t k) {
1535 llvm::MapVector<int, std::pair<int, Edge>> hops;
1536 for (const auto &path : paths)
1537 for (size_t l = k + 1; l + 1 < path.size(); l++)
1538 hops.insert({path[l + 1], t.find(path[l + 1])->second});
1539 return hops;
1540 };
1541 for (size_t k = 0; k < trunk; k++) {
1542 int at = paths[0][k];
1543 if ((at & 1) != In)
1544 continue;
1545 SetVector<int> next;
1546 for (const auto &path : paths)
1547 next.insert(path[k + 1]);
1548 if (!other && llvm::any_of(next, [&](int s) {
1549 return !isPending(s) && !(joins.count(s) && joinable(s, joins[s]));
1550 }))
1551 continue;
1552 auto taken = [&](int s) {
1553 return st.processedStamp[s] == st.curStamp || stops.count(s) ||
1554 partOff.count(s);
1555 };
1556 if (other && (llvm::any_of(next, taken) ||
1557 llvm::any_of(below(k), [&](const auto &hop) {
1558 const Edge &e = hop.second.second;
1559 return taken(hop.first) ||
1560 (e.sb->srcCoords == e.sb->dstCoords &&
1561 e.sb->circuitOnlyDst[e.j]);
1562 })))
1563 continue;
1564 SmallVector<int, 4> leaving;
1565 if (llvm::any_of(next, [&](int s) {
1566 bool ok = st.mayLeave(leaving, s);
1567 leaving.push_back(s);
1568 return !ok;
1569 }))
1570 continue;
1571 TileID tile = pf.nodes[stateNode(at)].coords;
1572 auto sb = pf.graph.find({tile, tile});
1573 if (sb == pf.graph.end())
1574 continue;
1575 for (Port port : sb->second.srcPorts) {
1576 auto node = pf.nodeIds.find({tile, port});
1577 if (node == pf.nodeIds.end())
1578 continue;
1579 int in = stateId(node->second, In);
1580 if (in == at || t.count(in))
1581 continue;
1582 Meeting m{in, {}, 0, k};
1583 for (int s : next)
1584 for (const Edge &e : pf.adjacency[node->second])
1585 if (e.dst == stateNode(s) && e.sb == &sb->second &&
1586 !(part.packetId && e.sb->circuitOnlyDst[e.j])) {
1587 m.hops.push_back({s, e});
1588 m.cost += pf.edgeWeight(e, part.packetId, avoid, nullptr);
1589 break;
1590 }
1591 if (m.hops.size() == next.size()) {
1592 meetings.push_back(std::move(m));
1593 targets.push_back(in);
1594 }
1595 }
1596 }
1597 if (meetings.empty())
1598 return;
1599 llvm::DenseSet<int> off;
1600 if (other)
1601 for (const auto &[s, _] : below(0))
1602 off.insert(s);
1603 search({}, targets, off);
1604 const Meeting *best = nullptr;
1605 for (const Meeting &m : meetings)
1606 if (pf.distance[m.in] < INF &&
1607 (!best ||
1608 pf.distance[m.in] + m.cost < pf.distance[best->in] + best->cost))
1609 best = &m;
1610 if (!best || !trace(best->in))
1611 return;
1612 st.unmet.erase({flow, owner});
1613 st.met.insert({flow, owner});
1614 int hops = treeHops[llvm::find(tree, best->in) - tree.begin()] + 1;
1615 auto add = [&](int s, int from, const Edge &e, int depth) {
1616 planned.insert({s, {from, e}});
1617 ++children[from];
1618 st.processedStamp[s] = st.curStamp;
1619 tree.push_back(s);
1620 treeHops.push_back(depth);
1621 };
1622 for (const auto &[s, e] : best->hops) {
1623 add(s, best->in, e, hops);
1624 if (other) {
1625 st.joinedHops[flow].push_back({owner, e});
1626 continue;
1627 }
1628 if (isPending(s)) {
1629 met.push_back(s);
1630 reach(s);
1631 continue;
1632 }
1633 const SmallVector<int, 4> &dsts = joins[s];
1634 reach(dsts);
1635 joinedAt.push_back({s, &dsts});
1636 ++children[s];
1637 }
1638 if (!other)
1639 return;
1640 for (const auto &[s, hop] : below(best->at)) {
1641 const auto &[from, e] = hop;
1642 add(s, from, e, treeHops[llvm::find(tree, from) - tree.begin()] + 1);
1643 st.joinedHops[flow].push_back({owner, e});
1644 }
1645 for (int dst : sharedDsts[owner]) {
1646 met.push_back(dst);
1647 reach(dst);
1648 }
1649}

1651// Dijkstra from the tree, less the states in `drop`, and not through
1652// those in `off`, until the states in `targets` are settled.
1653void Pathfinder::TreeBuilder::search(const llvm::DenseSet<int> &drop,
1654 ArrayRef<int> targets,
1655 const llvm::DenseSet<int> &off) {
1656 branchAvoid.clear();
1657 llvm::DenseMap<int, llvm::BitVector> onArbiter;
1658 if (avoid)
1659 for (const auto &[state, hop] : planned) {
1660 const auto &[from, e] = hop;
1661 if (e.sb->srcCoords != e.sb->dstCoords || drop.count(state))
1662 continue;
1663 llvm::BitVector &away =
1664 branchAvoid.try_emplace(from, st.conflicting.size()).first->second;
1665 llvm::BitVector &on =
1666 onArbiter.try_emplace(from, st.conflicting.size()).first->second;
1667 for (int other : e.sb->unitFlows(e.j)) {
1668 on.set(other);
1669 if (other != flow)
1670 away |= st.conflicting[other];
1671 }
1672 }
1673 // A flow sharing an arbiter with itself is no conflict, and flows
1674 // already on those arbiters share them whatever the branch does.
1675 for (auto &[from, away] : branchAvoid) {
1676 away.reset(onArbiter.find(from)->second);
1677 away.reset(flow);
1678 }
1679 seeds.clear();
1680 seedCosts.clear();
1681 for (auto [state, hops] : llvm::zip_equal(tree, treeHops))
1682 if (!drop.count(state) && !off.count(state)) {
1683 seeds.push_back(state);
1684 seedCosts.push_back(treeSeedFactor * hops);
1685 }
1686 llvm::DenseSet<int> blocked;
1687 if (!off.empty()) {
1688 blocked = stops;
1689 blocked.insert(off.begin(), off.end());
1690 }
1691 llvm::DenseMap<int, SmallVector<int, 4>> masters;
1692 bool limited = !st.overlayMasters.empty();
1693 if (limited)
1694 for (const auto &[state, hop] : planned)
1695 if ((hop.first & 1) == In && !drop.count(state))
1696 masters[hop.first].push_back(state);
1697 auto mayCross = [&](int from, int to) {
1698 auto it = masters.find(from);
1699 return st.mayLeave(it == masters.end() ? ArrayRef<int>{} : it->second, to);
1700 };
1701 pf.dijkstraShortestPaths(
1702 seeds, seedCosts, part.packetId, avoid, &branchAvoid,
1703 off.empty() ? &stops : &blocked, targets,
1704 limited ? llvm::function_ref<bool(int, int)>(mayCross) : nullptr);
1705}

1707// Trace the path Dijkstra found to `state` back to the tree.
1708bool Pathfinder::TreeBuilder::trace(int state) {
1709 size_t grown = tree.size();
1710 while (st.processedStamp[state] != st.curStamp) {
1711 // If Dijkstra never reached this node it has no predecessor; the
1712 // destination is unroutable under the current demand.
1713 int pred = pf.preds[state];
1714 if (pred < 0)
1715 return false;
1716 planned.insert({state, {pred, pf.predEdge[state]}});
1717 ++children[pred];
1718 st.processedStamp[state] = st.curStamp;
1719 tree.push_back(state);
1720 state = pred;
1721 }
1722 // The new hops were traced from the destination back to the tree.
1723 int hops = treeHops[llvm::find(tree, state) - tree.begin()] +
1724 static_cast<int>(tree.size() - grown);
1725 while (treeHops.size() < tree.size())
1726 treeHops.push_back(hops--);
1727 return true;
1728}

1730// The path to `dst` stays off the hops another destination's path
1731// takes below where the check split the two apart, and off the
1732// slave port it takes there too if one of them has to reach the tile
1733// on its own.
1734llvm::DenseSet<int> Pathfinder::TreeBuilder::splitOff(int dst) const {
1735 llvm::DenseSet<int> off = partOff;
1736 auto carries = [&](int d, int id) {
1737 return llvm::is_contained(st.idsTo(flow, d), id);
1738 };
1739 auto only = [&](int d, int id, int other) {
1740 return carries(d, id) && !carries(d, other);
1741 };
1742 for (auto [at, a, b, apart] : st.splitsOf[flow])
1743 for (int other : reached) {
1744 bool splits = !apart ? (only(dst, a, b) && only(other, b, a)) ||
1745 (only(dst, b, a) && only(other, a, b))
1746 : (only(dst, a, b) && other == *apart) ||
1747 (dst == *apart && only(other, a, b));
1748 if (!splits)
1749 continue;
1750 SmallVector<int, 8> below;
1751 int s = other;
1752 auto entersAt = [&, tile = at](int state) {
1753 return (state & 1) == In && pf.nodes[stateNode(state)].coords == tile;
1754 };
1755 for (auto *hop = planned.find(s); !entersAt(s) && hop != planned.end();
1756 hop = planned.find(s)) {
1757 below.push_back(s);
1758 s = hop->second.first;
1759 }
1760 if (!entersAt(s))
1761 continue;
1762 off.insert(below.begin(), below.end());
1763 if (apart)
1764 off.insert(s);
1765 }
1766 return off;
1767}

1769// The path to a trunked destination leaves the tree below the master port
1770// the others it has reached leave the source switchbox by.
1771llvm::DenseSet<int> Pathfinder::TreeBuilder::trunkOff(int dst) const {
1772 llvm::DenseSet<int> off;
1773 if (!trunkDsts.count(dst))
1774 return off;
1775 int root = tree.front();
1776 auto exitOf = [&](int s) {
1777 for (const auto *it = planned.find(s);
1778 it != planned.end() && it->second.first != root; it = planned.find(s))
1779 s = it->second.first;
1780 return s;
1781 };
1782 const auto *other = llvm::find_if(reached, [&](int d) {
1783 return d != dst && trunkDsts.count(d) && planned.count(d);
1784 });
1785 if (other == reached.end())
1786 return off;
1787 int exit = exitOf(*other);
1788 for (int s : tree)
1789 if (s == root || exitOf(s) != exit)
1790 off.insert(s);
1791 return off;
1792}

1794// Lay the tree along the route the flow takes alone.
1795llvm::Error Pathfinder::TreeBuilder::placePinned() {
1796 for (const auto &[from, to, joined] : *pinned) {
1797 if (toSelf && from == to)
1798 continue;
1799 bool intra = from.coords == to.coords;
1800 auto fromId = pf.nodeIds.find(from), toId = pf.nodeIds.find(to);
1801 const Edge *e = nullptr;
1802 if (fromId != pf.nodeIds.end() && toId != pf.nodeIds.end())
1803 for (const Edge &out : pf.adjacency[fromId->second])
1804 if (out.dst == toId->second &&
1805 (out.sb->srcCoords == out.sb->dstCoords) == intra)
1806 e = &out;
1807 if (!e)
1808 return llvm::make_error<RoutingFailure>(
1809 "the route packet flows from " +
1810 describeTilePort(part.src.coords, part.src.port) +
1811 " take alone does not fit this design: it goes from " +
1812 describeTilePort(from.coords, from.port) + " to " +
1813 describeTilePort(to.coords, to.port) +
1814 ", which the design's own switchboxes leave no connection for.",
1815 RoutingFaults{}, pf.flowLoc(part.src, nullptr));
1816 std::pair<int, std::pair<int, Edge>> hop{
1817 stateId(toId->second, intra ? Out : In),
1818 {stateId(fromId->second, intra ? In : Out), *e}};
1819 if (joined)
1820 pinnedJoins.push_back(hop);
1821 else
1822 planned.insert(hop);
1823 }
1824 int self = dstState(part.src);
1825 if (toSelf && llvm::is_contained(llvm::make_first_range(pinnedJoins), self)) {
1826 toSelf = false;
1827 pinnedJoinDsts.push_back(self);
1828 }
1829 for (const PathEndPoint &p : pending)
1830 (planned.count(dstState(p)) ? reached : pinnedJoinDsts)
1831 .push_back(dstState(p));
1832 pending.clear();
1833 return llvm::Error::success();
1834}

1836// Grow the tree one destination at a time: Dijkstra, given the current
1837// demand, from everything the tree reaches so far to the next destination,
1838// whose path is then traced back to the tree. Growing from the tree rather
1839// than the source lets destinations share hops; see treeSeedFactor for what
1840// a branch off the tree costs.
1841llvm::Error Pathfinder::TreeBuilder::grow() {
1842 // A path to another tree can leave the destination to reach first no way
1843 // to it, so the tree meets the others once it reaches it.
1844 bool metTrees = !first || !llvm::is_contained(pending, *first);
1845 if (metTrees)
1846 meet();
1847 while (!pending.empty()) {
1848 SmallVector<int, 8> targets;
1849 for (const PathEndPoint &p : pending)
1850 targets.push_back(dstState(p));
1851 for (const auto &[at, dsts] : joins)
1852 if (joinable(at, dsts))
1853 targets.push_back(at);
1854 search({}, targets);
1855 // The nearest destination joins the tree next, or the nearest join
1856 // brings every destination below it.
1857 auto *nearest =
1858 first && llvm::is_contained(pending, *first)
1859 ? llvm::find(pending, *first)
1860 : llvm::min_element(
1861 pending, [&](const PathEndPoint &a, const PathEndPoint &b) {
1862 return pf.distance[dstState(a)] < pf.distance[dstState(b)];
1863 });
1864 PathEndPoint endPoint = *nearest;
1865 int currId = dstState(endPoint);
1866 llvm::DenseSet<int> off = splitOff(currId);
1867 llvm::DenseSet<int> trunk = trunkOff(currId);
1868 off.insert(trunk.begin(), trunk.end());
1869 if (!off.empty()) {
1870 search({}, targets, off);
1871 if (pf.distance[currId] == INF)
1872 search({}, targets);
1873 }
1874 const SmallVector<int, 4> *joined = nullptr;
1875 for (const auto &[at, dsts] : joins)
1876 if (pf.distance[at] < pf.distance[currId] && joinable(at, dsts)) {
1877 currId = at;
1878 joined = &dsts;
1879 }
1880 if (joined) {
1881 reach(*joined);
1882 joinedAt.push_back({currId, joined});
1883 // The tree it joins goes on below.
1884 ++children[currId];
1885 } else {
1886 pending.erase(nearest);
1887 reached.push_back(currId);
1888 }
1889 if (!trace(currId))
1890 return llvm::make_error<RoutingFailure>(
1891 "no path leads from " +
1892 describeTilePort(part.src.coords, part.src.port) + " to " +
1893 describeTilePort(endPoint.coords, endPoint.port) +
1894 " through the connections the switchboxes allow and existing "
1895 "routing leaves free.",
1896 RoutingFaults{}, pf.flowLoc(part.src, &endPoint));
1897 if (!metTrees) {
1898 metTrees = true;
1899 meet();
1900 }
1901 }
1902 return llvm::Error::success();
1903}

1905// A destination's path was chosen before the tree reached the later
1906// ones, so reroute each destination's own branch from the rest of the
1907// tree where that is cheaper.
1908void Pathfinder::TreeBuilder::reroute() {
1909 for (int dst : reached) {
1910 llvm::DenseSet<int> branch;
1911 int top = dst;
1912 for (int below = 0; top != tree.front() && children.lookup(top) == below;
1913 below = 1) {
1914 branch.insert(top);
1915 top = planned.find(top)->second.first;
1916 }
1917 if (branch.empty())
1918 continue;
1919 llvm::DenseSet<int> off = splitOff(dst), trunk = trunkOff(dst);
1920 off.insert(trunk.begin(), trunk.end());
1921 search(branch, dst, off);
1922 double cost =
1923 treeSeedFactor * treeHops[llvm::find(tree, top) - tree.begin()];
1924 for (int s = dst; s != top;) {
1925 const auto &[from, e] = planned.find(s)->second;
1926 auto away = branchAvoid.find(from);
1927 cost +=
1928 pf.edgeWeight(e, part.packetId, avoid,
1929 away == branchAvoid.end() ? nullptr : &away->second);
1930 s = from;
1931 }
1932 if (pf.distance[dst] + rerouteMinSaving >= cost)
1933 continue;
1934 for (int s : branch) {
1935 st.processedStamp[s] = 0;
1936 planned.erase(s);
1937 children.erase(s);
1938 }
1939 --children[top];
1940 for (size_t k = tree.size(); k-- > 0;)
1941 if (branch.count(tree[k])) {
1942 tree.erase(tree.begin() + k);
1943 treeHops.erase(treeHops.begin() + k);
1944 }
1945 [[maybe_unused]] bool traced = trace(dst);
1946 assert(traced && "a rerouted branch reaches its destination");
1947 }
1948}

1950// Take the channels the tree's hops cross.
1951void Pathfinder::TreeBuilder::claim() {
1952 if (toSelf) {
1953 switchSettings[part.src.coords].srcs.push_back(part.src.port);
1954 switchSettings[part.src.coords].dsts.push_back(part.src.port);
1955 if (part.packetId)
1956 st.routing.packetTrees[part.src].push_back({part.src, part.src, false});
1957 }
1958 const std::optional<int> &packetId = part.packetId;
1959 int packetGroupId = part.packetGroupId;
1960 llvm::DenseMap<int, std::pair<SwitchboxConnect *, int>> branchPort;
1961 for (const auto &[currId, hop] : planned) {
1962 const auto &[predId, e] = hop;
1963 const PathEndPoint &curr = pf.nodes[stateNode(currId)];
1964 const PathEndPoint &pred = pf.nodes[stateNode(predId)];
1965 SwitchboxConnect &sb = *e.sb;
1966 int i = e.i;
1967 int j = e.j;
1968 SwitchboxConnect::Cell &cell = sb.at(i, j);
1969 if (packetId)
1970 st.treeOf[flow].try_emplace(currId, predId, e);
1971 cell.isPriority |= part.isPriorityFlow;
1972 // Two trees with the same id may merge onto one only where nothing fans out
1973 // after, or where both take the id to the same destinations, so a merged id
1974 // never fans back out to one tree's destinations alone. The flow's own tree
1975 // branching at a port never merges back.
1976 // packetGroupId only becomes >= 0 when packetId has a value (see
1977 // Pathfinder::addFlow), so the dereferences below are safe; the
1978 // checker just can't correlate the two across this loop's back edge.
1979 // NOLINTBEGIN(bugprone-unchecked-optional-access)
1980 auto seen =
1981 packetId ? cell.packetIds.find(*packetId) : cell.packetIds.end();
1982 auto mergeable = [&] {
1983 if (!isDirectional(sb.dstPorts[j].bundle))
1984 return true;
1985 auto reached = [&](const Flow &f) {
1986 std::set<PathEndPoint> dsts;
1987 for (const PathEndPoint &dst : f.dsts)
1988 if (auto ids = pf.packetIdsTo.find({f.src, dst});
1989 ids != pf.packetIdsTo.end() &&
1990 llvm::is_contained(ids->second, *packetId))
1991 dsts.insert(dst);
1992 return dsts;
1993 };
1994 return reached(part) == reached(st.parts[seen->second]);
1995 };
1996 bool sameGroupUnseen =
1997 packetGroupId >= 0 && packetId.has_value() &&
1998 (cell.packetGroupId == -1 || cell.packetGroupId == packetGroupId) &&
1999 (seen == cell.packetIds.end() || seen->second == flow || mergeable());
2000 if (sameGroupUnseen) {
2001 int packetIdValue = *packetId;
2002 // NOLINTEND(bugprone-unchecked-optional-access)
2003 auto claim = [&](SwitchboxConnect::Cell &c) {
2004 c.packetGroupId = packetGroupId;
2005 c.packetIds.try_emplace(packetIdValue, flow);
2006 };
2007 for (size_t k = 0; k < sb.srcPorts.size(); k++)
2008 claim(sb.at(k, j));
2009 for (size_t l = 0; l < sb.dstPorts.size(); l++)
2010 if (l != static_cast<size_t>(j))
2011 claim(sb.at(i, l));
2012 // maximum packet stream sharing per channel
2013 if (++cell.packetFlowCount >= maxPacketStreamCapacity) {
2014 cell.packetFlowCount = 0;
2015 cell.usedCapacity++;
2016 }
2017 } else {
2018 cell.usedCapacity++;
2019 }
2020 // if at capacity, bump demand to discourage using this Channel
2021 // this means the order matters!
2022 SwitchboxConnect::bumpDemand(cell);
2023 if (pred.coords == curr.coords) {
2024 switchSettings[pred.coords].srcs.push_back(pred.port);
2025 switchSettings[curr.coords].dsts.push_back(curr.port);
2026 if (packetId) {
2027 sb.addToUnit(j, flow);
2028 auto [it, first] = branchPort.try_emplace(predId, &sb, j);
2029 if (!first)
2030 sb.joinUnits(it->second.second, j);
2031 }
2032 }
2033 }
2034 if (packetId) {
2035 st.treeDsts[flow].append(reached.begin(), reached.end());
2036 st.treeDsts[flow].append(met.begin(), met.end());
2037 for (const auto &[currId, hop] : planned)
2038 st.routing.packetTrees[part.src].push_back(
2039 {pf.nodes[stateNode(hop.first)], pf.nodes[stateNode(currId)], false});
2040 }
2041}

2043// Below a join the flow's packets follow the tree they joined.
2044bool Pathfinder::TreeBuilder::join(int state, int predId, const Edge &e) {
2045 if (!st.treeOf[flow].try_emplace(state, predId, e).second)
2046 return false;
2047 const PathEndPoint &pred = pf.nodes[stateNode(predId)];
2048 const PathEndPoint &curr = pf.nodes[stateNode(state)];
2049 st.routing.packetTrees[part.src].push_back({pred, curr, true});
2050 if (pred.coords == curr.coords) {
2051 switchSettings[pred.coords].srcs.push_back(pred.port);
2052 switchSettings[curr.coords].dsts.push_back(curr.port);
2053 e.sb->addToUnit(e.j, flow);
2054 }
2055 return true;
2056}

2058void Pathfinder::TreeBuilder::joinTrees() {
2059 for (const auto &[joinAt, joined] : joinedAt) {
2060 for (int other : joinOwners[joinAt]) {
2061 st.joinedHops[flow].push_back(
2062 {other, planned.find(joinAt)->second.second});
2063 for (int dst : *joined) {
2064 const auto &t = st.treeOf[other];
2065 SmallVector<std::pair<int, std::pair<int, Edge>>, 8> path;
2066 auto it = t.find(dst);
2067 for (; it != t.end() && it->first != joinAt;
2068 it = t.find(it->second.first))
2069 path.push_back(*it);
2070 if (it == t.end())
2071 continue;
2072 for (const auto &[state, hop] : path)
2073 if (join(state, hop.first, hop.second))
2074 st.joinedHops[flow].push_back({other, hop.second});
2075 }
2076 }
2077 st.treeDsts[flow].append(joined->begin(), joined->end());
2078 }
2079 for (const auto &[state, hop] : pinnedJoins)
2080 join(state, hop.first, hop.second);
2081 st.treeDsts[flow].append(pinnedJoinDsts.begin(), pinnedJoinDsts.end());
2082}

2104llvm::Error Pathfinder::routePart(RouteState &st, int flow) {
2105 std::optional<TreeBuilder> tree;
2106 tree.emplace(*this, st, flow);
2107 llvm::Error err = tree->pinned ? tree->placePinned() : tree->grow();
2108 // Kept to the prioritized flows' master sets, the branch a tree takes first
2109 // can leave it no way to another destination; each is then tried first.
2110 if (err && !tree->pinned && !st.overlayMasters.empty())
2111 for (const PathEndPoint &dst : st.parts[flow].dsts) {
2112 for (auto *pairs : {&st.unmet, &st.met})
2113 pairs->erase(pairs->lower_bound({flow, 0}),
2114 pairs->lower_bound({flow + 1, 0}));
2115 st.joinedHops[flow].clear();
2116 tree.emplace(*this, st, flow);
2117 tree->first = dst;
2118 if (llvm::Error retry = tree->grow()) {
2119 llvm::consumeError(std::move(retry));
2120 continue;
2121 }
2122 llvm::consumeError(std::move(err));
2123 break;
2124 }
2125 if (err)
2126 return err;
2127 if (!tree->pinned && tree->reached.size() + tree->joinedAt.size() > 1)
2128 tree->reroute();
2129 tree->claim();
2130 tree->joinTrees();
2131 tree->record();
2132 return llvm::Error::success();
2133}

2135int Pathfinder::applyRoutingFaults(RouteState &st,
2136 const RoutingFaults &faults) {
2137 int illegalEdges = 0;
2138 for (const TreeSplit &split : faults.splits) {
2139 auto src = st.partsOf.find(split.src);
2140 if (src == st.partsOf.end())
2141 continue;
2142 auto partWith = [&](int id) {
2143 for (int k : src->second)
2144 if (st.flowIds[k].count(id))
2145 return k;
2146 return -1;
2147 };
2148 int flow = partWith(split.a), other = partWith(split.b);
2149 if (flow < 0 || other < 0)
2150 continue;
2151 std::optional<int> apart;
2152 if (split.apart) {
2153 auto it = nodeIds.find(*split.apart);
2154 if (it == nodeIds.end())
2155 continue;
2156 apart = stateId(it->second, Out);
2157 }
2158 // Whether a destination taking both ids is reached through the tile.
2159 auto inseparable = [&] {
2160 const Flow &f = st.parts[flow];
2161 for (const PathEndPoint &dst : f.dsts) {
2162 auto ids = packetIdsTo.find({f.src, dst});
2163 if (ids == packetIdsTo.end() ||
2164 !llvm::is_contained(ids->second, split.a) ||
2165 !llvm::is_contained(ids->second, split.b))
2166 continue;
2167 const auto &tree = st.treeOf[flow];
2168 auto it = tree.find(stateId(nodeIds.at(dst), Out));
2169 if (it == tree.end() && dst == f.src && dst.coords == split.at)
2170 return true;
2171 for (; it != tree.end(); it = tree.find(it->second.first)) {
2172 int up = it->second.first;
2173 if ((up & 1) == In && nodes[stateNode(up)].coords == split.at)
2174 return true;
2175 }
2176 }
2177 return false;
2178 };
2179 if (idsApart && flow == other && !split.apart &&
2180 !st.parts[flow].isPriorityFlow && inseparable())
2181 other = st.splitPart(flow, {split.b});
2182 if (flow != other) {
2183 if (st.partSplits[flow].insert({split.at, other}).second)
2184 illegalEdges++;
2185 st.partSplits[other].insert({split.at, flow});
2186 } else if (st.splitsOf[flow]
2187 .insert({split.at, split.a, split.b, apart})
2188 .second) {
2189 illegalEdges++;
2190 }
2191 }
2192 crowdedTiles.insert(faults.crowded.begin(), faults.crowded.end());
2193 // Trees with an id in common join where they meet, so the ids a source
2194 // sends that the other does not route apart as a part of their own.
2195 auto splitOwnIds = [&](const std::pair<PathEndPoint, PathEndPoint> &pair) {
2196 for (auto [src, other] : {pair, std::pair{pair.second, pair.first}}) {
2197 auto srcParts = st.partsOf.find(src);
2198 auto otherParts = st.partsOf.find(other);
2199 if (srcParts == st.partsOf.end() || otherParts == st.partsOf.end())
2200 continue;
2201 std::set<int> otherIds;
2202 for (int k : otherParts->second)
2203 otherIds.insert(st.flowIds[k].begin(), st.flowIds[k].end());
2204 for (int k : SmallVector<int, 2>(srcParts->second)) {
2205 std::set<int> own;
2206 for (int id : st.flowIds[k])
2207 if (!otherIds.count(id))
2208 own.insert(id);
2209 if (!st.parts[k].isPriorityFlow && !own.empty() &&
2210 own.size() < st.flowIds[k].size())
2211 st.splitPart(k, own);
2212 }
2213 }
2214 };
2215 bool related = false;
2216 for (const auto &pair : faults.apart) {
2217 if (!st.apartSources.insert(pair).second)
2218 continue;
2219 related = true;
2220 LLVM_DEBUG(llvm::dbgs()
2221 << "Route apart: "
2222 << describeTilePort(pair.first.coords, pair.first.port)
2223 << " and "
2224 << describeTilePort(pair.second.coords, pair.second.port)
2225 << "\n");
2226 if (splitShared)
2227 splitOwnIds(pair);
2228 }
2229 for (const auto &pair : faults.together) {
2230 if (!st.togetherSources.insert(pair).second)
2231 continue;
2232 related = true;
2233 LLVM_DEBUG(llvm::dbgs()
2234 << "Route together: "
2235 << describeTilePort(pair.first.coords, pair.first.port)
2236 << " and "
2237 << describeTilePort(pair.second.coords, pair.second.port)
2238 << "\n");
2239 splitOwnIds(pair);
2240 }
2241 if (related) {
2242 illegalEdges++;
2243 st.relateParts();
2244 }
2245 std::set<std::pair<int, int>> faulted;
2246 bool unjoined = false;
2247 for (const auto &[tile, conn] : faults.connections) {
2248 illegalEdges++;
2249 auto it = graph.find({tile, tile});
2250 if (it == graph.end())
2251 continue;
2252 SwitchboxConnect &sb = it->second;
2253 int i = sb.srcIndex(conn.src);
2254 int j = sb.dstIndex(conn.dst);
2255 if (i < 0 || j < 0)
2256 continue;
2257 sb.at(i, j).overCapacity += routingCheckPenalty;

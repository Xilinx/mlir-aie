65 AIEGenerateColumnControlOverlayPass> {
66 using Base::Base;
67 DenseMap<std::pair<int, int>, unsigned> packetDemand;

70 void runOnOperation() override {
71 llvm::MapVector<AIEDevice, SmallVector<DeviceOp>> groups;
72 for (DeviceOp device : getOperation().getOps<DeviceOp>())
73 if (getsColumnControlOverlay(device, configuration))
74 groups[device.getDevice()].push_back(device);

80 LogicalResult generate(ArrayRef<DeviceOp> users) {
81 ModuleOp module = getOperation();
82 AIEDialect::IsCtrlPktOverlayAttrHelper marker(&getContext());
83 DeviceOp first = users.front();
84 auto target = first.getDevice();
85 auto tokenTiles = collectTokenIssuingTiles(users);
86 std::set<int> columns;
87 if (configuration)
88 for (DeviceOp device : users)
89 for (TileOp tile : device.getOps<TileOp>())
90 columns.insert(tile.colIndex());
91 for (TileID tile : tokenTiles)
92 columns.insert(tile.col);
93 if (configuration) {
94 for (DeviceOp device : users)
95 device.walk([&](AIEX::NpuControlPacketOp packet) {
96 columns.insert(packet.getColumnFromAddr());
97 });
98 columns.insert(coverColumns.begin(), coverColumns.end());
99 collectShimMM2SUses(users);
100 }
101 SymbolTable symbols(module);
102 OpBuilder builder(module.getContext());
103 builder.setInsertionPointToEnd(module.getBody());
104 DeviceOp overlay =
105 DeviceOp::create(builder, module.getLoc(), target,
106 builder.getStringAttr("ctrl_pkt_overlay"), nullptr);
107 symbols.insert(overlay);
108 overlay.getRegion().emplaceBlock();
109 builder.setInsertionPointToEnd(overlay.getBody());
110 EndOp::create(builder, overlay.getLoc());
111 marker.setAttr(overlay, builder.getUnitAttr());
112 builder.setInsertionPointToStart(overlay.getBody());
113 for (DeviceOp device : users)
114 for (TileOp tile : device.getOps<TileOp>()) {
115 TileOp dest = TileOp::getOrCreate(builder, overlay, tile.colIndex(),
116 tile.rowIndex());
117 if (auto id = tile->getAttr("controller_id")) {
118 if (auto other = dest->getAttr("controller_id");
119 other && other != id) {
120 tile.emitOpError(
121 "has a different controller ID in another configuration");
122 return failure();
123 }
124 dest->setAttr("controller_id", id);
125 }
126 }
127 const auto &model = overlay.getTargetModel();
128 auto ids = model.getTileToControllerIdMap(true);
129 auto flow = [&](TileOp source, Port src, TileOp dest, Port dst,
130 TileOp controller) {
131 builder.setInsertionPoint(overlay.getBody()->getTerminator());
132 int id = ids[controller.getTileID()];
133 if (auto attr =
134 controller->getAttrOfType<PacketInfoAttr>("controller_id"))
135 id = attr.assignedId();
136 auto packet = PacketFlowOp::create(builder, controller.getLoc(), id,
137 builder.getBoolAttr(true),
138 builder.getBoolAttr(true));
139 OpBuilder::InsertionGuard guard(builder);
140 builder.createBlock(&packet.getPorts());
141 PacketSourceOp::create(builder, packet.getLoc(), source, src.bundle,
142 src.channel);
143 PacketDestOp::create(builder, packet.getLoc(), dest, dst.bundle,
144 dst.channel);
145 EndOp::create(builder, packet.getLoc());
146 };
147 std::map<int, int> controlChannel;
148 for (int col : columns) {
149 if (col < 0 || col >= model.columns()) {
150 overlay.emitOpError("control coverage column is outside the target");
151 return failure();
152 }
153 builder.setInsertionPointToStart(overlay.getBody());
154 TileOp shim = TileOp::getOrCreate(builder, overlay, col, 0);
155 flow(shim, {WireBundle::TileControl, 0}, shim, {WireBundle::South, 0},
156 shim);
157 for (TileID tile : tokenTiles)
158 if (tile.col == col && tile.row != 0) {
159 builder.setInsertionPointToStart(overlay.getBody());
160 TileOp source = TileOp::getOrCreate(builder, overlay, col, tile.row);
161 flow(source, {WireBundle::TileControl, 0}, shim,
162 {WireBundle::South, 0}, source);
163 }
164 if (!configuration)
165 continue;
166 int channel = -1;
167 unsigned best = std::numeric_limits<unsigned>::max();
168 int count =
169 adaptive
170 ? model.getNumSourceShimMuxConnections(col, 0, WireBundle::DMA)
171 : 1;
172 for (int ch = 0; ch < count; ++ch)
173 if (!fixedCircuits.lookup(col).contains(ch) &&
174 packetDemand.lookup({col, ch}) < best) {
175 channel = ch;
176 best = packetDemand.lookup({col, ch});
177 }
178 if (channel < 0) {
179 overlay.emitOpError("failed to generate control overlay: no available "
180 "shim MM2S channel in column ")
181 << col;
182 return failure();
183 }
184 controlChannel[col] = channel;
185 builder.setInsertionPoint(overlay.getBody()->getTerminator());
186 auto alloc = ShimDMAAllocationOp::create(
187 builder, shim.getLoc(), getCtrlPktAllocName(col, channel), shim,
188 DMAChannelDir::MM2S, channel, false, nullptr, nullptr);
189 marker.setAttr(alloc, builder.getUnitAttr());
190 for (int row = 0; row < model.rows(); ++row) {
191 builder.setInsertionPointToStart(overlay.getBody());
192 TileOp dest = TileOp::getOrCreate(builder, overlay, col, row);
193 flow(shim, {WireBundle::DMA, channel}, dest,
194 {WireBundle::TileControl, 0}, dest);
195 }
196 }
197 if (configuration && multicastSpan &&
198 failed(addMulticasts(builder, overlay, users, controlChannel)))
199 return failure();
200 for (DeviceOp device : users) {
201 device.setOverlayAttr(FlatSymbolRefAttr::get(overlay.getSymNameAttr()));
202 if (configuration)
203 AIEDialect::ReserveControllerPacketIdsAttrHelper(&getContext())
204 .setAttr(device, builder.getUnitAttr());
205 }
206 if (!configuration)
207 return success();
208 std::vector<std::pair<TileID, Port>> avoided;
209 if (adaptive && failed(applicationPorts(module, avoided)))
210 return failure();
211 // Detours around the application can fill a control channel's slave port
212 // with rules, leaving none for a data flow that shares the channel. Such
213 // an overlay routes again without the detours.
214 OwningOpRef<Operation *> unrouted;
215 if (!avoided.empty())
216 unrouted = overlay->clone();
217 if (failed(route(overlay, avoided)))
218 return failure();
219 if (unrouted && !controlChannelsHaveRuleSlots(overlay)) {
220 overlay.getRegion().takeBody(unrouted.get()->getRegion(0));
221 if (failed(route(overlay, {})))
222 return failure();
223 }
224 // Reprogramming another master set of an active arbiter can stall control
225 // packets.
226 for (auto box : overlay.getOps<SwitchboxOp>()) {
227 auto amsels = llvm::to_vector(box.getConnections().getOps<AMSelOp>());
228 std::set<std::pair<int, int>> slots;
229 for (auto amsel : amsels)
230 slots.emplace(amsel.arbiterIndex(), amsel.getMselValue());
231 builder.setInsertionPoint(box.getConnections().front().getTerminator());
232 for (auto amsel : amsels)
233 for (unsigned msel = 0; msel < model.getNumMselsPerArbiter(); ++msel)
234 if (slots.emplace(amsel.arbiterIndex(), msel).second)
235 AMSelOp::create(builder, amsel.getLoc(), amsel.arbiterIndex(),
236 msel);
237 }
238 overlay.walk([&](Operation *op) {
239 op->removeAttr("aie.priority_route");
240 if (!isa<DeviceOp, EndOp, PacketRulesOp>(op))
241 marker.setAttr(op, builder.getUnitAttr());
242 });
243 return success();
244 }

254 // Whether the slave port by which each control channel enters its
255 // switchbox has a packet rule slot left.
256 static bool controlChannelsHaveRuleSlots(DeviceOp overlay) {
257 unsigned slots = overlay.getTargetModel().getNumSlaveSlots();
258 std::map<TileID, SwitchboxOp> boxes;
259 for (auto box : overlay.getOps<SwitchboxOp>())
260 boxes[box.getTileOp().getTileID()] = box;
261 std::map<std::pair<TileID, int>, Port> entries;
262 for (auto mux : overlay.getOps<ShimMuxOp>())
263 for (auto connect : mux.getConnections().getOps<ConnectOp>())
264 if (connect.getSourceBundle() == WireBundle::DMA)
265 entries[{mux.getTileOp().getTileID(), connect.sourceIndex()}] = {
266 WireBundle::South, connect.destIndex()};
267 for (auto alloc : overlay.getOps<ShimDMAAllocationOp>()) {
268 TileID tile = alloc.getTileOp().getTileID();
269 auto entry = entries.find({tile, (int)alloc.getChannelIndex()});
270 auto box = boxes.find(tile);
271 if (entry == entries.end() || box == boxes.end())
272 continue;
273 for (auto rules : box->second.getConnections().getOps<PacketRulesOp>())
274 if (rules.sourcePort() == entry->second &&
275 llvm::range_size(rules.getRules().getOps<PacketRuleOp>()) >= slots)
276 return false;
277 }
278 return true;
279 }

281 // See the pass description's multicast-span. Every group shares one id: the
282 // router keeps two trees on one id apart unless they reach the same tiles,
283 // as it does the controllers' ids, which repeat in every column.
284 LogicalResult addMulticasts(OpBuilder &builder, DeviceOp overlay,
285 ArrayRef<DeviceOp> users,
286 const std::map<int, int> &controlChannel) {
287 std::set<TileID> cores;
288 for (DeviceOp device : users)
289 for (CoreOp core : device.getOps<CoreOp>())
290 cores.insert(core.getTileOp().getTileID());
291 if (cores.empty())
292 return success();
293 int base = cores.begin()->col;
294 std::map<int, SmallVector<TileID>> groups;
295 for (TileID t : cores)
296 groups[(t.col - base) / static_cast<int>(multicastSpan)].push_back(t);
297 for (auto it = groups.begin(); it != groups.end();)
298 it = it->second.size() < 2 ? groups.erase(it) : std::next(it);
299 if (groups.empty())
300 return success();
301
302 SmallVector<PacketIdSpace> spaces;
303 for (DeviceOp device : users)
304 spaces.emplace_back(device, /*reserveControllerIds=*/true);
305 // The router folds the ids headed for one arbiter into mask/value rules,
306 // so the highest free id, among the controllers' where one is free, keeps
307 // those masks narrow.
308 int id = overlay.getTargetModel().getMaxPacketId();
309 for (; id >= 0; --id)
310 if (llvm::none_of(spaces, [&](const PacketIdSpace &space) {
311 return space.isTaken(id);
312 }))
313 break;
314 if (id < 0)
315 return success();
316
317 MLIRContext *ctx = overlay.getContext();
318 SmallVector<Attribute> record;
319 for (auto &[g, members] : groups) {
320 int srcCol = members.front().col;
321 auto channel = controlChannel.find(srcCol);
322 if (channel == controlChannel.end())
323 return overlay.emitOpError("has no control channel in column ")
324 << srcCol << " for a control multicast";
325
326 builder.setInsertionPointToStart(overlay.getBody());
327 TileOp shim = TileOp::getOrCreate(builder, overlay, srcCol, 0);
328 SmallVector<TileOp> tiles;
329 SmallVector<Attribute> listed;
330 for (TileID t : members) {
331 tiles.push_back(TileOp::getOrCreate(builder, overlay, t.col, t.row));
332 listed.push_back(TileAttr::get(ctx, t.col, t.row));
333 }
334 builder.setInsertionPoint(overlay.getBody()->getTerminator());
335 auto flow = PacketFlowOp::create(builder, shim.getLoc(), id,
336 builder.getBoolAttr(true),
337 builder.getBoolAttr(true));
338 OpBuilder::InsertionGuard guard(builder);
339 builder.createBlock(&flow.getPorts());
340 PacketSourceOp::create(builder, flow.getLoc(), shim, WireBundle::DMA,
341 channel->second);
342 for (TileOp t : tiles)
343 PacketDestOp::create(builder, flow.getLoc(), t, WireBundle::TileControl,
344 0);
345 EndOp::create(builder, flow.getLoc());
346 record.push_back(DictionaryAttr::get(
347 ctx,
348 {builder.getNamedAttr(
349 "allocation",
350 FlatSymbolRefAttr::get(
351 ctx, getCtrlPktAllocName(srcCol, channel->second))),
352 builder.getNamedAttr("packet_id", builder.getI32IntegerAttr(id)),
353 builder.getNamedAttr("tiles", ArrayAttr::get(ctx, listed))}));
354 }
355 AIEDialect::CtrlPktMulticastAttrHelper(ctx).setAttr(
356 overlay, ArrayAttr::get(ctx, record));
357 return success();
358 }

360 void collectShimMM2SUses(ArrayRef<DeviceOp> devices) {
361 packetDemand.clear();
362 fixedCircuits.clear();
363 for (DeviceOp device : devices) {
364 ShimMM2SUses uses = shimMM2SUses(device, packetSwitched);
365 for (auto &[tile, channels] : uses.circuits)
366 if (std::optional<int> col = cast<TileLike>(tile).tryGetCol())
367 fixedCircuits[*col].insert(channels.begin(), channels.end());
368 for (auto &[at, ends] : uses.packetEnds)
369 if (std::optional<int> col = cast<TileLike>(at.first).tryGetCol())
370 packetDemand[{*col, at.second}] += ends.size();
371 for (auto &[at, flows] : uses.packetFlows)
372 if (std::optional<int> col = cast<TileLike>(at.first).tryGetCol())
373 packetDemand[{*col, at.second}] += flows;
374 }
375 }

377 static llvm::SmallSet<AIE::TileID, 4>
378 collectTokenIssuingTiles(ArrayRef<DeviceOp> devices) {
379 llvm::SmallSet<AIE::TileID, 4> tiles;
380 for (DeviceOp device : devices) {
381 for (TileOp tile : device.getOps<TileOp>())
382 if (tile.isShimNOCorPLTile())
383 tiles.insert(tile.getTileID());
384 SymbolTable symbols(device);
385 device.walk([&](Operation *op) {
386 bool issuesToken = false;
387 if (auto task = dyn_cast<AIEX::DMAConfigureTaskOp>(op))
388 issuesToken = bool(task.getIssueToken());
389 else if (auto task = dyn_cast<AIEX::DMAConfigureTaskForOp>(op))
390 issuesToken = bool(task.getIssueToken());
391 if (issuesToken)
392 if (auto tile = dyn_cast_or_null<TileOp>(
393 AIEX::getRuntimeDmaTile(op, symbols)))
394 tiles.insert(tile.getTileID());
395 if (auto push = dyn_cast<AIEX::NpuPushQueueOp>(op))
396 if (push.getIssueToken())
397 tiles.insert({static_cast<int>(push.getColumn()),
398 static_cast<int>(push.getRow())});
399 });
400 }
401 return tiles;
402 }

404 LogicalResult
405 applicationPorts(ModuleOp module,
406 std::vector<std::pair<TileID, Port>> &avoided) {
407 OwningOpRef<ModuleOp> trial = cast<ModuleOp>(module->clone());
408 for (DeviceOp device :
409 llvm::make_early_inc_range(trial->getOps<DeviceOp>())) {
410 if (device->hasAttr("aie.is_ctrl_pkt_overlay"))
411 device.erase();
412 else
413 device.removeOverlayAttr();
414 }
415 PassManager pm(&getContext());
416 auto &dpm = pm.nest<DeviceOp>();
417 dpm.addPass(createAIEAssignPacketIdsPass());
418 dpm.addPass(createAIEObjectFifoAllocatePass());
419 dpm.addPass(createAIEObjectFifoLowerDMAsPass());
420 if (failed(pm.run(*trial))) {
421 module.emitError("failed to analyze application resources for adaptive "
422 "control overlay");
423 return failure();
424 }
425 std::set<std::pair<TileID, Port>> ports;
426 for (DeviceOp device : trial->getOps<DeviceOp>()) {
427 for (SwitchboxOp box : device.getOps<SwitchboxOp>()) {
428 for (ConnectOp connect : box.getOps<ConnectOp>())
429 ports.insert({box.getTileOp().getTileID(), connect.destPort()});
430 for (MasterSetOp master : box.getOps<MasterSetOp>())
431 ports.insert({box.getTileOp().getTileID(), master.destPort()});
432 }
433 DynamicTileAnalysis analysis;
434 llvm::Error error = analysis.runAnalysis(device);
435 if (error) {
436 device.emitOpError("cannot plan adaptive control overlay: ")
437 << llvm::toString(std::move(error));
438 return failure();
439 }
440 auto collect = [&](const SwitchSettings &settings) {
441 for (const auto &[tile, setting] : settings)
442 for (Port output : setting.dsts)
443 ports.insert({tile, output});
444 };
445 for (const auto &[_, settings] : analysis.routing.settings)
446 collect(settings);
447 for (const auto &[_, settings] : analysis.routing.idSettings)
448 collect(settings);
449 }
450 avoided.assign(ports.begin(), ports.end());
451 return success();
452 }

455ShimMM2SUses AIE::shimMM2SUses(DeviceOp device, bool packetSwitched) {
456 ShimMM2SUses uses;
457 auto shim = [](Value tile) -> Operation * {
458 auto like = dyn_cast_or_null<TileLike>(tile.getDefiningOp());
459 return like && (like.getTileType() == AIETileType::ShimNOCTile ||
460 like.getTileType() == AIETileType::ShimPLTile)
461 ? like.getOperation()
462 : nullptr;
463 };
464 for (auto endpoint : device.getOps<RouteEndpointOp>()) {
465 Operation *tile = shim(endpoint.getTile());
466 std::optional<int> channel = endpoint.getChannelIndex();
467 if (!tile || endpoint.getBundle() != WireBundle::DMA ||
468 endpoint.getRouteDirection() != DMAChannelDir::MM2S || !channel)
469 continue;
470 bool packet = llvm::any_of(device.getOps<RouteOp>(), [&](RouteOp route) {
471 std::optional<PacketInfoAttr> header = route.getPacket();
472 return route.namesAsSource(endpoint.getSymName()) &&
473 isPacketSwitched(header.value_or(nullptr), packetSwitched);
474 });
475 if (packet)
476 uses.packetEnds[{tile, *channel}].push_back(endpoint.getSymNameAttr());
477 else
478 uses.circuits[tile].insert(*channel);
479 }
480 for (auto alloc : device.getOps<ShimDMAAllocationOp>())
481 if (alloc.getChannelDir() == DMAChannelDir::MM2S)
482 uses.circuits[alloc.getTile().getDefiningOp()].insert(
483 static_cast<int>(alloc.getChannelIndex()));
484 for (auto flow : device.getOps<FlowOp>())
485 if (flow.getSourceBundle() == WireBundle::DMA)
486 if (Operation *tile = shim(flow.getSource()))
487 uses.circuits[tile].insert(flow.getSourceChannel());
488 for (auto flow : device.getOps<PacketFlowOp>())
489 for (auto src : flow.getOps<PacketSourceOp>())
490 if (src.getBundle() == WireBundle::DMA)
491 if (Operation *tile = shim(src.getTile()))
492 ++uses.packetFlows[{tile, src.channelIndex()}];
493 for (auto mux : device.getOps<ShimMuxOp>())
494 for (auto connect : mux.getConnections().front().getOps<ConnectOp>())
495 if (connect.getSourceBundle() == WireBundle::DMA)
496 uses.circuits[mux.getTile().getDefiningOp()].insert(
497 connect.sourceIndex());
498 return uses;
499}

502 return !AIEDialect::IsCtrlPktOverlayAttrHelper(device.getContext())

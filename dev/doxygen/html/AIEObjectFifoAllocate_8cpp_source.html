102 if (a == b)
103 return true;
104 auto first = cast<TileLike>(a.getDefiningOp());
105 auto second = cast<TileLike>(b.getDefiningOp());
106 auto ac = first.tryGetCol(), ar = first.tryGetRow();

111 int64_t memoryUsed(Value tile, int64_t extraSize = 0, int extraCount = 0) {
112 SmallVector<BufferAllocation> layout;
113 int64_t alignment =
114 device.getTargetModel().getMemTileLoadStoreBusWidth() / 8;
115 // Generated buffers are inserted immediately after their tile, before
116 // existing buffers. Preserve that order for equal-sized aligned/unaligned
117 // buffers, including coordinate-equivalent tile references.
118 for (Operation &op : *device.getBody()) {
119 if (auto placed = dyn_cast<TileLike>(op);
120 placed && sameTile(tile, placed->getResult(0))) {
121 Value value = placed->getResult(0);
122 for (int64_t size : plannedMemory[value])
123 layout.push_back({size, alignment, std::nullopt});
124 if (value == tile)
125 for (int i = 0; i < extraCount; ++i)
126 layout.push_back({extraSize, alignment, std::nullopt});
127 } else if (auto buffer = dyn_cast<BufferOp>(op);
128 buffer && sameTile(tile, buffer.getTile())) {
129 layout.push_back({buffer.getAllocationSize(),
130 buffer.getAligned() ? alignment : 1,
131 buffer.getAddress()});
132 }
133 }
134 return assignSequentialBufferAddresses(layout);
135 }

152 bool canAccess(Value user, Value memory, bool localOnly = false) {
153 if (user == memory)
154 return true;
155 auto userTile = dyn_cast<TileLike>(user.getDefiningOp());
156 auto memoryTile = dyn_cast<TileLike>(memory.getDefiningOp());
157 if (!userTile || !memoryTile)
158 return false;
159 const auto &target = device.getTargetModel();
160 auto positions = [&](TileLike tile) {
161 SmallVector<TileID> compatible;
162 auto col = tile.tryGetCol(), row = tile.tryGetRow();
163 for (int c = 0; c < target.columns(); ++c) {
164 if (col && c != *col)
165 continue;
166 for (int r = 0; r < target.rows(); ++r) {
167 if ((!row || r == *row) &&
168 target.getTileType(c, r) == tile.getTileType())
169 compatible.push_back({c, r});
170 }
171 }
172 return compatible;
173 };
174 // Before placement, defer only accesses that some compatible physical
175 // positions could satisfy. Known coordinates can already disprove affinity.
176 auto userPositions = positions(userTile);
177 auto memoryPositions = positions(memoryTile);
178 for (TileID u : userPositions)
179 for (TileID m : memoryPositions) {
180 if (localOnly && u != m)
181 continue;
182 auto shared = target.getSharedMemory(u, m);
183 if (shared == AIETargetModel::SharedMemory::Second ||
184 shared == AIETargetModel::SharedMemory::Either)
185 return true;
186 }
187 return false;
188 }

215 /// Reserve mandatory-local objects before any other pool can spill into
216 /// their memory. Existing buffers are fixed and counted by identity, not by
217 /// how many pools or endpoints reference them.
218 LogicalResult planBuffers(ArrayRef<ObjectFifoPoolOp> pools) {
219 plannedMemory.clear();
220 bufferPlacements.clear();
221 bufferFailure = nullptr;
222 for (auto pool : pools) {
223 if (!localPools.contains(pool))
224 continue;
225 Value tile = localPools.lookup(pool);
226 int count = pool.getBuffers() ? 0 : bufferCount(pool);
227 int64_t size = bufferBytes(pool);
228 if (!canAccess(pool.getTile(), tile) ||
229 !canPlace(pool, tile, size, count)) {
230 bufferFailure = pool;
231 return failure();
232 }
233 plannedMemory[tile].append(count, size);
234 }
235 SmallVector<ObjectFifoPoolOp> generated;
236 for (auto pool : pools) {
237 if (pool.getBuffers()) {
238 for (auto buffer : pool.getBufferOps()) {
239 Value tile = buffer.getBufferTile();
240 if (tile && ((localPools.contains(pool) &&
241 !sameTile(tile, localPools.lookup(pool))) ||
242 !llvm::all_of(poolUsers[pool], [&](Value user) {
243 return canAccess(user, tile);
244 }))) {
245 bufferFailure = pool;
246 return failure();
247 }
248 bufferPlacements[pool].push_back(tile);
249 }
250 continue;
251 }
252 if (pool.getTileLike().isShimTile())
253 continue;
254 if (localPools.contains(pool))
255 bufferPlacements[pool].append(bufferCount(pool),
256 localPools.lookup(pool));
257 else
258 generated.append(bufferCount(pool), pool);
259 }
260 int retries = kSpillRetries;
261 return success(placeBuffers(generated, retries));
262 }

264 /// The first layout tried is the greedy one. When a buffer fits nowhere,
265 /// an earlier buffer may have spilled to the one neighbor it could use, or
266 /// stayed home where a later buffer that only home can hold needed the
267 /// room, so earlier choices are revisited, latest first, a bounded number of
268 /// times.
269 static constexpr int kSpillRetries = 256;
270
271 bool placeBuffers(ArrayRef<ObjectFifoPoolOp> buffers, int &retries) {
272 if (buffers.empty())
273 return true;
274 ObjectFifoPoolOp pool = buffers.front();
275 int64_t size = bufferBytes(pool);
276 Value homeTile = pool.getTileLike()->getResult(0);
277 bool fitsHome = canPlace(pool, homeTile, size);
278 SmallVector<Value> candidates;
279 if (fitsHome)
280 candidates.push_back(homeTile);
281 else
282 candidates = spillTargets(pool, size);
283 if (candidates.empty() &&
284 (!bufferFailure || buffers.size() < unplacedAtFailure)) {
285 bufferFailure = pool;
286 unplacedAtFailure = buffers.size();
287 }
288 auto tryTile = [&](Value tile) {
289 plannedMemory[tile].push_back(size);
290 bufferPlacements[pool].push_back(tile);
291 if (placeBuffers(buffers.drop_front(), retries))
292 return true;
293 plannedMemory[tile].pop_back();
294 bufferPlacements[pool].pop_back();
295 return false;
296 };
297 for (auto [index, tile] : llvm::enumerate(candidates)) {
298 if (index > 0 && retries-- <= 0)
299 return false;
300 if (tryTile(tile))
301 return true;
302 }
303 if (!fitsHome)
304 return false;
305 // Spill targets are only looked up once home has failed, so a layout that
306 // keeps every buffer home creates no neighbor tiles.
307 for (Value tile : spillTargets(pool, size)) {
308 if (retries-- <= 0)
309 return false;
310 if (tryTile(tile))
311 return true;
312 }
313 return false;
314 }

316 /// FIXME: choosing which tile a buffer lives on is the buffer allocator's
317 /// job, not this pass's. Tracking used memory here to make that choice
318 /// breaks the separation of concerns; --aie-assign-buffer-addresses should
319 /// instead be free to move buffers between tiles that share a memory module.
320 ///
321 /// A MemTile buffer that does not fit at home spills to a neighbor reachable
322 /// by its DMAs, preferring the emptier one so adjacent MemTiles
323 /// keep room for their own spills. Which tiles neighbor an unplaced one is
324 /// not yet known, so its buffers stay at home.
325 SmallVector<Value> spillTargets(ObjectFifoPoolOp pool, int64_t sizeBytes) {
326 TileLike home = pool.getTileLike();
327 auto &target = device.getTargetModel();
328 if (!home.isMemTile())
329 return {};
330
331 auto homeOp = dyn_cast<TileOp>(home.getOperation());
332 if (!homeOp) {
333 return {};
334 }
335
336 SmallVector<TileOp> neighbors;
337 for (int col : {homeOp.getCol() - 1, homeOp.getCol() + 1}) {
338 if (col < 0 || col >= target.columns()) {
339 continue;
340 }
341 int row = homeOp.getRow();
342 bool declared = llvm::any_of(device.getOps<TileOp>(), [&](TileOp t) {
343 return t.getCol() == col && t.getRow() == row;
344 });
345 TileOp neighbor =
346 TileOp::getOrCreate(builder, device, col, row, pool.getLoc());
347 if (!declared)
348 spillTiles.push_back(neighbor);
349 using SharedMemory = AIETargetModel::SharedMemory;
350 SharedMemory shared = sharedMemory(homeOp, neighbor);
351 if (shared == SharedMemory::Second || shared == SharedMemory::Either) {
352 neighbors.push_back(neighbor);
353 }
354 }
355 llvm::stable_sort(neighbors, [&](TileOp a, TileOp b) {
356 return memoryUsed(a.getResult()) < memoryUsed(b.getResult());
357 });
358 SmallVector<Value> fitting;
359 for (TileOp neighbor : neighbors) {
360 if (canPlace(pool, neighbor.getResult(), sizeBytes)) {
361 fitting.push_back(neighbor.getResult());
362 }
363 }
364 return fitting;
365 }

435 LogicalResult planLocks(ArrayRef<ObjectFifoPoolOp> pools) {
436 lockFailure = {};
437 lockAccessFailure = nullptr;
438 lockPlacements.clear();
439 DenseMap<std::pair<int, int>, int64_t> used;
440 auto reserve = [&](Value tile, int64_t count) {
441 auto like = cast<TileLike>(tile.getDefiningOp());
442 auto col = like.tryGetCol(), row = like.tryGetRow();
443 if (!col || !row)
444 return success();
445 if (used[{*col, *row}] + count >
446 device.getTargetModel().getNumLocks(*col, *row)) {
447 lockFailure = tile;
448 return failure();
449 }
450 used[{*col, *row}] += count;
451 return success();
452 };
453 for (auto lock : device.getOps<LockOp>())
454 if (failed(reserve(lock.getTile(), 1)))
455 return failure();
456 // Neighbors created to weigh a spill have no rank and come last.
457 SmallVector<TileLike> tiles(device.getOps<TileLike>());
458 llvm::stable_sort(tiles, [&](TileLike a, TileLike b) {
459 auto aRank = structuralRank.find(a), bRank = structuralRank.find(b);
460 if ((aRank == structuralRank.end()) != (bRank == structuralRank.end()))
461 return bRank == structuralRank.end();
462 return aRank != structuralRank.end() && aRank->second < bRank->second;
463 });
464 auto place = [&](ObjectFifoPoolOp pool, Operation *group, int count) {
465 SmallVector<Value> candidates{lockPlacement(pool, group)};
466 if (!localPools.contains(pool)) {
467 llvm::append_range(candidates, bufferPlacements[pool]);
468 for (Operation *user : lockUsers[group])
469 candidates.push_back(user->getOperand(0));
470 for (auto tile : tiles)
471 candidates.push_back(tile->getResult(0));
472 }
473 bool reachable = false;
474 for (Value tile : candidates) {
475 if (!tile || !llvm::all_of(lockUsers[group], [&](Operation *user) {
476 return canAccessLocks(user, tile);
477 }))
478 continue;
479 reachable = true;
480 if (succeeded(reserve(tile, count))) {
481 lockPlacements[group] = tile;
482 lockFailure = {};
483 return success();
484 }
485 }
486 if (!reachable && !lockUsers[group].empty()) {
487 lockFailure = {};
488 lockAccessFailure = lockUsers[group].front();
489 }
490 return failure();
491 };
492 for (auto pool : pools) {
493 if (!poolLocks->needsLocks(pool))
494 continue;
495 if (device.getTargetModel().getTargetArch() == AIEArch::AIE1) {
496 if (!pool.getLocks() && failed(place(pool, pool, pool.getDepth())))
497 return failure();
498 } else {
499 for (auto segment : pool.getSegmentOps())
500 if (!(segment.getProduceLock() && segment.getConsumeLock()) &&
501 failed(place(pool, segment, 2)))
502 return failure();
503 }
504 }
505 auto check = [&](auto endpoint) {
506 for (Value tile : endpointLockTiles(endpoint)) {
507 if (!canAccessLocks(endpoint, tile)) {
508 lockAccessFailure = endpoint;
509 return failure();
510 }
511 }
512 return success();
513 };
514 for (auto endpoint : device.getOps<ObjectFifoCoreEndpointOp>())
515 if (failed(check(endpoint)))
516 return failure();
517 for (auto endpoint : device.getOps<ObjectFifoDmaEndpointOp>())
518 if (failed(check(endpoint)))
519 return failure();
520 return success();
521 }

523 /// AIE1 guards each buffer with one binary lock that rotates with it; AIE2
524 /// gives each segment a counting pair, the producer's lock counting free
525 /// objects and the consumer's counting full ones.
526 void allocateLocks(ObjectFifoPoolOp pool) {
527 if (!poolLocks->needsLocks(pool))
528 return;
529 StringRef base = pool.getBaseName();
530 int depth = pool.getDepth();
531 auto initValues = pool.getInitValues();
532 int filled = initValues ? initValues->size() : 0;
533 int repeat = pool.getRepeatCount().value_or(1);
534
535 if (device.getTargetModel().getTargetArch() == AIEArch::AIE1) {
536 if (pool.getLocks()) {
537 return;
538 }
539 SmallVector<Attribute> names;
540 for (int i = 0; i < depth; i++) {
541 std::string name = (base + "__lock" + std::to_string(i)).str();
542 createLock(pool, lockPlacement(pool, pool), name, i < filled ? 1 : 0);
543 names.push_back(FlatSymbolRefAttr::get(builder.getContext(), name));
544 }
545 pool.setLocksAttr(builder.getArrayAttr(names));
546 return;
547 }
548
549 for (auto [index, segment] : llvm::enumerate(pool.getSegmentOps())) {
550 if (segment.getProduceLock() && segment.getConsumeLock()) {
551 continue;
552 }
553 std::string produce = (base + "__prodlock" + std::to_string(index)).str();
554 std::string consume = (base + "__conslock" + std::to_string(index)).str();
555 Value tile = lockPlacement(pool, segment);
556 createLock(pool, tile, produce, (depth - filled) * repeat);
557 createLock(pool, tile, consume, filled * repeat);
558 segment.setProduceLockAttr(
559 FlatSymbolRefAttr::get(builder.getContext(), produce));
560 segment.setConsumeLockAttr(
561 FlatSymbolRefAttr::get(builder.getContext(), consume));
562 }
563 }

585 /// Predict the locks core and DMA lowering will use, including hand-written
586 /// locks and those allocation will create. Only selected segments matter.
587 template <typename EndpointOp>
588 SmallVector<Value> endpointLockTiles(EndpointOp endpoint) {
589 SmallVector<Value> tiles;
590 auto pool = endpoint.getPoolOp();
591 if (pool.getDepth() == 0)
592 return tiles;
593 if (device.getTargetModel().getTargetArch() == AIEArch::AIE1) {
594 if (poolLocks->needsLocks(pool) && !pool.getLocks())
595 tiles.push_back(lockPlacement(pool, pool));
596 else
597 for (auto lock : pool.getLockOps())
598 tiles.push_back(lock.getTile());
599 return tiles;
600 }
601 for (auto segment : endpoint.getSelectedSegments()) {
602 if (poolLocks->needsLocks(pool) &&
603 !(segment.getProduceLock() && segment.getConsumeLock())) {
604 tiles.push_back(lockPlacement(pool, segment));
605 continue;
606 }
607 for (auto name :
608 {segment.getProduceLockAttr(), segment.getConsumeLockAttr()})
609 if (name)
610 if (auto lock = lookupNamedOp<LockOp>(device, name.getAttr()))
611 tiles.push_back(lock.getTile());
612 }
613 return tiles;
614 }

638 /// A DMA channel the design programs itself, by naming this endpoint from a

640 static bool isProgramOwned(RouteEndpoint endpoint) {
641 auto route = dyn_cast<RouteEndpointOp>(endpoint.getOperation());
642 return route && route.getBundle() == WireBundle::DMA &&
643 !route.getTileLike().isShimTile();

646 /// Blocks of the BD chain `start` begins, following next_bd.
647 static SmallVector<Block *> chainOf(DMAStartOp start) {
648 SmallVector<Block *> chain;
649 SmallPtrSet<Block *, 8> seen;
650 SmallVector<Block *> work{start.getDest()};
651 while (!work.empty()) {
652 Block *block = work.pop_back_val();
653 if (!seen.insert(block).second || block->getOps<DMABDOp>().empty())
654 continue;
655 chain.push_back(block);
656 if (auto next = dyn_cast<NextBDOp>(block->getTerminator()))
657 work.push_back(next.getDest());
658 }
659 return chain;
660 }

662 static int64_t countBDs(DMAStartOp start) {
663 int64_t bds = 0;
664 for (Block *block : chainOf(start))
665 bds += llvm::range_size(block->getOps<DMABDOp>());
666 return bds;
667 }

670 int64_t bds = 0;
671 taskBody.walk([&](DMABDOp) { bds++; });

691 /// Whether a program-owned MemTile channel's BDs touch a neighbor's buffers
692 /// or locks, which only the lower channels can.
693 bool programReachesAdjacentTile(RouteEndpointOp endpoint) {
694 if (!endpoint.getTileLike().isMemTile())
695 return false;
696 auto remote = [&](Operation *op) {
697 Value tile;
698 if (auto bd = dyn_cast<DMABDOp>(op)) {
699 if (auto buffer =
700 dyn_cast_or_null<BufferOp>(bd.getBuffer().getDefiningOp()))
701 tile = buffer.getTile();
702 } else if (auto use = dyn_cast<UseLockOp>(op)) {
703 if (auto lock = dyn_cast_or_null<LockOp>(use.getLock().getDefiningOp()))
704 tile = lock.getTile();
705 }
706 return tile && !sameTile(tile, endpoint.getTile());
707 };
708 for (DMAStartOp start : endpointStarts.lookup(endpoint))
709 for (Block *block : chainOf(start))
710 if (llvm::any_of(block->getOperations(),
711 [&](Operation &op) { return remote(&op); }))
712 return true;
713 for (auto task : endpointTasks.lookup(endpoint)) {
714 bool reaches = false;
715 task.getBody().walk([&](Operation *op) { reaches |= remote(op); });
716 if (reaches)
717 return true;
718 }
719 return false;
720 }

723 /// for on channels named by index.

726 endpointTasks.clear();

728 taskBDs.clear();

734 return std::nullopt;

762 /// Rank tiles, pools, endpoints and routes by their attributes and then, a
763 /// round at a time until no class splits, by the ranks of what they connect
764 /// to. Symbol names and op order play no part, so a permuted or renamed
765 /// design ranks alike, and ops still tied cannot be told apart.
766 void rankStructurally() {
767 StructuralGraph graph;
768 // A placed tile is one node however many ops name it.
769 DenseMap<std::pair<int, int>, Operation *> placed;
770 DenseMap<Operation *, Operation *> tileNode;
771 for (auto tile : device.getOps<TileLike>()) {
772 auto col = tile.tryGetCol(), row = tile.tryGetRow();
773 if (!col || !row) {
774 graph.add(tileNode[tile] = tile);
775 continue;
776 }
777 auto [it, first] = placed.try_emplace({*col, *row}, tile);
778 tileNode[tile] = it->second;
779 if (first) {
780 graph.nodes.push_back(tile);
781 llvm::raw_string_ostream(graph.features[tile])
782 << "tile " << *col << ' ' << *row << ' '
783 << stringifyAIETileType(tile.getTileType());
784 }
785 }
786 auto link = [&](Operation *from, Operation *to, StringRef label) {
787 if (Operation *tile = tileNode.lookup(to))
788 to = tile;
789 graph.link(from, to, label);
790 };
791 for (auto pool : device.getOps<ObjectFifoPoolOp>()) {
792 llvm::raw_string_ostream os(graph.add(pool));
793 for (auto segment : pool.getSegmentOps())
794 os << " [" << segment.getOffset() << ", " << segment.getSize() << "]";
795 link(pool, pool.getTile().getDefiningOp(), "tile");
796 }
797 for (auto endpoint : device.getOps<ObjectFifoCoreEndpointOp>()) {
798 llvm::raw_string_ostream os(graph.add(endpoint));
799 for (auto segment : endpoint.getSelectedSegments())
800 os << " @" << segment.getOffset();
801 link(endpoint, endpoint.getTile().getDefiningOp(), "tile");
802 link(endpoint, endpoint.getPoolOp(), "pool");
803 }
804 for (auto endpoint : device.getOps<RouteEndpoint>()) {
805 llvm::raw_string_ostream os(graph.add(endpoint));
806 if (auto dma =
807 dyn_cast<ObjectFifoDmaEndpointOp>(endpoint.getOperation())) {
808 for (auto segment : dma.getSelectedSegments())
809 os << " @" << segment.getOffset();
810 link(endpoint, dma.getPoolOp(), "pool");
811 }
812 link(endpoint, endpoint.getTile().getDefiningOp(), "tile");
813 }
814 for (auto route : device.getOps<RouteOp>()) {
815 graph.add(route);
816 for (StringRef source : route.getSourceNames())
817 link(route, lookupEndpoint(source), "source");
818 for (StringRef dest : route.getDestinationNames())
819 link(route, lookupEndpoint(dest), "dest");
820 }
821 DenseMap<Operation *, unsigned> rank = graph.rank();
822 for (auto [tile, node] : tileNode)
823 rank[tile] = rank[node];
824 structuralRank = std::move(rank);
825 }

827 /// BDs `channel` of the tile at (`col`, `row`) can use that `demand` leaves
828 /// over, counting every channel whose BD ids overlap its own. Negative when
829 /// overcommitted.
830 int64_t bdRoom(TileLike tile, int col, int row, int channel,
831 const DenseMap<std::tuple<int, int, int>, int64_t> &demand) {
832 const AIETargetModel &target = device.getTargetModel();
833 uint32_t numBDs = target.getNumBDs(col, row);
834 int numChannels = std::max(tile.getNumSourceConnections(WireBundle::DMA),
835 tile.getNumDestConnections(WireBundle::DMA));
836 auto sharesBDs = [&](int other) {
837 for (uint32_t bd = 0; bd < numBDs; ++bd)
838 if (target.isBdChannelAccessible(col, row, bd, channel) &&
839 target.isBdChannelAccessible(col, row, bd, other))
840 return true;
841 return false;
842 };
843 int64_t room = target.getNumBDsForChannel(col, row, channel);
844 for (int other = 0; other < numChannels; ++other)
845 if (sharesBDs(other))
846 if (auto it = demand.find({col, row, other}); it != demand.end())
847 room -= it->second;
848 return room;
849 }

851 /// On a tile whose BD ids are split between channels, the free channel whose
852 /// BDs have the most room left once every channel sharing them has its
853 /// demand; ties, and tiles without such a split, go to the lowest index.
854 /// Falls back to first-free where coordinates are unknown.
855 int chooseChannel(DMAChannelAnalysis &channels, RouteEndpoint endpoint,
856 const DenseMap<std::tuple<int, int, int>, int64_t> &demand,
857 bool adjacent) {
858 TileLike tile = tileOf(endpoint);
859 DMAChannelDir dir = endpoint.getRouteDirection();
860 std::optional<int> col = tile.tryGetCol(), row = tile.tryGetRow();
861 if (!col || !row)
862 return channels.getDMAChannelIndex(tile, dir, adjacent,
863 endpoint.getOperation());
864 int best = -1;
865 int64_t bestRoom = 0;
866 int limit = DMAChannelAnalysis::getDMAChannelLimit(tile, dir, adjacent);
867 for (int channel = 0; channel < limit; ++channel) {
868 if (!channels.isChannelFree(tile, dir, channel))
869 continue;
870 int64_t room = bdRoom(tile, *col, *row, channel, demand);
871 if (best < 0 || room > bestRoom) {
872 best = channel;
873 bestRoom = room;
874 }
875 }
876 if (best < 0)
877 return -1;
878 return channels.reservePinnedChannel(tile, dir, best,
879 endpoint.getOperation());
880 }

927 LogicalResult assignChannels(DMAChannelAnalysis &channels,
928 bool diagnose = true) {
929 channelAssignments.clear();
930 channelFailure = nullptr;
931 SmallVector<RouteEndpoint> pending;
932 // BDs per placed (col, row, channel) as channels are handed out, for
933 // choosing among a tile's channels by the BD ids they can use.
934 DenseMap<std::tuple<int, int, int>, int64_t> demand = fixedBDs;
935 auto noteDemand = [&](RouteEndpoint endpoint, int channel) {
936 TileLike tile = tileOf(endpoint);
937 auto col = tile.tryGetCol(), row = tile.tryGetRow();
938 if (!col || !row)
939 return;
940 int64_t bds = 0;
941 if (auto dma = dyn_cast<ObjectFifoDmaEndpointOp>(endpoint.getOperation()))
942 bds = dma.getNumBDs();
943 else if (isProgramOwned(endpoint))
944 bds = endpointBDs(endpoint.getOperation());
945 demand[{*col, *row, channel}] += bds;
946 };
947 for (auto endpoint : device.getOps<RouteEndpoint>()) {
948 DMAChannelDir dir = endpoint.getRouteDirection();
949 std::optional<int> channel = endpoint.getRouteChannel();
950 // A core's stream port is named by the design, not drawn from the tile's
951 // DMA channels; checkStreamPorts has claimed it.
952 if (endpoint.getRouteBundle() == WireBundle::Core ||
953 channelSharers.count(endpoint.getOperation()))
954 continue;
955
956 if (channel) {
957 if (reachesAdjacentTile(endpoint) &&
958 *channel >= DMAChannelAnalysis::getDMAChannelLimit(tileOf(endpoint),
959 dir, true)) {
960 channelFailure = endpoint;
961 if (!diagnose)
962 return failure();
963 return endpoint->emitOpError("pinned ")
964 << stringifyDMAChannelDir(dir) << " DMA channel " << *channel
965 << " cannot access adjacent MemTile buffers or locks";
966 }
967 if (channels.reservePinnedChannel(tileOf(endpoint), dir, *channel,
968 endpoint.getOperation()) < 0) {
969 if (!diagnose)
970 return failure();
971 auto diag = endpoint->emitOpError("pinned ");
972 diag << stringifyDMAChannelDir(dir) << " DMA channel " << *channel
973 << " is out of range or already in use on this tile";
974 if (Operation *owner =
975 channels.getDMAChannelOwner(tileOf(endpoint), dir, *channel))
976 noteChannelOwner(diag, owner, channel);
977 return failure();
978 }
979 noteDemand(endpoint, *channel);
980 continue;
981 }
982 pending.push_back(endpoint);
983 }
984
985 // Endpoints reaching a spilled buffer draw from a restricted channel
986 // range, so they are served before the unrestricted ones. Channels the
987 // design programs itself come after the objectFIFOs', whose BDs they are
988 // then chosen around, the most demanding first. The structural rank
989 // settles the rest, so the order ops were written in never does.
990 DenseMap<Operation *, int64_t> programBDs;
991 for (auto endpoint : pending)
992 if (isProgramOwned(endpoint))
993 programBDs[endpoint.getOperation()] =
994 endpointBDs(endpoint.getOperation());
995 llvm::stable_sort(pending, [&](RouteEndpoint a, RouteEndpoint b) {
996 bool aAdjacent = reachesAdjacentTile(a);
997 bool bAdjacent = reachesAdjacentTile(b);
998 if (aAdjacent != bAdjacent)
999 return aAdjacent;
1000 bool aProgram = isProgramOwned(a), bProgram = isProgramOwned(b);
1001 if (aProgram != bProgram)
1002 return bProgram;
1003 int64_t aBDs = programBDs.lookup(a.getOperation());
1004 int64_t bBDs = programBDs.lookup(b.getOperation());
1005 if (aBDs != bBDs)
1006 return aBDs > bBDs;
1007 return structuralRank.lookup(a.getOperation()) <
1008 structuralRank.lookup(b.getOperation());
1009 });
1010
1011 for (auto endpoint : pending) {
1012 DMAChannelDir dir = endpoint.getRouteDirection();
1013 TileLike tile = tileOf(endpoint);
1014 bool adjacent = reachesAdjacentTile(endpoint);
1015 // An objectFIFO end takes the first free channel, unless that channel's
1016 // BD ids have too few left for it: then the one with the most left.
1017 bool roomy = true;
1018 auto dma = dyn_cast<ObjectFifoDmaEndpointOp>(endpoint.getOperation());
1019 std::optional<int> col = tile.tryGetCol(), row = tile.tryGetRow();
1020 if (dma && col && row) {
1021 int limit = DMAChannelAnalysis::getDMAChannelLimit(tile, dir, adjacent);
1022 for (int free = 0; free < limit; ++free) {
1023 if (!channels.isChannelFree(tile, dir, free))
1024 continue;
1025 roomy = bdRoom(tile, *col, *row, free, demand) >= dma.getNumBDs();
1026 break;
1027 }
1028 }
1029 int channel = isProgramOwned(endpoint) || !roomy
1030 ? chooseChannel(channels, endpoint, demand, adjacent)
1031 : channels.getDMAChannelIndex(tile, dir, adjacent,
1032 endpoint.getOperation());
1033 if (channel < 0) {
1034 channelFailure = endpoint;
1035 if (!diagnose)
1036 return failure();
1037 int capacity =
1038 DMAChannelAnalysis::getDMAChannelLimit(tile, dir, adjacent);
1039 auto diag =
1040 tile.emitOpError(dir == DMAChannelDir::MM2S
1041 ? "number of output DMA channel exceeded!"
1042 : "number of input DMA channel exceeded!");
1043 diag << " requires at least " << capacity + 1 << " "
1044 << stringifyDMAChannelDir(dir) << " channels, but capacity is "
1045 << capacity;
1046 if (adjacent) {
1047 diag << " for adjacent MemTile access";
1048 }
1049 for (int i = 0; i < capacity; ++i)
1050 if (Operation *owner = channels.getDMAChannelOwner(tile, dir, i))
1051 noteChannelOwner(diag, owner, i);
1052 noteChannelOwner(diag, endpoint.getOperation());
1053 return failure();
1054 }
1055 channelAssignments[endpoint.getOperation()] = channel;
1056 noteDemand(endpoint, channel);
1057 }
1058 for (auto [sharer, owner] : channelSharers)
1059 channelAssignments[sharer] = channelAssignments.lookup(owner);
1060 // Runtime tasks did not steer the choice, but they hold BDs while they
1061 // run, which the overcommit warning counts.
1062 for (auto &[at, bds] : taskBDs)
1063 demand[at] += bds;
1064 for (auto endpoint : device.getOps<RouteEndpoint>()) {
1065 TileLike tile = tileOf(endpoint);
1066 std::optional<int> col = tile.tryGetCol(), row = tile.tryGetRow();
1067 std::optional<int> channel = endpoint.getRouteChannel();
1068 if (auto it = channelAssignments.find(endpoint.getOperation());
1069 it != channelAssignments.end())
1070 channel = it->second;
1071 if (isProgramOwned(endpoint) && channel && col && row)
1072 demand[{*col, *row, *channel}] +=
1073 endpointTaskBDs(endpoint.getOperation());
1074 }
1075 channelDemand = std::move(demand);
1076 return success();
1077 }

1116 /// walks its deepest plain pools with one descriptor a side instead, their
1117 /// objects side by side in one buffer and counted by one lock pair: the
1118 /// deepest first, until its descriptors fit.
1119 /// Whether one iterating descriptor a side could walk `pool`: a plain mem
1120 /// tile pool of one segment, deeper than one object and no deeper than a
1121 /// lock counts, whose ends neither dispatch nor take turns, but those in
1122 /// `regrouped`, whose turns are being chosen again.
1123 bool canWalkWhole(ObjectFifoPoolOp pool,
1124 ArrayRef<ObjectFifoDmaEndpointOp> regrouped = {}) {
1125 const AIETargetModel &target = device.getTargetModel();
1126 TileLike tile =
1127 pool ? dyn_cast<TileLike>(pool.getTile().getDefiningOp()) : TileLike();
1128 return tile && tile.isMemTile() && !pool.getContiguous() &&
1129 pool.getSegmentOps().size() == 1 && !pool.getRepeatCount() &&
1130 !pool.getBuffers() && pool.getDepth() > 1 &&
1131 pool.getDepth() <= static_cast<int>(target.getMaxLockValue()) &&
1132 llvm::all_of(device.getOps<ObjectFifoDmaEndpointOp>(),
1133 [&](ObjectFifoDmaEndpointOp end) {
1134 return end.getPoolOp() != pool ||
1135 (!end.getDispatch() &&
1136 (!end.getTurns() ||
1137 llvm::is_contained(regrouped, end)));
1138 });
1139 }

1141 void walkDeepPoolsWhole() {
1142 const AIETargetModel &target = device.getTargetModel();
1143 llvm::MapVector<Value, SmallVector<ObjectFifoDmaEndpointOp>> byTile;
1144 for (auto dma : device.getOps<ObjectFifoDmaEndpointOp>())
1145 if (dma.getTileLike() && dma.getTileLike().isMemTile())
1146 byTile[dma.getTile()].push_back(dma);
1147 for (SmallVector<ObjectFifoDmaEndpointOp> &ends :
1148 llvm::make_second_range(byTile)) {
1149 // A mem tile's BD ids are split between its even and odd channels, and
1150 // channels go to ends in an order of their own, each to the half with
1151 // room. That cannot fail while every end fits beside all the others in
1152 // either half: the ends' BDs and the largest of them within the BDs
1153 // plus one.
1154 TileLike tileLike = ends.front().getTileLike();
1155 std::optional<int> col = tileLike.tryGetCol(), row = tileLike.tryGetRow();
1156 int64_t numBDs = target.getNumBDs(AIETileType::MemTile);
1157 bool halved =
1158 col && row && target.getNumBDsForChannel(*col, *row, 0) < numBDs;
1159 auto fits = [&] {
1160 int64_t bds = 0, largest = 0;
1161 for (ObjectFifoDmaEndpointOp end : ends) {
1162 bds += end.getNumBDs();
1163 largest = std::max(largest, end.getNumBDs());
1164 }
1165 return halved ? bds + largest <= numBDs + 1 : bds <= numBDs;
1166 };
1167 SmallVector<ObjectFifoPoolOp> pools;
1168 for (ObjectFifoDmaEndpointOp end : ends) {
1169 ObjectFifoPoolOp pool = end.getPoolOp();
1170 if (!llvm::is_contained(pools, pool) && canWalkWhole(pool))
1171 pools.push_back(pool);
1172 }
1173 llvm::stable_sort(pools, [&](ObjectFifoPoolOp a, ObjectFifoPoolOp b) {
1174 if (a.getDepth() != b.getDepth())
1175 return a.getDepth() > b.getDepth();
1176 return structuralRank.lookup(a) < structuralRank.lookup(b);
1177 });
1178 for (ObjectFifoPoolOp pool : pools) {
1179 if (fits())
1180 break;
1181 pool.setContiguous(true);
1182 }
1183 }
1184 }

1229 /// Whether a chain moving `turns` over `ends` can walk each contiguous pool
1230 /// with one descriptor per turn of its unrolled passes, each stepping over
1231 /// the others' objects: they must divide the pool's objects.
1232 static bool contiguousWalkersFit(ArrayRef<ObjectFifoDmaEndpointOp> ends,
1233 ArrayRef<unsigned> turns) {
1234 SmallVector<std::pair<int64_t, int64_t>> counts;
1235 for (auto [i, constEnd] : llvm::enumerate(ends)) {
1236 ObjectFifoDmaEndpointOp end = constEnd;
1237 counts.push_back({llvm::count(turns, i), end.getPoolOp().getContiguous()
1238 ? 0
1239 : end.getNumDescriptors()});
1240 }
1241 int64_t passes = turnPasses(counts);
1242 for (auto [i, constEnd] : llvm::enumerate(ends)) {
1243 ObjectFifoDmaEndpointOp end = constEnd;
1244 if (end.getPoolOp().getContiguous() &&
1245 (end.walksStream() ||
1246 end.getPoolOp().getDepth() % (counts[i].first * passes)))
1247 return false;
1248 }
1249 return true;
1250 }

1268 /// Gives one more end of `exhausted` the channel of others whose objects
1269 /// the same core needs in a fixed order, read off its program, so only as
1270 /// many ends share as the tile needs. A circuit carries one stream, so the
1271 /// ends at the routes' other ends take the same turns.
1272 bool planTurnSharing(TileLike exhausted, DMAChannelDir dir) {
1273 // `coreEndOf`/`circuitPeer` each rescan the whole device; neither's
1274 // domain (core endpoints, routes) changes within this function (only
1275 // pool `contiguous` flags and endpoints' `turns` attrs do), so caching
1276 // them once here turns every lookup below from O(device size) into O(1).
1277 DenseMap<Operation *, ObjectFifoCoreEndpointOp> coreEndCache;
1278 DenseMap<Operation *, ObjectFifoDmaEndpointOp> circuitPeerCache;
1279 for (auto dma : device.getOps<ObjectFifoDmaEndpointOp>()) {
1280 coreEndCache[dma] = coreEndOf(dma);
1281 circuitPeerCache[dma] = circuitPeer(dma);
1282 }
1283 auto cachedCoreEnd = [&](ObjectFifoDmaEndpointOp dma) {
1284 return coreEndCache.lookup(dma);
1285 };
1286 auto cachedCircuitPeer = [&](ObjectFifoDmaEndpointOp dma) {
1287 return circuitPeerCache.lookup(dma);
1288 };
1289 SmallVector<ObjectFifoDmaEndpointOp> candidates;
1290 for (auto dma : device.getOps<ObjectFifoDmaEndpointOp>())
1291 if (sameTile(dma.getTile(), exhausted->getResult(0)) &&
1292 dma.getRouteDirection() == dir && cachedCoreEnd(dma) &&
1293 canTakeTurns(dma.getTurns() ? turnEndpoints(dma).front() : dma) &&
1294 canTakeTurns(cachedCircuitPeer(dma)))
1295 candidates.push_back(dma);
1296 llvm::stable_sort(candidates, [&](Operation *a, Operation *b) {
1297 return structuralRank.lookup(a) < structuralRank.lookup(b);
1298 });
1299 LLVM_DEBUG({
1300 llvm::dbgs() << "turns on " << exhausted->getLoc() << ": "
1301 << candidates.size() << " candidate ends\n";
1302 for (auto dma : device.getOps<ObjectFifoDmaEndpointOp>())
1303 if (sameTile(dma.getTile(), exhausted->getResult(0)) &&
1304 dma.getRouteDirection() == dir)
1305 llvm::dbgs() << " " << dma.getSymName() << ": core end "
1306 << bool(cachedCoreEnd(dma)) << ", takes turns "
1307 << canTakeTurns(dma) << ", peer takes turns "
1308 << canTakeTurns(cachedCircuitPeer(dma)) << "\n";
1309 });
1310 auto groupOf = [](ObjectFifoDmaEndpointOp end) {
1311 SmallVector<ObjectFifoDmaEndpointOp> group = turnEndpoints(end);
1312 return group.empty() ? SmallVector<ObjectFifoDmaEndpointOp>{end} : group;
1313 };
1314 for (auto [i, a] : llvm::enumerate(candidates)) {
1315 for (ObjectFifoDmaEndpointOp b : llvm::drop_begin(candidates, i + 1)) {
1316 SmallVector<ObjectFifoDmaEndpointOp> group = groupOf(a);
1317 if (llvm::is_contained(group, b))
1318 continue;
1319 llvm::append_range(group, groupOf(b));
1320 SmallVector<ObjectFifoCoreEndpointOp> cores;
1321 SmallVector<ObjectFifoDmaEndpointOp> peers;
1322 SmallVector<int64_t> depths;
1323 for (ObjectFifoDmaEndpointOp end : group) {
1324 cores.push_back(cachedCoreEnd(end));
1325 peers.push_back(cachedCircuitPeer(end));
1326 depths.push_back(end.getPoolOp().getDepth());
1327 }
1328 if (llvm::any_of(peers, [&](ObjectFifoDmaEndpointOp peer) {
1329 return !sameTile(peer.getTile(), peers.front().getTile()) ||
1330 llvm::any_of(turnEndpoints(peer), [&](auto other) {
1331 return !llvm::is_contained(peers, other);
1332 });
1333 })) {
1334 LLVM_DEBUG(llvm::dbgs()
1335 << " " << a.getSymName() << " + " << b.getSymName()
1336 << ": senders on different tiles\n");
1337 continue;
1338 }
1339 CoreTurns turns;
1340 std::string reason;
1341 if (succeeded(turns.read(cores, reason)) && !turns.keepsUp(depths))
1342 reason = "would stall";
1343 // Over plain pools the chain unrolls to whole passes over each; past
1344 // the peers' channel BDs, walking them whole takes one a turn.
1345 bool overflows = false;
1346 if (reason.empty()) {
1347 SmallVector<std::pair<int64_t, int64_t>> counts;
1348 for (auto [i, constPeer] : llvm::enumerate(peers)) {
1349 ObjectFifoDmaEndpointOp peer = constPeer;
1350 counts.push_back({llvm::count(turns.turns(), i),
1351 peer.getPoolOp().getContiguous()
1352 ? 0
1353 : peer.getNumDescriptors()});
1354 }
1355 TileLike tile = peers.front().getTileLike();
1356 std::optional<int> col = tile.tryGetCol(), row = tile.tryGetRow();
1357 overflows =
1358 col && row &&
1359 turnPasses(counts) * static_cast<int64_t>(turns.turns().size()) >
1360 device.getTargetModel().getNumBDsForChannel(*col, *row, 0);
1361 }
1362 // Beside a pool walked whole, the others' descriptors must come round
1363 // each pass, which walking them whole as well lines up.
1364 // A pool whose other end already takes turns may be walked whole too,
1365 // where those turns still line up.
1366 SmallVector<ObjectFifoPoolOp> walked;
1367 if (reason.empty() &&
1368 (overflows || !contiguousWalkersFit(peers, turns.turns())))
1369 for (ObjectFifoDmaEndpointOp peer : peers) {
1370 ObjectFifoPoolOp pool = peer.getPoolOp();
1371 SmallVector<ObjectFifoDmaEndpointOp> others;
1372 for (auto end : device.getOps<ObjectFifoDmaEndpointOp>())
1373 if (end.getPoolOp() == pool && end.getTurns() &&
1374 !llvm::is_contained(peers, end))
1375 others.push_back(end);
1376 SmallVector<ObjectFifoDmaEndpointOp> ignored(peers);
1377 llvm::append_range(ignored, others);
1378 if (!canWalkWhole(pool, ignored)) {
1379 LLVM_DEBUG(llvm::dbgs() << " " << pool.getSymName()
1380 << " cannot be walked whole\n");
1381 continue;
1382 }
1383 pool.setContiguous(true);
1384 bool linedUp =
1385 llvm::all_of(others, [&](ObjectFifoDmaEndpointOp end) {
1386 SmallVector<ObjectFifoDmaEndpointOp> ends =
1387 turnEndpoints(end);
1388 SmallVector<unsigned> order;
1389 for (auto name :
1390 end.getTurnsAttr().getAsRange<FlatSymbolRefAttr>())
1391 order.push_back(
1392 llvm::find_if(ends,
1393 [&](ObjectFifoDmaEndpointOp e) {
1394 return e.getSymNameAttr() ==
1395 name.getAttr();
1396 }) -
1397 ends.begin());
1398 return contiguousWalkersFit(ends, order);
1399 });
1400 LLVM_DEBUG(
1401 llvm::dbgs()
1402 << " " << pool.getSymName() << " walked whole"
1403 << (linedUp ? "" : ", but its other turns would not line up")
1404 << "\n");
1405 if (linedUp)
1406 walked.push_back(pool);
1407 else
1408 pool.setContiguous(false);
1409 }
1410 if (reason.empty() && (!contiguousWalkersFit(group, turns.turns()) ||
1411 !contiguousWalkersFit(peers, turns.turns()))) {
1412 reason = "a contiguous pool's objects do not divide among its turns";
1413 for (ObjectFifoPoolOp pool : walked)
1414 pool.setContiguous(false);
1415 }
1416 if (!reason.empty()) {
1417 LLVM_DEBUG(llvm::dbgs() << " " << a.getSymName() << " + "
1418 << b.getSymName() << ": " << reason << "\n");
1419 continue;
1420 }
1421 takeTurns(group, turns.turns());
1422 takeTurns(peers, turns.turns());
1423 return true;
1424 }
1425 }
1426 return false;
1427 }

1510 /// Keep a successful largest-first allocation unchanged. On channel
1511 /// exhaustion, try keeping an affected pool's resources local, cheapest
1512 /// first. Existing buffers and locks never move. Each DMA accesses every
1513 /// object in its pool, even when it selects only one segment. Reserving that
1514 /// pool once can therefore free several restricted channels. This is a
1515 /// bounded repair of the greedy buffer placement, not an exhaustive solver
1516 /// for tile placement or memory packing.
1517 LogicalResult planAllocation(ArrayRef<ObjectFifoPoolOp> pools,
1518 std::set<std::vector<unsigned>> &tried) {
1519 std::vector<unsigned> key;
1520 for (auto [index, pool] : llvm::enumerate(pools)) {
1521 if (auto tile = localPools.lookup(pool)) {
1522 auto placed = cast<TileLike>(tile.getDefiningOp());
1523 auto col = placed.tryGetCol(), row = placed.tryGetRow();
1524 assert(col && row && "locality repair requires resolved coordinates");
1525 key.push_back(index);
1526 key.push_back(*col);
1527 key.push_back(*row);
1528 }
1529 }
1530 if (tried.size() >= 64 || !tried.insert(key).second ||
1531 failed(planBuffers(pools)) || failed(planLocks(pools)))
1532 return failure();
1533 DMAChannelAnalysis channels(device);
1534 if (succeeded(assignChannels(channels, /*diagnose=*/false)))
1535 return success();
1536 if (!channelFailure || !tileOf(channelFailure).isMemTile() ||
1537 !tileOf(channelFailure).tryGetCol() ||
1538 !tileOf(channelFailure).tryGetRow())
1539 return failure();
1540
1541 Value localTile = channelFailure.getTile();
1542 SmallVector<ObjectFifoPoolOp> candidates;
1543 for (auto dma : device.getOps<ObjectFifoDmaEndpointOp>()) {
1544 auto endpoint = cast<RouteEndpoint>(dma.getOperation());
1545 auto pool = dma.getPoolOp();
1546 if (!pool || localPools.contains(pool) ||
1547 !sameTile(endpoint.getTile(), channelFailure.getTile()) ||
1548 endpoint.getRouteDirection() != channelFailure.getRouteDirection() ||
1549 (channelFailure.getRouteChannel() && endpoint != channelFailure) ||
1550 (endpoint.getRouteChannel() && endpoint != channelFailure) ||
1551 !reachesAdjacentTile(endpoint) ||
1552 llvm::is_contained(candidates, pool))
1553 continue;
1554 candidates.push_back(pool);
1555 }
1556 llvm::stable_sort(candidates, [&](ObjectFifoPoolOp a, ObjectFifoPoolOp b) {
1557 int64_t aBytes =
1558 a.getBuffers() ? 0 : a.getObjectSizeInBytes() * a.getDepth();
1559 int64_t bBytes =
1560 b.getBuffers() ? 0 : b.getObjectSizeInBytes() * b.getDepth();
1561 if (aBytes != bBytes)
1562 return aBytes < bBytes;
1563 return structuralRank.lookup(a) < structuralRank.lookup(b);
1564 });
1565 for (auto pool : candidates) {
1566 localPools[pool] = localTile;
1567 if (succeeded(planAllocation(pools, tried)))
1568 return success();
1569 localPools.erase(pool);
1570 }
1571 return failure();
1572 }

1890 /// A shim endpoint has no memory of its own, so the runtime needs its channel
1891 /// spelled out under the name the sequence refers to.
1892 void emitShimAllocations() {
1893 builder.setInsertionPoint(device.getBody()->getTerminator());
1894 // A fifo's first shim end takes the name the runtime sequence was pointed
1895 // at. Any further shim end of the same fifo, as when a core broadcasts to
1896 // two shim tiles, is a different tile and channel and needs a record of
1897 // its own; sharing the first would send the runtime to the wrong tile.
1898 llvm::StringMap<unsigned> shimEndsSeen;
1899 for (auto endpoint : device.getOps<RouteEndpoint>()) {
1900 std::optional<StringRef> fifoName = endpoint.getFifoName();
1901 std::optional<int> channel = endpoint.getRouteChannel();
1902 if (!tileOf(endpoint).isShimTile() || !fifoName || !channel) {
1903 continue;
1904 }
1905 std::string name = (*fifoName + "__shim_alloc").str();
1906 unsigned ordinal = shimEndsSeen[*fifoName]++;
1907 bool create = true;
1908 if (ordinal == 0) {
1909 // A record under this name may already be there, declared by hand.
1910 create = !SymbolTable::lookupNearestSymbolFrom<ShimDMAAllocationOp>(
1911 device, builder.getStringAttr(name));
1912 } else {
1913 for (unsigned suffix = ordinal - 1; device.lookupSymbol(name);
1914 suffix++) {
1915 name =
1916 (*fifoName + "__shim_alloc" + std::to_string(suffix + 1)).str();
1917 }
1918 }
1919 if (create) {
1920 ShimDMAAllocationOp::create(
1921 builder, endpoint.getLoc(), builder.getStringAttr(name),
1922 endpoint.getTile(),
1923 DMAChannelDirAttr::get(builder.getContext(),
1924 endpoint.getRouteDirection()),
1925 builder.getI64IntegerAttr(*channel),
1926 builder.getBoolAttr(endpoint.getRouteBundle() == WireBundle::PLIO),
1927 endpoint->getAttrOfType<PacketInfoAttr>("packet"));
1928 }
1929 // The runtime sequence reaches the fifo through this record.
1930 (void)SymbolTable::replaceAllSymbolUses(
1931 endpoint, builder.getStringAttr(name), device);
1932 }

1935 /// Alternative packing order: prioritize DMA demand within each home tile,
1936 /// but merge those lists by object size to avoid globally prioritizing small
1937 /// pools over large objects on unrelated tiles. This is only a heuristic;
1938 /// the same buffer, lock and channel planner validates both orders.
1939 SmallVector<ObjectFifoPoolOp>
1940 demandOrderedPools(ArrayRef<ObjectFifoPoolOp> pools) {
1941 SmallVector<ObjectFifoPoolOp> ordered(pools);
1942 SmallVector<size_t> memTileSlots;
1943 SmallVector<Value> memTileHomes;
1944 DenseMap<Value, SmallVector<ObjectFifoPoolOp>> memTilePools;
1945 DenseMap<Operation *, std::pair<int, int>> poolChannelDemand;
1946 for (auto endpoint : device.getOps<ObjectFifoDmaEndpointOp>()) {
1947 ObjectFifoPoolOp pool = endpoint.getPoolOp();
1948 if (!sameTile(endpoint.getTile(), pool.getTile()))
1949 continue;
1950 auto &demand = poolChannelDemand[pool];
1951 if (endpoint.getRouteDirection() == DMAChannelDir::S2MM)
1952 ++demand.first;
1953 else
1954 ++demand.second;
1955 }
1956 for (auto [index, pool] : llvm::enumerate(ordered)) {
1957 if (pool.getTileLike().isMemTile()) {
1958 memTileSlots.push_back(index);
1959 Value home = pool.getTile();
1960 for (Value known : memTileHomes)
1961 if (sameTile(home, known)) {
1962 home = known;
1963 break;
1964 }
1965 if (!memTilePools.count(home))
1966 memTileHomes.push_back(home);
1967 memTilePools[home].push_back(pool);
1968 }
1969 }
1970 DenseMap<Value, size_t> nextPool;
1971 auto demand = [&](ObjectFifoPoolOp pool) {
1972 auto [input, output] = poolChannelDemand.lookup(pool.getOperation());
1973 return std::max(input, output);
1974 };
1975 for (Value home : memTileHomes) {
1976 auto &tilePools = memTilePools[home];
1977 llvm::stable_sort(tilePools, [&](ObjectFifoPoolOp a, ObjectFifoPoolOp b) {
1978 if (demand(a) != demand(b))
1979 return demand(a) > demand(b);
1980 return a.getObjectSizeInBytes() > b.getObjectSizeInBytes();
1981 });
1982 }
1983 for (size_t slot : memTileSlots) {
1984 Value bestHome;
1985 ObjectFifoPoolOp bestPool;
1986 for (Value home : memTileHomes) {
1987 auto &tilePools = memTilePools[home];
1988 size_t index = nextPool[home];
1989 if (index >= tilePools.size())
1990 continue;
1991 ObjectFifoPoolOp pool = tilePools[index];
1992 if (!bestPool ||
1993 pool.getObjectSizeInBytes() > bestPool.getObjectSizeInBytes() ||
1994 (pool.getObjectSizeInBytes() == bestPool.getObjectSizeInBytes() &&
1995 demand(pool) > demand(bestPool))) {
1996 bestHome = home;
1997 bestPool = pool;
1998 }
1999 }
2000 ordered[slot] = bestPool;
2001 ++nextPool[bestHome];
2002 }
2003 return ordered;
2004 }

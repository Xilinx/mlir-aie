20/// The order one core moves objects through several pools, read off its
21/// program, so that one DMA channel can serve those pools in turn. The core's
22/// period is the innermost loop holding every acquire and release on the
23/// pools, with the loops of static trip count inside it unrolled and the
24/// branches on their counters folded; where a branch depends on that loop's
25/// own counter, the next loop out. Its order through them must not depend on
26/// data.
27class CoreTurns {
28public:
29 /// Reads the order off the program of the core working through `ends`, all
30 /// draining (the channel fills their pools) or all filling (the channel
31 /// drains them). Fails, saying why in `reason`, when the order is not
32 /// static.
33 mlir::LogicalResult read(llvm::ArrayRef<ObjectFifoCoreEndpointOp> ends,
34 std::string &reason);
35 /// The same, before split: `names` are objectFIFOs `core` acquires from
36 /// (`coreDrains`) or into.
37 mlir::LogicalResult read(CoreOp core, llvm::ArrayRef<mlir::StringAttr> names,
38 bool coreDrains, std::string &reason);
39
40 /// Indices into `ends`, one per object the channel moves in a period of the
41 /// core's steady state.
42 llvm::ArrayRef<unsigned> turns() const { return order; }
43
44 /// Whether a channel moving objects in `turns()` order forever, through
45 /// pools of `depths`, never has itself and the core both waiting. Settles
46 /// `turns()` on the shortest such order cut from the period's.
47 bool keepsUp(llvm::ArrayRef<int64_t> depths);
48
49private:
50 bool keepsUp(llvm::ArrayRef<unsigned> order,
51 llvm::ArrayRef<int64_t> depths) const;
52
53 struct Event {
54 unsigned end;
55 int64_t count;
56 bool acquire;
57 };
58
59 llvm::SmallVector<Event> period;
60 llvm::SmallVector<unsigned> order;
61 size_t ends = 0;
62 bool coreDrains = true;
63};

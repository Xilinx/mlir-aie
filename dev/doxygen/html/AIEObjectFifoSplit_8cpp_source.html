33std::optional<ObjectFifoLinkOp> getOptionalLinkOp(ObjectFifoCreateOp op) {
34 auto device = op->getParentOfType<DeviceOp>();
35 for (ObjectFifoLinkOp linkOp : device.getOps<ObjectFifoLinkOp>()) {
36 for (ObjectFifoCreateOp in : linkOp.getInputObjectFifos()) {
37 if (in == op) {
38 return linkOp;
39 }
40 }
41 for (ObjectFifoCreateOp out : linkOp.getOutputObjectFifos()) {
42 if (out == op) {
43 return linkOp;
44 }
45 }
46 }
47 return {};
48}

50/// The aie.objectfifo.allocate naming `op`, if any. Reports when several claim
51/// the same fifo, since only one delegate tile can hold its objects.
52std::optional<ObjectFifoAllocateOp>
53getOptionalAllocateOp(ObjectFifoCreateOp op) {
54 std::optional<ObjectFifoAllocateOp> found;
55 for (auto alloc :
56 op->getParentOfType<DeviceOp>().getOps<ObjectFifoAllocateOp>()) {
57 if (alloc.getObjFifoName() != op.name().getValue()) {
58 continue;
59 }
60 if (found) {
61 op.emitOpError("has more than one allocate operation");
62 }
63 found = alloc;
64 }
65 return found;
66}

68/// Whether `delegate`'s memory module is reachable from both ends of `op`.
69bool delegateReachesBothEnds(ObjectFifoCreateOp op, TileOp delegate) {
70 auto consumerTileOp = cast<TileOp>(op.getConsumerTiles()[0].getDefiningOp());
71 auto reachesDelegate = [&](TileOp end) {
72 using SharedMemory = AIETargetModel::SharedMemory;
73 SharedMemory shared = sharedMemory(delegate, end);
74 return shared == SharedMemory::First || shared == SharedMemory::Either;
75 };
76 return reachesDelegate(op.getProducerTileOp()) &&
77 reachesDelegate(consumerTileOp);
78}

80/// A fifo needs DMAs unless both ends reach one memory module and neither end
81/// asks the DMA to reshape the data on the way.
82bool requiresDMAs(ObjectFifoCreateOp createOp,
83 AIETargetModel::SharedMemory &shared) {
84 if (createOp.getVia_DMA() || createOp.getRepeatCount() ||
85 createOp.getAieStream() || createOp.getConsumerElemType()) {
86 return true;
87 }
88
89 if (createOp.getConsumerTiles().size() == 1 &&
90 createOp.getDimensionsToStream().empty()) {
91 auto consumerTileOp =
92 cast<TileOp>(createOp.getConsumerTiles()[0].getDefiningOp());
93 shared = sharedMemory(createOp.getProducerTileOp(), consumerTileOp);
94 }
95
96 if (shared == AIETargetModel::SharedMemory::None) {
97 return true;
98 }
99
100 for (BDDimLayoutArrayAttr dims :
101 createOp.getDimensionsFromStreamPerConsumer()) {
102 if (!dims.empty()) {
103 return true;
104 }
105 }
106
107 // A link passes objects between two fifos through a tile's DMAs, unless an
108 // aie.objectfifo.allocate names a delegate tile whose memory both ends of
109 // this fifo can reach directly.
110 if (getOptionalLinkOp(createOp)) {
111 if (auto alloc = getOptionalAllocateOp(createOp)) {
112 return !delegateReachesBothEnds(createOp, alloc->getDelegateTileOp());
113 }
114 return true;
115 }
116
117 return false;
118}

120/// Objects a fifo needs on `tile`: enough to cover the largest acquire made
121/// there, plus one so a core can hold an object while the next arrives. A fifo
122/// that states a depth per endpoint is taken at its word instead.
123int objectCountOn(DeviceOp device, Value tile, ObjectFifoCreateOp objFifo) {
124 if (objFifo.size() == 0) {
125 return 0;
126 }
127
128 int maxAcquire = 0;
129 for (auto coreOp : device.getOps<CoreOp>()) {
130 if (coreOp.getTile() == tile) {
131 coreOp.walk([&](ObjectFifoAcquireOp acqOp) {
132 if (acqOp.getObjectFifo() == objFifo) {
133 maxAcquire = std::max(maxAcquire, acqOp.acqNumber());
134 }
135 });
136 }
137 }
138
139 if (maxAcquire == 0) {
140 return objFifo.size();
141 }
142 if (maxAcquire == 1 && objFifo.size() == 1) {
143 return 1;
144 }
145 return maxAcquire + 1;
146}

148bool hasCoreAccess(DeviceOp device, Value tile, ObjectFifoCreateOp objFifo,
149 ObjectFifoPort port) {
150 for (auto coreOp : device.getOps<CoreOp>()) {
151 if (coreOp.getTile() != tile) {
152 continue;
153 }
154 bool found = false;
155 coreOp.walk([&](Operation *op) {
156 if (auto acq = dyn_cast<ObjectFifoAcquireOp>(op)) {
157 found |= acq.getObjectFifo() == objFifo && acq.getPort() == port;
158 } else if (auto rel = dyn_cast<ObjectFifoReleaseOp>(op)) {
159 found |= rel.getObjectFifo() == objFifo && rel.getPort() == port;
160 }
161 });
162 if (found) {
163 return true;
164 }
165 }
166 return false;
167}

229 ObjectFifoDmaEndpointOp createDmaEndpoint(Location loc, StringRef name,
230 Value tile, PoolRef ref,
231 ObjectFifoRole role,
232 ObjectFifoCreateOp from,
233 BDDimLayoutArrayAttr dims,
234 std::optional<int> channelIndex) {
235 // Both ends of the fifo carry `depth * repeat_count * iter_count` objects.
236 // The draining end spends `repeat_count` on descriptors, replaying each
237 // object in one pass of its chain; the filling end spends it on passes.
238 std::optional<int32_t> iterCount;
239 if (std::optional<int32_t> fifoIterations = from.getIterCount()) {
240 iterCount = role == ObjectFifoRole::Drain
241 ? *fifoIterations
242 : *fifoIterations * from.getRepeatCount().value_or(1);
243 }
244 ArrayAttr segments = selectedSegmentNames(ref);
245
246 BDDimLayoutArrayArrayAttr dimensions;
247 if (dims && !dims.empty()) {
248 SmallVector<BDDimLayoutArrayAttr> perSegment(ref.segments.size(), dims);
249 dimensions =
250 BDDimLayoutArrayArrayAttr::get(builder.getContext(), perSegment);
251 }
252
253 BDPadLayoutArrayArrayAttr padding;
254 std::optional<ArrayRef<BDPadLayoutAttr>> padDims = from.getPadDimensions();
255 if (role == ObjectFifoRole::Drain && padDims && !padDims->empty()) {
256 SmallVector<BDPadLayoutArrayAttr> perSegment(ref.segments.size(),
257 from.getPadDimensionsAttr());
258 padding =
259 BDPadLayoutArrayArrayAttr::get(builder.getContext(), perSegment);
260 }
261
262 return ObjectFifoDmaEndpointOp::create(
263 builder, loc, builder.getStringAttr(name), tile,
264 ObjectFifoRoleAttr::get(builder.getContext(), role),
265 FlatSymbolRefAttr::get(builder.getContext(), ref.pool.getSymName()),
266 segments,
267 channelIndex ? builder.getI32IntegerAttr(*channelIndex) : IntegerAttr(),
268 dimensions, padding,
269 from.getPadValue() ? builder.getI32IntegerAttr(from.getPadValue())
270 : IntegerAttr(),
271 iterCount ? builder.getI32IntegerAttr(*iterCount) : IntegerAttr(),
272 /*packet=*/PacketInfoAttr(),
273 builder.getStringAttr(from.name().getValue()));
274 }

290 /// External buffers registered against a fifo's shim end form a pool in DDR.
291 /// An unregistered end receives its address from the runtime at dispatch.
292 std::optional<PoolRef> registeredPool(ObjectFifoCreateOp fifo, Value tile,
293 StringRef name, MemRefType elemType) {
294 SmallVector<Attribute> names;
295 for (auto regOp : device.getOps<ObjectFifoRegisterExternalBuffersOp>()) {
296 if (regOp.getTile() != tile || regOp.getObjectFifo() != fifo) {
297 continue;
298 }
299 for (Value buffer : regOp.getExternalBuffers()) {
300 auto external = cast<ExternalBufferOp>(buffer.getDefiningOp());
301 names.push_back(SymbolRefAttr::get(external.getSymNameAttr()));
302 // The registered buffer's own extent is what the DMA moves, which need
303 // not be the fifo's object size.
304 elemType = cast<MemRefType>(external.getType());
305 }
306 }
307 if (names.empty()) {
308 return std::nullopt;
309 }
310
311 auto pool = ObjectFifoPoolOp::create(
312 builder, fifo.getLoc(), name, tile, (int)names.size(), elemType,
313 builder.getArrayAttr(names),
314 /*locks=*/ArrayAttr(), /*repeatCount=*/IntegerAttr(),
315 fifo.getDisableSynchronization(),
316 builder.getStringAttr(fifo.name().getValue()),
317 /*initValues=*/ArrayAttr());
318 createSegments(pool, fifo.getLoc(), {{0, elemType.getNumElements()}});
319 return PoolRef{pool, {0}};
320 }

322 /// Point a core's accesses at the endpoint it works through.
323 void retargetCoreAccesses(ObjectFifoCreateOp fifo, ObjectFifoPort port,
324 Value tile, StringRef endpointName) {
325 auto name = FlatSymbolRefAttr::get(builder.getContext(), endpointName);
326 // Each access moves to the endpoint for the port it named, so this cannot
327 // be a blanket replaceAllSymbolUses: the two ports of one fifo become two
328 // different symbols.
329 for (auto coreOp : device.getOps<CoreOp>()) {
330 if (coreOp.getTile() != tile) {
331 continue;
332 }
333 coreOp.walk([&](Operation *op) {
334 if (auto acq = dyn_cast<ObjectFifoAcquireOp>(op)) {
335 if (acq.getObjectFifo() == fifo && acq.getPort() == port) {
336 acq.setObjFifoNameAttr(name);
337 acq.removePortAttr();
338 }
339 } else if (auto rel = dyn_cast<ObjectFifoReleaseOp>(op)) {
340 if (rel.getObjectFifo() == fifo && rel.getPort() == port) {
341 rel.setObjFifoNameAttr(name);
342 rel.removePortAttr();
343 }
344 }
345 });
346 }

351 /// Every pool sits on a concrete tile, so the fifo's ends must be placed.
352 LogicalResult verifyTilesArePlaced() {
353 for (auto fifo : device.getOps<ObjectFifoCreateOp>()) {
354 auto placed = [](Value tile) {
355 return isa_and_nonnull<TileOp>(tile.getDefiningOp());
356 };
357 if (!placed(fifo.getProducerTile())) {
358 return fifo.emitOpError("producer tile is not a placed aie.tile; run "
359 "--aie-place-tiles before this pass");
360 }
361 for (Value consumer : fifo.getConsumerTiles()) {
362 if (!placed(consumer)) {
363 return fifo.emitOpError("consumer tile is not a placed aie.tile; run "
364 "--aie-place-tiles before this pass");
365 }
366 }
367 }
368 return success();
369 }

371 /// A delegate tile can only hold a fifo's objects if both ends reach its
372 /// memory module. A fifo that meets a link elsewhere has no objects of its
373 /// own to place, so its delegate is moot.
374 LogicalResult verifyAllocateDelegates() {
375 for (auto fifo : device.getOps<ObjectFifoCreateOp>()) {
376 if (getOptionalLinkOp(fifo) && !linkPoolOwner.contains(fifo)) {
377 continue;
378 }
379 auto alloc = getOptionalAllocateOp(fifo);
380 if (alloc && !delegateReachesBothEnds(fifo, alloc->getDelegateTileOp())) {
381 return alloc->emitOpError("objectfifo has no shared memory access to "
382 "delegate tile's memory module");
383 }
384 }
385 return success();
386 }
387
388 /// A fifo end wired to a Core stream port holds no objects, so there is
389 /// nothing for a core to acquire or release there.
390 LogicalResult verifyStreamPortAccesses() {
391 auto check = [&](Operation *op, ObjectFifoCreateOp fifo,
392 std::optional<ObjectFifoPort> port, StringRef verb) {
393 if (!fifo || !port || !fifo.getAieStream()) {
394 return success();
395 }
396 int streamEnd = *fifo.getAieStream();
397 int end = *port == ObjectFifoPort::Produce ? 0 : 1;
398 if (streamEnd != 2 && streamEnd != end) {
399 return success();
400 }
401 return LogicalResult(op->emitOpError("cannot ")
402 << verb << " objectfifo stream port");
403 };
404 for (auto coreOp : device.getOps<CoreOp>()) {
405 auto result = coreOp.walk([&](Operation *op) {
406 LogicalResult ok = success();
407 if (auto acq = dyn_cast<ObjectFifoAcquireOp>(op)) {
408 ok = check(op, acq.getObjectFifo(), acq.getPort(), "acquire from");
409 } else if (auto rel = dyn_cast<ObjectFifoReleaseOp>(op)) {
410 ok = check(op, rel.getObjectFifo(), rel.getPort(), "release from");
411 }
412 return failed(ok) ? WalkResult::interrupt() : WalkResult::advance();
413 });
414 if (result.wasInterrupted()) {
415 return failure();
416 }
417 }
418 return success();
419 }
420
421 /// The DMAs on either side of a link own the shared pool between them, so a
422 /// core sharing the link point has no claim on the objects passing through.
423 LogicalResult verifyLinkAccesses() {
424 for (auto linkOp : device.getOps<ObjectFifoLinkOp>()) {
425 auto sharedTile = linkOp.getOptionalSharedTile();
426 if (!sharedTile) {
427 continue;
428 }
429 auto reject = [&](ObjectFifoCreateOp fifo) {
430 return LogicalResult(
431 linkOp.emitOpError("core on the shared tile cannot access '")
432 << fifo.name().getValue() << "': a link moves objects by DMA");
433 };
434 for (ObjectFifoCreateOp in : linkOp.getInputObjectFifos()) {
435 if (hasCoreAccess(device, *sharedTile, in, ObjectFifoPort::Consume)) {
436 return reject(in);
437 }
438 }
439 for (ObjectFifoCreateOp out : linkOp.getOutputObjectFifos()) {
440 if (hasCoreAccess(device, *sharedTile, out, ObjectFifoPort::Produce)) {
441 return reject(out);
442 }
443 }
444 }
445 return success();
446 }

448 /// Ends of a linked fifo that resolve to the link's shared pool.
449 DenseMap<Operation *, PoolRef> linkedProducerEnd;
450 DenseMap<Operation *, PoolRef> linkedConsumerEnd;
451 /// Fifos whose own objects a link's shared pool holds.
452 DenseSet<Operation *> linkPoolOwner;
453 /// Shim endpoint each fifo is driven through, where it has one.
454 DenseMap<Operation *, std::string> shimEndpointName;

459/// A link's participants meet on one pool on the shared tile: the fifo carrying
460/// whole objects owns its size, and each of the other side's fifos occupies one
461/// segment.
462void AIEObjectFifoSplitPass::createLinkPools() {
463 for (auto linkOp : device.getOps<ObjectFifoLinkOp>()) {
464 auto sharedTile = linkOp.getOptionalSharedTile();
465 if (!sharedTile) {
466 continue;
467 }
468
469 std::vector<ObjectFifoCreateOp> ins = linkOp.getInputObjectFifos();
470 std::vector<ObjectFifoCreateOp> outs = linkOp.getOutputObjectFifos();
471 bool isJoin = linkOp.isJoin();
472 bool isDistribute = linkOp.isDistribute();
473
474 ObjectFifoCreateOp owner = isJoin ? outs[0] : ins[0];
475 if (!isJoin && !isDistribute) {
476 auto inType = cast<MemRefType>(
477 cast<AIEObjectFifoType>(ins[0].getElemType()).getElementType());
478 auto outType = cast<MemRefType>(
479 cast<AIEObjectFifoType>(outs[0].getElemType()).getElementType());
480 // Padding is applied as the objects leave, so the pool holds what
481 // arrives.
482 if (outs[0].getInitValues() ||
483 (outType.getNumElements() > inType.getNumElements() &&
484 !outs[0].getPadDimensions())) {
485 owner = outs[0];
486 }
487 }
488
489 auto elemType = cast<MemRefType>(
490 cast<AIEObjectFifoType>(owner.getElemType()).getElementType());
491 builder.setInsertionPoint(owner);
492
493 SmallVector<std::pair<int64_t, int64_t>> extents;
494 if (isJoin || isDistribute) {
495 ArrayAttr offsets =
496 isJoin ? linkOp.getSrcOffsets() : linkOp.getDstOffsets();
497 std::vector<ObjectFifoCreateOp> &side = isJoin ? ins : outs;
498 for (auto [index, participant] : llvm::enumerate(side)) {
499 int64_t offset = *getConstantIntValue(offsets[index]);
500 int64_t next = index + 1 < side.size()
501 ? *getConstantIntValue(offsets[index + 1])
502 : elemType.getNumElements();
503 int64_t size = next - offset;
504 if (size <= 0) {
505 size = cast<MemRefType>(
506 cast<AIEObjectFifoType>(participant.getElemType())
507 .getElementType())
508 .getNumElements();
509 }
510 extents.emplace_back(offset, size);
511 }
512 } else {
513 extents.emplace_back(0, elemType.getNumElements());
514 }
515
516 bool ownerIsOutput = llvm::is_contained(outs, owner);
517 std::string name;
518 int depth;
519 if (ownerIsOutput) {
520 name = (owner.name().getValue() + "_pool").str();
521 depth = owner.size();
522 } else {
523 auto consumers = owner.getConsumerTiles();
524 int index =
525 std::distance(consumers.begin(), llvm::find(consumers, *sharedTile));
526 name = (owner.name().getValue() +
527 (consumers.size() > 1 ? "_" + std::to_string(index) + "_cons_pool"
528 : "_cons_pool"))
529 .str();
530 depth = isa<ArrayAttr>(owner.getElemNumber())
531 ? owner.size(index + 1)
532 : objectCountOn(device, *sharedTile, owner);
533 }
534
535 auto pool = createPool(
536 linkOp.getLoc(), name, *sharedTile, depth, elemType, owner, extents,
537 /*holdsInitialContents=*/ownerIsOutput, linkOp.getRepeatCount());
538 linkPoolOwner.insert(owner);
539
540 SmallVector<int32_t> allSegments;
541 for (int32_t index = 0; index < (int32_t)extents.size(); ++index) {
542 allSegments.push_back(index);
543 }
544 for (auto [index, in] : llvm::enumerate(ins)) {
545 linkedConsumerEnd[in] = PoolRef{
546 pool, isJoin ? SmallVector<int32_t>{static_cast<int32_t>(index)}
547 : allSegments};
548 }
549 for (auto [index, out] : llvm::enumerate(outs)) {
550 linkedProducerEnd[out] = PoolRef{
551 pool, isDistribute ? SmallVector<int32_t>{static_cast<int32_t>(index)}
552 : allSegments};
553 }
554 }

557void AIEObjectFifoSplitPass::runOnOperation() {
558 device = getOperation();
559 builder = OpBuilder(device.getContext());
560 builder.setInsertionPoint(device.getBody()->getTerminator());
561
562 SmallVector<ObjectFifoCreateOp> fifos(device.getOps<ObjectFifoCreateOp>());
563
564 if (failed(verifyTilesArePlaced()) || failed(verifyStreamPortAccesses()) ||
565 failed(verifyLinkAccesses())) {
566 return signalPassFailure();
567 }
568
569 createLinkPools();
570
571 if (failed(verifyAllocateDelegates())) {
572 return signalPassFailure();
573 }
574
575 for (ObjectFifoCreateOp fifo : fifos) {
576 builder.setInsertionPoint(fifo);
577 Location loc = fifo.getLoc();
578 StringRef fifoName = fifo.name().getValue();
579 auto fifoType = cast<AIEObjectFifoType>(fifo.getElemType());
580 auto elemType = cast<MemRefType>(fifoType.getElementType());
581 auto consElemType = cast<MemRefType>(
582 cast<AIEObjectFifoType>(fifo.getConsumerElemTypeOrDefault())
583 .getElementType());
584
585 auto sharedModule = AIETargetModel::SharedMemory::None;
586 bool shared = !requiresDMAs(fifo, sharedModule);
587
588 if (shared) {
589 PoolRef ref;
590 if (auto linked = linkedProducerEnd.find(fifo);
591 linked != linkedProducerEnd.end()) {
592 ref = linked->second;
593 } else if (auto linked = linkedConsumerEnd.find(fifo);
594 linked != linkedConsumerEnd.end()) {
595 ref = linked->second;
596 } else {
597 Value tile = sharedModule == AIETargetModel::SharedMemory::Second
598 ? fifo.getConsumerTiles()[0]
599 : fifo.getProducerTile();
600 // An aie.objectfifo.allocate names the tile whose memory is to hold the
601 // objects, in place of either end's own.
602 if (auto alloc = getOptionalAllocateOp(fifo)) {
603 if (!delegateReachesBothEnds(fifo, alloc->getDelegateTileOp())) {
604 alloc->emitOpError("objectfifo has no shared memory access to "
605 "delegate tile's memory module");
606 return signalPassFailure();
607 }
608 tile = alloc->getDelegateTile();
609 }
610 ref = PoolRef{
611 createPool(loc, (fifoName + "_pool").str(), tile, fifo.size(),
612 elemType, fifo, {{0, elemType.getNumElements()}},
613 /*holdsInitialContents=*/true, fifo.getRepeatCount()),
614 {0}};
615 }
616
617 if (hasCoreAccess(device, fifo.getProducerTile(), fifo,
618 ObjectFifoPort::Produce)) {
619 auto name = (fifoName + "_prod").str();
620 createCoreEndpoint(loc, name, fifo.getProducerTile(), ref,
621 ObjectFifoRole::Fill);
622 retargetCoreAccesses(fifo, ObjectFifoPort::Produce,
623 fifo.getProducerTile(), name);
624 }
625 if (hasCoreAccess(device, fifo.getConsumerTiles()[0], fifo,
626 ObjectFifoPort::Consume)) {
627 auto name = (fifoName + "_cons").str();
628 createCoreEndpoint(loc, name, fifo.getConsumerTiles()[0], ref,
629 ObjectFifoRole::Drain);
630 retargetCoreAccesses(fifo, ObjectFifoPort::Consume,
631 fifo.getConsumerTiles()[0], name);
632 }
633 continue;
634 }
635
636 Value prodTile = fifo.getProducerTile();
637 // A fifo end wired straight to a Core stream port has no objects of its
638 // own: whatever the core writes goes out on the port.
639 int streamEnd = fifo.getAieStream().value_or(-1);
640 std::optional<int> prodStreamPort;
641 std::optional<int> consStreamPort;
642 if (streamEnd == 0 || streamEnd == 2) {
643 prodStreamPort = fifo.getAieStreamPort();
644 }
645 if (streamEnd == 1 || streamEnd == 2) {
646 consStreamPort = fifo.getAieStreamPort();
647 }
648
649 bool prodIsShim = cast<TileOp>(prodTile.getDefiningOp()).isShimTile();
650 std::optional<PoolRef> prodRef;
651 // Objects entering a repeating link arrive that many at a time.
652 if (auto linked = linkedProducerEnd.find(fifo);
653 linked != linkedProducerEnd.end()) {
654 prodRef = linked->second;
655 } else if (prodIsShim) {
656 if (!fifo.getPlio()) {
657 prodRef = registeredPool(fifo, prodTile, (fifoName + "_pool").str(),
658 elemType);
659 }
660 } else if (!prodStreamPort) {
661 int depth = fifo.getInitValues() ? fifo.size()
662 : objectCountOn(device, prodTile, fifo);
663 auto pool =
664 createPool(loc, (fifoName + "_pool").str(), prodTile, depth, elemType,
665 fifo, {{0, elemType.getNumElements()}},
666 /*holdsInitialContents=*/true, fifo.getRepeatCount());
667 prodRef = PoolRef{pool, {0}};
668 }
669
670 if (prodRef &&
671 hasCoreAccess(device, prodTile, fifo, ObjectFifoPort::Produce)) {
672 auto name = (fifoName + "_prod").str();
673 createCoreEndpoint(loc, name, prodTile, *prodRef, ObjectFifoRole::Fill);
674 retargetCoreAccesses(fifo, ObjectFifoPort::Produce, prodTile, name);
675 }
676
677 std::string prodDmaName = (fifoName + "_prod_dma").str();
678 if (prodRef) {
679 createDmaEndpoint(
680 loc, prodDmaName, prodTile, *prodRef, ObjectFifoRole::Drain, fifo,
681 fifo.getDimensionsToStreamAttr(), fifo.getProdDmaChannel());
682 } else {
683 createRouteEndpoint(
684 loc, prodDmaName, prodTile,
685 prodStreamPort ? WireBundle::Core
686 : fifo.getPlio() ? WireBundle::PLIO
687 : WireBundle::DMA,
688 prodStreamPort ? prodStreamPort : fifo.getProdDmaChannel(), fifo);
689 }
690 if (prodIsShim) {
691 shimEndpointName[fifo] = prodDmaName;
692 }
693
694 SmallVector<Attribute> destinations;
695 ArrayRef<BDDimLayoutArrayAttr> consumerDims =
696 fifo.getDimensionsFromStreamPerConsumer();
697 int consumerIndex = 0;
698 for (Value consumerTile : fifo.getConsumerTiles()) {
699 std::string suffix = fifo.getConsumerTiles().size() > 1
700 ? ("_" + std::to_string(consumerIndex) + "_cons")
701 : "_cons";
702 bool consIsShim = cast<TileOp>(consumerTile.getDefiningOp()).isShimTile();
703
704 std::optional<PoolRef> consRef;
705 if (auto linked = linkedConsumerEnd.find(fifo);
706 linked != linkedConsumerEnd.end() &&
707 consumerTile == linked->second.pool.getTile()) {
708 consRef = linked->second;
709 } else if (consIsShim) {
710 if (!fifo.getPlio()) {
711 consRef =
712 registeredPool(fifo, consumerTile,
713 (fifoName + suffix + "_pool").str(), consElemType);
714 }
715 } else if (!consStreamPort) {
716 int depth = isa<ArrayAttr>(fifo.getElemNumber())
717 ? fifo.size(consumerIndex + 1)
718 : objectCountOn(device, consumerTile, fifo);
719 auto pool = createPool(loc, (fifoName + suffix + "_pool").str(),
720 consumerTile, depth, consElemType, fifo,
721 {{0, consElemType.getNumElements()}},
722 /*holdsInitialContents=*/false,
723 /*repeatCount=*/std::nullopt);
724 consRef = PoolRef{pool, {0}};
725 }
726
727 if (consRef &&
728 hasCoreAccess(device, consumerTile, fifo, ObjectFifoPort::Consume)) {
729 auto name = (fifoName + suffix).str();
730 createCoreEndpoint(loc, name, consumerTile, *consRef,
731 ObjectFifoRole::Drain);
732 retargetCoreAccesses(fifo, ObjectFifoPort::Consume, consumerTile, name);
733 }
734
735 std::optional<int> pinned;
736 if (auto consChannels = fifo.getConsDmaChannels();
737 consChannels && consumerIndex < (int)consChannels->size() &&
738 (*consChannels)[consumerIndex] >= 0) {
739 pinned = (*consChannels)[consumerIndex];
740 }
741
742 std::string consDmaName = (fifoName + suffix + "_dma").str();
743 if (consRef) {
744 createDmaEndpoint(loc, consDmaName, consumerTile, *consRef,
745 ObjectFifoRole::Fill, fifo,
746 consumerDims.empty() ? BDDimLayoutArrayAttr()
747 : consumerDims[consumerIndex],
748 pinned);
749 } else {
750 createRouteEndpoint(loc, consDmaName, consumerTile,
751 consStreamPort ? WireBundle::Core
752 : fifo.getPlio() ? WireBundle::PLIO
753 : WireBundle::DMA,
754 consStreamPort ? consStreamPort : pinned, fifo);
755 }
756 destinations.push_back(
757 FlatSymbolRefAttr::get(builder.getContext(), consDmaName));
758 if (consIsShim) {
759 shimEndpointName[fifo] = consDmaName;
760 }
761 consumerIndex++;
762 }
763
764 RouteOp::create(builder, loc,
765 FlatSymbolRefAttr::get(builder.getContext(), prodDmaName),
766 builder.getArrayAttr(destinations),
767 fifo.getPacket() ? builder.getUnitAttr() : UnitAttr(),
768 fifo.getPacketIdAttr());
769 }
770
771 SmallVector<Operation *> toErase;
772 for (ObjectFifoCreateOp fifo : fifos) {
773 // The runtime sequence drives a fifo through its shim end.
774 auto shim = shimEndpointName.find(fifo);
775 if (shim != shimEndpointName.end()) {
776 (void)SymbolTable::replaceAllSymbolUses(
777 fifo, builder.getStringAttr(shim->second), device);
778 }
779 toErase.push_back(fifo);
780 }
781 for (auto linkOp : device.getOps<ObjectFifoLinkOp>()) {
782 toErase.push_back(linkOp);
783 }
784 for (auto allocOp : device.getOps<ObjectFifoAllocateOp>()) {
785 toErase.push_back(allocOp);
786 }
787 for (auto regOp : device.getOps<ObjectFifoRegisterExternalBuffersOp>()) {
788 toErase.push_back(regOp);
789 }
790 for (Operation *op : toErase) {
791 op->erase();
792 }
793}

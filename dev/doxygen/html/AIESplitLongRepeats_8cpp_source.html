33struct AIESplitLongRepeatsPass
34 : xilinx::AIEX::impl::AIESplitLongRepeatsBase<AIESplitLongRepeatsPass> {
35 void runOnOperation() override {
36 AIE::DeviceOp device = getOperation();
37 uint32_t maxRepeat = device.getTargetModel().getMaxRepeatCount();
38 int64_t queueDepth = device.getTargetModel().getDmaTaskQueueDepth();
39 // A target without a repeat field has nothing to split into; its push
40 // verifier reports the count instead.
41 if (maxRepeat == 0)
42 return;
43 SmallVector<DMAStartTaskOp> starts;
44 device.walk([&](DMAStartTaskOp start) { starts.push_back(start); });
45 for (DMAStartTaskOp start : starts) {
46 // A start's own count needs no configure. Otherwise the count is the
47 // task's, read through any control flow that carries the task.
48 std::optional<int64_t> rc;
49 if (std::optional<uint32_t> own = start.getRepeatCount()) {
50 rc = *own;
51 } else if (Value ownVal = start.getRepeatCountVal()) {
52 rc = getConstantIntValue(ownVal);
53 } else {
54 DMAConfigureTaskOp cfg = start.getTaskOp();
55 if (!cfg)
56 cfg = getUniqueReachableConfigure(start.getTask());
57 if (!cfg)
58 continue;
59 rc = getConstantIntValue(start.getPushRepeatCount(cfg));
60 }
61 if (!rc || *rc <= maxRepeat)
62 continue;
63 int64_t pushes = (*rc + maxRepeat + 1) / (maxRepeat + 1);
64 if (pushes > static_cast<int64_t>(maxPushes)) {
65 start.emitOpError("repeat count ")
66 << *rc << " needs " << pushes
67 << " queue pushes, more than max-pushes (" << maxPushes.getValue()
68 << ")";
69 return signalPassFailure();
70 }
71 if (queueDepth > 0 && pushes > queueDepth)
72 start.emitWarning("repeat count ")
73 << *rc << " needs " << pushes << " queue pushes, more than the "
74 << queueDepth
75 << " the channel's queue holds, so the sequence waits for this "
76 "channel to drain before going on. A transfer this one waits "
77 "on must be started before it, or the wait never ends.";
78 // Leading starts withhold the token, so an await on the task still
79 // returns only after the last pass.
80 OpBuilder b(start);
81 int64_t runs = *rc + 1;
82 for (; runs > maxRepeat + 1; runs -= maxRepeat + 1)
83 DMAStartTaskOp::create(b, start.getLoc(), start.getTask(),
84 b.getI32IntegerAttr(maxRepeat),
85 /*repeat_count_val=*/nullptr,
86 /*no_token=*/b.getUnitAttr());
87 start.setRepeatCountAttr(b.getI32IntegerAttr(runs - 1));
88 start.getRepeatCountValMutable().clear();
89 }
90 }
91};

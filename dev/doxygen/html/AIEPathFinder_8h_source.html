41// The connections from the source ports to the destination ports of a
42// switchbox, if `srcCoords == dstCoords`, or of the wires from one switchbox
43// to its neighbour at `dstCoords`.
44struct SwitchboxConnect {
45 // A connection from a source port to a destination port.
46 struct Cell {
47 bool available = false;
48 // weight of Dijkstra's shortest path
49 double demand = 0;
50 // iterations the connection was over capacity
51 int overCapacity = 0;
52 // circuit streams using the connection
53 int usedCapacity = 0;
54 // packet streams using the connection
55 int packetFlowCount = 0;
56 // only packet streams of this group may share the connection
57 int packetGroupId = -1;
58 // packet ids routed through the connection (and its crossbar row and
59 // column), by the flow routing each; it may be shared only among distinct
60 // ids.
61 llvm::SmallDenseMap<int, int, 2> packetIds;
62 // whether a prioritized flow uses the connection
63 bool isPriority = false;
64 };
65
66 SwitchboxConnect() = default;
67 SwitchboxConnect(TileID coords) : srcCoords(coords), dstCoords(coords) {}
68 SwitchboxConnect(TileID srcCoords, TileID dstCoords)
69 : srcCoords(srcCoords), dstCoords(dstCoords) {}
70
71 TileID srcCoords, dstCoords;
72 std::vector<Port> srcPorts;
73 std::vector<Port> dstPorts;
74 // The connection from srcPorts[i] to dstPorts[j] is cells[i * dstPorts.size()
75 // + j].
76 std::vector<Cell> cells;
77 // Units of dst ports tied to one arbiter (see planArbiters in
78 // AIECreatePathFindFlows.cpp); a unit's leader lists the packet flows
79 // (indices into the router's flows) leaving by any of its ports
80 llvm::IntEqClasses dstUnits;
81 std::vector<llvm::SmallVector<int, 2>> unitPacketFlows;
82 // source ports the design already gives packet rules, which circuit streams
83 // cannot enter
84 llvm::BitVector packetOnlySrc;
85 // dst ports packet streams may no longer take
86 llvm::BitVector circuitOnlyDst;
87
88 Cell &at(size_t i, size_t j) { return cells[i * dstPorts.size() + j]; }
89 const Cell &at(size_t i, size_t j) const {
90 return cells[i * dstPorts.size() + j];
91 }
92
93 // The index of `p` in srcPorts or dstPorts, or -1.
94 int srcIndex(Port p) const { return indexIn(srcPorts, p); }
95 int dstIndex(Port p) const { return indexIn(dstPorts, p); }
96
97 // Size the cells and the per-port state to srcPorts and dstPorts.
98 void resize() {
99 cells.assign(srcPorts.size() * dstPorts.size(), Cell());
100 packetOnlySrc = llvm::BitVector(srcPorts.size());
101 circuitOnlyDst = llvm::BitVector(dstPorts.size());
102 unitPacketFlows.resize(dstPorts.size());
103 resetUnits();
104 }
105
106 void resetUnits() {
107 dstUnits.clear();
108 dstUnits.grow(dstPorts.size());
109 for (auto &flows : unitPacketFlows)
110 flows.clear();
111 }
112
113 llvm::ArrayRef<int> unitFlows(int j) const {
114 return unitPacketFlows[dstUnits.findLeader(j)];
115 }
116
117 void addToUnit(int j, int flow) {
118 auto &flows = unitPacketFlows[dstUnits.findLeader(j)];
119 if (!llvm::is_contained(flows, flow))
120 flows.push_back(flow);
121 }
122
123 void joinUnits(int j, int k) {
124 j = dstUnits.findLeader(j);
125 k = dstUnits.findLeader(k);
126 if (j == k)
127 return;
128 int leader = dstUnits.join(j, k);
129 int other = leader == j ? k : j;
130 for (int flow : unitPacketFlows[other])
131 addToUnit(leader, flow);
132 unitPacketFlows[other].clear();
133 }
134
135 // update demand at the beginning of each dijkstraShortestPaths iteration
136 void updateDemand();
137
138 // Bumps an over-capacity cell's demand, all but ruling it out if a
139 // prioritized flow uses it, to keep prioritized flows' routes.
140 static void bumpDemand(Cell &c);
141
142private:
143 static int indexIn(llvm::ArrayRef<Port> ports, Port p) {
144 const Port *it = llvm::find(ports, p);
145 return it == ports.end() ? -1 : it - ports.begin();
146 }
147};

228// Whether packet flows from the two sources, with ids among the two sets, can
229// deadlock if they share an arbiter; see StreamConflicts::conflict.
230using PacketConflict =
231 std::function<bool(const PathEndPoint &, const std::set<int> &,

234// Packets `src` sends with ids `a` and `b` share a master port they leave
235// tile `at` by, which puts them on one arbiter. If `apart` is set, the
236// packets with id `b` for that destination on the tile reach it by a slave
237// port of their own instead.
238struct TreeSplit {
239 PathEndPoint src;
240 TileID at;
241 int a, b;
242 std::optional<PathEndPoint> apart;
243
244 bool operator<(const TreeSplit &rhs) const {
245 return std::tie(src, at, a, b, apart) <
246 std::tie(rhs.src, rhs.at, rhs.a, rhs.b, rhs.apart);
247 }

341// Congestion-negotiated routing: each iteration routes every flow by
342// Dijkstra, given each connection's demand, and raises the demand of the
343// connections used over capacity, until no connection is.
344class Pathfinder {
345public:
346 void initialize(int maxCol, int maxRow, const AIETargetModel &targetModel);
347 // Adds a flow from `src` to `dst`; failures to route it are reported at
348 // `loc` when given.
349 void addFlow(TileID srcCoords, Port srcPort, TileID dstCoords, Port dstPort,
350 std::optional<int> packetId, bool isPriorityFlow,
351 std::optional<mlir::Location> loc = std::nullopt);
352 void sortFlows();
353 // Reserves the connections `switchboxOp` already makes, so routing avoids
354 // them. Fails if it makes one the switchbox cannot.
355 mlir::LogicalResult addFixedConnection(SwitchboxOp switchboxOp);
356 // Keeps routes out of the switchbox at `coords`.
357 void excludeTile(TileID coords);
358 // A RoutingFailure if no legal routing is found in `maxIterations`.
359 llvm::Expected<Routing> findPaths(int maxIterations);
360 void setPacketConstraints(PacketConstraints c) { constraints = std::move(c); }
361 // Loosens the packet constraints by one step after findPaths found no
362 // routing; false once no step is left or no packet stream took part in the
363 // failure. Packet flows start sharing channels only within their
364 // destination group. The steps, each skipped if it changes nothing: share
365 // channels with any packet flow; cap the fan-out of tiles the routing check
366 // found out of packet rules or msels to packetFanoutCap channels per
367 // direction; route the second id of an unbranchable split apart, which
368 // restarts sharing and caps; then share and cap again; route the ids a
369 // source sends that a source it must keep apart from does not as a part of
370 // their own, which restarts sharing and caps; then share and cap again.
371 bool relax();
372
373private:
374 // A directed edge in the dense routing graph: from some node to node `dst`,
375 // realized by cell (i, j) of `sb`. `sb` points into `graph`, so demand reads
376 // always see the current iteration's weights.
377 struct Edge {
378 int dst;
379 SwitchboxConnect *sb;
380 int i;
381 int j;
382 };
383
384 // A `Port` is (bundle, channel) with no direction, so one dense node stands
385 // for both a switchbox port's input side and its output side. Dijkstra
386 // therefore searches over states, not nodes: a state is a node paired with
387 // the side of that port the stream is currently on. A stream enters a
388 // switchbox on an input port, crosses the crossbar once to an output port,
389 // and then rides the wire to the neighbour's input port -- so In only ever
390 // takes intra-switchbox edges and Out only ever takes inter-switchbox ones.
391 // Without the split, Dijkstra can chain two crossbar hops through one port
392 // and turn the stream around inside a switchbox; the settings it emits then
393 // dead-end and AIECreatePathFindFlows reports the flow as unroutable.
394 enum PortSide : int { In = 0, Out = 1 };
395 static int stateId(int nodeId, PortSide side) { return 2 * nodeId + side; }
396 static int stateNode(int state) { return state >> 1; }
397
398 // Build the dense integer node numbering and per-node adjacency from `graph`
399 // and `flows`. Topology is fixed across congestion iterations, so this runs
400 // once. A node's edges are in the order of the PathEndPoints they reach, so
401 // Dijkstra breaks ties the same way every run.
402 void buildRoutingGraph();
403
404 // The cost of taking `e`, as dijkstraShortestPaths weighs it.
405 double edgeWeight(const Edge &e, std::optional<int> packetId,
406 const llvm::BitVector *avoid,
407 const llvm::BitVector *avoidBranch) const;
408
409 // Dijkstra over the dense graph from the states in `seeds`, each starting at
410 // its cost in `seedCosts`. Fills `preds` (predecessor state id, or -1) and
411 // `predEdge` (the edge taken to reach each state). Reuses the scratch buffers
412 // below. Each edge costs its edgeWeight given `packetId`, `avoid` and, from a
413 // state in `branchAvoid`, the flows it maps to. States in `stops` are reached
414 // but not left, and a crossbar hop is taken only if `mayCross` allows it. The
415 // search ends once every state in `targets` is settled; only their `distance`
416 // and paths are final then.
417 void dijkstraShortestPaths(
418 llvm::ArrayRef<int> seeds, llvm::ArrayRef<double> seedCosts,
419 std::optional<int> packetId, const llvm::BitVector *avoid = nullptr,
420 const llvm::DenseMap<int, llvm::BitVector> *branchAvoid = nullptr,
421 const llvm::DenseSet<int> *stops = nullptr,
422 llvm::ArrayRef<int> targets = {},
423 llvm::function_ref<bool(int, int)> mayCross = nullptr);
424
425 struct RouteState;
426 struct TreeBuilder;
427 // Route `flow`'s tree around those routed before it this iteration. A
428 // RoutingFailure if it reaches no path to a destination.
429 llvm::Error routePart(RouteState &st, int flow);
430 // Steer the next iteration away from what the routing check faulted.
431 // Returns the illegal edges the faults count as.
432 int applyRoutingFaults(RouteState &st, const RoutingFaults &faults);
433 // A RoutingFailure saying why the last iteration's routing does not fit.
434 llvm::Error explainNoRouting(const RouteState &st) const;
435 bool hasRoom(const SwitchboxConnect &sb) const;
436 // The steps of relax.
437 bool shareAllChannels();
438 bool capCrowdedFanOut();
439 bool routeIdsApart();
440 bool splitSharedIds();
441
442 // Flows to be routed
443 std::vector<Flow> flows;
444 // The packet ids each source sends each destination.
445 std::map<std::pair<PathEndPoint, PathEndPoint>, llvm::SmallVector<int, 2>>
446 packetIdsTo;
447 // The packet ids each source sends, to any destination.
448 std::map<PathEndPoint, std::set<int>> idsBySource() const;
449 // The packet ids each source sends in prioritized flows.
450 std::map<PathEndPoint, std::set<int>> priorityIds;
451 // Where each source's flow to each destination was declared.
452 std::map<PathEndPoint, std::map<PathEndPoint, mlir::Location>> flowLocs;
453 std::optional<mlir::Location> flowLoc(const PathEndPoint &src,
454 const PathEndPoint *dst) const;
455 // The routing graph, by the tiles a SwitchboxConnect connects: a tile to
456 // itself for its switchbox, or to a neighbour for the wires between them.
457 std::map<std::pair<TileID, TileID>, SwitchboxConnect> graph;
458
459 // Dense routing graph (built once by buildRoutingGraph()).
460 bool graphBuilt = false;
461 std::map<PathEndPoint, int> nodeIds; // PathEndPoint -> dense id
462 std::vector<PathEndPoint> nodes; // dense id -> PathEndPoint
463 std::vector<std::vector<Edge>> adjacency; // dense id -> out-edges
464
465 // Dijkstra scratch, indexed by state id (2 * nodes.size()) and reused across
466 // calls.
467 std::vector<double> distance;
468 std::vector<uint64_t> indexInHeap;
469 std::vector<int8_t> colors;
470 std::vector<int> preds;
471 std::vector<Edge> predEdge;
472
473 int getOrAddNodeId(const PathEndPoint &pep);
474
475 PacketConstraints constraints;
476 // Why the routing check rejected the last routing it rejected.
477 std::string checkReason;
478 // Whether packet streams took part in the last failure: the routing check
479 // rejected a routing, or packet streams cross a link left overused.
480 bool packetsFailed = true;
481 bool shareChannels = false, idsApart = false, splitShared = false;
482 size_t relaxStep = 0;
483 llvm::DenseSet<TileID> crowdedTiles, cappedTiles;

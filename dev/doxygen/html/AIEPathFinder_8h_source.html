43// The connections from the source ports to the destination ports of a
44// switchbox, if `srcCoords == dstCoords`, or of the wires from one switchbox
45// to its neighbour at `dstCoords`.
46struct SwitchboxConnect {
47 // A connection from a source port to a destination port.
48 struct Cell {
49 bool available = false;
50 // weight of Dijkstra's shortest path
51 double demand = 0;
52 // iterations the connection was over capacity
53 int overCapacity = 0;
54 // circuit streams using the connection
55 int usedCapacity = 0;
56 // packet streams using the connection
57 int packetFlowCount = 0;
58 // only packet streams of this group may share the connection
59 int packetGroupId = -1;
60 // packet ids routed through the connection (and its crossbar row and
61 // column), by the flow routing each; it may be shared only among distinct
62 // ids.
63 llvm::SmallDenseMap<int, int, 2> packetIds;
64 // whether a prioritized flow uses the connection
65 bool isPriority = false;
66 // added to the demand, so routes keep off the connection where they can
67 double avoidance = 0;
68 };
69
70 SwitchboxConnect() = default;
71 SwitchboxConnect(TileID coords) : srcCoords(coords), dstCoords(coords) {}
72 SwitchboxConnect(TileID srcCoords, TileID dstCoords)
73 : srcCoords(srcCoords), dstCoords(dstCoords) {}
74
75 TileID srcCoords, dstCoords;
76 std::vector<Port> srcPorts;
77 std::vector<Port> dstPorts;
78 // The connection from srcPorts[i] to dstPorts[j] is cells[i * dstPorts.size()
79 // + j].
80 std::vector<Cell> cells;
81 // Units of dst ports tied to one arbiter (see planArbiters in
82 // AIECreatePathFindFlows.cpp); a unit's leader lists the packet flows
83 // (indices into the router's flows) leaving by any of its ports
84 llvm::IntEqClasses dstUnits;
85 std::vector<llvm::SmallVector<int, 2>> unitPacketFlows;
86 // source ports the design already gives packet rules, which circuit streams
87 // cannot enter
88 llvm::BitVector packetOnlySrc;
89 // dst ports packet streams may no longer take
90 llvm::BitVector circuitOnlyDst;
91
92 Cell &at(size_t i, size_t j) { return cells[i * dstPorts.size() + j]; }
93 const Cell &at(size_t i, size_t j) const {
94 return cells[i * dstPorts.size() + j];
95 }
96
97 // The index of `p` in srcPorts or dstPorts, or -1.
98 int srcIndex(Port p) const { return indexIn(srcPorts, p); }
99 int dstIndex(Port p) const { return indexIn(dstPorts, p); }
100
101 // Size the cells and the per-port state to srcPorts and dstPorts.
102 void resize() {
103 cells.assign(srcPorts.size() * dstPorts.size(), Cell());
104 packetOnlySrc = llvm::BitVector(srcPorts.size());
105 circuitOnlyDst = llvm::BitVector(dstPorts.size());
106 unitPacketFlows.resize(dstPorts.size());
107 resetUnits();
108 }
109
110 void resetUnits() {
111 dstUnits.clear();
112 dstUnits.grow(dstPorts.size());
113 for (auto &flows : unitPacketFlows)
114 flows.clear();
115 }
116
117 llvm::ArrayRef<int> unitFlows(int j) const {
118 return unitPacketFlows[dstUnits.findLeader(j)];
119 }
120
121 void addToUnit(int j, int flow) {
122 auto &flows = unitPacketFlows[dstUnits.findLeader(j)];
123 if (!llvm::is_contained(flows, flow))
124 flows.push_back(flow);
125 }
126
127 void joinUnits(int j, int k) {
128 j = dstUnits.findLeader(j);
129 k = dstUnits.findLeader(k);
130 if (j == k)
131 return;
132 int leader = dstUnits.join(j, k);
133 int other = leader == j ? k : j;
134 for (int flow : unitPacketFlows[other])
135 addToUnit(leader, flow);
136 unitPacketFlows[other].clear();
137 }

139 // update demand at the beginning of each dijkstraShortestPaths iteration
140 void updateDemand();
141
142 // Bumps an over-capacity cell's demand, all but ruling it out if a
143 // prioritized flow uses it, to keep prioritized flows' routes.
144 static void bumpDemand(Cell &c);
145
146private:
147 static int indexIn(llvm::ArrayRef<Port> ports, Port p) {
148 const Port *it = llvm::find(ports, p);
149 return it == ports.end() ? -1 : it - ports.begin();
150 }
151};

249// Packets `src` sends with ids `a` and `b` share a master port they leave

366// What packet flows are routed under.
367struct PacketConstraints {
368 // Routings the check rejects count as illegal; the connections it names
369 // are penalized like overused channels so later iterations avoid them, and
370 // the trees it splits branch where it says from then on. A routing it
371 // rejects as a DeadlockProneRouting is kept, and found if no better one is
372 // soon.
373 RoutingCheck check;
374 // Packet flows that conflict are steered off each other's master ports; see
375 // edgeWeight.
376 PacketConflict conflict;
377 PacketsApartAcross apartAcross;
378 // The prioritized packet flows from these sources take these trees
379 // instead of being routed.
380 PacketTrees pinned;
381 // The packet ids each pinned tree carries. A pinned source's other flows
382 // route with the rest of the design, prioritized or not.
383 std::map<PathEndPoint, std::set<int>> pinnedIds;
384 // Whether prioritized packet flows not pinned route first, keeping
385 // other flows off their channels; otherwise they route like the others.
386 bool prioritize = true;

474 // A directed edge in the dense routing graph: from some node to node `dst`,

484 // A `Port` is (bundle, channel) with no direction, so one dense node stands
485 // for both a switchbox port's input side and its output side. Dijkstra
486 // therefore searches over states, not nodes: a state is a node paired with
487 // the side of that port the stream is currently on. A stream enters a
488 // switchbox on an input port, crosses the crossbar once to an output port,
489 // and then rides the wire to the neighbour's input port -- so In only ever
490 // takes intra-switchbox edges and Out only ever takes inter-switchbox ones.
491 // Without the split, Dijkstra can chain two crossbar hops through one port
492 // and turn the stream around inside a switchbox; the settings it emits then
493 // dead-end and AIECreatePathFindFlows reports the flow as unroutable.
494 enum PortSide : int { In = 0, Out = 1 };
495 static int stateId(int nodeId, PortSide side) { return 2 * nodeId + side; }
496 static int stateNode(int state) { return state >> 1; }

510 // Dijkstra over the dense graph from the states in `seeds`, each starting at
511 // its cost in `seedCosts`. Fills `preds` (predecessor state id, or -1) and
512 // `predEdge` (the edge taken to reach each state). Reuses the scratch buffers
513 // below. Each edge costs its edgeWeight given `packetId`, `avoid` and, from a
514 // state in `branchAvoid`, the flows it maps to. States in `stops` are reached
515 // but not left, and a crossbar hop is taken only if `mayCross` allows it. The
516 // search ends once every state in `targets` is settled; only their `distance`
517 // and paths are final then.

522 const llvm::DenseSet<int> *stops = nullptr,

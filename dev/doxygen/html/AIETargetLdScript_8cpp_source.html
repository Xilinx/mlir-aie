20// Output the memorymap in gnu linker format for the given buffer operations,
21// with the given offset. The offset is different depending on where the buffers
22// are accessed from.
23static void writeLDScriptMap(raw_ostream &output, BufferOp buf, int offset) {
24 std::string bufName(buf.name().getValue());
25 int bufferBaseAddr = getBufferBaseAddress(buf);
26 int numBytes = buf.getAllocationSize();
27 output << ". = 0x" << llvm::utohexstr(offset + bufferBaseAddr) << ";\n";
28 output << bufName << " = .;\n";
29 output << ". += 0x" << llvm::utohexstr(numBytes) << ";\n";
30}

32///// ld.script format:
33//
34// MEMORY
35// {
36// program (RX) : ORIGIN = 0, LENGTH = 0x0020000
37// data (!RX) : ORIGIN = 0x20000, LENGTH = 0x0020000
38// }
39// ENTRY(__start)
40// INPUT(something.o)
41// SECTIONS
42// {
43// . = 0x0;
44// .text : {
45// // the __start symbol has to come at address zero.
46// *crt0.o(.text*)
47// __ctors_start__ = .;
48// __init_array_start = .;
49// KEEP(SORT(*)(.init_array))
50// __ctors_end__ = .;
51// __init_array_end = .;
52// __dtors_start__ = .;
53// __dtors_end__ = .;
54// *(.text*)
55// } > program
56// .data : { *(.data*) } > data
57// . = 0x20000;
58// _sp_start_value_DM_stack = .;
59// . = 0x24000;
60// a = .;
61// . += 1024;
62// .bss : { *(.bss*) } > data
63// }
64// PROVIDE(main = core_3_3);

66LogicalResult xilinx::AIE::AIETranslateToLdScript(ModuleOp module,
67 raw_ostream &output,
68 int tileCol, int tileRow,
69 llvm::StringRef deviceName,
70 bool probe) {
71 DenseMap<TileID, Operation *> tiles;
72 DenseMap<Operation *, SmallVector<BufferOp, 4>> buffers;
73
74 DeviceOp targetOp =
75 AIE::DeviceOp::getForSymbolInModuleOrError(module, deviceName);
76
77 if (!targetOp) {
78 return failure();
79 }
80
81 collectTiles(targetOp, tiles);
82 collectBuffers(targetOp, buffers);
83
84 for (auto tile : targetOp.getOps<TileOp>()) {
85 if (tile.colIndex() == tileCol && tile.rowIndex() == tileRow) {
86 TileID srcCoord = {tile.colIndex(), tile.rowIndex()};
87 // The "data" region below holds the sections the core compiler generates
88 // itself (.data, .rodata and .bss) and that nothing places explicitly. It
89 // is one contiguous region, so its size bounds what the core can link,
90 // and the total free memory on the tile does not.
91 const auto &targetModel = getTargetModel(tile);
92 MemoryRun stackRun;
93 if (auto core = tile.getCoreOp()) {
94 if (!probe && core.getStackBank() && !core.getStackAddress())
95 return core.emitOpError(
96 "stack_bank has no assigned stack_address; run "
97 "--aie-assign-buffer-addresses with bank-aware allocation");
98 stackRun = core.getStackRun();
99 }
100 MemoryRun dataRun =
101 probe ? MemoryRun{stackRun.end(),
102 targetModel.getLocalMemorySize() - stackRun.end()}
103 : coreDataRegion(tile, buffers[tiles[srcCoord]]);
104 // The checks below all ask whether placement is stale. A probe link runs
105 // before placement, so there is nothing yet to be stale.
106 if (!probe) {
107 if (dataRun.start < 0 ||
108 dataRun.end() > targetModel.getLocalMemorySize())
109 return tile.emitOpError(
110 "data region runs past this tile's local memory; the buffer "
111 "allocator's placement is stale. Re-run "
112 "--aie-assign-buffer-addresses");
113 if (dataRun.size > 0 && stackRun.size > 0 &&
114 dataRun.start < stackRun.end() && stackRun.start < dataRun.end())
115 return tile.emitOpError(
116 "data region overlaps the stack; the buffer allocator's "
117 "placement is stale. Re-run --aie-assign-buffer-addresses");
118
119 // A pass that adds or moves a buffer after the allocator ran leaves the
120 // region aliasing it, and the core compiler and that buffer would then
121 // write over each other. Report the alias here.
122 for (auto buf : buffers[tiles[srcCoord]]) {
123 if (buf.getCoreData() || buf.getAllocationSize() == 0 ||
124 dataRun.size == 0) {
125 continue;
126 }
127 int64_t bufStart = getBufferBaseAddress(buf);
128 int64_t bufEnd = bufStart + buf.getAllocationSize();
129 if (bufStart < dataRun.end() && dataRun.start < bufEnd) {
130 return tile.emitOpError("data region 0x")
131 << llvm::utohexstr(dataRun.start) << "-0x"
132 << llvm::utohexstr(dataRun.end() - 1) << " overlaps buffer '"
133 << buf.name().getValue() << "' at 0x"
134 << llvm::utohexstr(bufStart)
135 << "; the buffer allocator's placement is stale. Re-run "
136 "--aie-assign-buffer-addresses";
137 }
138 }
139 }
140
141 // Was hardcoded to 0x20000 -- eight times the real 0x4000 -- which let
142 // an overflowing core link cleanly and fail much later in aie-rt's ELF
143 // loader instead of here, at the linker, naming the section.
144 int origin =
145 targetModel.getMemInternalBaseAddress(srcCoord) + dataRun.start;
146 int length = dataRun.size;
147 bool reservedData =
148 !probe && llvm::any_of(buffers[tiles[srcCoord]], [](BufferOp buf) {
149 return buf.getCoreData();
150 });
151 llvm::SmallVector<MemoryRun> bankRuns;
152 if (probe) {
153 // A probe measures; it does not judge. Section sizes come from the
154 // objects and the garbage collection, never from how much room a region
155 // offers, so every region is given the whole tile and the regions are
156 // allowed to overlap. A design too big to link still probes cleanly,
157 // and the real link reports it with the diagnostic that belongs there.
158 int numBanks = targetModel.getNumBanks(srcCoord.col, srcCoord.row);
159 int64_t bankSize =
160 numBanks > 0 ? targetModel.getLocalMemorySize() / numBanks : 0;
161 for (int bank = 0; bank < numBanks; ++bank)
162 bankRuns.push_back(
163 MemoryRun{bankSize * bank, targetModel.getLocalMemorySize()});
164 } else {
165 bankRuns = coreBankRegions(tile, buffers[tiles[srcCoord]],
166 reservedData ? dataRun : MemoryRun{});
167 }
168 std::string dataOrigin = "0x" + llvm::utohexstr(origin);
169 std::string dataEnd = "0x" + llvm::utohexstr(origin + length);
170 if (!reservedData) {
171 // Bank sections have fixed regions. LLD resolves their sizes before
172 // assigning ordinary data, including BSS-only cores and empty banks.
173 for (auto [bank, run] : llvm::enumerate(bankRuns)) {
174 int64_t bankStart =
175 targetModel.getMemInternalBaseAddress(srcCoord) + run.start;
176 if (run.size == 0 || bankStart >= origin + length ||
177 bankStart + run.size <= origin) {
178 continue;
179 }
180 std::string sec = bankSectionName(bank);
181 std::string end = "(ADDR(" + sec + ") + SIZEOF(" + sec + "))";
182 dataOrigin = "MAX(" + dataOrigin + ", (SIZEOF(" + sec +
183 ") != 0 ? MIN(" + dataEnd + ", " + end + ") : 0))";
184 }
185 }
186 output << R"THESCRIPT(
187MEMORY
188{
189)THESCRIPT";
190 output << " program (RX) : ORIGIN = 0, LENGTH = 0x"
191 << llvm::utohexstr(targetModel.getProgramMemorySize()) << "\n";
192 output << " data (!RX) : ORIGIN = " << dataOrigin << ", LENGTH = ";
193 if (reservedData) {
194 output << "0x" << llvm::utohexstr(length);
195 } else {
196 output << dataEnd << " - " << dataOrigin;
197 }
198 output << "\n";
199 // One region per bank, for statics a kernel pins with a bank attribute.
200 // A bank with nothing spare gets a zero-length region rather than being
201 // left out, so a section aimed at it overflows by name instead of
202 // becoming an orphan.
203 for (auto [bank, run] : llvm::enumerate(bankRuns)) {
204 output << " " << bankRegionName(bank) << " (!RX) : ORIGIN = 0x"
205 << llvm::utohexstr(
206 targetModel.getMemInternalBaseAddress(srcCoord) +
207 run.start)
208 << ", LENGTH = 0x" << llvm::utohexstr(run.size) << "\n";
209 }
210 output << R"THESCRIPT(}
211ENTRY(__start)
212SECTIONS
213{
214 . = 0x0;
215 .text : {
216 /* the __start symbol has to come at address zero. */
217 *crt0.o(.text*)
218 _ctors_start = .;
219 _init_array_start = .;
220 KEEP(SORT(*.init_array))
221 _ctors_end = .;
222 _init_array_end = .;
223 _dtors_start = .;
224 _dtors_end = .;
225 *(.text*)
226 } > program
227)THESCRIPT";
228 // Ahead of .data and .bss: lld assigns an input section to the first
229 // description that matches it, and the wildcards below would otherwise
230 // take the bank-pinned ones.
231 for (auto [bank, run] : llvm::enumerate(bankRuns)) {
232 std::string sec = bankSectionName(bank);
233 output << " " << sec << " : { *(" << sec << " " << sec << ".*)"
234 << " *(*.DM_bank" << bankLetter(bank) << "*) } > "
235 << bankRegionName(bank) << "\n";
236 }
237 output << R"THESCRIPT( .data : {
238 *(.data*)
239 *(.rodata*)
240 } > data
241 .comment : {
242 *(.comment*)
243 }
244 .symtab : {
245 *(.symtab)
246 }
247 .shstrtab : {
248 *(.shstrtab)
249 }
250 .strtab : {
251 *(.strtab)
252 }
253 .stack_sizes : {
254 *(.stack_sizes)
255 }
256 /* Bitcode an object carries for aiecc's own analysis. It describes the core
257 rather than running on it, and --orphan-handling=error rejects any section
258 with no home, so name it here to drop it. */
259 /DISCARD/ : {
260 *(.llvmbc)
261 *(.llvmcmd)
262 }
263
264)THESCRIPT";
265 auto doBuffer = [&](std::optional<TileID> tile, int offset,
266 const std::string &dir) {
267 if (tile) {
268 // A probe has no addresses to write; --defsym stands the symbols up
269 // so the link resolves without claiming where anything lives.
270 if (tiles.count(*tile) && !probe) {
271 for (auto buf : buffers[tiles[*tile]]) {
272 if (!buf.getCoreData()) {
273 writeLDScriptMap(output, buf, offset);
274 }
275 }
276 }
277 } else {
278 output << "/* No tile with memory exists to the " << dir << ". */\n";
279 output << ". = 0x" << llvm::utohexstr(offset) << ";\n";
280 uint32_t localMemSize = targetModel.getLocalMemorySize();
281 output << ". += 0x" << llvm::utohexstr(localMemSize) << ";\n";
282 }
283 };
284
285 // Stack
286 if (auto core = tile.getCoreOp()) {
287 MemoryRun stackRun = core.getStackRun();
288 output << ". = 0x"
289 << llvm::utohexstr(
290 targetModel.getMemInternalBaseAddress(srcCoord) +
291 stackRun.start)
292 << ";\n";
293 output << "_sp_start_value_DM_stack = .;\n";
294 output << ". += 0x" << llvm::utohexstr(stackRun.size)
295 << "; /* stack */\n";
296 } else {
297 output << ". = 0x"
298 << llvm::utohexstr(
299 targetModel.getMemInternalBaseAddress(srcCoord))
300 << ";\n";
301 output << "_sp_start_value_DM_stack = .;\n";
302 output << "/* no stack allocated */\n";
303 }
304
305 doBuffer(targetModel.getMemSouth(srcCoord),
306 targetModel.getMemSouthBaseAddress(), std::string("south"));
307 doBuffer(targetModel.getMemWest(srcCoord),
308 targetModel.getMemWestBaseAddress(), std::string("west"));
309 doBuffer(targetModel.getMemNorth(srcCoord),
310 targetModel.getMemNorthBaseAddress(), std::string("north"));
311 doBuffer(targetModel.getMemEast(srcCoord),
312 targetModel.getMemEastBaseAddress(), std::string("east"));
313
314 output << " .bss : { *(.bss*) } > data\n";
315 // INPUT() directives must follow the closing brace of SECTIONS; placing
316 // them inside SECTIONS is invalid linker script syntax.
317 output << "}\n";
318 if (auto coreOp = tile.getCoreOp()) {
319 // `link_files` holds the ordinary final-link inputs (object files)
320 if (auto filesAttr = coreOp.getLinkFiles()) {
321 // Canonical path: link_files populated by aie-assign-core-link-files.
322 for (auto f : filesAttr->getAsRange<mlir::StringAttr>()) {
323 output << "INPUT(" << f.getValue() << ")\n";
324 }
325 } else if (auto fileAttr = coreOp.getLinkWith()) {
326 // Deprecated fallback: core-level link_with was not migrated by
327 // aie-assign-core-link-files (e.g., the pass was not run). It carries
328 // no mode, so it is always an ordinary link input.
329 output << "INPUT(" << fileAttr.value().str() << ")\n";
330 }
331
332 output << "PROVIDE(main = core_" << tile.getCol() << "_"
333 << tile.getRow() << ");\n";
334 }
335 }
336 }
337 return success();
338}

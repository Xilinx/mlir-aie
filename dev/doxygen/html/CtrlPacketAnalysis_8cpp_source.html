15SmallVector<PacketInterval> AIEX::packetIntervals(Block &block) {
16 SmallVector<PacketInterval> intervals;
17 PacketInterval current;
18 ArrayAttr multicasts;
19 auto finish = [&] {
20 if (!current.packets.empty()) {
21 intervals.push_back(current);
22 current.packets.clear();
23 }
24 };
25 for (Operation &operation : block) {
26 if (auto packet = dyn_cast<NpuControlPacketOp>(operation)) {
27 bool ordinary = packet.isOrdinaryWrite();
28 if (auto id = packet.getMulticastId()) {
29 auto group = multicasts && *id < multicasts.size()
30 ? dyn_cast<DictionaryAttr>(multicasts[*id])
31 : DictionaryAttr{};
32 auto tiles = group ? group.getAs<ArrayAttr>("tiles") : ArrayAttr{};
33 ordinary =
34 ordinary && tiles && !tiles.empty() &&
35 llvm::all_of(tiles, [&](Attribute attribute) {
36 auto tile = dyn_cast<AIE::TileAttr>(attribute);
37 return tile &&
38 llvm::all_of(
39 llvm::seq<size_t>(0, packet.getDataAttr().size()),
40 [&](size_t word) {
41 return AIE::getTargetModel(packet)
42 .isOrdinaryWriteAddress(
43 tile.getTileID(),
44 packet.getLocalAddress() + 4 * word);
45 });
46 });
47 }
48 if (ordinary != current.ordinary) {
49 finish();
50 current.ordinary = ordinary;
51 }
52 current.packets.push_back(packet);
53 continue;
54 }
55 finish();
56 auto barrier = dyn_cast<ControlPacketBarrierOp>(operation);
57 current.ordinary = false;
58 current.independent =
59 barrier && (barrier.getIndependent() ||
60 (barrier.getCompilerOnly() && current.independent));
61 if (!barrier || !barrier.getCompilerOnly())
62 ++current.delivery;
63 if (auto context = dyn_cast<SetDeviceOp>(operation)) {
64 current.device = context.getDeviceRefAttr();
65 auto module = context->getParentOfType<ModuleOp>();
66 auto device = module.lookupSymbol<AIE::DeviceOp>(current.device);
67 auto overlay =
68 device && device.getOverlayAttr()
69 ? module.lookupSymbol<AIE::DeviceOp>(device.getOverlayAttr())
70 : AIE::DeviceOp{};
71 multicasts =
72 overlay
73 ? AIE::AIEDialect::CtrlPktMulticastAttrHelper(module.getContext())
74 .getAttr(overlay)
75 : ArrayAttr{};
76 } else if (isa<NpuLoadPdiOp>(operation)) {
77 current.device = {};
78 multicasts = {};
79 }
80 }
81 finish();
82 return intervals;
83}

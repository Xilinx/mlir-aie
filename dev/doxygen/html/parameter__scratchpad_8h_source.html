51 /// Construct from an XRT run handle (C++ usage).
52 ParameterScratchpad(xrt::run &run, const std::string &paramsPath) {
53 parseParams(paramsPath);
54 scratchpadBo = run.get_ctrl_scratchpad_bo();
55 boMap = scratchpadBo.map<uint32_t *>();
56 if (scratchpadBo.size() < scratchpadSizeBytes)
57 throw std::runtime_error("ParameterScratchpad: BO size (" +
58 std::to_string(scratchpadBo.size()) +
59 ") < required scratchpad size (" +
60 std::to_string(scratchpadSizeBytes) + ")");
61 clear();
62 }

65 /// Construct from a raw buffer and its size in bytes (Python bindings /
66 /// testing). The size is required: clear() and every write index off
67 /// scratchpadSizeBytes, which parseParams derives from the params file, not
68 /// from the buffer.
69 ParameterScratchpad(uint32_t *buffer, size_t bufferSizeBytes,
70 const std::string &paramsPath)
71 : boMap(buffer) {
72 parseParams(paramsPath);
73 if (bufferSizeBytes < scratchpadSizeBytes)
74 throw std::runtime_error("ParameterScratchpad: buffer size (" +
75 std::to_string(bufferSizeBytes) +
76 ") < required scratchpad size (" +
77 std::to_string(scratchpadSizeBytes) + ")");
78 clear();
79 }

81 /// Write raw bytes (up to 4) by name, interpreted as a little-endian
82 /// uint32. For core-kind parameters, the bits are left-shifted by 2
83 /// (firmware requirement).
84 /// For addr-kind parameters, the value is written raw (no shift).
85 void writeBytes(const std::string &name, const void *data, size_t len) {
86 uint32_t bits = 0;
87 std::memcpy(&bits, data, std::min(len, sizeof(bits)));
88 writeBits(name, bits);
89 }
90
91 /// Write a raw 32-bit value by name. For core-kind parameters, the bits
92 /// are left-shifted by 2 (firmware requirement). For addr-kind parameters,
93 /// the value is written directly (no shift). A DMA offset or length
94 /// parameter's value, read as an int32, must lie in the range the compiler
95 /// gave it, which keeps its transfers within their buffers.
96 void writeBits(const std::string &name, uint32_t bits) {
97 auto it = paramMap.find(name);
98 if (it == paramMap.end()) {
99 throw std::runtime_error("ParameterScratchpad: unknown parameter '" +
100 name + "'");
101 }
102 if (auto range = ranges.find(name); range != ranges.end()) {
103 int32_t value = static_cast<int32_t>(bits);
104 if (value < range->second.first || value > range->second.second)
105 throw std::invalid_argument(
106 "ParameterScratchpad: value " + std::to_string(value) + " of '" +
107 name + "' is outside [" + std::to_string(range->second.first) +
108 ", " + std::to_string(range->second.second) +
109 "], which keeps its DMA transfers within their buffers");
110 values[name] = value;
111 }
112 uint8_t idx = it->second;
113 uint32_t encoded = bits;
114 if (coreParams.count(name)) {
115 // core parameters require shift-2 to survive masking of lowest bits by
116 // firmware op
117 encoded = bits << 2;
118 }
119 boMap[idx] = encoded;
120 }
121
122 /// Write a typed parameter value. For core-kind parameters, the raw bits
123 /// are left-shifted by 2 as required by the firmware's UPDATE_REG Incr mode,
124 /// and the core right-shifts (unsigned) by 2 after reading. The round trip
125 /// therefore zeroes the top 2 bits of the value (the bottom 2 bits are
126 /// preserved). Types that fit in 30 bits (e.g. uint16_t, int16_t,
127 /// std::bfloat16_t, and uint32_t values < 2^30) round-trip losslessly.
128 /// `float` is not supported as a core-kind parameter (the verifier in
129 /// `--aie-lower-scratchpad-parameters` rejects it).
130 template <typename T>
131 void write(const std::string &name, T value) {
132 static_assert(sizeof(T) <= 4, "Parameter values must be at most 32 bits");
133 uint32_t bits = 0;
134 std::memcpy(&bits, &value, sizeof(T));
135 writeBits(name, bits);
136 }
137
138 /// Check the written values of each offset and length parameter pair that
139 /// one DMA transfer uses. Throws std::invalid_argument if they take the
140 /// transfer past the end of its buffer.
141 void validate() const {
142 for (const JointBound &bound : jointBounds) {
143 int64_t offset = values.at(bound.offset);
144 int64_t length = values.at(bound.length);
145 int64_t end = offset + bound.lengthStep * length;
146 if (end > bound.max)
147 throw std::invalid_argument(
148 "ParameterScratchpad: '" + bound.offset +
149 "' = " + std::to_string(offset) + " plus " +
150 std::to_string(bound.lengthStep) + " * '" + bound.length +
151 "' = " + std::to_string(length) + " is " + std::to_string(end) +
152 ", above " + std::to_string(bound.max) +
153 ", so a DMA transfer using both runs past its buffer");
154 }
155 }

158 /// Validate, then sync the scratchpad buffer to device. Call after all writes
159 /// for this run.

161 validate();
162 scratchpadBo.sync(XCL_BO_SYNC_BO_TO_DEVICE);
163 }

166 /// Read back a parameter's current encoded value (for debugging).
167 uint32_t read(const std::string &name) const {
168 auto it = paramMap.find(name);
169 if (it == paramMap.end()) {
170 throw std::runtime_error("ParameterScratchpad: unknown parameter '" +
171 name + "'");
172 }
173 return boMap[it->second];
174 }

176private:
177#if TEST_UTILS_USE_XRT
178 xrt::bo scratchpadBo;

180 uint32_t *boMap = nullptr;
181 size_t scratchpadSizeBytes = 0;
182 std::unordered_map<std::string, uint8_t> paramMap;
183 std::unordered_set<std::string> coreParams; // params with kind="core"
184 std::unordered_map<std::string, std::pair<int32_t, int32_t>> ranges;
185 std::unordered_map<std::string, int32_t> values; // of the ranged params
186 struct JointBound {
187 std::string length, offset;
188 int64_t lengthStep, max;
189 };
190 std::vector<JointBound> jointBounds;
191
192 void clear() {
193 for (size_t i = 0; i < scratchpadSizeBytes / 4; i++) {
194 boMap[i] = 0;
195 }
196 }

198 void parseParams(const std::string &path) {
199 std::ifstream file(path);
200 if (!file.is_open()) {
201 throw std::runtime_error("ParameterScratchpad: cannot open '" + path +
202 "'");
203 }
204
205 // Format:
206 // <num_parameters>
207 // <name> <state_table_idx> <type> <kind> <min> <max>
208 // ...
209 // <num_joint_bounds>
210 // <length> <offset> <length_step> <max>
211 // ...
212 // where kind is "core" or "addr", and <min> <max> is "- -" for a
213 // parameter with no range. A joint bound requires
214 // offset + length_step * length <= max.
215 unsigned numParams = 0;
216 file >> numParams;
217 scratchpadSizeBytes = numParams * 4;
218
219 for (unsigned i = 0; i < numParams; i++) {
220 std::string name, type, kind, min, max;
221 unsigned idx;
222 file >> name >> idx >> type >> kind >> min >> max;
223 if (!file)
224 throw std::runtime_error("ParameterScratchpad: malformed entry " +
225 std::to_string(i) + " in '" + path + "'");
226 if (idx > 255)
227 throw std::runtime_error("ParameterScratchpad: state_table_idx " +
228 std::to_string(idx) + " for '" + name +
229 "' exceeds uint8_t range");
230 if (paramMap.count(name))
231 throw std::runtime_error(
232 "ParameterScratchpad: duplicate parameter name '" + name + "'");
233 paramMap[name] = static_cast<uint8_t>(idx);
234 if (kind != "core" && kind != "addr") {
235 throw std::runtime_error("ParameterScratchpad: invalid kind '" + kind +
236 "' for parameter '" + name + "'");
237 } else if (kind == "core") {
238 coreParams.insert(name);
239 }
240 if (min != "-") {
241 try {
242 ranges[name] = {std::stoi(min), std::stoi(max)};
243 } catch (const std::logic_error &) {
244 throw std::runtime_error("ParameterScratchpad: malformed range '" +
245 min + " " + max + "' for parameter '" +
246 name + "' in '" + path + "'");
247 }
248 values[name] = 0;
249 }
250 }
251
252 unsigned numJointBounds = 0;
253 file >> numJointBounds;
254 if (!file)
255 throw std::runtime_error(
256 "ParameterScratchpad: missing joint bound count in '" + path + "'");
257 for (unsigned i = 0; i < numJointBounds; i++) {
258 JointBound bound;
259 file >> bound.length >> bound.offset >> bound.lengthStep >> bound.max;
260 if (!file)
261 throw std::runtime_error("ParameterScratchpad: malformed joint bound " +
262 std::to_string(i) + " in '" + path + "'");
263 for (const std::string &name : {bound.length, bound.offset})
264 if (!ranges.count(name))
265 throw std::runtime_error("ParameterScratchpad: joint bound " +
266 std::to_string(i) + " names '" + name +
267 "', which has no range, in '" + path + "'");
268 jointBounds.push_back(bound);
269 }
270 }
271};

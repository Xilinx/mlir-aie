18/// Which DMA channels of each tile are already spoken for, so that a channel is
19/// handed out at most once across everything that programs one.
20class DMAChannelAnalysis {
21 /// A channel or stream port is either spoken for or free, so membership is
22 /// the whole state.
23 mlir::DenseSet<std::tuple<mlir::Value, DMAChannelDir, int>> usedChannels;
24 mlir::DenseSet<std::tuple<mlir::Value, DMAChannelDir, int>> usedStreams;
25
26public:
27 DMAChannelAnalysis(DeviceOp &device);
28
29 /// Next free channel of `tile` in `dir`, or -1 when the tile has none left.
30 /// A channel reaching an adjacent MemTile's memory must come from the lower
31 /// half of the range, which only a placed tile can bound.
32 int getDMAChannelIndex(TileLike tile, DMAChannelDir dir,
33 bool requiresAdjacentTileAccessChannels);
34
35 /// Claim `channel` for (`tile`, `dir`) so first-free assignment cannot take
36 /// it. Returns the channel, or -1 when it is out of range or already
37 /// claimed; the caller reports, since it knows which endpoint asked.
38 int reservePinnedChannel(TileLike tile, DMAChannelDir dir, int channel);
39
40 /// Claim a raw stream port, reporting on `tile` when it is already taken.
41 void checkAIEStreamIndex(TileLike tile, DMAChannel chan);
42};

18/// Which DMA channels of each tile are already spoken for, so that a channel is
19/// handed out at most once across everything that programs one.
20class DMAChannelAnalysis {
21 /// Fully resolved tile aliases name the same hardware resources. Unresolved
22 /// tiles retain their SSA identity until placement establishes co-location.
23 mlir::DenseMap<std::pair<int, int>, mlir::Value> tilesByCoordinate;
24 mlir::Value getTileKey(mlir::Value tile);
25
26 /// Keep the reserving operation so diagnostics retain its MLIR location.
27 mlir::DenseMap<std::tuple<mlir::Value, DMAChannelDir, int>, mlir::Operation *>
28 usedChannels;
29 /// DMA channels an explicit flow routes. First-free assignment skips them;
30 /// a pinned request may still claim one.
31 mlir::DenseMap<std::tuple<mlir::Value, DMAChannelDir, int>, mlir::Operation *>
32 streamedChannels;
33 /// The first claimant of a core stream port, and whether only packets use
34 /// it; circuit occupancy is exclusive.
35 struct StreamClaim {
36 mlir::Operation *owner;
37 bool packet;
38 };
39 mlir::DenseMap<std::tuple<mlir::Value, DMAChannelDir, int>, StreamClaim>
40 usedStreams;
41
42public:
43 DMAChannelAnalysis(DeviceOp &device);
44
45 /// Exclusive upper bound on channel indices eligible for this transfer.
46 static int getDMAChannelLimit(TileLike tile, DMAChannelDir dir,
47 bool requiresAdjacentTileAccessChannels);
48
49 /// Next free channel of `tile` in `dir`, or -1 when the tile has none left.
50 /// A channel an explicit flow routes is not free.
51 /// A channel reaching an adjacent MemTile's memory must come from the
52 /// target's restricted range at every compatible physical position.
53 int getDMAChannelIndex(TileLike tile, DMAChannelDir dir,
54 bool requiresAdjacentTileAccessChannels,
55 mlir::Operation *owner = nullptr);
56
57 /// Whether first-free assignment could hand out `channel`: nothing reserves
58 /// it and no explicit flow routes it.
59 bool isChannelFree(TileLike tile, DMAChannelDir dir, int channel);
60
61 /// Claim `channel` for (`tile`, `dir`) so first-free assignment cannot take
62 /// it. Returns the channel, or -1 when it is out of range or already
63 /// claimed; the caller reports, since it knows which endpoint asked.
64 int reservePinnedChannel(TileLike tile, DMAChannelDir dir, int channel,
65 mlir::Operation *owner = nullptr);
66
67 /// Operation reserving this channel, or else the flow routing it; null if
68 /// neither.
69 mlir::Operation *getDMAChannelOwner(TileLike tile, DMAChannelDir dir,
70 int channel);
71
72 /// Claim a raw stream port for `user`, reporting on it when the port is
73 /// taken; packet users may share it with other packets.
74 mlir::LogicalResult checkAIEStreamIndex(TileLike tile, DMAChannel chan,
75 mlir::Operation *user,
76 bool packet = false);
77};

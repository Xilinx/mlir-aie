25/// Which DMA channels of each tile are already spoken for, so that a channel is
26/// handed out at most once across everything that programs one.
27class DMAChannelAnalysis {
28 /// Fully resolved tile aliases name the same hardware resources. Unresolved

33 /// Keep the reserving operation so diagnostics retain its MLIR location.

35 usedChannels;

37 /// a pinned request may still claim one.

39 streamedChannels;
40 /// Channels whose claimants all share them. Several packet-switched sources
41 /// coexist on one; a circuit claim is exclusive.

44 shareableChannels;

109 /// A DMA end of one tile: its direction, the BD ids it takes, the
110 /// exclusive upper bound on the channels it may take, and the channel it
111 /// pins, if any, which every end pinning it shares.
112 struct BDPlanEnd {
113 BDPlanEnd(DMAChannelDir dir, int64_t bds, int limit,
114 std::optional<int> pinned)
115 : dir(dir), bds(bds), limit(limit), pinned(pinned) {}
116 DMAChannelDir dir;
117 int64_t bds;
118 int limit;
119 std::optional<int> pinned;
120 };

122 /// Channels among `0..numChannels-1` at (`col`, `row`), grouped by whichever
123 /// BD ids any two in a group share, transitively.
124 struct BDGroup {
125 llvm::BitVector ids;
126 llvm::SmallVector<int> channels;
127 };
128 static llvm::SmallVector<BDGroup> bdGroups(const AIETargetModel &target,
129 int col, int row, int numChannels);

131 /// A channel of the tile at (`col`, `row`) for each of `ends`: its pinned
132 /// one, else one `free` leaves that no other end of its direction takes;
133 /// and, for every set of channels sharing BD ids, the ends on them plus the
134 /// ids `spent` on each channel within those ids fit those ids. Nothing when
135 /// no assignment fits.

137 planBDChannels(const AIETargetModel &target, int col, int row,

148 /// A pool whose ends a BD plan weighs: how many objects it holds, the bytes
149 /// of one, and whether it can be walked whole, its ends then taking one BD
150 /// id each.
151 struct BDWalkPool {
152 int64_t depth;
153 int64_t objectBytes;
154 bool walkable;
155 };
156 /// walkSetsThatFit's result; `channelShort` when the ends do not fit the
157 /// tile's channels even taking no BD ids, `fits` false when no set does.
158 struct BDWalkSets {
159 llvm::SmallVector<llvm::SmallVector<unsigned>> sets;
160 /// The bytes each set walks whole.
161 llvm::SmallVector<int64_t> bytes;
162 bool channelShort = false;
163 bool fits = true;
164 };
165
166 /// The walkable pool not yet `whole` to walk next: the deepest, then the one
167 /// with the smallest objects, then the one whose ends take the most BD ids;
168 /// `pools` order decides only between pools alike in all three. Nothing
169 /// when none is left.
170 static std::optional<unsigned>
171 nextPoolToWalk(llvm::ArrayRef<BDPlanEnd> ends,
172 llvm::ArrayRef<unsigned> poolOf,
173 llvm::ArrayRef<BDWalkPool> pools, llvm::ArrayRef<bool> whole);
174
175 /// Every set of walkable `pools` whose walk lets planBDChannels fit `ends`
176 /// (`poolOf` gives each end's pool), ordered by the bytes the set walks
177 /// whole, then its size, then its members in nextPoolToWalk's order; at
178 /// most `maxSets` of them, cheapest first (unbounded by default). Empty
179 /// when the ends fit unwalked. At most 12 pools are walkable: each draws a
180 /// channel of its own, and more ends than channels make `channelShort`.
181 /// aie-objectfifo-allocate and the SA placer both walk by this; the SA only
182 /// ever takes the first set, so it should pass `maxSets=1`.
183 static BDWalkSets
184 walkSetsThatFit(const AIETargetModel &target, int col, int row,
185 llvm::ArrayRef<BDPlanEnd> ends,
186 llvm::ArrayRef<unsigned> poolOf,
187 llvm::ArrayRef<BDWalkPool> pools,
188 llvm::function_ref<bool(DMAChannelDir, int)> free,
189 llvm::function_ref<int64_t(int)> spent,
190 unsigned maxSets = std::numeric_limits<unsigned>::max());
191
192 /// One walkSetsThatFit set per mem tile of a spill component
193 /// (memTileSpillComponents), by index; `placed` when its buffers place,
194 /// `stopped` when the search gave up at `maxTries` tries.
195 struct WalkChoice {
196 llvm::SmallVector<unsigned> sets;
197 bool placed = false;
198 bool stopped = false;
199 };
200 /// Tries the choices of one set per tile, `bytes[tile]` giving each of its
201 /// sets' bytes walked whole, in order of total bytes walked, then the
202 /// tiles' set indices in turn, until `places` (the caller's buffer search
203 /// over the component) accepts one; the first choice when none does.
204 static WalkChoice
205 searchWalkProduct(llvm::ArrayRef<llvm::SmallVector<int64_t>> bytes,
206 llvm::function_ref<bool(llvm::ArrayRef<unsigned>)> places,
207 unsigned maxTries);
208 static constexpr unsigned kMaxWalkPlacements = 1024;

238 };

42/// Returns the aie.buffer that `v` is, or that `v` reshapes as a whole, or

44static AIE::BufferOp rootBuffer(Value v) {
45 while (Operation *d = v.getDefiningOp()) {
46 if (auto b = dyn_cast<AIE::BufferOp>(d))

48 if (!isa<memref::CollapseShapeOp, memref::ExpandShapeOp, memref::CastOp>(d))

61/// Returns a known divisor of the affine expression `e`, whose dims and
62/// symbols are `operands`.
63static int64_t exprDivisor(AffineExpr e, ValueRange operands, unsigned nDims,
64 int depth) {
65 if (auto c = dyn_cast<AffineConstantExpr>(e))
66 return c.getValue() ? std::abs(c.getValue()) : kMaxDiv;
67 if (auto d = dyn_cast<AffineDimExpr>(e))
68 return knownDivisor(operands[d.getPosition()], depth + 1);
69 if (auto s = dyn_cast<AffineSymbolExpr>(e))
70 return knownDivisor(operands[nDims + s.getPosition()], depth + 1);
71 auto bin = cast<AffineBinaryOpExpr>(e);
72 int64_t l = exprDivisor(bin.getLHS(), operands, nDims, depth);
73 int64_t r = exprDivisor(bin.getRHS(), operands, nDims, depth);
74 switch (bin.getKind()) {
75 case AffineExprKind::Add:
76 return std::gcd(l, r);
77 case AffineExprKind::Mul:
78 return mulCapped(l, r);
79 case AffineExprKind::Mod:
80 // x mod c = x - c * (x floordiv c)
81 if (auto c = dyn_cast<AffineConstantExpr>(bin.getRHS()))
82 return std::gcd(l, std::abs(c.getValue()));
83 return 1;
84 default:
85 return 1;
86 }
87}

89/// Returns a divisor of the index `v` that follows from how it is computed,
90/// or 1.
91static int64_t knownDivisor(Value v, int depth) {
92 if (depth > 16)
93 return 1;
94 if (std::optional<int64_t> c = getConstantIntValue(v))
95 return *c ? std::abs(*c) : kMaxDiv;
96 if (auto arg = dyn_cast<BlockArgument>(v)) {
97 auto loop = dyn_cast<scf::ForOp>(arg.getOwner()->getParentOp());
98 if (!loop)
99 return 1;
100 if (arg == loop.getInductionVar())
101 return std::gcd(knownDivisor(loop.getLowerBound(), depth + 1),
102 knownDivisor(loop.getStep(), depth + 1));
103 // An iter_arg incremented by the same value every iteration, such as a
104 // pointer the loop advances.
105 OpOperand *init = loop.getTiedLoopInit(arg);
106 Value next = loop.getTiedLoopYieldedValue(arg)->get();
107 auto add = next.getDefiningOp<arith::AddIOp>();
108 if (!init || !add || (add.getLhs() != arg && add.getRhs() != arg))
109 return 1;
110 Value step = add.getLhs() == arg ? add.getRhs() : add.getLhs();
111 return std::gcd(knownDivisor(init->get(), depth + 1),
112 knownDivisor(step, depth + 1));
113 }
114 Operation *d = v.getDefiningOp();
115 if (isa<arith::AddIOp, arith::SubIOp>(d))
116 return std::gcd(knownDivisor(d->getOperand(0), depth + 1),
117 knownDivisor(d->getOperand(1), depth + 1));
118 if (isa<arith::MulIOp>(d))
119 return mulCapped(knownDivisor(d->getOperand(0), depth + 1),
120 knownDivisor(d->getOperand(1), depth + 1));
121 if (isa<arith::IndexCastOp, arith::IndexCastUIOp>(d))
122 return knownDivisor(d->getOperand(0), depth + 1);
123 if (auto ap = dyn_cast<affine::AffineApplyOp>(d))
124 return exprDivisor(ap.getAffineMap().getResult(0), ap.getMapOperands(),
125 ap.getAffineMap().getNumDims(), depth);
126 return 1;
127}

147/// Returns the read that `conv` converts if it reads a contiguous,
148/// tile-aligned 8x8 tile of an identity-layout aie.buffer.
149static std::optional<TileRead> getTileRead(ToBFP16Op conv) {
150 VectorType srcTy = conv.getSource().getType();
151 if (srcTy.getNumElements() != kTile)
152 return std::nullopt;
153 Value src = conv.getSource();
154 while (auto sc = src.getDefiningOp<vector::ShapeCastOp>())
155 src = sc.getSource();
156 auto read = src.getDefiningOp<vector::TransferReadOp>();
157 if (!read || read.getMask() || !read.getPermutationMap().isMinorIdentity())
158 return std::nullopt;
159 if (llvm::any_of(read.getInBoundsValues(), [](bool b) { return !b; }))
160 return std::nullopt;
161 auto memTy = dyn_cast<MemRefType>(read.getBase().getType());
162 VectorType vecTy = read.getVectorType();
163 if (!memTy || !memTy.hasStaticShape() || !memTy.getLayout().isIdentity() ||
164 vecTy.getNumElements() != kTile)
165 return std::nullopt;
166 // Contiguous: all vector dims but the outermost cover the corresponding
167 // memref dims entirely.
168 int64_t vr = vecTy.getRank(), mr = memTy.getRank();
169 if (vr > mr)
170 return std::nullopt;
171 for (int64_t i = 1; i < vr; ++i)
172 if (vecTy.getDimSize(i) != memTy.getDimSize(mr - vr + i))
173 return std::nullopt;
174 AIE::BufferOp buf = rootBuffer(read.getBase());
175 if (!buf)
176 return std::nullopt;
177 auto bufTy = cast<MemRefType>(buf.getType());
178 if (!bufTy.getLayout().isIdentity() || !bufTy.getElementType().isBF16() ||
179 bufTy.getNumElements() % kTile)
180 return std::nullopt;
181 // Tile-aligned: the element offset is a multiple of the tile size.
182 ArrayRef<int64_t> shape = memTy.getShape();
183 int64_t stride = 1, div = kMaxDiv;
184 for (int64_t i = mr - 1; i >= 0; --i) {
185 div = std::gcd(div, mulCapped(knownDivisor(read.getIndices()[i]), stride));
186 stride *= shape[i];
187 }
188 if (div % kTile)
189 return std::nullopt;
190 return TileRead{read, buf};
191}

193/// Builds the element offset of `read` in its identity-layout memref.
194static Value linearOffset(OpBuilder &b, Location loc,
195 vector::TransferReadOp read) {
196 auto memTy = cast<MemRefType>(read.getBase().getType());
197 Value off = arith::ConstantIndexOp::create(b, loc, 0);
198 int64_t stride = 1;
199 for (int64_t i = memTy.getRank() - 1; i >= 0; --i) {
200 Value s = arith::ConstantIndexOp::create(b, loc, stride);
201 off = arith::AddIOp::create(
202 b, loc, off,
203 arith::MulIOp::create(b, loc, read.getIndices()[i], s).getResult());
204 stride *= memTy.getDimSize(i);
205 }
206 return off;
207}

209/// The locks that guard a buffer: for each DMA buffer descriptor that moves
210/// it, the lock acquired last before it and the lock released first after
211/// it, along the descriptor chain (blocks joined by aie.next_bd). If another
212/// core uses the buffer, any lock may guard it.
213struct BufferLocks {
214 DenseSet<Value> locks;
215 bool any = false;
216
217 bool contains(Value lock) const { return any || locks.contains(lock); }
218};
219
220static BufferLocks bufferLocks(AIE::BufferOp buf, Operation *core) {
221 BufferLocks bl;
222 SmallVector<Value> work = {buf.getBuffer()};
223 while (!work.empty()) {
224 Value v = work.pop_back_val();
225 for (Operation *user : v.getUsers()) {
226 if (isa<ViewLikeOpInterface>(user))
227 work.push_back(user->getResult(0));
228 else if (auto c = user->getParentOfType<AIE::CoreOp>(); c && c != core)
229 bl.any = true;
230 }
231 }
232 auto dev = buf->getParentOfType<AIE::DeviceOp>();
233 if (!dev)
234 return bl;
235 // The chain's next and previous descriptor blocks.
236 auto next = [](Block *b) -> Block * {
237 auto n = dyn_cast<AIE::NextBDOp>(b->getTerminator());
238 return n ? n.getDest() : nullptr;
239 };
240 auto prev = [&](Block *b) -> Block * {
241 for (Block *p : b->getPredecessors())
242 if (next(p) == b)
243 return p;
244 return nullptr;
245 };
246 // The first lock op that `want` accepts from `bd` on, backward or forward
247 // along the chain.
248 auto scan = [&](AIE::DMABDOp bd, bool forward,
249 function_ref<bool(AIE::UseLockOp)> want) -> Value {
250 Block *blk = bd->getBlock();
251 Operation *op = forward ? bd->getNextNode() : bd->getPrevNode();
252 for (unsigned n = 0; blk && n < 64; ++n) {
253 for (; op; op = forward ? op->getNextNode() : op->getPrevNode())
254 if (auto ul = dyn_cast<AIE::UseLockOp>(op); ul && want(ul))
255 return ul.getLock();
256 blk = forward ? next(blk) : prev(blk);
257 if (blk)
258 op = forward ? &blk->front() : &blk->back();
259 }
260 return Value();
261 };
262 dev.walk([&](AIE::DMABDOp bd) {
263 if (rootBuffer(bd.getBuffer()) != buf)
264 return;
265 for (Value lock : {scan(bd, /*forward=*/false,
266 [](AIE::UseLockOp ul) {
267 return ul.acquire() || ul.acquireGE();
268 }),
269 scan(bd, /*forward=*/true,
270 [](AIE::UseLockOp ul) { return ul.release(); })})
271 if (lock)
272 bl.locks.insert(lock);
273 });
274 return bl;
275}

277/// MLIR's local alias analysis, which also knows that aie.buffer ops (which
278/// declare no allocation effect) are distinct memory: from each other and
279/// from allocations.
280class BufferAliasAnalysis : public LocalAliasAnalysis {
281protected:
282 AliasResult aliasImpl(Value lhs, Value rhs) override {
283 AliasResult r = LocalAliasAnalysis::aliasImpl(lhs, rhs);
284 if (!r.isMay())
285 return r;
286 auto root = [](Value v) {
287 while (auto view = v.getDefiningOp<ViewLikeOpInterface>())
288 v = view.getViewSource();
289 return v;
290 };
291 Operation *a = root(lhs).getDefiningOp(), *b = root(rhs).getDefiningOp();
292 auto isBuffer = [](Operation *op) {
293 return isa_and_nonnull<AIE::BufferOp>(op);
294 };
295 auto isMemory = [](Operation *op) {
296 return isa_and_nonnull<AIE::BufferOp, memref::AllocOp, memref::AllocaOp,
297 memref::GetGlobalOp>(op);
298 };
299 if (a != b && (isBuffer(a) || isBuffer(b)) && isMemory(a) && isMemory(b))
300 return AliasResult::NoAlias;
301 return r;
302 }
303};

313/// Returns true if `op` may modify `buf`: it writes memory that may alias it,
314/// has unknown effects, or acquires a lock of a DMA that fills `buf`.
315static bool mayChange(Operation *op, Value mem, const BufferLocks &locks) {
316 if (auto ul = dyn_cast<AIE::UseLockOp>(op))
317 return (ul.acquire() || ul.acquireGE()) && locks.contains(ul.getLock());
318 return mayWrite(op, mem);

332static int64_t runsPerExecution(Operation *op, scf::ForOp loop) {
333 int64_t n = 1;

335 l = l->getParentOfType<scf::ForOp>()) {

362/// Returns the index `v` as an affine expression whose symbols are the values
363/// in `syms`. Values that are not affine in others become new symbols.
364static AffineExpr toExpr(Value v, SmallVectorImpl<Value> &syms, int depth = 0) {
365 MLIRContext *ctx = v.getContext();
366 if (std::optional<int64_t> c = getConstantIntValue(v))
367 return getAffineConstantExpr(*c, ctx);
368 if (depth < 12) {
369 if (auto ap = v.getDefiningOp<affine::AffineApplyOp>()) {
370 AffineMap m = ap.getAffineMap();
371 ValueRange ops = ap.getMapOperands();
372 SmallVector<AffineExpr> dims, symbols;
373 for (unsigned i = 0; i < m.getNumDims(); ++i)
374 dims.push_back(toExpr(ops[i], syms, depth + 1));
375 for (unsigned i = 0; i < m.getNumSymbols(); ++i)
376 symbols.push_back(toExpr(ops[m.getNumDims() + i], syms, depth + 1));
377 return m.getResult(0).replaceDimsAndSymbols(dims, symbols);
378 }
379 if (auto add = v.getDefiningOp<arith::AddIOp>())
380 return toExpr(add.getLhs(), syms, depth + 1) +
381 toExpr(add.getRhs(), syms, depth + 1);
382 if (auto mul = v.getDefiningOp<arith::MulIOp>()) {
383 if (std::optional<int64_t> c = getConstantIntValue(mul.getRhs()))
384 return toExpr(mul.getLhs(), syms, depth + 1) * *c;
385 if (std::optional<int64_t> c = getConstantIntValue(mul.getLhs()))
386 return toExpr(mul.getRhs(), syms, depth + 1) * *c;
387 }
388 }
389 auto it = llvm::find(syms, v);
390 if (it == syms.end()) {
391 syms.push_back(v);
392 it = std::prev(syms.end());
393 }
394 return getAffineSymbolExpr(it - syms.begin(), ctx);
395}

397/// Splits `v` into a value plus a constant.
398static std::pair<Value, int64_t> splitConstant(Value v) {
399 int64_t c = 0;
400 while (auto add = v.getDefiningOp<arith::AddIOp>()) {
401 if (std::optional<int64_t> x = getConstantIntValue(add.getRhs())) {
402 c += *x;
403 v = add.getLhs();
404 } else if (std::optional<int64_t> x = getConstantIntValue(add.getLhs())) {
405 c += *x;
406 v = add.getRhs();
407 } else {
408 break;
409 }
410 }
411 return {v, c};
412}

414/// Evaluates `exprs` for every combination of values of `syms`, which must all
415/// be induction variables of loops with static bounds. Returns nullopt if
416/// that is not the case or if there are too many combinations.
417static std::optional<SmallVector<SmallVector<int64_t>>>
418valuesOver(ArrayRef<AffineExpr> exprs, ArrayRef<Value> syms) {
419 SmallVector<SmallVector<int64_t>> ranges;
420 int64_t total = 1;
421 for (Value v : syms) {
422 auto arg = dyn_cast<BlockArgument>(v);
423 auto loop =
424 arg ? dyn_cast<scf::ForOp>(arg.getOwner()->getParentOp()) : nullptr;
425 if (!loop || loop.getInductionVar() != arg)
426 return std::nullopt;
427 std::optional<int64_t> lb = getConstantIntValue(loop.getLowerBound());
428 std::optional<int64_t> st = getConstantIntValue(loop.getStep());
429 std::optional<int64_t> n = staticTrips(loop);
430 if (!lb || !st || !n || *n <= 0 || *n > 4096 / total)
431 return std::nullopt;
432 total *= *n;
433 SmallVector<int64_t> r;
434 for (int64_t i = 0; i < *n; ++i)
435 r.push_back(*lb + i * *st);
436 ranges.push_back(std::move(r));
437 }
438 SmallVector<SmallVector<int64_t>> out;
439 SmallVector<size_t> at(syms.size(), 0);
440 for (int64_t n = 0; n < total; ++n) {
441 SmallVector<AffineExpr> vals;
442 for (size_t i = 0; i < syms.size(); ++i)
443 vals.push_back(
444 getAffineConstantExpr(ranges[i][at[i]], syms[i].getContext()));
445 SmallVector<int64_t> row;
446 for (AffineExpr e : exprs) {
447 auto c = dyn_cast<AffineConstantExpr>(
448 simplifyAffineExpr(e.replaceSymbols(vals), 0, 0));
449 if (!c)
450 return std::nullopt;
451 row.push_back(c.getValue());
452 }
453 out.push_back(std::move(row));
454 for (size_t i = 0; i < syms.size(); ++i) {
455 if (++at[i] < ranges[i].size())
456 break;
457 at[i] = 0;
458 }
459 }
460 return out;
461}

463/// The order of the T blocks in a bfp16 copy. A loop of K iterations that
464/// reads r tiles per iteration, d tiles apart, and advances s tiles per
465/// iteration, reads tile t = s*k + d*i + rest as block rest*r*K + r*k + i, so
466/// it reads the blocks in order. With s = 0, tile t is block t.
467struct Layout {
468 int64_t s = 0, K = 1, d = 1, r = 1, T = 0;
469
470 bool operator==(const Layout &o) const {
471 return s == o.s && K == o.K && d == o.d && r == o.r && T == o.T;
472 }
473 bool identity() const { return s == 0; }
474
475 /// Returns the (stride, extent) of the lower and the higher of the two
476 /// digits of t that the loop walks.
477 std::pair<std::pair<int64_t, int64_t>, std::pair<int64_t, int64_t>>
478 digits() const {
479 std::pair<int64_t, int64_t> k = {s, K}, i = {d, r};
480 return s < d ? std::pair(k, i) : std::pair(i, k);
481 }
482
483 static std::optional<Layout> make(int64_t s, int64_t K, int64_t d, int64_t r,
484 int64_t T) {
485 Layout l{s, K, d, r, T};
486 if (s <= 0 || K <= 0 || d <= 0 || r <= 0 || T <= 0)
487 return std::nullopt;
488 auto [lo, hi] = l.digits();
489 if (hi.first % (lo.first * lo.second) || T % (hi.first * hi.second))
490 return std::nullopt;
491 return l;
492 }

494 /// Builds the block index of tile `t`.
495 Value block(OpBuilder &b, Location loc, Value t) const {
496 if (identity())
497 return t;
498 auto c = [&](int64_t v) -> Value {
499 return arith::ConstantIndexOp::create(b, loc, v);
500 };
501 auto div = [&](Value x, int64_t v) -> Value {
502 return v == 1 ? x : arith::DivUIOp::create(b, loc, x, c(v)).getResult();
503 };
504 auto rem = [&](Value x, int64_t v) -> Value {
505 return arith::RemUIOp::create(b, loc, x, c(v));
506 };
507 auto mul = [&](Value x, int64_t v) -> Value {
508 return v == 1 ? x : arith::MulIOp::create(b, loc, x, c(v)).getResult();
509 };
510 auto add = [&](Value x, Value y) -> Value {
511 return arith::AddIOp::create(b, loc, x, y);
512 };
513 auto [lo, hi] = digits();
514 int64_t m = hi.first / (lo.first * lo.second);
515 Value k = rem(div(t, s), K), i = rem(div(t, d), r);
516 Value rest =
517 add(rem(t, lo.first), mul(add(rem(div(t, lo.first * lo.second), m),
518 mul(div(t, hi.first * hi.second), m)),
519 lo.first));
520 return add(mul(rest, r * K), add(mul(k, r), i));
521 }
522};

533static std::optional<SteppedRead> steppedRead(vector::TransferReadOp rd) {
534 auto loop = dyn_cast<scf::ForOp>(rd->getParentOp());
535 if (!loop || rd.getIndices().size() != 1)
536 return std::nullopt;
537 auto arg = dyn_cast<BlockArgument>(rd.getIndices()[0]);
538 if (!arg || arg.getOwner() != loop.getBody() || arg.getArgNumber() == 0)
539 return std::nullopt;
540 auto [base, c] = splitConstant(loop.getTiedLoopYieldedValue(arg)->get());
541 std::optional<int64_t> trips = staticTrips(loop);
542 if (base != arg || c <= 0 || c % kTile || !trips || *trips <= 0)
543 return std::nullopt;
544 return SteppedRead{loop, loop.getTiedLoopInit(arg)->get(), c / kTile, *trips};
545}

555/// Picks the block order for a copy of T tiles so that a loop advancing

559chooseLayout(ArrayRef<vector::TransferReadOp> reads, int64_t T) {
560 llvm::MapVector<Operation *, SmallVector<vector::TransferReadOp>> byLoop;
561 for (vector::TransferReadOp rd : reads)
562 if (std::optional<SteppedRead> sr = steppedRead(rd))
563 byLoop[sr->loop].push_back(rd);
564 std::optional<Layout> chosen;
565 SmallVector<Stream> streams;
566 for (auto &[loopOp, rds] : byLoop) {
567 SteppedRead first = *steppedRead(rds.front());
568 SmallVector<Value> syms;
569 SmallVector<AffineExpr> exprs;
570 bool ok = true;
571 for (vector::TransferReadOp rd : rds) {
572 SteppedRead sr = *steppedRead(rd);
573 ok &= sr.step == first.step;
574 exprs.push_back(toExpr(sr.init, syms));
575 }
576 if (!ok)
577 continue;
578 std::optional<SmallVector<SmallVector<int64_t>>> vals =
579 valuesOver(exprs, syms);
580 if (!vals)
581 continue;
582 // Each read must be at a fixed tile offset from the first one.
583 SmallVector<int64_t> off;
584 for (size_t j = 0; j < rds.size(); ++j) {
585 int64_t o = (*vals)[0][j] - (*vals)[0][0];
586 for (auto &row : *vals)
587 ok &= row[j] - row[0] == o && row[j] % kTile == 0;
588 off.push_back(o / kTile);
589 }
590 int64_t lo = *llvm::min_element(off);
591 SmallVector<int64_t> distinct;
592 for (int64_t &o : off) {
593 o -= lo;
594 if (!llvm::is_contained(distinct, o))
595 distinct.push_back(o);
596 }
597 llvm::sort(distinct);
598 int64_t r = distinct.size(), d = r > 1 ? distinct[1] : 1;
599 for (int64_t j = 0; j < r; ++j)
600 ok &= distinct[j] == j * d;
601 if (!ok || r > 4)
602 continue;
603 std::optional<Layout> layout =
604 Layout::make(first.step, first.trips, d, r, T);
605 if (!layout || (chosen && !(*chosen == *layout)))
606 continue;
607 // Every entry into the loop must start at the first block of a group.
608 int64_t lowest = 0;
609 for (size_t j = 0; j < rds.size(); ++j)
610 if (off[j] == 0)
611 lowest = j;
612 for (auto &row : *vals) {
613 int64_t t = row[lowest] / kTile;
614 ok &= t >= 0 && t < T && (t / layout->s) % layout->K == 0 &&
615 (t / layout->d) % layout->r == 0;
616 }
617 if (!ok)
618 continue;
619 chosen = layout;
620 Stream st{cast<scf::ForOp>(loopOp), {}, steppedRead(rds[lowest])->init};
621 for (size_t j = 0; j < rds.size(); ++j)
622 st.reads.push_back({rds[j], off[j] / d});
623 streams.push_back(std::move(st));
624 }
625 if (!chosen)
626 return {Layout{0, 1, 1, 1, T}, {}};
627 return {*chosen, std::move(streams)};
628}

630/// Returns the number of load streams already carried by `loop`.
631static int64_t loadStreams(scf::ForOp loop) {
632 int64_t n = 0;
633 for (Operation &op : *loop.getBody())
634 if (auto pop = dyn_cast<FifoLdPopBFP16Op>(op))
635 if (auto arg = dyn_cast<BlockArgument>(pop.getAddr()))
636 n += arg.getOwner() == loop.getBody();
637 return n;
638}

640struct HoistBFP16ConversionPattern : public OpRewritePattern<ToBFP16Op> {
641 using OpRewritePattern::OpRewritePattern;
642
643 LogicalResult matchAndRewrite(ToBFP16Op conv,
644 PatternRewriter &rewriter) const override {
645 std::optional<TileRead> tr = getTileRead(conv);
646 if (!tr)
647 return rewriter.notifyMatchFailure(conv, "not a tile read of a buffer");
648 AIE::BufferOp buf = tr->buf;
649 auto core = conv->getParentOfType<AIE::CoreOp>();
650 if (!core)
651 return rewriter.notifyMatchFailure(conv, "not in a core");
652
653 // Find the outermost loop around the conversion that does not modify
654 // the buffer.
655 BufferLocks locks = bufferLocks(buf, core);
656 scf::ForOp loop;
657 for (auto l = conv->getParentOfType<scf::ForOp>();
658 l && core->isProperAncestor(l); l = l->getParentOfType<scf::ForOp>()) {
659 if (loopMayChange(l, buf.getBuffer(), locks))
660 break;
661 loop = l;
662 }
663 if (!loop)
664 return rewriter.notifyMatchFailure(conv, "no loop keeps the buffer");
665
666 // Collect the conversions of the buffer's tiles in that loop. Only
667 // operands of matmuls whose other operand is already loaded as bfp16
668 // blocks are considered, so that the matmul needs no conversion at all.
669 auto partnered = [](ToBFP16Op c) {
670 return !c->use_empty() && llvm::all_of(c->getUses(), [&](OpOperand &u) {
671 auto mm = dyn_cast<MatMulBFP16Op_AIE2P>(u.getOwner());
672 if (!mm)
673 return false;
674 Value other = u.getOperandNumber() < 2 ? mm.getRhsMantissas()
675 : mm.getLhsMantissas();
676 return isa_and_nonnull<LoadBFP16Op, FifoLdPopBFP16Op>(
677 other.getDefiningOp());
678 });
679 };
680 if (!partnered(conv))
681 return rewriter.notifyMatchFailure(conv, "the other operand converted");
682 SmallVector<std::pair<ToBFP16Op, vector::TransferReadOp>> group;
683 int64_t runs = 0;
684 loop.getBody()->walk([&](ToBFP16Op c) {
685 if (c.getTranspose() != conv.getTranspose() || !partnered(c))
686 return;
687 std::optional<TileRead> t = getTileRead(c);
688 if (!t || t->buf != buf || !loop->isAncestor(t->read))
689 return;
690 group.push_back({c, t->read});
691 runs = std::min(kMaxDiv, runs + runsPerExecution(c, loop));
692 });
693 // The copy is built each time the loop starts, at about the cost of one
694 // conversion per tile, so require at least two conversions per tile in
695 // one execution of the loop.
696 auto bufTy = cast<MemRefType>(buf.getType());
697 int64_t tiles = bufTy.getNumElements() / kTile;
698 if (runs < 2 * tiles)
699 return rewriter.notifyMatchFailure(conv, "too few conversions per tile");
700 if (!fitsInCoreMemory(buf, tiles * (kTile + kExpBytes)))
701 return rewriter.notifyMatchFailure(conv, "no room for the bfp16 copy");
702
703 // The copy holds one bfp16 block per tile, in the order the loops read
704 // them.
705 Location loc = conv.getLoc();
706 Attribute space = bufTy.getMemorySpace();
707 Type i8 = rewriter.getI8Type(), i32 = rewriter.getI32Type();
708 auto stateTy = VectorType::get({32}, i32);
709 auto mantTy = VectorType::get({kTile}, i8);
710 auto expTy = VectorType::get({kExpBytes}, i8);
711 AIE::BufferOp copy;
712 {
713 OpBuilder::InsertionGuard g(rewriter);
714 rewriter.setInsertionPointAfter(buf);
715 std::string base =
716 (buf.getSymName() ? buf.getSymName()->str() : std::string("buf")) +
717 "_bfp16";
718 DenseSet<StringRef> taken;
719 if (auto dev = buf->getParentOfType<AIE::DeviceOp>())
720 dev.walk([&](Operation *op) {
721 if (auto sym = op->getAttrOfType<StringAttr>(
722 SymbolTable::getSymbolAttrName()))
723 taken.insert(sym.getValue());
724 });
725 std::string name = base;
726 for (int i = 0; taken.contains(name); ++i)
727 name = base + "_" + std::to_string(i);
728 copy = AIE::BufferOp::create(
729 rewriter, buf.getLoc(),
730 MemRefType::get({tiles * (kTile + kExpBytes)}, i8, nullptr, space),
731 buf.getTile(), rewriter.getStringAttr(name), nullptr, nullptr,
732 nullptr, nullptr, nullptr, nullptr);
733 }
734 SmallVector<vector::TransferReadOp> reads;
735 for (auto &[c, rd] : group)
736 if (!llvm::is_contained(reads, rd))
737 reads.push_back(rd);
738 auto [layout, streams] = chooseLayout(reads, tiles);
739 auto cIdx = [&](OpBuilder &b, Location l, int64_t v) -> Value {
740 return arith::ConstantIndexOp::create(b, l, v);
741 };
742 auto blockOffset = [&](OpBuilder &b, Location l, Value t) -> Value {
743 return arith::MulIOp::create(b, l, layout.block(b, l, t),
744 cIdx(b, l, kTile + kExpBytes));
745 };
746
747 // Before the loop, convert every tile and write the blocks in order
748 // through a store stream. The loop nest follows the layout's digits so
749 // that tile indices are affine in the induction variables.
750 rewriter.setInsertionPoint(loop);
751 SmallVector<ReassociationIndices> all(1);
752 for (int64_t i = 0; i < bufTy.getRank(); ++i)
753 all[0].push_back(i);
754 Value flat = bufTy.getRank() == 1 ? Value(buf.getBuffer())
755 : memref::CollapseShapeOp::create(
756 rewriter, loc, buf.getBuffer(), all)
757 .getResult();
758 SmallVector<std::pair<int64_t, int64_t>> nest; // (extent, tile stride)
759 if (layout.identity()) {
760 nest.push_back({tiles, 1});
761 } else {
762 auto [lo, hi] = layout.digits();
763 int64_t m = hi.first / (lo.first * lo.second);
764 nest.push_back({tiles / (hi.first * hi.second), hi.first * hi.second});
765 nest.push_back({m, lo.first * lo.second});
766 nest.push_back({lo.first, 1});
767 nest.push_back({layout.K, layout.s});
768 nest.push_back({layout.r, layout.d});
769 }
770 // Handle U tiles per innermost iteration and read them all before the
771 // first store; later reads are not moved above the stores.
772 while (nest.size() > 1 && nest.back().first == 1)
773 nest.pop_back();
774 int64_t inner = nest.back().first;
775 int64_t U = inner % 4 == 0 ? 4 : inner % 2 == 0 ? 2 : 1;
776 Value bfZero = arith::ConstantOp::create(
777 rewriter, loc, rewriter.getZeroAttr(rewriter.getBF16Type()));
778 auto tileTy = VectorType::get({kTile}, rewriter.getBF16Type());
779 VectorType convTy = conv.getSource().getType();
780 UnitAttr transpose = conv.getTransposeAttr();
781 std::function<SmallVector<Value>(OpBuilder &, Location, unsigned, Value,
782 ValueRange)>
783 build = [&](OpBuilder &b, Location l, unsigned level, Value t,
784 ValueRange st) -> SmallVector<Value> {
785 if (level == nest.size()) {
786 int64_t stride = nest.back().second;
787 SmallVector<Value> tileVals;
788 for (int64_t u = 0; u < U; ++u) {
789 Value tu = arith::AddIOp::create(b, l, t, cIdx(b, l, u * stride));
790 Value at = arith::MulIOp::create(b, l, tu, cIdx(b, l, kTile));
791 tileVals.push_back(
792 vector::TransferReadOp::create(b, l, tileTy, flat, ValueRange{at},
793 bfZero, ArrayRef<bool>{true}));
794 }
795 SmallVector<Value> cur(st.begin(), st.end());
796 for (Value v : tileVals) {
797 auto c = ToBFP16Op::create(
798 b, l, mantTy, expTy, vector::ShapeCastOp::create(b, l, convTy, v),
799 transpose);
800 auto push = FifoStPushBFP16Op::create(
801 b, l, i32, stateTy, i32, cur[0], cur[1], cur[2], c.getMantissas(),
802 c.getExponents());
803 cur = SmallVector<Value>(push->getResults());
804 }
805 return cur;
806 }
807 auto [extent, stride] = nest[level];
808 bool last = level + 1 == nest.size();
809 int64_t trips = last ? extent / U : extent;
810 int64_t step = last ? stride * U : stride;
811 if (trips == 1)
812 return build(b, l, level + 1, t, st);
813 auto nloop = scf::ForOp::create(
814 b, l, cIdx(b, l, 0), cIdx(b, l, trips), cIdx(b, l, 1), st,
815 [&](OpBuilder &ib, Location il, Value iv, ValueRange ist) {
816 Value ti = arith::AddIOp::create(
817 ib, il, t,
818 arith::MulIOp::create(ib, il, iv, cIdx(ib, il, step)));
819 scf::YieldOp::create(ib, il, build(ib, il, level + 1, ti, ist));
820 });
821 return SmallVector<Value>(nloop.getResults());
822 };
823 auto start = FifoStartOp::create(rewriter, loc, i32, stateTy, i32,
824 copy.getBuffer(), cIdx(rewriter, loc, 0));
825 (void)build(rewriter, loc, 0, cIdx(rewriter, loc, 0),
826 ValueRange(start->getResults()));
827
828 // Replaces the conversions of `rd` with the block (m, e).
829 auto setBFP16 = [&](vector::TransferReadOp rd, Value m, Value e) {
830 for (auto &[c, r] : group)
831 if (r == rd) {
832 c.getMantissas().replaceAllUsesWith(m);
833 c.getExponents().replaceAllUsesWith(e);
834 }
835 };
836 // Read through load streams where possible, at most two per loop (one per
837 // load FIFO). The pops of an iteration are emitted together; a fill
838 // provides enough data for the next 8 pops.
839 SmallPtrSet<Operation *, 16> streamed;
840 for (Stream &st : streams) {
841 if (loadStreams(st.loop) >= 2)
842 continue;
843 rewriter.setInsertionPoint(st.loop);
844 Location l = st.loop.getLoc();
845 Value t0 = arith::DivUIOp::create(rewriter, l, st.init,
846 cIdx(rewriter, l, kTile));
847 auto begin =
848 FifoStartOp::create(rewriter, l, i32, stateTy, i32, copy.getBuffer(),
849 blockOffset(rewriter, l, t0));
850 int64_t r = layout.r;
851 (void)st.loop.replaceWithAdditionalYields(
852 rewriter, begin->getResults(), /*replaceInitOperandUsesInLoop=*/false,
853 [&](OpBuilder &b, Location, ArrayRef<BlockArgument> args) {
854 Operation *first = st.reads.front().first;
855 for (auto &[rd, _] : st.reads)
856 if (rd->isBeforeInBlock(first))
857 first = rd;
858 OpBuilder::InsertionGuard g(b);
859 b.setInsertionPoint(first);
860 Value addr = args[0], state = args[1], count = args[2];
861 for (int64_t i = 0; i < r; ++i) {
862 auto pop = FifoLdPopBFP16Op::create(
863 b, first->getLoc(), i32, stateTy, i32, mantTy, expTy, addr,
864 state, count,
865 r % 8 == 0 && i % 8 ? b.getUnitAttr() : UnitAttr());
866 for (auto &[rd, place] : st.reads)
867 if (place == i)
868 setBFP16(rd, pop.getMantissas(), pop.getExponents());
869 addr = pop.getNext();
870 state = pop.getStateOut();
871 count = pop.getCountOut();
872 }
873 return SmallVector<Value>{addr, state, count};
874 });
875 for (auto &[rd, _] : st.reads)
876 streamed.insert(rd);
877 }
878 // Load the remaining blocks individually.
879 for (vector::TransferReadOp rd : reads) {
880 if (streamed.contains(rd))
881 continue;
882 rewriter.setInsertionPoint(rd);
883 Location l = rd.getLoc();
884 Value t = arith::DivUIOp::create(
885 rewriter, l, linearOffset(rewriter, l, rd), cIdx(rewriter, l, kTile));
886 auto ld =
887 LoadBFP16Op::create(rewriter, l, mantTy, expTy, copy.getBuffer(),
888 blockOffset(rewriter, l, t));
889 setBFP16(rd, ld.getMantissas(), ld.getExponents());
890 }
891 for (auto &[c, rd] : group)
892 rewriter.eraseOp(c);
893 return success();
894 }
895};

898/// with step 1.

903 if (!loop || loop.getInductionVar() != arg ||

905 getConstantIntValue(loop.getStep()) != 1 ||

907 return nullptr;
908 return loop;

911/// A loop nest that decodes the bfp16ebs8 blocks in `bytes` ([..., 8, 9]) into
912/// the bf16 tiles of `tiles` ([..., 8, 8]):
913/// tiles[i..., r, c] = bf16(sitofp(bytes[p(i...), r, c + 1]) *
914/// bitcast<f32>((zext(bytes[p(i...), r, 0]) - 6) <<
915/// 23))
916/// for all elements of `tiles`. `bytes` may have extra unit dims, indexed
917/// with 0. Dim j of the block grid is dim perm[j] of the tile grid.
918struct Decode {
919 AIE::BufferOp bytes, tiles;
920 scf::ForOp nest;
921 SmallVector<unsigned> perm;
922 SmallVector<int64_t> blocks;
923};

925static std::optional<Decode> matchDecode(memref::StoreOp st) {
926 auto trunc = st.getValueToStore().getDefiningOp<arith::TruncFOp>();
927 auto mul = trunc ? trunc.getIn().getDefiningOp<arith::MulFOp>() : nullptr;
928 if (!mul || !trunc.getType().isBF16())
929 return std::nullopt;
930 arith::SIToFPOp man;
931 arith::BitcastOp scale;
932 for (auto [x, y] : {std::pair(mul.getLhs(), mul.getRhs()),
933 std::pair(mul.getRhs(), mul.getLhs())})
934 if (x.getDefiningOp<arith::SIToFPOp>() &&
935 y.getDefiningOp<arith::BitcastOp>()) {
936 man = x.getDefiningOp<arith::SIToFPOp>();
937 scale = y.getDefiningOp<arith::BitcastOp>();
938 }
939 if (!man)
940 return std::nullopt;
941 auto shl = scale.getIn().getDefiningOp<arith::ShLIOp>();
942 auto sub = shl ? shl.getLhs().getDefiningOp<arith::SubIOp>() : nullptr;
943 auto ext = sub ? sub.getLhs().getDefiningOp<arith::ExtUIOp>() : nullptr;
944 auto expLd = ext ? ext.getIn().getDefiningOp<memref::LoadOp>() : nullptr;
945 auto manLd = man.getIn().getDefiningOp<memref::LoadOp>();
946 if (!expLd || !manLd || getConstantIntValue(shl.getRhs()) != 23 ||
947 getConstantIntValue(sub.getRhs()) != 6 || !scale.getType().isF32() ||
948 !man.getType().isF32() || expLd.getMemref() != manLd.getMemref())
949 return std::nullopt;
950 auto bBuf = expLd.getMemref().getDefiningOp<AIE::BufferOp>();
951 auto xBuf = st.getMemref().getDefiningOp<AIE::BufferOp>();
952 if (!bBuf || !xBuf)
953 return std::nullopt;
954 auto bt = cast<MemRefType>(bBuf.getType());
955 auto xt = cast<MemRefType>(xBuf.getType());
956 int64_t n = xt.getRank(), nb = bt.getRank();
957 if (!bt.hasStaticShape() || !xt.hasStaticShape() ||
958 !bt.getLayout().isIdentity() || !xt.getLayout().isIdentity() || nb < n ||
959 n < 3 || !bt.getElementType().isInteger(8) ||
960 !xt.getElementType().isBF16() || bt.getDimSize(nb - 2) != 8 ||
961 bt.getDimSize(nb - 1) != 9 || xt.getDimSize(n - 2) != 8 ||
962 xt.getDimSize(n - 1) != 8)
963 return std::nullopt;
964 // Each store index must be the induction variable of a loop over the whole
965 // corresponding dim.
966 SmallVector<scf::ForOp> loops;
967 for (int64_t i = 0; i < n; ++i) {
968 scf::ForOp l = loopOver(st.getIndices()[i], xt.getDimSize(i));
969 if (!l || llvm::is_contained(loops, l))
970 return std::nullopt;
971 loops.push_back(l);
972 }
973 scf::ForOp nest = loops.front();
974 for (scf::ForOp l : loops)
975 if (l->isProperAncestor(nest))
976 nest = l;
977 // The nest is removed, so nothing may use what it returns.
978 if (!nest->use_empty())
979 return std::nullopt;
980 // Both loads read the same block and row: the exponent at column 0 and the
981 // mantissa at column c + 1.
982 Decode d{bBuf, xBuf, nest, {}, {}};
983 ValueRange tileIdx = st.getIndices().take_front(n - 2);
984 for (int64_t j = 0; j < nb - 2; ++j) {
985 Value v = manLd.getIndices()[j];
986 if (expLd.getIndices()[j] != v)
987 return std::nullopt;
988 if (bt.getDimSize(j) == 1 && getConstantIntValue(v) == 0)
989 continue;
990 auto it = llvm::find(tileIdx, v);
991 if (it == tileIdx.end() ||
992 llvm::is_contained(d.perm, unsigned(it - tileIdx.begin())))
993 return std::nullopt;
994 d.perm.push_back(it - tileIdx.begin());
995 d.blocks.push_back(bt.getDimSize(j));
996 }
997 if (d.perm.size() != size_t(n - 2) ||
998 manLd.getIndices()[nb - 2] != st.getIndices()[n - 2] ||
999 expLd.getIndices()[nb - 2] != st.getIndices()[n - 2] ||
1000 getConstantIntValue(expLd.getIndices()[nb - 1]) != 0)
1001 return std::nullopt;
1002 SmallVector<Value> syms;
1003 AffineExpr off = toExpr(manLd.getIndices()[nb - 1], syms) -
1004 toExpr(st.getIndices()[n - 1], syms);
1005 auto c =
1006 dyn_cast<AffineConstantExpr>(simplifyAffineExpr(off, 0, syms.size()));
1007 if (!c || c.getValue() != 1)
1008 return std::nullopt;
1009 // Nothing else in the nest may access memory.
1010 WalkResult clean = nest->walk([&](Operation *op) {
1011 if (op == st || op == manLd || op == expLd || isPure(op) ||
1012 isa<scf::ForOp, scf::YieldOp>(op))
1013 return WalkResult::advance();
1014 return WalkResult::interrupt();
1015 });
1016 if (clean.wasInterrupted())
1017 return std::nullopt;
1018 return d;
1019}

1021/// Builds the index of the block in `d.bytes` that holds tile `t`.
1022static Value blockOf(OpBuilder &b, Location loc, Decode &d, Value t) {
1023 auto xt = cast<MemRefType>(d.tiles.getType());
1024 auto c = [&](int64_t v) -> Value {
1025 return arith::ConstantIndexOp::create(b, loc, v);
1026 };
1027 int64_t n = xt.getRank() - 2;
1028 SmallVector<Value> dig(n);
1029 for (int64_t i = n - 1; i >= 0; --i) {
1030 dig[i] =
1031 i == 0 ? t : arith::RemUIOp::create(b, loc, t, c(xt.getDimSize(i)));
1032 if (i)
1033 t = arith::DivUIOp::create(b, loc, t, c(xt.getDimSize(i)));
1034 }
1035 Value blk;
1036 for (size_t j = 0; j < d.blocks.size(); ++j)
1037 blk = blk ? arith::AddIOp::create(
1038 b, loc, arith::MulIOp::create(b, loc, blk, c(d.blocks[j])),
1039 dig[d.perm[j]])
1040 .getResult()
1041 : dig[d.perm[j]];
1042 return blk;
1043}

1045/// Makes matmuls load the bfp16 blocks that a core would otherwise decode into
1046/// bf16 tiles, when the tiles are only read whole by untransposed
1047/// conversions. The decode loop and the tile buffer are removed, and the
1048/// release of the block buffer's lock moves after the last read.
1049struct ReadDecodedBFP16Pattern : public OpRewritePattern<memref::StoreOp> {
1050 using OpRewritePattern::OpRewritePattern;
1051
1052 LogicalResult matchAndRewrite(memref::StoreOp st,
1053 PatternRewriter &rewriter) const override {
1054 std::optional<Decode> d = matchDecode(st);
1055 if (!d)
1056 return rewriter.notifyMatchFailure(st, "not a bfp16 decode");
1057 auto core = st->getParentOfType<AIE::CoreOp>();
1058 BufferLocks bytesLocks = bufferLocks(d->bytes, core);
1059 BufferLocks tileLocks = bufferLocks(d->tiles, core);
1060 if (bytesLocks.any || bytesLocks.locks.empty() || tileLocks.any ||
1061 !tileLocks.locks.empty() || d->tiles.getInitialValue())
1062 return rewriter.notifyMatchFailure(st, "not DMA bytes into core tiles");
1063
1064 // Every other use of the tiles must be a whole-tile read, after the
1065 // decode, that only untransposed conversions use.
1066 SmallVector<Operation *> views;
1067 SmallVector<std::pair<ToBFP16Op, vector::TransferReadOp>> convs;
1068 SmallVector<Value> work = {d->tiles.getBuffer()};
1069 Block *blk = d->nest->getBlock();
1070 Operation *last = d->nest;
1071 while (!work.empty()) {
1072 Value v = work.pop_back_val();
1073 for (Operation *user : v.getUsers()) {
1074 if (user == st.getOperation())
1075 continue;
1076 if (isa<memref::CollapseShapeOp, memref::ExpandShapeOp, memref::CastOp>(
1077 user)) {
1078 views.push_back(user);
1079 work.push_back(user->getResult(0));
1080 continue;
1081 }
1082 auto rd = dyn_cast<vector::TransferReadOp>(user);
1083 Operation *anc = rd ? blk->findAncestorOpInBlock(*rd) : nullptr;
1084 if (!anc || !d->nest->isBeforeInBlock(anc))
1085 return rewriter.notifyMatchFailure(st, "another use of the tiles");
1086 for (Operation *u : rd->getUsers()) {
1087 while (isa<vector::ShapeCastOp>(u) && u->hasOneUse())
1088 u = *u->getUsers().begin();
1089 auto conv = dyn_cast<ToBFP16Op>(u);
1090 std::optional<TileRead> tr = conv ? getTileRead(conv) : std::nullopt;
1091 if (!conv || conv.getTranspose() || !tr || tr->read != rd)
1092 return rewriter.notifyMatchFailure(st, "a tile read not converted");
1093 convs.push_back({conv, rd});
1094 }
1095 if (last->isBeforeInBlock(anc))
1096 last = anc;
1097 }
1098 }
1099 if (convs.empty())
1100 return rewriter.notifyMatchFailure(st, "no conversions");
1101
1102 // Find the release of the block buffer after the decode. It has to move
1103 // after the last read, so no other use of its locks may come between.
1104 AIE::UseLockOp release;
1105 for (Operation *op = d->nest->getNextNode(); op; op = op->getNextNode()) {
1106 bool usesLock = op->walk([&](AIE::UseLockOp ul) {
1107 return bytesLocks.contains(ul.getLock())
1108 ? WalkResult::interrupt()
1109 : WalkResult::advance();
1110 }).wasInterrupted();
1111 auto ul = dyn_cast<AIE::UseLockOp>(op);
1112 if (ul && ul.release() && bytesLocks.contains(ul.getLock()) && !release)
1113 release = ul;
1114 else if (usesLock)
1115 return rewriter.notifyMatchFailure(st, "the bytes' lock in between");
1116 bool writes = op->walk([&](Operation *o) {
1117 return mayChange(o, d->bytes.getBuffer(), {})
1118 ? WalkResult::interrupt()
1119 : WalkResult::advance();
1120 }).wasInterrupted();
1121 if (writes)
1122 return rewriter.notifyMatchFailure(st, "the bytes may change");
1123 if (op == last)
1124 break;
1125 }
1126
1127 // Load the blocks from a flat view of the block buffer.
1128 rewriter.setInsertionPoint(d->nest);
1129 Location loc = st.getLoc();
1130 auto bt = cast<MemRefType>(d->bytes.getType());
1131 SmallVector<ReassociationIndices> all(1);
1132 for (int64_t i = 0; i < bt.getRank(); ++i)
1133 all[0].push_back(i);
1134 Value flat = memref::CollapseShapeOp::create(rewriter, loc,
1135 d->bytes.getBuffer(), all);
1136 // One load per tile read, where the read is, which all its conversions
1137 // use.
1138 DenseMap<Operation *, LoadBFP16Op> loads;
1139 for (auto [conv, rd] : convs) {
1140 LoadBFP16Op &ld = loads[rd];
1141 if (!ld) {
1142 rewriter.setInsertionPointAfter(rd);
1143 Location l = rd.getLoc();
1144 Value t = arith::DivUIOp::create(
1145 rewriter, l, linearOffset(rewriter, l, rd),
1146 arith::ConstantIndexOp::create(rewriter, l, kTile));
1147 Value off = arith::MulIOp::create(
1148 rewriter, l, blockOf(rewriter, l, *d, t),
1149 arith::ConstantIndexOp::create(rewriter, l, kTile + kExpBytes));
1150 ld = LoadBFP16Op::create(rewriter, l, conv.getMantissas().getType(),
1151 conv.getExponents().getType(), flat, off);
1152 }
1153 rewriter.replaceOp(conv, ld.getResults());
1154 }
1155 if (release && release->isBeforeInBlock(last))
1156 rewriter.moveOpAfter(release, last);
1157 rewriter.eraseOp(d->nest);
1158 // Erase the reads and their shape casts, the views and the tile buffer
1159 // once they are unused.
1160 for (auto &[rd, ld] : loads) {
1161 for (Operation *u : llvm::make_early_inc_range(rd->getUsers()))
1162 if (u->use_empty())
1163 rewriter.eraseOp(u);
1164 if (rd->use_empty())
1165 rewriter.eraseOp(rd);
1166 }
1167 for (Operation *v : llvm::reverse(views))
1168 if (v->use_empty())
1169 rewriter.eraseOp(v);
1170 if (d->tiles->use_empty())
1171 rewriter.eraseOp(d->tiles);
1172 return success();
1173 }

1176/// Evaluates index computations in `scope`, given the values in `env` for
1177/// induction variables and iter_args. Supports constants, integer arithmetic

1185 return it->second;

1187 return c;
1188 Operation *d = v.getDefiningOp();

1190 return std::nullopt;

1192 std::optional<int64_t> a = eval(d->getOperand(0), depth + 1);
1193 std::optional<int64_t> b = eval(d->getOperand(1), depth + 1);

1204 if (isa<arith::MulIOp>(d))

1206 [](int64_t a, int64_t b) -> std::optional<int64_t> { return a * b; });
1207 if (isa<arith::DivUIOp, arith::DivSIOp>(d))
1208 return bin([](int64_t a, int64_t b) -> std::optional<int64_t> {
1209 return b > 0 && a >= 0 ? std::optional<int64_t>(a / b) : std::nullopt;
1210 });
1211 if (isa<arith::RemUIOp, arith::RemSIOp>(d))
1212 return bin([](int64_t a, int64_t b) -> std::optional<int64_t> {
1213 return b > 0 && a >= 0 ? std::optional<int64_t>(a % b) : std::nullopt;
1214 });
1215 if (isa<arith::IndexCastOp, arith::IndexCastUIOp>(d))
1216 return eval(d->getOperand(0), depth + 1);
1217 if (auto ap = dyn_cast<affine::AffineApplyOp>(d)) {
1218 SmallVector<int64_t> vals;
1219 for (Value o : ap.getMapOperands()) {
1220 std::optional<int64_t> x = eval(o, depth + 1);
1221 if (!x)
1222 return std::nullopt;
1223 vals.push_back(*x);
1224 }
1225 AffineMap m = ap.getAffineMap();
1226 SmallVector<AffineExpr> dims, syms;
1227 MLIRContext *ctx = ap.getContext();
1228 for (unsigned i = 0; i < m.getNumDims(); ++i)
1229 dims.push_back(getAffineConstantExpr(vals[i], ctx));
1230 for (unsigned i = 0; i < m.getNumSymbols(); ++i)
1231 syms.push_back(getAffineConstantExpr(vals[m.getNumDims() + i], ctx));
1232 auto c = dyn_cast<AffineConstantExpr>(
1233 m.getResult(0).replaceDimsAndSymbols(dims, syms));
1234 return c ? std::optional<int64_t>(c.getValue()) : std::nullopt;
1235 }
1236 return std::nullopt;

1240/// Returns the values from outside `loop` that the loop body's `vals` depend
1241/// on, directly or through the values the loop carries.

1243 auto yield = cast<scf::YieldOp>(loop.getBody()->getTerminator());
1244 SmallVector<Value> outside, work(vals.begin(), vals.end());
1245 DenseSet<Value> seen;

1255 yield.getOperand(arg.getArgNumber() - loop.getNumInductionVars()));

1257 } else if (Operation *d = v.getDefiningOp(); d && loop->isAncestor(d)) {

1260 outside.push_back(v);

1288/// Turns the r loads of one buffer in a loop body into one load stream when
1289/// iteration k loads blocks n + r*k ... n + r*k + r - 1 in program order. n
1290/// may depend on the induction variables of loops with static bounds around
1291/// the loop; the order is checked for each of their values.
1292/// The stream starts empty before the loop; its first pop fills it.
1293struct BFP16LoadStreamPattern : public OpRewritePattern<scf::ForOp> {
1294 using OpRewritePattern::OpRewritePattern;
1295
1296 LogicalResult matchAndRewrite(scf::ForOp loop,
1297 PatternRewriter &rewriter) const override {
1298 std::optional<int64_t> lb = getConstantIntValue(loop.getLowerBound());
1299 std::optional<int64_t> st = getConstantIntValue(loop.getStep());
1300 std::optional<int64_t> n = staticTrips(loop);
1301 if (!lb || !st || !n || *n <= 0)
1302 return rewriter.notifyMatchFailure(loop, "trips not known");
1303 int64_t trips = *n;
1304 // Group the loads in the body by buffer. Loads or streams in nested
1305 // regions are not handled.
1306 llvm::MapVector<Value, SmallVector<LoadBFP16Op>> byBase;
1307 for (Operation &op : *loop.getBody())
1308 if (auto ld = dyn_cast<LoadBFP16Op>(op))
1309 byBase[ld.getBase()].push_back(ld);
1310 bool nested = false;
1311 loop.getBody()->walk([&](Operation *op) {
1312 if (op->getParentOp() != loop &&
1313 isa<LoadBFP16Op, FifoStartOp, FifoLdPopBFP16Op>(op))
1314 nested = true;
1315 });
1316 if (byBase.empty() || nested)
1317 return rewriter.notifyMatchFailure(loop, "no loads to stream");
1318
1319 // Evaluate each load's offset in every iteration to check the order, for
1320 // each value of the outer induction variables the offsets depend on.
1321 Operation *scope = loop->getParentOfType<AIE::CoreOp>();
1322 if (!scope)
1323 scope = loop->getParentOp();
1324 SmallVector<Value> offsets;
1325 for (auto &[base, loads] : byBase)
1326 for (LoadBFP16Op ld : loads)
1327 offsets.push_back(ld.getOffset());
1328 std::optional<SmallVector<Value>> ivs =
1329 outerIVs(loop, valuesFromOutside(loop, offsets));
1330 std::optional<SmallVector<SmallVector<int64_t>>> combos;
1331 if (ivs) {
1332 SmallVector<AffineExpr> syms;
1333 for (unsigned i = 0; i < ivs->size(); ++i)
1334 syms.push_back(getAffineSymbolExpr(i, loop.getContext()));
1335 combos = valuesOver(syms, *ivs);
1336 }
1337 // Each offset is evaluated once per trip and outer combination.
1338 constexpr int64_t kMaxEvals = 1 << 16;
1339 if (combos && (int64_t(offsets.size()) * trips > kMaxEvals ||
1340 int64_t(combos->size()) >
1341 kMaxEvals / (int64_t(offsets.size()) * trips)))
1342 combos.reset();
1343 SmallVector<std::pair<Value, SmallVector<LoadBFP16Op>>> streams;
1344 for (auto &[base, loads] : byBase) {
1345 // The pops read ahead of the loads they replace, so the loop must not
1346 // change the buffer.
1347 AIE::BufferOp root = rootBuffer(base);
1348 BufferLocks locks;
1349 locks.any = true;
1350 if (root)
1351 locks = bufferLocks(root, loop->getParentOfType<AIE::CoreOp>());
1352 bool changes =
1353 loop.getBody()
1354 ->walk([&](Operation *op) {
1355 auto ul = dyn_cast<AIE::UseLockOp>(op);
1356 bool c = ul ? (ul.acquire() || ul.acquireGE()) &&
1357 locks.contains(ul.getLock())
1358 : mayWrite(op, base);
1359 return c ? WalkResult::interrupt() : WalkResult::advance();
1360 })
1361 .wasInterrupted();
1362 if (!loop.isDefinedOutsideOfLoop(base) || !combos || changes)
1363 continue;
1364 int64_t r = loads.size();
1365 bool ok = true;
1366 for (ArrayRef<int64_t> combo : *combos) {
1367 DenseMap<Value, int64_t> outer;
1368 for (auto [iv, x] : llvm::zip(*ivs, combo))
1369 outer[iv] = x;
1370 TripEvaluator ev{scope, outer};
1371 // Track the iter_arg values from one iteration to the next.
1372 SmallVector<std::optional<int64_t>> carried;
1373 for (Value init : loop.getInitArgs())
1374 carried.push_back(ev.eval(init));
1375 std::optional<int64_t> start;
1376 for (int64_t k = 0; k < trips && ok; ++k) {
1377 ev.env = outer;
1378 ev.env[loop.getInductionVar()] = *lb + k * *st;
1379 for (auto [arg, val] : llvm::zip(loop.getRegionIterArgs(), carried))
1380 if (val)
1381 ev.env[arg] = *val;
1382 for (int64_t j = 0; j < r && ok; ++j) {
1383 std::optional<int64_t> off = ev.eval(loads[j].getOffset());
1384 if (!off || *off % (kTile + kExpBytes)) {
1385 ok = false;
1386 break;
1387 }
1388 if (!start)
1389 start = *off;
1390 ok = *off == *start + (kTile + kExpBytes) * (r * k + j);
1391 }
1392 auto yield = cast<scf::YieldOp>(loop.getBody()->getTerminator());
1393 for (auto [i, y] : llvm::enumerate(yield.getOperands()))
1394 carried[i] = ev.eval(y);
1395 }
1396 if (!ok || !start) {
1397 ok = false;
1398 break;
1399 }
1400 }
1401 if (ok)
1402 streams.push_back({base, loads});
1403 }
1404 // A core has two load FIFOs.
1405 if (streams.empty() || int64_t(streams.size()) + loadStreams(loop) > 2)
1406 return rewriter.notifyMatchFailure(loop, "no streams, or too many");
1407
1408 Location loc = loop.getLoc();
1409 rewriter.setInsertionPoint(loop);
1410 SmallVector<Value> inits;
1411 for (auto &[base, loads] : streams) {
1412 // The first offset, computed before the loop as in its first iteration.
1413 IRMapping first;
1414 first.map(loop.getInductionVar(), loop.getLowerBound());
1415 first.map(loop.getRegionIterArgs(), loop.getInitArgs());
1416 std::function<Value(Value)> before = [&](Value v) -> Value {
1417 if (Value m = first.lookupOrNull(v))
1418 return m;
1419 Operation *d = v.getDefiningOp();
1420 if (!d || !loop->isAncestor(d))
1421 return v;
1422 for (Value o : d->getOperands())
1423 before(o);
1424 return rewriter.clone(*d, first)->getResult(
1425 cast<OpResult>(v).getResultNumber());
1426 };
1427 auto begin = FifoStartOp::create(
1428 rewriter, loc, rewriter.getI32Type(),
1429 VectorType::get({32}, rewriter.getI32Type()), rewriter.getI32Type(),
1430 base, before(loads.front().getOffset()));
1431 inits.append(begin->getResults().begin(), begin->getResults().end());
1432 }
1433 (void)loop.replaceWithAdditionalYields(
1434 rewriter, inits, /*replaceInitOperandUsesInLoop=*/false,
1435 [&](OpBuilder &b, Location l, ArrayRef<BlockArgument> args) {
1436 SmallVector<Value> yields;
1437 for (auto [s, entry] : llvm::enumerate(streams)) {
1438 Value addr = args[3 * s], state = args[3 * s + 1],
1439 count = args[3 * s + 2];
1440 // Emit the pops of an iteration together. A fill provides enough
1441 // data for the next 8 pops, so with a multiple of 8 blocks per
1442 // iteration only every eighth pop fills; otherwise every pop
1443 // fills.
1444 int64_t r = entry.second.size();
1445 Operation *first = entry.second.front();
1446 for (LoadBFP16Op ld : entry.second)
1447 if (ld->isBeforeInBlock(first))
1448 first = ld;
1449 OpBuilder::InsertionGuard g(b);
1450 b.setInsertionPoint(first);
1451 for (auto [j, ld] : llvm::enumerate(entry.second)) {
1452 auto pop = FifoLdPopBFP16Op::create(
1453 b, ld.getLoc(), b.getI32Type(), state.getType(),
1454 b.getI32Type(), ld.getMantissas().getType(),
1455 ld.getExponents().getType(), addr, state, count,
1456 r % 8 == 0 && j % 8 ? b.getUnitAttr() : UnitAttr());
1457 addr = pop.getNext();
1458 state = pop.getStateOut();
1459 count = pop.getCountOut();
1460 ld.getMantissas().replaceAllUsesWith(pop.getMantissas());
1461 ld.getExponents().replaceAllUsesWith(pop.getExponents());
1462 }
1463 yields.append({addr, state, count});
1464 }
1465 return yields;
1466 });
1467 for (auto &[base, loads] : streams)
1468 for (LoadBFP16Op ld : loads)
1469 rewriter.eraseOp(ld);
1470 return success();
1471 }
1472};

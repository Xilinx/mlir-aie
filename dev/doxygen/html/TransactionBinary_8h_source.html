65 TransactionBinaryOperation(XAie_TxnOpcode opc, uint32_t mask, uint64_t addr,
66 uint32_t value, const uint8_t *data,
67 uint32_t size) {
68 cmd.Opcode = opc;
69 cmd.Mask = mask;
70 cmd.RegOff = addr;

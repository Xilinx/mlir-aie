172 matchAndRewrite(FlowOp flowOp, OpAdaptor adaptor,
173 ConversionPatternRewriter &rewriter) const override {
174 Operation *Op = flowOp.getOperation();
175 DeviceOp d = flowOp->getParentOfType<DeviceOp>();
176 if (!d) {
177 flowOp->emitOpError("This operation must be contained within a device");
178 return failure();

180 rewriter.setInsertionPoint(d.getBody()->getTerminator());
181
182 auto srcTile = cast<TileOp>(flowOp.getSource().getDefiningOp());
183 TileID srcCoords = {srcTile.colIndex(), srcTile.rowIndex()};
184 auto srcBundle = flowOp.getSourceBundle();
185 auto srcChannel = flowOp.getSourceChannel();
186 Port srcPort = {srcBundle, srcChannel};
187
188 LLVM_DEBUG({
189 auto dstTile = cast<TileOp>(flowOp.getDest().getDefiningOp());
190 llvm::dbgs() << "\n\t---Begin rewrite() for flowOp: (" << srcCoords.col
191 << ", " << srcCoords.row << ")"
192 << stringifyWireBundle(srcBundle) << srcChannel << " -> ("
193 << dstTile.colIndex() << ", " << dstTile.rowIndex() << ")"
194 << stringifyWireBundle(flowOp.getDestBundle())
195 << flowOp.getDestChannel() << "\n\t";
196 });
197
198 // if the flow (aka "net") for this FlowOp hasn't been processed yet,
199 // add all switchbox connections to implement the flow
200 TileID srcSbId = {srcCoords.col, srcCoords.row};
201 PathEndPoint srcPoint = {srcSbId, srcPort};
202 if (analyzer.processedFlows.count(srcPoint)) {
203 // This FlowOp is a broadcast sibling of a flow whose route was already
204 // materialized (the analyzer merges all destinations sharing a source
205 // into one net, so the first sibling emitted connections for every
206 // destination). Erase it and report success so the erase is committed.
207 LLVM_DEBUG(llvm::dbgs() << "Flow already processed!\n");
208 rewriter.eraseOp(Op);
209 return success();
210 }
211 const SwitchSettings &settings = analyzer.routing.settings.at(srcPoint);
212 // add connections for all the Switchboxes in SwitchSettings
213 for (const auto &[tileId, setting] : settings) {
214 int col = tileId.col;
215 int row = tileId.row;
216 SwitchboxOp swOp = analyzer.getSwitchbox(rewriter, col, row);
217 int shimCh = srcChannel;
218 bool isShim = analyzer.getTile(rewriter, tileId).isShimNOCorPLTile();
219
220 // TODO: must reserve N3, N7, S2, S3 for DMA connections
221 if (isShim && tileId == srcSbId) {
222
223 shimCh = shimMuxChannelFrom(srcPort);
224 ShimMuxOp shimMuxOp = analyzer.getShimMux(rewriter, col);
225 addConnection(rewriter, cast<Interconnect>(shimMuxOp.getOperation()),
226 flowOp, srcBundle, srcChannel, WireBundle::North, shimCh);
227 }
228 assert(setting.srcs.size() == setting.dsts.size());
229 for (size_t i = 0; i < setting.srcs.size(); i++) {
230 Port src = setting.srcs[i];
231 Port dest = setting.dsts[i];
232
233 // A flow can start and end at one shim (see shimMuxChannelFrom).
234 if (isShim && tileId == srcSbId && src == srcPort)
235 src = {WireBundle::South, shimCh};
236 if (isShim && (dest.bundle == WireBundle::DMA ||
237 dest.bundle == WireBundle::PLIO ||
238 dest.bundle == WireBundle::NOC)) {
239 int destCh = shimMuxChannelTo(dest);
240
241 ShimMuxOp shimMuxOp = analyzer.getShimMux(rewriter, col);
242 addConnection(rewriter, cast<Interconnect>(shimMuxOp.getOperation()),
243 flowOp, WireBundle::North, destCh, dest.bundle,
244 dest.channel);
245 dest = {WireBundle::South, destCh};
246 }
247 addConnection(rewriter, cast<Interconnect>(swOp.getOperation()), flowOp,

250

253 }
254
255 LLVM_DEBUG(llvm::dbgs()
256 << "\n\t\tFinished adding ConnectOps to implement flowOp.\n");
257
258 analyzer.processedFlows.insert(srcPoint);
259 rewriter.eraseOp(Op);
260 return success();

305/// Whether the connection out of `out` at `tile` leads, through `settings`, to
306/// `finalPort` at `finalTile`.
307static bool findPathToDest(const SwitchSettings &settings, TileID tile,
308 Port out, TileID finalTile, Port finalPort) {
309 if (tile == finalTile && out == finalPort)
310 return true;
311 std::optional<std::pair<TileID, Port>> next = linkedInput(tile, out);
312 if (!next)
313 return false;
314 TileID neighbour = next->first;
315 Port in = next->second;
316 auto setting = settings.find(neighbour);
317 if (setting == settings.end())
318 return false;
319 return llvm::any_of(
320 llvm::zip(setting->second.srcs, setting->second.dsts), [&](auto link) {
321 auto [src, dest] = link;
322 return src == in &&
323 findPathToDest(settings, neighbour, dest, finalTile, finalPort);
324 });
325}

327/// Every node `successors` reaches from `start`, with the node a breadth-first
328/// search first reaches it from; `start` maps to itself. Stops once `goal` is
329/// reached, if given.
330template <typename Node, typename Successors>
331static std::map<Node, Node>
332breadthFirstParents(Node start, Successors successors,
333 std::optional<Node> goal = std::nullopt) {
334 std::map<Node, Node> parent{{start, start}};
335 std::deque<Node> work{start};
336 while (!work.empty() && !(goal && parent.count(*goal))) {
337 Node u = work.front();
338 work.pop_front();
339 for (Node v : successors(u))
340 if (parent.try_emplace(v, u).second)
341 work.push_back(v);
342 }
343 return parent;
344}

355// Minimum set cover by branch-and-bound: find the fewest `cubes` whose coverage
356// (cube.cov) unions to every match id still set in `uncovered`. `sel` is the
357// current partial pick; `bestSel`/`bestSize` hold the smallest full cover so
358// far and are updated in place.
359static void bnbMinCover(ArrayRef<CoverCube> cubes,
360 const llvm::SmallBitVector &uncovered,
361 SmallVectorImpl<int> &sel, int &bestSize,
362 SmallVectorImpl<int> &bestSel) {
363
364 // base case
365 if (uncovered.none()) {
366 if (static_cast<int>(sel.size()) < bestSize) {
367 bestSize = static_cast<int>(sel.size());
368 bestSel.assign(sel.begin(), sel.end());
369 }
370 return;
371 }
372
373 // bound: adding a cube would only tie the best
374 if (static_cast<int>(sel.size()) + 1 >= bestSize) {
375 return;
376 }
377
378 // most constrained variable: find the least-covered, uncovered matchId
379 int pick = -1;
380 int pickCount = static_cast<int>(cubes.size()) + 1;
381
382 for (int id = uncovered.find_first(); id != -1;
383 id = uncovered.find_next(id)) {
384 int c = 0;
385
386 for (const CoverCube &cb : cubes) {
387 if (cb.cov.test(id)) {
388 c++;
389 }
390 }
391
392 if (c < pickCount) {
393 pickCount = c;
394 pick = id;
395 }
396 }
397
398 // branch: try each cube that covers the chosen id, recursing on the rest.
399 for (int i = 0; i < static_cast<int>(cubes.size()); ++i) {
400 if (!cubes[i].cov.test(pick)) {
401 continue;
402 }
403
404 sel.push_back(i);
405 llvm::SmallBitVector next = uncovered;
406 next.reset(cubes[i].cov); // next &= ~cov
407 bnbMinCover(cubes, next, sel, bestSize, bestSel);
408 sel.pop_back();
409 }
410}

412static bool cubesIntersect(std::pair<int, int> a, std::pair<int, int> b) {
413 return ((a.second ^ b.second) & a.first & b.first) == 0;
414}

416// Cover `matchIds` (a group's packet ids) with the fewest (mask, value) rules
417// that match every specified id and no cube in `avoidCubes` (what the other
418// groups on the same slave port claim). A rule matches id x iff
419// (x & mask) == value; ids no cube claims are don't-cares, free to over-claim.
420//
421// Based on Quine-McCluskey:
422// 1. Enumerate prime implicants: maximal cubes (a cube is one (mask, value))
423// that match >=1 matchIds and no avoid cube.
424// 2. Pick the fewest of those cubes that cover all match ids (bnbMinCover).
425// 3. Tighten each chosen cube to the smallest enclosing cube of the ids it
426// took, so it claims no more don't-cares than necessary; the one-cube case
427// reduces to the old common-bits mask.
428static SmallVector<std::pair<int, int>>
429computeSubcubeCover(const SmallVector<int, 4> &matchIds,
430 ArrayRef<std::pair<int, int>> avoidCubes, int idBits) {
431
432 // id space and full mask, sized from the target's packet-id width.
433 const int numIds = 1 << idBits;
434 const int idMask = numIds - 1;
435
436 auto hitsAvoid = [&](int mask, int value) {
437 return llvm::any_of(avoidCubes, [&](std::pair<int, int> c) {
438 return cubesIntersect({mask, value}, c);
439 });
440 };
441
442 llvm::SmallBitVector matchMask(numIds);
443 for (int id : matchIds) {
444 matchMask.set(id);
445 }
446
447 // phase 1: enumerate prime implicants. small (3^idBits), so brute force.
448 SmallVector<CoverCube> primes;
449 for (int mask = 0; mask < numIds; ++mask) {
450 for (int value = 0; value < numIds; ++value) {
451 // valid cube := value covered by mask, avoids properly
452 if ((value & mask) != value) {
453 continue;
454 }
455 if (hitsAvoid(mask, value)) {
456 continue;
457 }
458
459 // record covering
460 llvm::SmallBitVector cov(numIds);
461 for (int id : matchIds) {
462 if ((id & mask) == value) {
463 cov.set(id);
464 }
465 }
466
467 if (cov.none()) {
468 continue;
469 }
470
471 // prime = maximal, i.e., no checked mask bit can be dropped without
472 // covering an avoidId.
473 bool isPrime = true;
474 for (int b = 0; b < idBits; ++b) {
475 if (!((mask >> b) & 1)) {
476 continue;
477 }
478
479 if (!hitsAvoid(mask & ~(1 << b), value & ~(1 << b))) {
480 isPrime = false;
481 break;
482 }
483 }
484
485 if (isPrime) {
486 primes.push_back({mask, value, std::move(cov)});
487 }
488 }
489 }
490
491 // phase 2: pick the fewest primes that cover every match id (bnb).
492 SmallVector<int> sel, bestSel;
493 int bestSize = static_cast<int>(matchIds.size()) + 1;
494
495 bnbMinCover(primes, matchMask, sel, bestSize, bestSel);
496
497 SmallVector<std::pair<int, int>> chosen;
498 for (int i : bestSel) {
499 chosen.push_back({primes[i].mask, primes[i].value});
500 }
501
502 if (chosen.empty()) {
503 // unreachable: a single-id cube always covers
504 for (int id : matchIds) {
505 chosen.push_back({idMask, id});
506 }
507 }
508
509 // phase 3: tighten each chosen cube to match only the ids it took.
510 // maximal -> over-claim don't-cares -> more mask bits narrow the cube.
511 //
512 // e.g., {2,4} avoid {0}:
513 // prime (mask 00010, value 00010): 2 & 00010 = 00010 == value -> hit
514 // 6 & 00010 = 00010 == value -> also hit
515 // tight (mask 11111, value 00010): 2 & 11111 = 00010 == value -> hit
516 // 6 & 11111 = 00110 != value -> dropped
517 // 0 & 11111 = 00000 != value -> avoided
518
519 SmallVector<SmallVector<int, 4>, 4> assigned(chosen.size());
520 for (int id : matchIds) {
521 for (int c = 0; c < static_cast<int>(chosen.size()); ++c) {
522 if ((id & chosen[c].first) == chosen[c].second) {
523 assigned[c].push_back(id);
524 break;
525 }
526 }
527 }
528
529 SmallVector<std::pair<int, int>> cover;
530 for (const SmallVector<int, 4> &ids : assigned) {
531 if (ids.empty()) {
532 continue;
533 }
534
535 int mask = idMask;
536 for (int i = 0; i < idBits; ++i) {
537 int bit = (ids.front() >> i) & 1;
538 for (int id : ids) {
539 if (((id >> i) & 1) != bit) {
540 mask &= ~(1 << i);
541 break;
542 }
543 }
544 }
545 cover.push_back({mask, ids.front() & mask});
546 }
547
548 return cover;
549}

551// A group's rules (see GroupClaims): the stated ones, then a cover of the
552// derived ids that avoids them and `avoid`, whatever else claims ids on the
553// port.
554static SmallVector<std::pair<int, int>>
555groupRules(ArrayRef<std::pair<int, int>> stated, ArrayRef<int> derived,
556 ArrayRef<std::pair<int, int>> avoid, int idBits) {
557 SmallVector<std::pair<int, int>> rules(stated);
558 if (derived.empty())
559 return rules;
560 SmallVector<std::pair<int, int>> derivedAvoid(avoid);
561 derivedAvoid.append(stated.begin(), stated.end());
562 llvm::append_range(rules, computeSubcubeCover(SmallVector<int, 4>(derived),
563 derivedAvoid, idBits));
564 return rules;
565}

585static uint64_t cubeIds(int mask, int value, int idBits) {
586 uint64_t ids = 0;
587 for (int id = 0; id < (1 << idBits); ++id)
588 if ((id & mask) == (value & mask))
589 ids |= uint64_t(1) << id;
590 return ids;

593// The fewest rules, at most `budget`, that send every id a group claims to
594// that group, where a packet takes the first rule it matches. A rule may then
595// claim ids of another group that an earlier rule already takes.
596static std::optional<SmallVector<PortRule>>
597orderedRules(ArrayRef<GroupClaims> groups,
598 ArrayRef<std::pair<int, int>> existing, int idBits,
599 size_t budget) {
600 if (idBits > 6)
601 return std::nullopt;
602 const int idMask = (1 << idBits) - 1;
603 SmallVector<uint64_t> claims;
604 for (const GroupClaims &group : groups) {
605 uint64_t ids = 0;
606 for (auto [mask, value] : group.stated)
607 ids |= cubeIds(mask, value, idBits);
608 for (int id : group.derived)
609 ids |= uint64_t(1) << id;
610 claims.push_back(ids);
611 }
612 uint64_t claimed = 0;
613 for (uint64_t ids : claims)
614 claimed |= ids;
615 uint64_t taken = 0;
616 for (auto [mask, value] : existing)
617 taken |= cubeIds(mask, value, idBits);
618 if (taken & claimed)
619 return std::nullopt;
620 SmallVector<uint64_t> cubes;
621 for (int mask = 0; mask <= idMask; ++mask)
622 for (int value = 0; value <= idMask; ++value)
623 if ((value & mask) == value)
624 cubes.push_back(cubeIds(mask, value, idBits));
625
626 // A rule is worth only the claimed ids it newly takes, all of one group, so
627 // the tightest cube around them serves, and fewer ids of the same group
628 // never serve better.
629 SmallVector<PortRule> rules;
630 std::set<std::pair<uint64_t, size_t>> dead;
631 auto search = [&](auto &self, uint64_t taken, size_t left) -> bool {
632 uint64_t open = claimed & ~taken;
633 size_t openGroups =
634 llvm::count_if(claims, [&](uint64_t ids) { return ids & open; });
635 if (openGroups == 0)
636 return true;
637 if (openGroups > left || dead.count({taken, left}))
638 return false;
639 SmallVector<std::pair<uint64_t, size_t>> moves;
640 for (uint64_t cube : cubes)
641 for (auto [g, ids] : llvm::enumerate(claims))
642 if ((cube & open) && (cube & open & ~ids) == 0 &&
643 !llvm::is_contained(moves, std::pair{cube & open, g}))
644 moves.push_back({cube & open, g});
645 llvm::stable_sort(moves, [](auto a, auto b) {
646 return llvm::popcount(a.first) > llvm::popcount(b.first);
647 });
648 SmallVector<std::pair<uint64_t, size_t>> tried;
649 for (auto [own, g] : moves) {
650 if (llvm::any_of(tried, [&, own = own, g = g](auto t) {
651 return t.second == g && (t.first & own) == own;
652 }))
653 continue;
654 tried.push_back({own, g});
655 int all = idMask, any = 0;
656 for (int id = 0; id <= idMask; ++id)
657 if ((own >> id) & 1) {
658 all &= id;
659 any |= id;
660 }
661 int mask = idMask & ~(all ^ any);
662 rules.push_back({mask, all, g});
663 if (self(self, taken | own, left - 1))
664 return true;
665 rules.pop_back();
666 }
667 dead.insert({taken, left});
668 return false;
669 };
670 for (size_t length = 1; length <= budget; ++length)
671 if (search(search, taken, length))
672 return rules;
673 return std::nullopt;
674}

676// The rules of a slave port, in slot order after the `existing` ones. Each
677// group takes its own cover, which claims no id another group claims, unless
678// those outnumber the free slots and first-match order fits them.
679static SmallVector<PortRule> portRules(ArrayRef<GroupClaims> groups,
680 ArrayRef<std::pair<int, int>> existing,
681 int idBits, size_t slots) {
682 const int idMask = (1 << idBits) - 1;
683 SmallVector<PortRule> rules;
684 for (auto [g, group] : llvm::enumerate(groups)) {
685 SmallVector<std::pair<int, int>> avoid(existing);
686 for (auto [o, other] : llvm::enumerate(groups)) {
687 if (o == g)
688 continue;
689 avoid.append(other.stated.begin(), other.stated.end());
690 for (int id : other.derived)
691 avoid.push_back({idMask, id});
692 }
693 for (auto [mask, value] :
694 groupRules(group.stated, group.derived, avoid, idBits))
695 rules.push_back({mask, value, g});
696 }
697 if (existing.size() + rules.size() <= slots || existing.size() >= slots)
698 return rules;
699 if (std::optional<SmallVector<PortRule>> ordered =
700 orderedRules(groups, existing, idBits, slots - existing.size()))
701 return *ordered;
702 return rules;
703}
704
705// The rules of a slave port, in slot order after the `existing` ones. The
706// overlay's groups take the rules they take alone, in the first slots (see
707// has_ctrl_pkt_overlay), and the other groups' rules follow.
708static SmallVector<PortRule>
709slavePortRules(ArrayRef<GroupClaims> groups,
710 ArrayRef<std::pair<int, int>> existing, int idBits,
711 size_t slots) {
712 SmallVector<std::pair<int, int>> taken(existing);
713 SmallVector<PortRule> rules;
714 for (bool overlay : {true, false}) {
715 SmallVector<size_t, 4> members;
716 SmallVector<GroupClaims> claims;
717 for (auto [g, group] : llvm::enumerate(groups))
718 if (group.overlay == overlay) {
719 members.push_back(g);
720 claims.push_back(group);
721 }
722 if (claims.empty())
723 continue;
724 for (const PortRule &r : portRules(claims, taken, idBits, slots))
725 rules.push_back({r.mask, r.value, members[r.group]});
726 for (const PortRule &r : rules)
727 if (groups[r.group].overlay == overlay)
728 taken.push_back({r.mask, r.value});
729 }
730 return rules;
731}

761/// Chooses arbiters for a switchbox's slave flows so that no two flows that
762/// can deadlock (`conflict`) and enter on different slave ports share one.
763/// A master port is tied to one arbiter, so every master port a slave flow
764/// leaves by takes that flow's arbiter; the master ports linked that way form
765/// a unit, and each master set in a unit takes an msel. A flow never takes
766/// an arbiter `excluded` rules out for it. Units are colored onto arbiters by
767/// backtracking, which is exact up to a step budget. On failure, fills
768/// `blocking` with the conflicting pairs that stood in the way; it stays empty
769/// when the master sets alone outnumber the free msels.
770std::optional<ArbiterPlan>
771planArbiters(const AIETargetModel &targetModel, ArrayRef<SlaveFlow> flows,
772 llvm::function_ref<bool(size_t, size_t)> conflict,
773 llvm::function_ref<bool(size_t, int)> excluded,
774 const std::set<int> &reservedAmsels,
775 SmallVectorImpl<std::pair<size_t, size_t>> &blocking) {
776 const int numArbiters = targetModel.getNumArbiters();
777 const int numMselsPerArbiter = targetModel.getNumMselsPerArbiter();
778 auto amselOf = [&](int arbiter, int msel) {
779 return arbiter + msel * numArbiters;
780 };
781
782 llvm::EquivalenceClasses<Port> tied = tiedMasters(flows);
783
784 struct Unit {
785 SmallVector<size_t, 4> flows;
786 SmallVector<SmallVector<Port, 4>, 2> masterSets;
787 bool isCtrlPkt = false;
788 };
789 std::vector<Unit> units;
790 std::map<Port, size_t> unitOf;
791 SmallVector<size_t, 8> flowUnit;
792 for (auto [i, f] : llvm::enumerate(flows)) {
793 auto [it, inserted] =
794 unitOf.try_emplace(tied.getLeaderValue(f.masters.front()), 0);
795 if (inserted) {
796 it->second = units.size();
797 units.emplace_back();
798 }
799 Unit &unit = units[it->second];
800 flowUnit.push_back(it->second);
801 unit.flows.push_back(i);
802 unit.isCtrlPkt |= f.isCtrlPkt;
803 if (!llvm::is_contained(unit.masterSets, f.masters))
804 unit.masterSets.push_back(f.masters);
805 }
806 for (Unit &unit : units) {
807 auto *ctrlEnd = std::stable_partition(
808 unit.masterSets.begin(), unit.masterSets.end(),
809 [&](const SmallVector<Port, 4> &masters) {
810 return llvm::any_of(unit.flows, [&](size_t f) {
811 return flows[f].isCtrlPkt && flows[f].masters == masters;
812 });
813 });
814 std::sort(unit.masterSets.begin(), ctrlEnd);
815 }
816
817 auto clash = [&](size_t a, size_t b) {
818 return flows[a].slave != flows[b].slave && conflict(a, b);
819 };
820 for (const Unit &unit : units)
821 for (auto [k, a] : llvm::enumerate(unit.flows))
822 for (size_t b : llvm::drop_begin(unit.flows, k + 1))
823 if (clash(a, b))
824 blocking.push_back({a, b});
825 if (!blocking.empty())
826 return std::nullopt;
827
828 std::vector<llvm::SmallDenseSet<size_t, 4>> neighbors(units.size());
829 SmallVector<std::pair<size_t, size_t>, 4> crossPairs;
830 for (size_t a = 0; a < flows.size(); a++)
831 for (size_t b = a + 1; b < flows.size(); b++)
832 if (flowUnit[a] != flowUnit[b] && clash(a, b)) {
833 neighbors[flowUnit[a]].insert(flowUnit[b]);
834 neighbors[flowUnit[b]].insert(flowUnit[a]);
835 crossPairs.push_back({a, b});
836 }
837
838 SmallVector<SmallVector<int, 4>, 6> freeMsels(numArbiters);
839 SmallVector<SmallVector<size_t, 4>, 6> excludedUnits(numArbiters);
840 for (int a = 0; a < numArbiters; a++) {
841 for (int m = 0; m < numMselsPerArbiter; m++)
842 if (!reservedAmsels.count(amselOf(a, m)))
843 freeMsels[a].push_back(m);
844 for (size_t f = 0; f < flows.size(); f++)
845 if (excluded(f, a) && !llvm::is_contained(excludedUnits[a], flowUnit[f]))
846 excludedUnits[a].push_back(flowUnit[f]);
847 }
848 size_t mostFree = 0;
849 for (const SmallVector<int, 4> &free : freeMsels)
850 mostFree = std::max(mostFree, free.size());
851 if (llvm::any_of(units, [&](const Unit &unit) {
852 return unit.masterSets.size() > mostFree;
853 }))
854 return std::nullopt;
855
856 // Most constrained first.
857 SmallVector<size_t, 8> order(units.size());
858 std::iota(order.begin(), order.end(), 0);
859 llvm::stable_sort(order, [&](size_t a, size_t b) {
860 return std::make_pair(neighbors[a].size(), units[a].masterSets.size()) >
861 std::make_pair(neighbors[b].size(), units[b].masterSets.size());
862 });
863
864 SmallVector<int, 8> arbiterOf(units.size(), -1);
865 SmallVector<size_t, 6> load(numArbiters, 0);
866 int steps = 0;
867 auto place = [&](auto &self, size_t depth) -> bool {
868 if (depth == order.size())
869 return true;
870 if (++steps > maxSearchSteps) {
871 LLVM_DEBUG(llvm::dbgs() << "Arbiter search gave up after "
872 << maxSearchSteps << " steps\n");
873 return false;
874 }
875 size_t u = order[depth];
876 SmallVector<int, 6> candidates(numArbiters);
877 std::iota(candidates.begin(), candidates.end(), 0);
878 if (units[u].isCtrlPkt)
879 std::reverse(candidates.begin(), candidates.end());
880 else
881 llvm::stable_sort(candidates, [&](int a, int b) {
882 return load[a] + numMselsPerArbiter - freeMsels[a].size() <
883 load[b] + numMselsPerArbiter - freeMsels[b].size();
884 });
885 std::set<std::pair<size_t, SmallVector<size_t, 4>>> triedEmpty;
886 for (int a : candidates) {
887 if (load[a] + units[u].masterSets.size() > freeMsels[a].size())
888 continue;
889 if (llvm::is_contained(excludedUnits[a], u) ||
890 llvm::any_of(neighbors[u],
891 [&](size_t v) { return arbiterOf[v] == a; }))
892 continue;
893 // Empty arbiters with as many free msels, ruled out for the same units,
894 // are interchangeable.
895 if (load[a] == 0 &&
896 !triedEmpty.insert({freeMsels[a].size(), excludedUnits[a]}).second)
897 continue;
898 arbiterOf[u] = a;
899 load[a] += units[u].masterSets.size();
900 if (self(self, depth + 1))
901 return true;
902 load[a] -= units[u].masterSets.size();
903 arbiterOf[u] = -1;
904 }
905 return false;
906 };
907 if (!place(place, 0)) {
908 blocking.append(crossPairs.begin(), crossPairs.end());
909 return std::nullopt;
910 }
911
912 // Control packets take the highest msels, as they do elsewhere, in master
913 // port order so the overlay's msels don't depend on the design's flows (see
914 // has_ctrl_pkt_overlay).
915 SmallVector<size_t, 8> mselOrder(units.size());
916 std::iota(mselOrder.begin(), mselOrder.end(), 0);
917 llvm::stable_sort(mselOrder, [&](size_t a, size_t b) {
918 if (!units[a].isCtrlPkt || !units[b].isCtrlPkt)
919 return units[a].isCtrlPkt > units[b].isCtrlPkt;
920 return units[a].masterSets.front() < units[b].masterSets.front();
921 });
922 ArbiterPlan plan;
923 SmallVector<size_t, 6> low(numArbiters, 0), high(numArbiters, 0);
924 for (size_t u : mselOrder) {
925 const Unit &unit = units[u];
926 int a = arbiterOf[u];
927 std::map<SmallVector<Port, 4>, int> setAmsel;
928 for (const SmallVector<Port, 4> &masters : unit.masterSets) {
929 int msel = unit.isCtrlPkt
930 ? freeMsels[a][freeMsels[a].size() - 1 - high[a]++]
931 : freeMsels[a][low[a]++];
932 setAmsel[masters] = amselOf(a, msel);
933 plan.amselMasters[amselOf(a, msel)] = masters;
934 }
935 for (size_t f : unit.flows)
936 plan.slaveAmsels[{flows[f].slave, flows[f].id}] =
937 setAmsel.at(flows[f].masters);
938 }
939 return plan;
940}

942/// The tiles other than `src` and `dst` that every route between them passes.
943SmallVector<TileID> cutTiles(const AIETargetModel &targetModel, TileID src,
944 TileID dst) {
945 auto neighbors = [&](TileID t) {
946 SmallVector<TileID, 4> next;
947 for (auto [bundle, dc, dr] : {std::tuple{WireBundle::North, 0, 1},
948 {WireBundle::South, 0, -1},
949 {WireBundle::East, 1, 0},
950 {WireBundle::West, -1, 0}}) {
951 TileID n{t.col + dc, t.row + dr};
952 if (n.col >= 0 && n.col < targetModel.columns() && n.row >= 0 &&
953 n.row < targetModel.rows() &&
954 targetModel.getNumDestSwitchboxConnections(t.col, t.row, bundle) > 0)
955 next.push_back(n);
956 }
957 return next;
958 };
959 auto reach = [&](std::optional<TileID> avoid) {
960 return breadthFirstParents(
961 src,
962 [&](TileID t) {
963 SmallVector<TileID, 4> next = neighbors(t);
964 llvm::erase(next, avoid);
965 return next;
966 },
967 std::optional{dst});
968 };
969 std::map<TileID, TileID> via = reach(std::nullopt);
970 if (!via.count(dst))
971 return {};
972 SmallVector<TileID> interior;
973 for (TileID t = via.at(dst); t != src; t = via.at(t))
974 interior.push_back(t);
975 llvm::erase_if(interior, [&](TileID t) { return reach(t).count(dst) > 0; });
976 return interior;
977}

999/// The largest set of `candidates` that must be kept apart pairwise
1000/// (StreamConflicts::mustSeparate). The search stops once it finds one larger
1001/// than `limit` or runs out of steps.
1002SmallVector<size_t, 8> largestApartSet(const StreamConflicts &conflicts,
1003 ArrayRef<size_t> candidates,
1004 size_t limit) {
1005 SmallVector<size_t, 8> clique, best;
1006 int steps = 0;
1007 auto grow = [&](auto &self, ArrayRef<size_t> cands) -> void {
1008 if (clique.size() > best.size())
1009 best = clique;
1010 for (auto [k, s] : llvm::enumerate(cands)) {
1011 if (best.size() > limit || ++steps > maxSearchSteps ||
1012 clique.size() + cands.size() - k <= best.size())
1013 return;
1014 SmallVector<size_t, 8> next;
1015 for (size_t t : cands.drop_front(k + 1))
1016 if (conflicts.mustSeparate(s, t))
1017 next.push_back(t);
1018 clique.push_back(s);
1019 self(self, next);
1020 clique.pop_back();
1021 }
1022 };
1023 grow(grow, candidates);
1024 return best;
1025}

1040/// Packet streams take an arbiter at the tile they end at whatever the
1041/// routing, and where `pinsHops` says hops cannot be circuit switched, or the
1042/// stream is prioritized, at the tile they start at and every tile each of
1043/// their routes passes too. Two that must be kept apart pass any tile on
1044/// different slave ports -- sharing one means they merged, unsafely, upstream
1045/// -- so a set of them that must be kept apart pairwise needs an arbiter
1046/// apiece. Says why no routing can work if some tile has such a set larger
1047/// than its free arbiters.
1048std::optional<std::string>
1049tooFewArbiters(DeviceOp device, const StreamConflicts &conflicts,
1050 llvm::function_ref<bool(TileID)> pinsHops,
1051 const PrioritizedPackets &prioritized) {
1052 const AIETargetModel &targetModel = device.getTargetModel();
1053 const int numArbiters = targetModel.getNumArbiters();
1054 const int numMselsPerArbiter = targetModel.getNumMselsPerArbiter();
1055 std::map<TileID, SmallVector<size_t, 8>> pinned;
1056 for (auto [i, s] : llvm::enumerate(conflicts.getRequestedStreams())) {
1057 if (!s.packetID)
1058 continue;
1059 pinned[s.dst.tile].push_back(i);
1060 if (s.src.tile == s.dst.tile)
1061 continue;
1062 bool circuitless = prioritized.contains(s);
1063 if (circuitless || pinsHops(s.src.tile))
1064 pinned[s.src.tile].push_back(i);
1065 for (TileID t : cutTiles(targetModel, s.src.tile, s.dst.tile))
1066 if (circuitless || pinsHops(t))
1067 pinned[t].push_back(i);
1068 }
1069 std::map<TileID, std::set<int>> reserved = configuredAmsels(device);
1070
1071 ArrayRef<RoutedStream> streams = conflicts.getStreams();
1072 for (const auto &[tileId, candidates] : pinned) {
1073 const std::set<int> &taken = reserved[tileId];
1074 size_t free = 0;
1075 for (int a = 0; a < numArbiters; a++)
1076 free += llvm::any_of(llvm::seq(numMselsPerArbiter), [&](int m) {
1077 return !taken.count(a + m * numArbiters);
1078 });
1079 // Streams that must be kept apart have distinct sources and destinations.
1080 std::set<std::pair<TileID, Port>> srcs, dsts;
1081 for (size_t s : candidates) {
1082 srcs.insert({streams[s].src.tile, streams[s].src.port});
1083 dsts.insert({streams[s].dst.tile, streams[s].dst.port});
1084 }
1085 if (std::min(srcs.size(), dsts.size()) <= free)
1086 continue;
1087 SmallVector<size_t, 8> best = largestApartSet(conflicts, candidates, free);
1088 if (best.size() <= free)
1089 continue;
1090
1091 std::string reason;
1092 llvm::raw_string_ostream os(reason);
1093 if (best.size() == 1) {
1094 os << "at tile (" << tileId.col << ", " << tileId.row << "), "
1095 << describeStream(streams[best[0]])
1096 << " takes an arbiter whatever the routing, but the switchbox has "
1097 "none free";
1098 return reason;
1099 }
1100 os << "at tile (" << tileId.col << ", " << tileId.row << "), no two of ";
1101 llvm::interleave(
1102 best, os, [&](size_t s) { os << describeStream(streams[s]); }, ", ");
1103 os << " can share an arbiter, and each takes one there whatever the "
1104 "routing, but the switchbox has "
1105 << free << " free. For example, " << conflicts.explain(best[0], best[1]);
1106 return reason;
1107 }
1108 return std::nullopt;
1109}

1119/// Where each stream leaves each tile on the trees in `pinnedTrees`, and the
1120/// tiles those trees pass.
1121std::pair<std::map<TileID, SmallVector<Leaving, 8>>, std::set<TileID>>
1122leavingPinnedTrees(const StreamConflicts &conflicts,
1123 const PacketTrees &pinnedTrees,
1124 const PrioritizedPackets &prioritized) {
1125 // Each hop of a pinned tree by the hop before it, unset where two lead to it.
1126 std::map<PathEndPoint, std::map<PathEndPoint, std::optional<PathEndPoint>>>
1127 preds;
1128 for (const auto &[src, hops] : pinnedTrees)
1129 for (const TreeHop &hop : hops) {
1130 auto [it, first] = preds[src].try_emplace(hop.to, hop.from);
1131 if (!first && it->second && !(*it->second == hop.from))
1132 it->second.reset();
1133 }
1134 std::map<TileID, SmallVector<Leaving, 8>> leaving;
1135 std::set<TileID> onTrees;
1136 for (auto [i, s] : llvm::enumerate(conflicts.getRequestedStreams())) {
1137 if (!s.packetID)
1138 continue;
1139 // Only the prioritized packets of a source take its tree.
1140 auto tree = prioritized.contains(s) ? preds.find({s.src.tile, s.src.port})
1141 : preds.end();
1142 SmallVector<std::pair<Port, PathEndPoint>, 8> hops;
1143 PathEndPoint at{s.dst.tile, s.dst.port}, src{s.src.tile, s.src.port};
1144 for (size_t n = 0;
1145 tree != preds.end() && !(at == src) && n <= tree->second.size(); n++) {
1146 auto pred = tree->second.find(at);
1147 if (pred == tree->second.end() || !pred->second)
1148 break;
1149 if (pred->second->coords == at.coords)
1150 hops.push_back({pred->second->port, at});
1151 at = *pred->second;
1152 }
1153 if (!(at == src)) {
1154 leaving[s.dst.tile].push_back({i, std::nullopt, s.dst.port});
1155 continue;
1156 }
1157 for (auto [slave, hop] : hops) {
1158 leaving[hop.coords].push_back({i, slave, hop.port});
1159 onTrees.insert(hop.coords);
1160 }
1161 }
1162 return {std::move(leaving), std::move(onTrees)};
1163}

1173/// Says why `a` and `b`, which must be kept apart, end up on one arbiter at
1174/// `tileId` through the visits in `chain`.
1175std::string explainSharedArbiter(TileID tileId, size_t a, size_t b,
1176 ArrayRef<ArbiterLink> chain,
1177 ArrayRef<Leaving> here,
1178 const StreamConflicts &conflicts) {
1179 ArrayRef<RoutedStream> streams = conflicts.getStreams();
1180 std::string reason;
1181 llvm::raw_string_ostream os(reason);
1182 os << "at tile (" << tileId.col << ", " << tileId.row
1183 << "), the routes prioritized flows (priority_route) keep put "
1184 << describeStream(streams[a]) << " and " << describeStream(streams[b])
1185 << " on one arbiter: " << describeStream(streams[a]);
1186 for (auto [k, link] : llvm::enumerate(chain))
1187 os << (k == 0 ? "" : ", which")
1188 << (link.rule ? " takes one packet rule on " : " leaves by ")
1189 << describePort(link.port) << " with "
1190 << describeStream(streams[here[link.visit].stream]);
1191 os << ", and a master port or packet rule takes one arbiter. "
1192 << conflicts.explain(a, b);
1193 return reason;
1194}

1196/// Says why no routing can work if two streams that must be kept apart leave
1197/// `tileId` by master ports that streams leaving by one each join, which puts
1198/// them on one arbiter.
1199std::optional<std::string> sharedArbiterAt(TileID tileId,
1200 ArrayRef<Leaving> here,
1201 const StreamConflicts &conflicts) {
1202 ArrayRef<RoutedStream> streams = conflicts.getStreams();
1203 // A stream packet switched at a master port takes its one arbiter, and
1204 // packets with one id on a slave port take one packet rule's arbiter.
1205 // The visits on one arbiter with each. A stream passing a tile twice takes
1206 // an arbiter on each visit, so the nodes are visits.
1207 std::map<size_t, SmallVector<ArbiterLink, 4>> with;
1208 for (auto [x, l] : llvm::enumerate(here))
1209 for (auto [y, m] : llvm::enumerate(here)) {
1210 if (x == y)
1211 continue;
1212 if (l.master == m.master)
1213 with[x].push_back({y, l.master, false});
1214 else if (l.slave && m.slave && *l.slave == *m.slave &&
1215 streams[l.stream].packetID == streams[m.stream].packetID)
1216 with[x].push_back({y, *l.slave, true});
1217 }
1218 for (size_t x : llvm::make_first_range(with)) {
1219 // The chain of visits that joins `x` to each visit it reaches.
1220 std::map<size_t, size_t> via = breadthFirstParents(x, [&](size_t u) {
1221 return llvm::map_range(
1222 with.at(u), [](const ArbiterLink &link) { return link.visit; });
1223 });
1224 size_t a = here[x].stream;
1225 for (size_t y : llvm::make_first_range(via)) {
1226 size_t b = here[y].stream;
1227 if (b <= a || !conflicts.mustSeparate(a, b))
1228 continue;
1229 SmallVector<ArbiterLink, 4> chain;
1230 for (size_t v = y; v != x; v = via.at(v))
1231 chain.push_back(
1232 *llvm::find_if(with.at(via.at(v)), [&](const ArbiterLink &link) {
1233 return link.visit == v;
1234 }));
1235 std::reverse(chain.begin(), chain.end());
1236 return explainSharedArbiter(tileId, a, b, chain, here, conflicts);
1237 }
1238 }
1239 return std::nullopt;
1240}

1242/// A RoutingFailure if no routing can work because the arbiters packet streams
1243/// take whatever the routing do not fit: too few free ones at a tile, or, as a
1244/// PinnedRoutingFailure, two streams that must be kept apart put on one by the
1245/// trees in `pinnedTrees`, which prioritized sources keep.
1246llvm::Error unroutableArbiters(DeviceOp device,
1247 const StreamConflicts &conflicts,
1248 llvm::function_ref<bool(TileID)> pinsHops,
1249 const PacketTrees &pinnedTrees) {
1250 PrioritizedPackets prioritized = prioritizedPackets(device);
1251 if (std::optional<std::string> reason =
1252 tooFewArbiters(device, conflicts, pinsHops, prioritized))
1253 return llvm::make_error<RoutingFailure>(std::move(*reason));
1254 if (pinnedTrees.empty())
1255 return llvm::Error::success();
1256 auto [leaving, onTrees] =
1257 leavingPinnedTrees(conflicts, pinnedTrees, prioritized);
1258 for (TileID tileId : onTrees)
1259 if (std::optional<std::string> reason =
1260 sharedArbiterAt(tileId, leaving[tileId], conflicts))
1261 return llvm::make_error<PinnedRoutingFailure>(std::move(*reason));
1262 return llvm::Error::success();
1263}

1492void PacketFlowRouting::collectFlows() {
1493 for (auto [i, s] : llvm::enumerate(conflicts.getStreams()))
1494 if (s.packetID)
1495 packetStreamIndex[{{s.src.tile, s.src.port},
1496 {s.dst.tile, s.dst.port},
1497 *s.packetID}]
1498 .push_back(i);
1499 for (const RoutedStream &s : conflicts.getStreams())
1500 routes.push_back(s.hops);
1501
1502 if (routeCircuit)
1503 for (FlowOp flow : device.getOps<FlowOp>())
1504 circuitSources.insert(sourceOf(flow));
1505
1506 for (PacketFlowOp pktFlowOp : device.getOps<PacketFlowOp>()) {
1507 Region &r = pktFlowOp.getPorts();
1508 Block &b = r.front();
1509 int flowID = pktFlowOp.IDInt();
1510 SmallVector<std::pair<TileID, Port>, 4> sources;
1511
1512 // Pass 1: collect all sources (order-independent; supports fan-in).
1513 for (Operation &Op : b.getOperations()) {
1514 if (auto pktSource = dyn_cast<PacketSourceOp>(Op)) {
1515 auto srcTile = cast<TileOp>(pktSource.getTile().getDefiningOp());
1516 sources.push_back(
1517 {{srcTile.colIndex(), srcTile.rowIndex()}, pktSource.port()});
1518 }
1519 }
1520 // Pass 2: lower each (source, destination) pair so fan-in flows lay
1521 // down switchbox connections for every source, not just the last one.
1522 for (Operation &Op : b.getOperations()) {
1523 auto pktDest = dyn_cast<PacketDestOp>(Op);
1524 if (!pktDest)
1525 continue;
1526 auto destTile = cast<TileOp>(pktDest.getTile().getDefiningOp());
1527 Port destPort = pktDest.port();
1528 TileID destCoords = {destTile.colIndex(), destTile.rowIndex()};
1529 // Assign "keep_pkt_header flag"
1530 auto keep = pktFlowOp.getKeepPktHeader();
1531 keepPktHeaderAttr[{destTile.getTileID(), destPort}] =
1532 keep ? BoolAttr::get(Op.getContext(), *keep) : nullptr;
1533
1534 for (auto &[srcCoords, srcPort] : sources) {
1535 TileID srcSB = {srcCoords.col, srcCoords.row};
1536 PathEndPoint srcPoint = {srcSB, srcPort};
1537 if (circuitSources.count(srcPoint))
1538 continue;
1539 const SwitchSettings &settings = settingsOf(srcPoint, flowID);
1540 // The requested streams this flow asks for here; with none, the
1541 // stream the design already routes.
1542 SmallVector<size_t, 1> flowStreams;
1543 if (auto stream = packetStreamIndex.find(
1544 {srcPoint, {destCoords, destPort}, flowID});
1545 stream != packetStreamIndex.end()) {
1546 int mask = static_cast<int>(pktFlowOp.getMask().value_or(~0));
1547 for (size_t i : stream->second)
1548 if (i < conflicts.getRequestedStreams().size() &&
1549 conflicts.getStreams()[i].packetMask == mask)
1550 flowStreams.push_back(i);
1551 if (flowStreams.empty())
1552 flowStreams.push_back(stream->second.front());
1553 }
1554 for (size_t stream : flowStreams) {
1555 SmallVector<StreamHop, 8> &hops = routes[stream];
1556 hops.clear();
1557 // A route may pass a switchbox more than once, but no connection.
1558 size_t connections = 0;
1559 for (const auto &[_, setting] : settings)
1560 connections += setting.srcs.size();
1561 std::optional<std::pair<TileID, Port>> at{{srcSB, srcPort}};
1562 while (at && hops.size() <= connections) {
1563 auto [tile, input] = *at;
1564 hops.push_back({tile, input, std::nullopt});
1565 at.reset();
1566 auto setting = settings.find(tile);
1567 if (setting == settings.end())
1568 break;
1569 for (auto [src, dest] :
1570 llvm::zip(setting->second.srcs, setting->second.dsts))
1571 if (src == input && !(tile == destCoords && dest == destPort) &&
1572 findPathToDest(settings, tile, dest, destCoords, destPort)) {
1573 at = linkedInput(tile, dest);
1574 break;
1575 }
1576 }
1577 }
1578 // Track whether the source's own switch connection survives routing.
1579 bool srcRouted = false;
1580 // add connections for all the Switchboxes in SwitchSettings
1581 for (const auto &[curr, setting] : settings) {
1582 assert(setting.srcs.size() == setting.dsts.size());
1583 TileID currTile = {curr.col, curr.row};
1584 for (size_t i = 0; i < setting.srcs.size(); i++) {
1585 Port src = setting.srcs[i];
1586 Port dest = setting.dsts[i];
1587 // reject false broadcast
1588 if (!findPathToDest(settings, currTile, dest, destCoords, destPort))
1589 continue;
1590 if (currTile == srcSB && src.bundle == srcPort.bundle &&
1591 src.channel == srcPort.channel)
1592 srcRouted = true;
1593 Connect connect = {{src.bundle, src.channel},
1594 {dest.bundle, dest.channel}};
1595 if (std::find(
1596 switchboxes[currTile].begin(), switchboxes[currTile].end(),
1597 std::pair{connect, flowID}) == switchboxes[currTile].end())
1598 switchboxes[currTile].push_back({connect, flowID});
1599 // Keyed by slave port as well as ID, since IDs are reused across
1600 // unrelated flows.
1601 PhysPort slavePort = {currTile, {src.bundle, src.channel}};
1602 std::pair<PhysPort, int> slaveFlow = {slavePort, flowID};
1603 for (size_t stream : flowStreams)
1604 if (!llvm::is_contained(slaveFlowStreams[slaveFlow], stream))
1605 slaveFlowStreams[slaveFlow].push_back(stream);
1606 slaveFlowSources[slaveFlow][srcPoint].insert(dest);
1607 if (std::optional<uint8_t> mask = pktFlowOp.getMask()) {
1608 pinnedMasks[{{currTile, {src.bundle, src.channel}}, flowID}]
1609 .insert(*mask);
1610 }
1611 if (pktFlowOp.getPriorityRoute().value_or(false)) {
1612 bool overlay = !reloadConfigures(pktFlowOp);
1613 if (overlay)
1614 markedFlows.insert(slaveFlow);
1615 if (prioritize || pinned.trees.count(srcPoint)) {
1616 prioritizedFlows.insert(slaveFlow);
1617 if (overlay)
1618 overlayFlows.insert(slaveFlow);
1619 if (slavePort == PhysPort{srcSB, srcPort})
1620 prioritizedSourcePorts.insert(slavePort);
1621 }
1622 }
1623 }
1624 }
1625 if (!srcRouted && !incomplete)
1626 incomplete = {
1627 pktFlowOp.getLoc(),
1628 "packet flow source " + describeTilePort(srcCoords, srcPort) +
1629 " could not be routed to destination " +
1630 describeTilePort(destCoords, destPort) +
1631 "; the pathfinder produced an incomplete routing for this "
1632 "placement."};
1633 }
1634 }
1635 }
1636}

1663void PacketFlowRouting::findCircuitHops() {
1664 // Seed the reserved set from the packet-switch configuration the switchboxes
1665 // already carry, so this run allocates around it instead of over it.
1666 reservedAmsels = configuredAmsels(device);
1667
1668 // Ports the input's own aie.switchbox ops already drive, circuit or packet.
1669 std::set<PhysPort> claimedSlaves, claimedMasters;
1670 for (auto swbox : device.getOps<SwitchboxOp>()) {
1671 TileID tileId = swbox.getTileOp().getTileID();
1672 Region &ops = swbox.getConnections();
1673 for (auto connect : ops.getOps<ConnectOp>()) {
1674 claimedSlaves.insert({tileId, connect.sourcePort()});
1675 claimedMasters.insert({tileId, connect.destPort()});
1676 }
1677 for (auto rules : ops.getOps<PacketRulesOp>())
1678 claimedSlaves.insert({tileId, rules.sourcePort()});
1679 for (auto masterSet : ops.getOps<MasterSetOp>())
1680 claimedMasters.insert({tileId, masterSet.destPort()});
1681 }
1682 // Circuits not yet lowered claim their ports all the same.
1683 if (routeCircuit)
1684 for (FlowOp flow : device.getOps<FlowOp>())
1685 for (const auto &[tileId, setting] : settingsOf(sourceOf(flow))) {
1686 for (Port p : setting.srcs)
1687 claimedSlaves.insert({tileId, p});
1688 for (Port p : setting.dsts)
1689 claimedMasters.insert({tileId, p});
1690 }
1691 for (auto &[tileId, connects] : switchboxes) {
1692 if (!circuitSwitchHops ||
1693 targetModel.isShimNOCorPLTile(tileId.col, tileId.row))
1694 continue;
1695 std::set<Port> masters;
1696 for (const auto &[conn, flowID] : connects)
1697 masters.insert(conn.dst);
1698 SmallVector<Port, 8> slaves;
1699 for (const auto &[conn, flowID] : connects)
1700 if (!llvm::is_contained(slaves, conn.src))
1701 slaves.push_back(conn.src);
1702 const std::set<int> &reserved = reservedAmsels[tileId];
1703 size_t freeArbiters = llvm::count_if(llvm::seq(0, numArbiters), [&](int a) {
1704 return llvm::none_of(llvm::seq(0, numMselsPerArbiter), [&](int m) {
1705 return reserved.count(a + m * numArbiters);
1706 });
1707 });
1708 for (Port slave : slaves) {
1709 if (masters.size() <= freeArbiters)
1710 break;
1711 std::set<Port> reached;
1712 std::map<int, std::set<Port>> reachedByID;
1713 for (const auto &[conn, flowID] : connects)
1714 if (conn.src == slave) {
1715 reached.insert(conn.dst);
1716 reachedByID[flowID].insert(conn.dst);
1717 }
1718 bool exclusive =
1719 !claimedSlaves.count({tileId, slave}) &&
1720 llvm::all_of(reached,
1721 [&, tileId = tileId](Port m) {
1722 return isDirectional(m.bundle) &&
1723 !claimedMasters.count({tileId, m});
1724 }) &&
1725 llvm::all_of(
1726 reachedByID,
1727 [&](const auto &ids) { return ids.second == reached; }) &&
1728 llvm::all_of(connects, [&, tileId = tileId](const auto &entry) {
1729 const auto &[other, otherID] = entry;
1730 if (!reached.count(other.dst))
1731 return true;
1732 return other.src == slave &&
1733 !prioritizedFlows.contains({{tileId, slave}, otherID});
1734 });
1735 if (!exclusive)
1736 continue;
1737 for (Port m : reached) {
1738 circuitHops[tileId].push_back({slave, m});
1739 masters.erase(m);
1740 }
1741 llvm::erase_if(connects, [&](const auto &entry) {
1742 return entry.first.src == slave;
1743 });
1744 }
1745 }

1803std::optional<ArbiterPlan> PacketFlowRouting::planTile(
1804 TileID tileId, SmallVectorImpl<std::pair<size_t, size_t>> &blocking) {
1805 ArrayRef<SlaveFlow> flows = tileFlows.at(tileId);
1806 auto key = [&](size_t f) { return FlowKey{flows[f].slave, flows[f].id}; };
1807 auto conflict = [&](size_t a, size_t b) {
1808 return apart.count(
1809 {tileId, std::min(key(a), key(b)), std::max(key(a), key(b))}) ||
1810 conflictingStreams(tileId, flows[a], flows[b]);
1811 };
1812 auto excluded = [&](size_t f, int arbiter) {
1813 return offArbiter.count({tileId, key(f), arbiter}) > 0;
1814 };
1815 SmallVector<size_t, 8> overlay, others;
1816 for (auto [i, f] : llvm::enumerate(flows))
1817 (f.isCtrlPkt ? overlay : others).push_back(i);
1818 if (overlay.empty() || (others.empty() && pinned.rules.empty()))
1819 return planArbiters(targetModel, flows, conflict, excluded,
1820 reservedAmsels[tileId], blocking);
1821
1822 auto subset = [&](ArrayRef<size_t> members) {
1823 SmallVector<SlaveFlow, 8> sub;
1824 for (size_t m : members)
1825 sub.push_back(flows[m]);
1826 return sub;
1827 };
1828 // The pinned amsels go first where they fit and the rest plan around them
1829 // (see has_ctrl_pkt_overlay); without a reload, ones the design rules out
1830 // are planned anew.
1831 auto keepPinned = [&]() -> std::optional<ArbiterPlan> {
1832 std::string reason;
1833 llvm::raw_string_ostream os(reason);
1834 auto unkept = [&](size_t f) -> llvm::raw_ostream & {
1835 return os << "packet flow " << flows[f].id << " into "
1836 << describePort(flows[f].slave) << ' ';
1837 };
1838 ArbiterPlan kept;
1839 for (size_t f : overlay) {
1840 // The overlay's rules name a shim DMA by the mux port it enters by.
1841 Port slave = flows[f].slave;
1842 if (slave.bundle == WireBundle::DMA &&
1843 targetModel.isShimNOCTile(tileId.col, tileId.row))
1844 slave = {WireBundle::South, shimMuxChannelFrom(slave)};
1845 auto rules = pinned.rules.find({tileId, slave});
1846 const std::tuple<int, int, int> *rule = nullptr;
1847 if (rules != pinned.rules.end())
1848 rule = llvm::find_if(rules->second, [&](const auto &r) {
1849 auto [mask, value, amsel] = r;
1850 return (flows[f].id & mask) == (value & mask);
1851 });
1852 if (!rule || rule == rules->second.end()) {
1853 unkept(f) << "takes no packet rule there alone.";
1854 } else if (int amsel = std::get<2>(*rule);
1855 excluded(f, amsel % numArbiters)) {
1856 unkept(f) << "takes arbiter " << amsel % numArbiters
1857 << " alone, and here can deadlock holding it.";
1858 } else if (reservedAmsels[tileId].count(amsel)) {
1859 unkept(f) << "takes amsel<" << amsel % numArbiters << "> ("
1860 << amsel / numArbiters
1861 << ") alone, and the design's own switchbox takes it.";
1862 } else if (kept.amselMasters.try_emplace(amsel, flows[f].masters)
1863 .first->second != flows[f].masters) {
1864 unkept(f) << "takes amsel<" << amsel % numArbiters << "> ("
1865 << amsel / numArbiters
1866 << ") alone, to other master ports than here.";
1867 } else {
1868 kept.slaveAmsels[key(f)] = amsel;
1869 continue;
1870 }
1871 unkeptOverlay[tileId] = std::move(reason);
1872 return std::nullopt;
1873 }
1874 for (auto [k, a] : llvm::enumerate(overlay))
1875 for (size_t b : llvm::drop_begin(overlay, k + 1))
1876 if (kept.slaveAmsels.at(key(a)) % numArbiters ==
1877 kept.slaveAmsels.at(key(b)) % numArbiters &&
1878 flows[a].slave != flows[b].slave && conflict(a, b)) {
1879 unkept(a) << "and packet flow " << flows[b].id << " into "
1880 << describePort(flows[b].slave)
1881 << " share an arbiter alone, and here can deadlock on it.";
1882 unkeptOverlay[tileId] = std::move(reason);
1883 return std::nullopt;
1884 }
1885 return kept;
1886 };
1887 SmallVector<std::pair<size_t, size_t>, 4> stuck;
1888 std::optional<ArbiterPlan> plan;
1889 if (!pinned.rules.empty())
1890 plan = keepPinned();
1891 if (!plan && keepOverlay)
1892 return std::nullopt;
1893 if (!plan) {
1894 plan = planArbiters(
1895 targetModel, subset(overlay),
1896 [&](size_t a, size_t b) { return conflict(overlay[a], overlay[b]); },
1897 [&](size_t f, int arbiter) { return excluded(overlay[f], arbiter); },
1898 reservedAmsels[tileId], stuck);
1899 if (!plan) {
1900 for (auto [a, b] : stuck)
1901 blocking.push_back({overlay[a], overlay[b]});
1902 return std::nullopt;
1903 }
1904 }
1905 if (others.empty())
1906 return plan;
1907
1908 // A flow that leaves by the overlay's master ports alone takes the amsel
1909 // that selects them (checkRules lets no other flow reach them).
1910 auto around = [&](ArbiterPlan plan, ArrayRef<size_t> planned,
1911 ArrayRef<size_t> pending,
1912 SmallVectorImpl<std::pair<size_t, size_t>> &blocking)
1913 -> std::optional<ArbiterPlan> {
1914 auto arbiterOf = [&](size_t f) {
1915 return plan.slaveAmsels.at(key(f)) % numArbiters;
1916 };
1917 SmallVector<size_t, 8> placed(planned), rest;
1918 for (size_t f : pending) {
1919 auto same = llvm::find_if(plan.amselMasters, [&](const auto &entry) {
1920 return entry.second == flows[f].masters;
1921 });
1922 if (same == plan.amselMasters.end()) {
1923 rest.push_back(f);
1924 continue;
1925 }
1926 for (size_t g : placed)
1927 if (arbiterOf(g) == same->first % numArbiters &&
1928 flows[g].slave != flows[f].slave && conflict(f, g))
1929 blocking.push_back({g, f});
1930 if (excluded(f, same->first % numArbiters))
1931 blocking.push_back({planned.front(), f});
1932 plan.slaveAmsels[key(f)] = same->first;
1933 placed.push_back(f);
1934 }
1935 if (!blocking.empty())
1936 return std::nullopt;
1937 if (rest.empty())
1938 return plan;
1939
1940 // A master port is tied to one arbiter, so a flow leaving by a master port
1941 // the plan has tied takes that arbiter.
1942 llvm::EquivalenceClasses<Port> tied = tiedMasters(flows);
1943 std::map<Port, std::set<int>> tiedArbiters;
1944 std::set<int> reserved = reservedAmsels[tileId];
1945 for (const auto &[amsel, masters] : plan.amselMasters) {
1946 reserved.insert(amsel);
1947 tiedArbiters[tied.getLeaderValue(masters.front())].insert(amsel %
1948 numArbiters);
1949 }
1950 SmallVector<std::pair<size_t, size_t>, 4> stuck;
1951 std::optional<ArbiterPlan> rested = planArbiters(
1952 targetModel, subset(rest),
1953 [&](size_t a, size_t b) { return conflict(rest[a], rest[b]); },
1954 [&](size_t f, int arbiter) {
1955 auto arbiters = tiedArbiters.find(
1956 tied.getLeaderValue(flows[rest[f]].masters.front()));
1957 return excluded(rest[f], arbiter) ||
1958 (arbiters != tiedArbiters.end() &&
1959 (arbiters->second.size() > 1 ||
1960 !arbiters->second.count(arbiter))) ||
1961 llvm::any_of(placed, [&](size_t g) {
1962 return arbiterOf(g) == arbiter &&
1963 flows[g].slave != flows[rest[f]].slave &&
1964 conflict(rest[f], g);
1965 });
1966 },
1967 reserved, stuck);
1968 if (!rested) {
1969 for (auto [a, b] : stuck)
1970 blocking.push_back({rest[a], rest[b]});
1971 for (size_t f : rest)
1972 for (size_t g : placed)
1973 if (flows[g].slave != flows[f].slave && conflict(f, g))
1974 blocking.push_back({g, f});
1975 return std::nullopt;
1976 }
1977 plan.slaveAmsels.insert(rested->slaveAmsels.begin(),
1978 rested->slaveAmsels.end());
1979 plan.amselMasters.insert(rested->amselMasters.begin(),
1980 rested->amselMasters.end());
1981 return plan;
1982 };
1983 SmallVector<std::pair<size_t, size_t>, 4> first;
1984 if (std::optional<ArbiterPlan> placed = around(*plan, overlay, others, first))
1985 return placed;
1986 if (keepOverlay) {
1987 blocking.append(first.begin(), first.end());
1988 return std::nullopt;
1989 }
1990
1991 // The flows that take the overlay's arbiters may conflict where the
1992 // overlay's flows do not, so they are planned with them.
1993 llvm::EquivalenceClasses<Port> tied = tiedMasters(flows);
1994 SmallVector<size_t, 8> joint(overlay), rest;
1995 for (size_t f : others) {
1996 bool tiedToOverlay = llvm::any_of(overlay, [&](size_t g) {
1997 return tied.isEquivalent(flows[g].masters.front(),
1998 flows[f].masters.front());
1999 });
2000 (tiedToOverlay ? joint : rest).push_back(f);
2001 }
2002 stuck.clear();
2003 std::optional<ArbiterPlan> together;
2004 if (joint.size() > overlay.size())
2005 together = planArbiters(
2006 targetModel, subset(joint),
2007 [&](size_t a, size_t b) { return conflict(joint[a], joint[b]); },
2008 [&](size_t f, int arbiter) { return excluded(joint[f], arbiter); },
2009 reservedAmsels[tileId], stuck);
2010 if (!together) {
2011 blocking.append(first.begin(), first.end());
2012 return std::nullopt;
2013 }
2014 return around(*together, joint, rest, blocking);
2015}

2051void PacketFlowRouting::splitFlow(TileID tileId, FlowKey key,
2052 ArrayRef<FlowKey> partners) {
2053 auto flows = tileSlaveFlows.find(tileId);
2054 if (flows == tileSlaveFlows.end())
2055 return;
2056 auto self = flows->second.find(key);
2057 if (self == flows->second.end())
2058 return;
2059 auto onlySource = [&](FlowKey k) -> std::optional<PathEndPoint> {
2060 auto sources = slaveFlowSources.find({{tileId, k.first}, k.second});
2061 if (sources == slaveFlowSources.end() || sources->second.size() != 1)
2062 return std::nullopt;
2063 return sources->second.begin()->first;
2064 };
2065 std::optional<PathEndPoint> src = onlySource(key);
2066 if (!src)
2067 return;
2068 auto sharesMaster = [&](const auto &f, const std::set<Port> &ports) {
2069 return llvm::any_of(f.masters, [&](Port m) { return ports.count(m); });
2070 };
2071 std::set<Port> own(self->second.masters.begin(), self->second.masters.end());
2072 std::set<FlowKey> siblings;
2073 for (const auto &[other, f] : flows->second)
2074 if (other != key && other.first == key.first && sharesMaster(f, own) &&
2075 onlySource(other) == src)
2076 siblings.insert(other);
2077 // The masters tied to `partner`'s arbiter, not through `key`, nor through
2078 // its siblings if `skipSiblings`.
2079 auto tied = [&](FlowKey partner, bool skipSiblings) {
2080 llvm::EquivalenceClasses<Port> classes =
2081 tiedMasters(llvm::make_filter_range(
2082 llvm::make_second_range(flows->second), [&](const SlaveFlow &f) {
2083 FlowKey other{f.slave, f.id};
2084 return other != key && !(skipSiblings && siblings.count(other));
2085 }));
2086 auto unit = classes.members(flows->second.at(partner).masters.front());
2087 return std::set<Port>(unit.begin(), unit.end());
2088 };
2089 auto split = [&](FlowKey other,
2090 std::optional<PathEndPoint> apart = std::nullopt) {
2091 int a = key.second, b = other.second;
2092 if (!apart && a > b)
2093 std::swap(a, b);
2094 hazardSplits.insert({*src, tileId, a, b, apart});
2095 };
2096 for (FlowKey partner : partners) {
2097 if (partner == key || !flows->second.count(partner))
2098 continue;
2099 if (siblings.count(partner)) {
2100 split(partner);
2101 continue;
2102 }
2103 std::set<Port> partnerUnit = tied(partner, true);
2104 if (sharesMaster(self->second, partnerUnit))
2105 continue;
2106 std::set<Port> unit = tied(partner, false);
2107 for (FlowKey other : siblings) {
2108 const SlaveFlow &f = flows->second.find(other)->second;
2109 if (!sharesMaster(f, unit))
2110 continue;
2111 // Where the sibling only ties `key` to the partner's arbiter by
2112 // destinations on this tile, those can reach it by a slave port of
2113 // their own, and `key` keeps sharing the rest of the tree.
2114 SmallVector<PathEndPoint, 2> apart;
2115 for (Port m : f.masters)
2116 if (partnerUnit.count(m))
2117 apart.push_back({tileId, m});
2118 if (PathEndPoint{tileId, key.first} == *src || apart.empty() ||
2119 !llvm::all_of(apart, [&](const PathEndPoint &p) {
2120 return packetStreamIndex.count({*src, p, other.second});
2121 })) {
2122 split(other);
2123 continue;
2124 }
2125 for (const PathEndPoint &p : apart)
2126 split(other, p);
2127 }
2128 }
2129}

2163void PacketFlowRouting::checkRules() {
2164 // A slave port holds a few packet rules, and how many its flows need depends
2165 // on which ids leave by the same master ports.
2166 std::map<PhysPort, SmallVector<std::pair<int, int>>> existingCubes;
2167 for (auto swbox : device.getOps<SwitchboxOp>())
2168 for (auto rules : swbox.getConnections().getOps<PacketRulesOp>())
2169 for (auto rule : rules.getRules().getOps<PacketRuleOp>())
2170 existingCubes[{swbox.getTileOp().getTileID(), rules.sourcePort()}]
2171 .push_back({rule.maskInt(), rule.valueInt()});
2172 struct RuleGroup {
2173 SmallVector<std::pair<int, int>, 4> stated;
2174 SmallVector<int, 4> derived;
2175 };
2176 for (const auto &[slaveFlow, sources] : slaveFlowSources) {
2177 const auto &[first, firstMasters] = *sources.begin();
2178 auto other = llvm::find_if(
2179 sources, [&, &firstMasters = firstMasters](const auto &source) {
2180 return source.second != firstMasters;
2181 });
2182 if (other == sources.end())
2183 continue;
2184 const auto &[slavePort, id] = slaveFlow;
2185 TileID tileId = slavePort.first;
2186 moveFlow(tileId, {slavePort.second, id});
2187 if (planFailure)
2188 continue;
2189 const PathEndPoint &second = other->first;
2190 planFailure = llvm::formatv(
2191 "at tile ({0}, {1}), packets with id {2} from {3} and {4} enter on {5} "
2192 "and leave by different ports; a switchbox routes on the id alone, so "
2193 "each source's packets would also go where the other's do.",
2194 tileId.col, tileId.row, id, describeTilePort(first.coords, first.port),
2195 describeTilePort(second.coords, second.port),
2196 describePort(slavePort.second));
2197 }
2198 for (const auto &[tileId, byFlow] : tileSlaveFlows) {
2199 // The packet rules the flows entering on `slave` add, without the packets
2200 // `without` sends to a master port if set, or only the control overlay's.
2201 // `groupMasters`, if set, gets the master ports each rule group selects.
2202 auto newRules =
2203 [&, tileId = tileId, &byFlow = byFlow](
2204 Port slave, std::optional<std::pair<PathEndPoint, Port>> without,
2205 bool overlayOnly,
2206 SmallVector<SmallVector<Port, 4>> *groupMasters = nullptr) {
2207 std::map<std::pair<bool, SmallVector<Port, 4>>, RuleGroup> groups;
2208 for (const auto &[key, f] : byFlow) {
2209 if (f.slave != slave || (overlayOnly && !f.isCtrlPkt))
2210 continue;
2211 SmallVector<Port, 4> masters = f.masters;
2212 auto sources = slaveFlowSources.find({{tileId, slave}, f.id});
2213 if (without && sources != slaveFlowSources.end()) {
2214 std::set<Port> kept;
2215 for (const auto &[src, ports] : sources->second)
2216 for (Port m : ports)
2217 if (!(src == without->first) || m != without->second)
2218 kept.insert(m);
2219 masters.assign(kept.begin(), kept.end());
2220 }
2221 if (masters.empty())
2222 continue;
2223 RuleGroup &group = groups[{f.isCtrlPkt, masters}];
2224 auto cubes = statedCubes({{tileId, f.slave}, f.id}, idMask);
2225 if (cubes.empty())
2226 group.derived.push_back(f.id);
2227 for (auto cube : cubes)
2228 if (!llvm::is_contained(group.stated, cube))
2229 group.stated.push_back(cube);
2230 }
2231 SmallVector<GroupClaims> claims;
2232 for (const auto &[masters, group] : groups) {
2233 claims.push_back({group.stated, group.derived, masters.first});
2234 if (groupMasters)
2235 groupMasters->push_back(masters.second);
2236 }
2237 return slavePortRules(claims, existingCubes[{tileId, slave}], idBits,
2238 targetModel.getNumSlaveSlots());
2239 };
2240 auto rulesNeeded =
2241 [&, tileId = tileId](
2242 Port slave,
2243 std::optional<std::pair<PathEndPoint, Port>> without = {}) {
2244 return existingCubes[{tileId, slave}].size() +
2245 newRules(slave, without, false).size();
2246 };
2247 // Packets for a destination on the tile that reach it by a slave port of
2248 // their own take their rules with them. The source's tree splits apart the
2249 // destination that frees the most rules on `slave`, if one frees any.

2490std::optional<std::string> PacketFlowRouting::breakHoldCycles() {
2491 // A packet holds every arbiter it has taken until its tail passes, so waits
2492 // chain from switchbox to switchbox, and planning each alone can close a
2493 // cycle of them. Breaking any one shared arbiter of such a cycle breaks it,
2494 // so the search tries each in turn, up to a budget.
2495 auto arbitrate = [&]() {
2496 for (size_t s = 0; s < conflicts.getRequestedStreams().size(); s++) {
2497 if (!conflicts.getStreams()[s].packetID)
2498 continue;
2499 for (StreamHop &hop : routes[s]) {
2500 hop.arbiter.reset();
2501 auto plan = plans.find(hop.tile);
2502 if (plan == plans.end())
2503 continue;
2504 auto amsel = plan->second.slaveAmsels.find(
2505 {hop.input, *conflicts.getStreams()[s].packetID});
2506 if (amsel != plan->second.slaveAmsels.end())
2507 hop.arbiter = amsel->second % numArbiters;
2508 }
2509 }
2510 };
2511 ArrayRef<RoutedStream> streams = conflicts.getStreams();
2512 size_t numRequested = conflicts.getRequestedStreams().size();
2513 // Every cycle the search meets: the routing moves the flows of each, as the
2514 // first may run only through flows it cannot move.
2515 SmallVector<HoldCycle, 2> cycles;
2516 int replans = 0;
2517 bool forcedWaits = false, definite = false;
2518 auto search = [&](auto &self) -> bool {
2519 arbitrate();
2520 std::optional<HoldCycle> cycle =
2521 conflicts.holdCycle(routes, forcedWaits, definite);
2522 // Waits `definite` drops still count where no receiver is shared, so a
2523 // replan must not close a cycle the first search broke.
2524 if (!cycle && definite)
2525 cycle = conflicts.holdCycle(routes);
2526 if (!cycle)
2527 return true;
2528 cycles.push_back(*cycle);
2529 LLVM_DEBUG(llvm::dbgs()
2530 << "Hold cycle: " << conflicts.explain(*cycle) << '\n');
2531 for (const HoldCycle::Step &step : cycle->steps) {
2532 if (step.wait != HoldCycle::Wait::Arbiter || step.forced)
2533 continue;
2534 FlowKey sharer{step.sharerInput, *streams[step.sharer].packetID};
2535 FlowKey holder{step.holderInput, *streams[step.holding].packetID};
2536 bool sharerNew = step.sharer < numRequested;
2537 bool holderNew = step.holding < numRequested;
2538 std::optional<std::tuple<TileID, FlowKey, FlowKey>> pair;
2539 std::optional<std::tuple<TileID, FlowKey, int>> off;
2540 if (sharerNew && holderNew)
2541 pair = {step.tile, std::min(sharer, holder), std::max(sharer, holder)};
2542 else if (sharerNew || holderNew)
2543 off = {step.tile, sharerNew ? sharer : holder, step.arbiter};
2544 else
2545 continue;
2546 if (replans == maxHoldCycleReplans)
2547 return false;
2548 if ((pair && !apart.insert(*pair).second) ||
2549 (off && !offArbiter.insert(*off).second))
2550 continue;
2551 replans++;
2552 ArbiterPlan saved = plans.at(step.tile);
2553 SmallVector<std::pair<size_t, size_t>, 4> blocking;
2554 if (std::optional<ArbiterPlan> plan = planTile(step.tile, blocking)) {
2555 plans[step.tile] = std::move(*plan);
2556 if (self(self))
2557 return true;
2558 plans[step.tile] = std::move(saved);
2559 }
2560 if (pair)
2561 apart.erase(*pair);
2562 if (off)
2563 offArbiter.erase(*off);
2564 }
2565 return false;
2566 };
2567 // The routing moves the flows of each cycle the search met, but for those
2568 // waiting at a receiver they share.
2569 auto move = [&] {
2570 arbitrate();
2571 for (const HoldCycle &cycle : cycles)
2572 for (const HoldCycle::Step &step : cycle.steps) {
2573 if (step.wait == HoldCycle::Wait::Arbiter && step.forced &&
2574 step.sharer < numRequested && step.holding < numRequested &&
2575 streams[step.sharer].packetID && streams[step.holding].packetID) {
2576 PathEndPoint a{streams[step.sharer].src.tile,
2577 streams[step.sharer].src.port},
2578 b{streams[step.holding].src.tile, streams[step.holding].src.port};
2579 if (!(a == b))
2580 hazardTogether.insert(std::minmax(a, b));
2581 }
2582 if (step.wait != HoldCycle::Wait::Drain && !step.forced) {
2583 SmallVector<FlowKey, 3> keys;
2584 for (auto [s, input] : {std::pair{step.waiting, step.sharerInput},
2585 std::pair{step.sharer, step.sharerInput},
2586 std::pair{step.holding, step.holderInput}})
2587 if (std::optional<int> id = streams[s].packetID;
2588 s < numRequested && id)
2589 keys.push_back({input, *id});
2590 for (FlowKey key : keys) {
2591 moveUnit(step.tile, key);
2592 splitFlow(step.tile, key, keys);
2593 }
2594 if (step.sharer >= numRequested || step.holding >= numRequested ||
2595 !streams[step.sharer].packetID || !streams[step.holding].packetID)
2596 continue;
2597 PathEndPoint a{streams[step.sharer].src.tile,
2598 streams[step.sharer].src.port},
2599 b{streams[step.holding].src.tile, streams[step.holding].src.port};
2600 if (!(a == b))
2601 hazardApart.insert(std::minmax(a, b));
2602 }
2603 }
2604 };
2605 if (!search(search)) {
2606 move();
2607 return "packet flows can deadlock holding arbiters across "
2608 "switchboxes, and no arbiter assignment found avoids it. " +
2609 conflicts.explain(cycles.front());
2610 }
2611 // Trees into a receiver they share wait on each other at its master port
2612 // whatever the plan, so a cycle through such waits can only be broken
2613 // elsewhere along it. Where none of it can, the routing fails, unless some
2614 // plan has such cycles only by assumption.
2615 cycles.clear();
2616 replans = 0;
2617 forcedWaits = true;
2618 if (!search(search)) {
2619 LLVM_DEBUG(llvm::dbgs() << "Hold cycle at a shared receiver: "
2620 << conflicts.explain(cycles.front()) << '\n');
2621 cycles.clear();
2622 replans = 0;
2623 definite = true;
2624 if (!search(search)) {
2625 move();
2626 sharedReceiverCycle = cycles.front();
2627 }
2628 }
2629 return std::nullopt;
2630}

2775 builder, tileLoc, builder.getIndexType(), master.second.bundle,

2777 if (ctrlPktOverlayMasterPorts.contains(master))
2778 overlay.setAttr(msOp, unit);
2779 }

2781 // Generate the packet rules, adding to any the switchbox already has.
2782 DenseMap<Port, PacketRulesOp> slaveRules;
2783 DenseMap<Port, SmallVector<PacketRuleOp>> existingRules;
2784 struct PortPlan {
2785 SmallVector<size_t, 4> groups;
2786 SmallVector<PortRule> rules;
2787 size_t emitted = 0;
2788 };
2789 std::map<Port, PortPlan> portPlans;
2790 for (auto rulesOp : b.getOps<PacketRulesOp>()) {
2791 slaveRules[rulesOp.sourcePort()] = rulesOp;
2792 llvm::append_range(existingRules[rulesOp.sourcePort()],
2793 rulesOp.getRules().getOps<PacketRuleOp>());
2794 }
2795 for (size_t gi = 0; gi < slaveGroups.size(); ++gi) {
2796 const auto &group = slaveGroups[gi];
2797 builder.setInsertionPoint(b.getTerminator());
2798
2799 auto port = group.front().first;
2800 if (tileId != port.first)
2801 continue;
2802
2803 Port slave = port.second;
2804
2805 SmallVector<int, 4> matchIds =
2806 llvm::to_vector<4>(llvm::make_second_range(group));
2807 for (int id : matchIds)
2808 if (id > static_cast<int>(maxPacketId))
2809 return mlir::emitError(tileLoc)
2810 << "packet id " << id << " exceeds the maximum of "
2811 << maxPacketId;
2812
2813 // Rules the switchbox already has, e.g. hand-authored, match first.
2814 for (PacketRuleOp rule : existingRules.lookup(slave))
2815 for (int id : matchIds)
2816 if ((id & rule.maskInt()) == rule.valueInt()) {
2817 rule->emitOpError("can lead to false packet id match for id ")
2818 << id << ", which is not supposed to pass through this port.";
2819 rule->emitRemark("Please consider changing all uses of packet id ")
2820 << id << " to avoid deadlock.";
2821 return failure();
2822 }
2823
2824 // Groups on one slave port carry different destination sets, so
2825 // checkRules let no two of them claim the same id.
2826 assert(llvm::all_of(
2827 llvm::seq<size_t>(0, slaveGroups.size()),
2828 [&](size_t oi) {
2829 return oi == gi || slaveGroups[oi].front().first != port ||
2830 llvm::none_of(claimsOf(gi), [&](auto own) {
2831 return llvm::any_of(claimsOf(oi), [&](auto other) {
2832 return cubesIntersect(own, other);
2833 });
2834 });
2835 }) &&
2836 "groups on one slave port claim the same id");
2837
2838 // The rules of the slave port, planned at its first group, go out in
2839 // order: each group adds those up to its last.
2840 auto [planIt, fresh] = portPlans.try_emplace(slave);
2841 PortPlan &plan = planIt->second;
2842 if (fresh) {
2843 SmallVector<GroupClaims> claims;
2844 for (size_t oi = 0; oi < slaveGroups.size(); ++oi) {
2845 if (slaveGroups[oi].front().first != port)
2846 continue;
2847 plan.groups.push_back(oi);
2848 claims.push_back({statedRules[oi], derivedIds[oi],
2849 overlayFlows.contains(slaveGroups[oi].front())});
2850 }
2851 SmallVector<std::pair<int, int>> existing;
2852 for (PacketRuleOp rule : existingRules.lookup(slave))
2853 existing.push_back({rule.maskInt(), rule.valueInt()});
2854 plan.rules = slavePortRules(claims, existing, idBits,
2855 targetModel.getNumSlaveSlots());
2856 LLVM_DEBUG({
2857 llvm::dbgs() << "packet rules " << describePort(slave) << ":";
2858 for (const PortRule &r : plan.rules)
2859 llvm::dbgs() << " rule(" << r.mask << ", " << r.value
2860 << ") -> group " << plan.groups[r.group];
2861 llvm::dbgs() << '\n';
2862 });
2863 }
2864
2865 // Every id the group claims takes a rule to its amsel first: its own,
2866 // or one of the control overlay's for the same destinations.
2867 assert(
2868 llvm::all_of(
2869 llvm::seq(0, idMask + 1),
2870 [&](int id) {
2871 bool own = llvm::is_contained(derivedIds[gi], id) ||
2872 llvm::any_of(statedRules[gi], [&](auto c) {
2873 return (id & c.first) == (c.second & c.first);
2874 });
2875 const auto *first =
2876 llvm::find_if(plan.rules, [&](const PortRule &r) {
2877 return (id & r.mask) == r.value;
2878 });
2879 return !own ||
2880 (first != plan.rules.end() &&
2881 slaveAMSels.at(
2882 slaveGroups[plan.groups[first->group]].front()) ==
2883 slaveAMSels.at(slaveGroups[gi].front()));
2884 }) &&
2885 "packet rules send a claimed id elsewhere");
2886
2887 size_t last = plan.emitted;
2888 for (size_t r = plan.emitted; r < plan.rules.size(); ++r)
2889 if (plan.groups[plan.rules[r].group] == gi)
2890 last = r + 1;
2891
2892 PacketRulesOp packetrules = slaveRules.lookup(slave);
2893 if (!packetrules) {
2894 packetrules = PacketRulesOp::create(builder, tileLoc, slave.bundle,
2895 slave.channel);
2896 PacketRulesOp::ensureTerminator(packetrules.getRules(), builder,
2897 tileLoc);
2898 slaveRules[slave] = packetrules;
2899 } else {
2900 // After the amsels its new rules use.
2901 packetrules->moveBefore(b.getTerminator());
2902 }
2903
2904 Block &rules = packetrules.getRules().front();
2905
2906 assert(llvm::range_size(rules.getOps<PacketRuleOp>()) +
2907 (last - plan.emitted) <=
2908 targetModel.getNumSlaveSlots() &&
2909 "checkRules lets a slave port take more rules than it holds");
2910
2911 builder.setInsertionPoint(rules.getTerminator());
2912 for (; plan.emitted < last; ++plan.emitted) {
2913 const PortRule &r = plan.rules[plan.emitted];
2914 const auto &ruleGroup = slaveGroups[plan.groups[r.group]];
2915 auto rule = PacketRuleOp::create(
2916 builder, tileLoc, r.mask, r.value,
2917 amselOps.at(slaveAMSels.at(ruleGroup.front())));
2918 if (llvm::all_of(ruleGroup, [&](const auto &member) {
2919 return markedFlows.contains(member);
2920 }))
2921 overlay.setAttr(rule, unit);
2922 if (prioritizedSourcePorts.contains(port) &&
2923 llvm::any_of(ruleGroup, [&](const auto &member) {
2924 return prioritizedFlows.contains(member);
2925 }))
2926 priorityRoute.setAttr(rule, unit);
2927 }
2928 }
2929
2930 // A reload skips the overlay's rules. A port of its rules alone says so
2931 // once; a port it shares says so rule by rule.
2932 for (auto rulesOp : b.getOps<PacketRulesOp>()) {
2933 auto rules = rulesOp.getRules().getOps<PacketRuleOp>();
2934 bool blockTagged = overlay.isAttrPresent(rulesOp);
2935 auto tagged = [&](PacketRuleOp rule) {
2936 return blockTagged || overlay.isAttrPresent(rule);
2937 };
2938 bool all = llvm::all_of(rules, tagged);
2939 for (PacketRuleOp rule : rules) {
2940 if (!all && tagged(rule))
2941 overlay.setAttr(rule, unit);
2942 else
2943 rule->removeDiscardableAttr(overlay.getName());
2944 }
2945 if (all && !rules.empty())
2946 overlay.setAttr(rulesOp, unit);
2947 else
2948 rulesOp->removeDiscardableAttr(overlay.getName());
2949 }
2950 }
2951 return success();
2952}

3020llvm::Expected<PacketPlan> AIEPathfinderPass::route(
3021 DeviceOp d, DynamicTileAnalysis &analyzer, const StreamConflicts &conflicts,
3022 const PinnedOverlay &pinned, bool circuitSwitchHops, bool prioritize) {
3023 // Packet flows that can deadlock must not share an arbiter. Every routing
3024 // the router finds is checked by planning the arbiters on it, and one that
3025 // cannot be planned counts as illegal, so routing and allocation agree.
3026 std::map<PathEndPoint, SmallVector<size_t, 4>> streamsFrom;
3027 PacketConstraints constraints;
3028 constraints.pinned = pinned.trees;
3029 constraints.prioritize = prioritize;
3030 if (clRoutePacket && !d.getOps<PacketFlowOp>().empty()) {
3031 const AIETargetModel &targetModel = d.getTargetModel();
3032 if (llvm::Error err = unroutableArbiters(
3033 d, conflicts,
3034 [&](TileID tile) {
3035 return !circuitSwitchHops ||
3036 targetModel.isShimNOCorPLTile(tile.col, tile.row);
3037 },
3038 pinned.trees))
3039 return std::move(err);
3040 for (auto [i, s] : llvm::enumerate(conflicts.getRequestedStreams()))
3041 if (s.packetID)
3042 streamsFrom[{s.src.tile, s.src.port}].push_back(i);
3043 constraints.conflict = [&](const PathEndPoint &a, const std::set<int> &aIds,
3044 const PathEndPoint &b,
3045 const std::set<int> &bIds) {
3046 auto as = streamsFrom.find(a), bs = streamsFrom.find(b);
3047 if (as == streamsFrom.end() || bs == streamsFrom.end())
3048 return false;
3049 ArrayRef<RoutedStream> streams = conflicts.getStreams();
3050 for (size_t s : as->second)
3051 for (size_t t : bs->second)
3052 if (aIds.count(*streams[s].packetID) &&
3053 bIds.count(*streams[t].packetID) && conflicts.conflict(s, t))
3054 return true;
3055 return false;
3056 };
3057 constraints.check = [&](const Routing &routing) -> llvm::Error {
3058 PacketFlowRouting check(d, routing, conflicts, pinned, clRouteCircuit,
3059 circuitSwitchHops, prioritize);
3060 llvm::Expected<PacketPlan> plan = check.plan();
3061 if (!plan)
3062 return plan.takeError();
3063 RoutingFaults all = check.faults(), faults;
3064 faults.connections = std::move(all.connections);
3065 faults.apart = std::move(all.apart);
3066 faults.together = std::move(all.together);
3067 if (!plan->sharedReceiverCycle ||
3068 (faults.connections.empty() && faults.apart.empty() &&
3069 faults.together.empty()))
3070 return llvm::Error::success();
3071 return llvm::make_error<DeadlockProneRouting>(
3072 conflicts.explain(*plan->sharedReceiverCycle), std::move(faults));
3073 };
3074 }
3075 analyzer.pathfinder.setPacketConstraints(std::move(constraints));
3076 llvm::Error routed = analyzer.runAnalysis(d);
3077 analyzer.pathfinder.setPacketConstraints({});
3078 if (routed)
3079 return std::move(routed);
3080 if (!clRoutePacket)
3081 return PacketPlan();
3082 PacketFlowRouting found(d, analyzer.routing, conflicts, pinned,
3083 clRouteCircuit, circuitSwitchHops, prioritize);
3084 llvm::Expected<PacketPlan> plan = found.plan();
3085 if (found.incomplete) {
3086 llvm::consumeError(plan.takeError());
3087 return llvm::make_error<RoutingFailure>(std::move(found.incomplete->second),
3088 RoutingFaults{},
3089 found.incomplete->first);
3090 }
3091 // The routing found may still leave a hold cycle through receivers flows
3092 // share, as the check lets one through where it names nothing to move.
3093 if (!plan || clAllowDeadlockProne)
3094 return plan;
3095 const std::optional<HoldCycle> &cycle = plan->sharedReceiverCycle;
3096 if (!cycle)
3097 return plan;
3098 return llvm::make_error<DeadlockProneRouting>(
3099 (llvm::Twine(sharedReceiverCycleReason) + conflicts.explain(*cycle) +
3100 allowDeadlockProneHint)
3101 .str());
3102}

3105
3106 // create analysis pass with routing graph for entire device
3107 LLVM_DEBUG(llvm::dbgs() << "---Begin AIEPathfinderPass---\n");
3108
3109 DeviceOp d = getOperation();
3110 if (failed(verifyDMAChannelsResolved(d)))
3111 return signalPassFailure();

3113 if (clRoutePacket)
3114 unmuxShimDMAPacketPorts(d);
3115
3116 StreamConflicts conflicts(d);
3117 if (auto pairs = conflicts.unavoidable(); !pairs.empty()) {
3118 InFlightDiagnostic diag =
3119 (clAllowDeadlockProne ? emitWarning(d.getLoc()) : emitError(d.getLoc()))
3120 << "Flows can deadlock however they are routed: "
3121 << conflicts.explain(pairs[0].first, pairs[0].second);

3123 diag << " So can " << pairs.size() - 1 << " other pair"

3126 diag << allowDeadlockProneHint;
3127 return signalPassFailure();
3128 }

3130 DynamicTileAnalysis analyzer;
3131 // A prioritized flow keeps the route it takes alone, so route the
3132 // prioritized flows without the rest of the design first, and pin their
3133 // trees for the rest to route around. Packets sharing a source and id with
3134 // a prioritized flow go with it (see prioritizedFlows); other packets from
3135 // their sources route freely.
3136 auto sourcesOf = [](PacketFlowOp flow) {
3137 return llvm::map_range(
3138 flow.getPorts().getOps<PacketSourceOp>(), [](PacketSourceOp src) {
3139 return PathEndPoint{
3140 cast<TileOp>(src.getTile().getDefiningOp()).getTileID(),
3141 src.port()};
3142 });
3143 };

3146 for (PacketFlowOp flow : d.getOps<PacketFlowOp>())
3147 if (flow.getPriorityRoute().value_or(false))
3148 for (PathEndPoint src : sourcesOf(flow)) {
3149 prioritized.insert(src);
3150 prioritizedIds.insert({src, flow.IDInt()});
3151 }

3153 return llvm::any_of(sourcesOf(flow), [&](const PathEndPoint &src) {
3154 return prioritizedIds.count({src, flow.IDInt()}) > 0;
3155 });
3156 };

3158 // the overlay's.

3161 for (PacketFlowOp flow : d.getOps<PacketFlowOp>())
3162 for (PacketDestOp dst : flow.getPorts().getOps<PacketDestOp>()) {
3163 PathEndPoint port{cast<TileOp>(dst.getTile().getDefiningOp()).getTileID(),
3164 dst.port()};

3168 prio = flow;

3173 keepsPktHeader(dst.coords, dst.port, prio.getKeepPktHeader()) ==
3174 keepsPktHeader(dst.coords, dst.port, last.getKeepPktHeader()))

3180 "there, but the prioritized flows (the control "
3181 "overlay) ending there set it otherwise, and a "
3182 "control-packet reload keeps their setting.";

3308 prioritized,
3309 [](const PathEndPoint &src) {
3310 return describeTilePort(src.coords, src.port);

3312 ", and the other flows route only if it moves. Around "
3313 "it, " +

3317 emitError(f.loc.value_or(d.getLoc())) << f.message();
3318 });
3319 signalPassFailure();

3321 }

3329 signalPassFailure();

3335 rules.try_emplace(port);

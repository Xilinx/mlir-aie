102 if (analyzer.processedFlows[srcPoint]) {
103 LLVM_DEBUG(llvm::dbgs() << "Flow already processed!\n");

120 // shim DMAs at start of flows

125 // must be NOC0/NOC1 -> N2/N3 or NOC2/NOC3 -> N6/N7

127 else if (srcBundle == WireBundle::PLIO)

134 assert(setting.srcs.size() == setting.dsts.size());

136 Port src = setting.srcs[i];
137 Port dest = setting.dsts[i];

141 addConnection(rewriter, cast<Interconnect>(swOp.getOperation()),
142 flowOp, WireBundle::South, shimCh, dest.bundle,
143 dest.channel);

146 dest.bundle == WireBundle::NOC)) {

151 shimCh = dest.channel == 0 ? 2 : 3;
152 else if (dest.bundle == WireBundle::NOC)
153 // must be either N2/3/4/5 -> NOC0/1/2/3
154 shimCh = dest.channel + 2;
155 else if (dest.bundle == WireBundle::PLIO)
156 shimCh = dest.channel;

181 rewriter.eraseOp(Op);

226bool AIEPathfinderPass::findPathToDest(const SwitchSettings &settings,
227 TileID currTile,
228 WireBundle currDestBundle,
229 int currDestChannel, TileID finalTile,
230 WireBundle finalDestBundle,
231 int finalDestChannel) {
232
233 if ((currTile == finalTile) && (currDestBundle == finalDestBundle) &&
234 (currDestChannel == finalDestChannel)) {
235 return true;
236 }
237
238 WireBundle neighbourSourceBundle;
239 TileID neighbourTile;
240 if (currDestBundle == WireBundle::East) {
241 neighbourSourceBundle = WireBundle::West;
242 neighbourTile = {currTile.col + 1, currTile.row};
243 } else if (currDestBundle == WireBundle::West) {
244 neighbourSourceBundle = WireBundle::East;
245 neighbourTile = {currTile.col - 1, currTile.row};
246 } else if (currDestBundle == WireBundle::North) {
247 neighbourSourceBundle = WireBundle::South;
248 neighbourTile = {currTile.col, currTile.row + 1};
249 } else if (currDestBundle == WireBundle::South) {
250 neighbourSourceBundle = WireBundle::North;
251 neighbourTile = {currTile.col, currTile.row - 1};
252 } else {
253 return false;
254 }
255
256 int neighbourSourceChannel = currDestChannel;
257 for (const auto &[sbNode, setting] : settings) {
258 TileID tile = {sbNode.col, sbNode.row};
259 if (tile == neighbourTile) {
260 assert(setting.srcs.size() == setting.dsts.size());
261 for (size_t i = 0; i < setting.srcs.size(); i++) {
262 Port src = setting.srcs[i];
263 Port dest = setting.dsts[i];
264 if ((src.bundle == neighbourSourceBundle) &&
265 (src.channel == neighbourSourceChannel)) {
266 if (findPathToDest(settings, neighbourTile, dest.bundle, dest.channel,
267 finalTile, finalDestBundle, finalDestChannel)) {
268 return true;
269 }
270 }
271 }
272 }
273 }
274
275 return false;
276}

287// Minimum set cover by branch-and-bound: find the fewest `cubes` whose coverage
288// (cube.cov) unions to every match id still set in `uncovered`. `sel` is the
289// current partial pick; `bestSel`/`bestSize` hold the smallest full cover so
290// far and are updated in place.
291static void bnbMinCover(ArrayRef<CoverCube> cubes,
292 const llvm::SmallBitVector &uncovered,
293 SmallVectorImpl<int> &sel, int &bestSize,
294 SmallVectorImpl<int> &bestSel) {
295
296 // base case
297 if (uncovered.none()) {
298 if (static_cast<int>(sel.size()) < bestSize) {
299 bestSize = static_cast<int>(sel.size());
300 bestSel.assign(sel.begin(), sel.end());
301 }
302 return;
303 }
304
305 // bound: adding a cube would only tie the best
306 if (static_cast<int>(sel.size()) + 1 >= bestSize) {
307 return;
308 }
309
310 // most constrained variable: find the least-covered, uncovered matchId
311 int pick = -1;
312 int pickCount = static_cast<int>(cubes.size()) + 1;
313
314 for (int id = uncovered.find_first(); id != -1;
315 id = uncovered.find_next(id)) {
316 int c = 0;
317
318 for (const CoverCube &cb : cubes) {
319 if (cb.cov.test(id)) {
320 c++;
321 }
322 }
323
324 if (c < pickCount) {
325 pickCount = c;
326 pick = id;
327 }
328 }
329
330 // branch: try each cube that covers the chosen id, recursing on the rest.
331 for (int i = 0; i < static_cast<int>(cubes.size()); ++i) {
332 if (!cubes[i].cov.test(pick)) {
333 continue;
334 }
335
336 sel.push_back(i);
337 llvm::SmallBitVector next = uncovered;
338 next.reset(cubes[i].cov); // next &= ~cov
339 bnbMinCover(cubes, next, sel, bestSize, bestSel);
340 sel.pop_back();
341 }
342}

344// Cover `matchIds` (a group's packet ids) with the fewest (mask, value) rules
345// that match every specified id and no `avoidIds` (other ids on the same slave
346// port). A rule matches id x iff (x & mask) == value; ids in neither set are
347// don't-cares, free to over-claim.
348//
349// Based on Quine-McCluskey:
350// 1. Enumerate prime implicants: maximal cubes (a cube is one (mask, value))
351// that match >=1 matchIds and no avoidIds.
352// 2. Pick the fewest of those cubes that cover all match ids (bnbMinCover).
353// 3. Tighten each chosen cube to the smallest enclosing cube of the ids it
354// took, so it claims no more don't-cares than necessary; the one-cube case
355// reduces to the old common-bits mask.
356static SmallVector<std::pair<int, int>>
357computeSubcubeCover(const SmallVector<int, 4> &matchIds,
358 const llvm::SmallSet<int, 8> &avoidIds, int idBits) {
359
360 // id space and full mask, sized from the target's packet-id width.
361 const int numIds = 1 << idBits;
362 const int idMask = numIds - 1;
363
364 auto hitsAvoid = [&](int mask, int value) {
365 return llvm::any_of(avoidIds, [&](int o) { return (o & mask) == value; });
366 };
367
368 llvm::SmallBitVector matchMask(numIds);
369 for (int id : matchIds) {
370 matchMask.set(id);
371 }
372
373 // phase 1: enumerate prime implicants. small (3^idBits), so brute force.
374 SmallVector<CoverCube> primes;
375 for (int mask = 0; mask < numIds; ++mask) {
376 for (int value = 0; value < numIds; ++value) {
377 // valid cube := value covered by mask, avoids properly
378 if ((value & mask) != value) {
379 continue;
380 }
381 if (hitsAvoid(mask, value)) {
382 continue;
383 }
384
385 // record covering
386 llvm::SmallBitVector cov(numIds);
387 for (int id : matchIds) {
388 if ((id & mask) == value) {
389 cov.set(id);
390 }
391 }
392
393 if (cov.none()) {
394 continue;
395 }
396
397 // prime = maximal, i.e., no checked mask bit can be dropped without
398 // covering an avoidId.
399 bool isPrime = true;
400 for (int b = 0; b < idBits; ++b) {
401 if (!((mask >> b) & 1)) {
402 continue;
403 }
404
405 if (!hitsAvoid(mask & ~(1 << b), value & ~(1 << b))) {
406 isPrime = false;
407 break;
408 }
409 }
410
411 if (isPrime) {
412 primes.push_back({mask, value, std::move(cov)});
413 }
414 }
415 }
416
417 // phase 2: pick the fewest primes that cover every match id (bnb).
418 SmallVector<int> sel, bestSel;
419 int bestSize = static_cast<int>(matchIds.size()) + 1;
420
421 bnbMinCover(primes, matchMask, sel, bestSize, bestSel);
422
423 SmallVector<std::pair<int, int>> chosen;
424 for (int i : bestSel) {
425 chosen.push_back({primes[i].mask, primes[i].value});
426 }
427
428 if (chosen.empty()) {
429 // unreachable: a single-id cube always covers
430 for (int id : matchIds) {
431 chosen.push_back({idMask, id});
432 }
433 }
434
435 // phase 3: tighten each chosen cube to match only the ids it took.
436 // maximal -> over-claim don't-cares -> more mask bits narrow the cube.
437 //
438 // e.g., {2,4} avoid {0}:
439 // prime (mask 00010, value 00010): 2 & 00010 = 00010 == value -> hit
440 // 6 & 00010 = 00010 == value -> also hit
441 // tight (mask 11111, value 00010): 2 & 11111 = 00010 == value -> hit
442 // 6 & 11111 = 00110 != value -> dropped
443 // 0 & 11111 = 00000 != value -> avoided
444
445 SmallVector<SmallVector<int, 4>, 4> assigned(chosen.size());
446 for (int id : matchIds) {
447 for (int c = 0; c < static_cast<int>(chosen.size()); ++c) {
448 if ((id & chosen[c].first) == chosen[c].second) {
449 assigned[c].push_back(id);
450 break;
451 }
452 }
453 }
454
455 SmallVector<std::pair<int, int>> cover;
456 for (const SmallVector<int, 4> &ids : assigned) {
457 if (ids.empty()) {
458 continue;
459 }
460
461 int mask = idMask;
462 for (int i = 0; i < idBits; ++i) {
463 int bit = (ids.front() >> i) & 1;
464 for (int id : ids) {
465 if (((id >> i) & 1) != bit) {
466 mask &= ~(1 << i);
467 break;
468 }
469 }
470 }
471 cover.push_back({mask, ids.front() & mask});
472 }
473
474 return cover;
475}

486/// What a slave port's traffic does at one switchbox. A port carries one
487/// ordered stream, so a stall at its head holds up every packet behind it:
488/// hazards are properties of the port, not of one flow on it.
489struct PortTraffic {
490 /// Some flow on this port ends at a DMA, here or on another tile.
491 bool feedsDma = false;
492 /// Some flow on this port has a non-DMA destination here: in transit to
493 /// another tile, or ending at a local control port (which backs up when its
494 /// config queue fills). Only a local DMA is taken to drain unconditionally.
495 bool leaves = false;
496 /// Some flow on this port sends packets the receiving DMA takes in more than
497 /// one buffer descriptor.
498 bool spansConsumerBds = false;
499};

501/// Ways flows sharing an arbiter can hang. OR'd together and compared as a
502/// number, so a higher bit outranks lower ones and more bits are worse. An
503/// oversubscribed switchbox has to pick the least bad arbiter.
504enum HazardKind : unsigned {
505 /// A shape that stalls, with no known cycle closing it back: packets
506 /// outlasting the receiving DMA's descriptors, or a stalling source sharing
507 /// with a co-tenant that still has to leave this switchbox.
508 StallShape = 1,
509 /// A cycle closed through intermediate pipeline stages, all of which must
510 /// saturate at once.
511 RelayedCycle = 2,
512 /// The same cycle with nothing in between: one flow ends at the tile the
513 /// other starts from. Tightest loop, so it outranks any pile of the others.
514 DirectCycle = 4,
515};

517/// A placed flow the flow being allocated can deadlock against on one arbiter.
518struct ArbiterHazard {
519 int conflictingFlowID;
520 /// HazardKinds this arbiter trips; the allocator falls back to the lowest.

522};

525LogicalResult
526AIEPathfinderPass::runOnPacketFlow(DeviceOp device, OpBuilder &builder,
527 DynamicTileAnalysis &analyzer) {
528
529 ConversionTarget target(getContext());
530
531 std::map<TileID, mlir::Operation *> tiles;
532
533 // Map from a port and flowID to
534 std::map<std::pair<PhysPort, int>, SmallVector<PhysPort, 4>> packetFlows;
535 std::map<std::pair<PhysPort, int>, SmallVector<PhysPort, 4>> ctrlPacketFlows;
536 SmallVector<std::pair<PhysPort, int>, 4> slavePorts;
537 DenseMap<std::pair<PhysPort, int>, int> slaveAMSels;
538 // Flag to keep packet header at packet flow destination
539 DenseMap<PhysPort, BoolAttr> keepPktHeaderAttr;
540 // Map from tileID and master ports to flags labelling control packet flows
541 DenseMap<std::pair<PhysPort, int>, bool> ctrlPktFlows;
542 // Set of master ports that belong to control packet overlay flows
543 DenseSet<PhysPort> ctrlPktOverlayMasterPorts;
544 // DMA endpoints of each slave flow, consulted when choosing its arbiter.
545 DenseMap<std::pair<PhysPort, int>, DmaEndpoints> slaveFlowDmaEndpoints;
546 // Aggregate of the above over all flows sharing a slave port.
547 DenseMap<PhysPort, PortTraffic> slavePortTraffic;
548
549 for (auto tileOp : device.getOps<TileOp>()) {
550 int col = tileOp.colIndex();
551 int row = tileOp.rowIndex();
552 tiles[{col, row}] = tileOp;
553 }
554
555 // Tile-level dataflow graph: an edge P -> C per packet flow from a DMA of P
556 // to a DMA of C. C must run to free its input buffer, so everything C emits
557 // waits behind that flow.
558 DenseMap<TileID, llvm::SmallDenseSet<TileID, 4>> tileConsumers;
559 for (PacketFlowOp pktFlowOp : device.getOps<PacketFlowOp>()) {
560 Block &b = pktFlowOp.getPorts().front();
561 SmallVector<TileID, 4> srcTiles;
562 for (PacketSourceOp src : b.getOps<PacketSourceOp>())
563 if (src.port().bundle == WireBundle::DMA)
564 srcTiles.push_back(
565 cast<TileOp>(src.getTile().getDefiningOp()).getTileID());
566 for (PacketDestOp dst : b.getOps<PacketDestOp>()) {
567 if (dst.port().bundle != WireBundle::DMA)
568 continue;
569 TileID dstTile = cast<TileOp>(dst.getTile().getDefiningOp()).getTileID();
570 for (TileID srcTile : srcTiles)
571 if (srcTile != dstTile)
572 tileConsumers[srcTile].insert(dstTile);
573 }
574 }
575
576 // Tiles the root's draining depends on: itself, what it feeds, transitively.
577 // Shim tiles are reached but not expanded through: they drain to DDR in an
578 // order only the host knows, and following them would make nearly every tile
579 // downstream of every other.
580 const AIETargetModel &targetModel = device.getTargetModel();
581 DenseMap<TileID, llvm::SmallDenseSet<TileID, 8>> downstreamTiles;
582 auto downstreamOf =
583 [&](TileID root) -> const llvm::SmallDenseSet<TileID, 8> & {
584 auto cached = downstreamTiles.find(root);
585 if (cached != downstreamTiles.end())
586 return cached->second;
587 llvm::SmallDenseSet<TileID, 8> reached;
588 SmallVector<TileID, 8> work{root};
589 while (!work.empty()) {
590 TileID tile = work.pop_back_val();
591 if (!reached.insert(tile).second)
592 continue;
593 if (targetModel.isShimNOCorPLTile(tile.col, tile.row))
594 continue;
595 auto succs = tileConsumers.find(tile);
596 if (succs == tileConsumers.end())
597 continue;
598 work.append(succs->second.begin(), succs->second.end());
599 }
600 return downstreamTiles[root] = std::move(reached);
601 };
602
603 // Largest buffer descriptor each DMA channel issues, in bytes. Only channels
604 // with a body here appear; ones programmed from outside (e.g. shim channels,
605 // once the runtime sequence is lowered away) are absent and draw no
606 // conclusion below.
607 DenseMap<std::pair<TileID, int>, uint64_t> mm2sBdBytes, s2mmBdBytes;
608 auto scanDmaBds = [&](TileID tileId, Region &region) {
609 for (Block &block : region)
610 for (DMAStartOp startOp : block.getOps<DMAStartOp>()) {
611 Block *head = startOp.getDest();
612 if (!head)
613 continue;
614 // Walk the aie.next_bd chain from the start block. A block holding a
615 // dma_start belongs to another channel; dma_start terminates its
616 // block, so none of this chain's descriptors can be in it.
617 uint64_t largest = 0;
618 SmallVector<Block *, 8> work{head};
619 SmallPtrSet<Block *, 8> seen;
620 seen.insert(head);
621 while (!work.empty()) {
622 Block *b = work.pop_back_val();
623 if (!b->getOps<DMAStartOp>().empty())
624 continue;
625 for (DMABDOp bd : b->getOps<DMABDOp>())
626 largest = std::max(largest, bd.getLenInBytes());
627 for (Block *succ : b->getSuccessors())
628 if (seen.insert(succ).second)
629 work.push_back(succ);
630 }
631 if (largest == 0)
632 continue;
633 auto &sizes = startOp.isSend() ? mm2sBdBytes : s2mmBdBytes;
634 uint64_t &slot = sizes[{tileId, startOp.getChannelIndex()}];
635 slot = std::max(slot, largest);
636 }
637 };
638 for (MemOp mem : device.getOps<MemOp>())
639 scanDmaBds(mem.getTileOp().getTileID(), mem->getRegion(0));
640 for (MemTileDMAOp mem : device.getOps<MemTileDMAOp>())

645 // Stripped at the destination, so a send descriptor is this much longer than
646 // the receive descriptor for the same transfer -- unless keep_pkt_header.
647 constexpr uint64_t packetHeaderBytes = 4;

649 // One send descriptor is one packet under one arbiter grant, so if the
650 // receiving DMA takes it in smaller descriptors the grant is held across all
651 // of them. Comparing largest against largest under-reports (a mixed-size
652 // receive chain can pair a large send with a small descriptor); comparing
653 // against the receiver's smallest flags nearly everything. A descriptor-by-
654 // descriptor comparison is the accurate answer, left for later.
655 auto sendSpansReceiveBds = [&](TileID srcTile, Port srcPort, TileID destTile,
656 Port destPort, bool keepsPktHeader) {
657 auto send = mm2sBdBytes.find({srcTile, srcPort.channel});
658 auto recv = s2mmBdBytes.find({destTile, destPort.channel});
659 if (send == mm2sBdBytes.end() || recv == s2mmBdBytes.end())
660 return false;
661 uint64_t slack = keepsPktHeader ? 0 : packetHeaderBytes;
662 return send->second > recv->second + slack;
663 };

868 // True when both flows leave by the same, single master port and it is a DMA
869 // of this tile. One destination each is deliberate: a flow with a second
870 // destination can be held up there instead.
871 auto fansIntoSameLocalDma = [&](const std::pair<PhysPort, int> &flow,
872 const std::pair<PhysPort, int> &other) {
873 auto a = packetFlows.find(flow);
874 auto b = packetFlows.find(other);
875 if (a == packetFlows.end() || b == packetFlows.end())
876 return false;
877 if (a->second.size() != 1 || b->second.size() != 1)
878 return false;
879 return a->second[0] == b->second[0] &&
880 a->second[0].second.bundle == WireBundle::DMA;
881 };

883 // An arbiter holds its grant until tlast, so two slave ports on one arbiter
884 // serialize even on separate master ports. Three ways that deadlocks:
885 //
886 // - A port whose packets outlast one receive descriptor holds the grant
887 // across the consumer's whole lock cycle. Nothing may share with it.
888 // - A stalling port (see portStalls) blocks any co-tenant that still has to
889 // leave this switchbox. Co-tenants ending here are safe: the local DMA
890 // drains them.
891 // - Flow F ends at a tile flow G's producer waits on, so F stalls on a full
892 // buffer while holding the grant G needs to drain it. Asked of the flow
893 // graph (see couples), not just of the two flows' endpoints.
894 //
895 // Returns the worst hazard on this arbiter, or nullopt if there is none.
896 // Only the third exhibits an actual cycle; the first two recognize a shape
897 // that stalls, so they rank below either cycle.
898 auto arbiterHazard = [&](TileID tileId, int arbiter,
899 const std::pair<PhysPort, int> &flow)
900 -> std::optional<ArbiterHazard> {
901 auto placed = arbiterSlaveFlows.find({tileId, arbiter});
902 if (placed == arbiterSlaveFlows.end())
903 return std::nullopt;
904 const DmaEndpoints &f = dmaEndpointsOf(flow);
905 const PortTraffic &fp = trafficOf(flow.first);
906 bool flowStalls = portStalls(flow.first);
907 ArbiterHazard worst{0, 0};
908 auto raise = [&](int conflictingFlowID, HazardKind kind) {
909 // Name the flow behind the strongest evidence, not the first found.
910 if (kind > worst.kinds)
911 worst.conflictingFlowID = conflictingFlowID;
912 worst.kinds |= kind;
913 };
914 for (const auto &other : placed->second) {
915 // Flows on one slave port are serialized by the port and never
916 // arbitrate against each other.
917 if (other.first == flow.first)
918 continue;
919 // Nor do flows fanning into one local DMA channel and going nowhere
920 // else: that DMA takes them one packet at a time regardless, and they
921 // must share an arbiter anyway since a master port is tied to one.
922 // Checked first -- the rules below would flag exactly this shape.
923 if (fansIntoSameLocalDma(flow, other))
924 continue;
925 const DmaEndpoints &g = dmaEndpointsOf(other);
926 // Nothing outranks a direct cycle, so stop at the first one.
927 if (sharesTile(g.consumers, f.producers) ||
928 sharesTile(g.producers, f.consumers))
929 return ArbiterHazard{other.second, HazardKind::DirectCycle};
930 if (couples(g.consumers, f.producers) ||
931 couples(f.consumers, g.producers))
932 raise(other.second, HazardKind::RelayedCycle);
933 const PortTraffic &gp = trafficOf(other.first);
934 if (fp.spansConsumerBds || gp.spansConsumerBds ||
935 (flowStalls && gp.leaves) || (fp.leaves && portStalls(other.first)))
936 raise(other.second, HazardKind::StallShape);
937 }
938 if (!worst.kinds)
939 return std::nullopt;
940 return worst;
941 };

943 // Reported wherever a flow lands on an arbiter it can deadlock against. The
944 // tile op's location is rarely the switchbox in question, so name the tile.
945 auto warnSharedArbiter = [](TileOp tileOp, int flowID, int arbiter,
946 int conflictingFlowID, StringRef why) {
947 tileOp->emitWarning() << "at tile (" << tileOp.colIndex() << ", "
948 << tileOp.rowIndex() << "), packet flow " << flowID
949 << " shares arbiter " << arbiter
950 << " with packet flow " << conflictingFlowID
951 << ", which it can deadlock against; " << why;

954 // Get a new unique amsel from masterAMSels on tile op. Prioritize on
955 // incrementing arbiter id, before incrementing msel, skipping arbiters that
956 // would deadlock. Only an oversubscribed switchbox ever skips, since msel 0
957 // fills across all arbiters first and an unused arbiter has no hazard.
958 // Reached only with no master port of this flow pinned.
959 auto getNewUniqueAmsel =
960 [&](const std::map<std::pair<TileID, int>, SmallVector<Port, 4>>
961 &masterAMSels,
962 TileOp tileOp, bool isCtrlPkt, const std::pair<PhysPort, int> &flow) {
963 TileID tileId = tileOp.getTileID();
964
965 // Control packets take the highest amsels first.
966 SmallVector<int, 32> candidates;
967 for (int i = 0; i < numMselsPerArbiter; i++)
968 for (int a = 0; a < numArbiters; a++) {
969 int msel = isCtrlPkt ? numMselsPerArbiter - 1 - i : i;
970 int arbiter = isCtrlPkt ? numArbiters - 1 - a : a;
971 candidates.push_back(getAmselFromArbiterIDAndMsel(arbiter, msel));
972 }
973
974 // Take the first hazard-free amsel; if every free one is hazardous,
975 // fall back to the lowest-ranked. Ties go to the earliest, keeping the
976 // old scan order wherever the ranking has no opinion.
977 int hazardous = INVALID_AMSEL_VALUE;
978 ArbiterHazard worst{0, 0};
979 for (int amsel : candidates) {
980 if (masterAMSels.count({tileId, amsel}))
981 continue;
982 std::optional<ArbiterHazard> hazard =
983 arbiterHazard(tileId, getArbiterIDFromAmsel(amsel), flow);
984 if (!hazard)
985 return amsel;
986 if (hazardous == INVALID_AMSEL_VALUE || hazard->kinds < worst.kinds) {
987 hazardous = amsel;
988 worst = *hazard;
989 }
990 }
991
992 if (hazardous == INVALID_AMSEL_VALUE) {
993 tileOp->emitOpError(
994 "tile op has used up all arbiter-msel combinations");
995 return INVALID_AMSEL_VALUE;
996 }
997 warnSharedArbiter(tileOp, flow.second, getArbiterIDFromAmsel(hazardous),
998 worst.conflictingFlowID,
999 "no independent arbiter is free");
1000 return hazardous;
1001 };
1002 // Get a new unique amsel from masterAMSels on tile op with given arbiter id
1003 auto getNewUniqueAmselPerArbiterID =
1004 [&](const std::map<std::pair<TileID, int>, SmallVector<Port, 4>>
1005 &masterAMSels,
1006 TileOp tileOp, int arbiter) {
1007 for (int i = 0; i < numMselsPerArbiter; i++)
1008 if (!masterAMSels.count({tileOp.getTileID(),
1009 getAmselFromArbiterIDAndMsel(arbiter, i)}))
1010 return getAmselFromArbiterIDAndMsel(arbiter, i);
1011 tileOp->emitOpError("tile op arbiter ")
1012 << std::to_string(arbiter) << " has used up all its msels";
1013 return INVALID_AMSEL_VALUE;
1014 };

1083 // value pair for each flow
1084 for (const auto &packetFlow : sortedPacketFlows) {
1085 // The Source Tile of the flow
1086 TileID tileId = packetFlow.first.first.first;
1087 TileOp tileOp = analyzer.getTile(builder, tileId);
1088 if (amselValues.count(tileOp) == 0)
1089 amselValues[tileOp] = 0;

1092 // arb1: 6*0+1, 6*1+1, 6*2+1, 6*3+1
1093 // arb2: 6*0+2, 6*1+2, 6*2+2, 6*3+2
1094 // arb3: 6*0+3, 6*1+3, 6*2+3, 6*3+3
1095 // arb4: 6*0+4, 6*1+4, 6*2+4, 6*3+4
1096 // arb5: 6*0+5, 6*1+5, 6*2+5, 6*3+5

1101 // Find existing arbiter and amsel assignments for this flow
1102 // Strategy: Look for existing amsel entries that match the flow's
1103 // destinations
1104 // - Complete match: reuse existing amsel
1105 // - Partial match: create new amsel on same arbiter
1106 // - No match: create new amsel (on existing arbiter if ports already
1107 // assigned)
1108 bool hasMatchingAmselEntry = false;
1109 int partialMatchArbiterID = INVALID_ARBITER_VALUE;
1110 // Set on the one path that chooses an arbiter, and so reports its own
1111 // hazard. Other paths take a pinned arbiter and are checked below.
1112 bool hazardChecked = false;
1113
1114 // Check if any ports in this flow already have arbiter assignments
1115 int existingArbiter = findExistingArbiter(tileId, packetFlow.second);
1116
1117 // Search for matching amsel entries, prioritizing those that match the
1118 // existing arbiter
1119 for (const auto &amselEntry : masterAMSels) {
1120 if (amselEntry.first.first != tileId)
1121 continue;
1122 amselValue = amselEntry.first.second;
1123 int thisArbiter = getArbiterIDFromAmsel(amselValue);
1124
1125 // If we have an existing arbiter assignment, only consider amsels from
1126 // that arbiter
1127 if (existingArbiter != INVALID_ARBITER_VALUE &&
1128 existingArbiter != thisArbiter) {
1129 continue;
1130 }
1131
1132 // Check if destinations match (completely or partially)
1133 const SmallVector<Port, 4> &existingPorts =
1134 masterAMSels[{tileId, amselValue}];
1135 bool hasOverlap = false;
1136 bool hasNonOverlap = false;
1137
1138 for (auto dest : packetFlow.second) {
1139 Port port = dest.second;
1140 if (llvm::find(existingPorts, port) == existingPorts.end())
1141 hasNonOverlap = true;
1142 else
1143 hasOverlap = true;
1144 }
1145
1146 if (hasOverlap) {
1147 hasMatchingAmselEntry = true;
1148 // Partial match if some ports don't overlap or sizes differ
1149 if (hasNonOverlap || existingPorts.size() != packetFlow.second.size())
1150 partialMatchArbiterID = thisArbiter;
1151 break;
1152 }
1153 }
1154
1155 if (!hasMatchingAmselEntry) {
1156 // This packet flow switchbox's output ports completely mismatches with
1157 // any existing amsel. Creating a new amsel.
1158
1159 // Determine target arbiter: use existing if available, otherwise allocate
1160 // based on priority
1161 int targetArbiter = existingArbiter;
1162
1163 if (targetArbiter == INVALID_ARBITER_VALUE) {
1164 // No existing assignment, choose based on control packet priority
1165 bool isCtrlPkt =
1166 llvm::any_of(packetFlow.second, [&](PhysPort destPhysPort) {
1167 Port port = destPhysPort.second;
1168 return ctrlPktFlows[{{tileId, port}, packetFlow.first.second}];
1169 });
1170
1171 amselValue = getNewUniqueAmsel(masterAMSels, tileOp, isCtrlPkt,
1172 packetFlow.first);
1173 if (amselValue == INVALID_AMSEL_VALUE)
1174 return failure();
1175 hazardChecked = true;
1176 } else {
1177 // Use existing arbiter to maintain consistency
1178 amselValue =
1179 getNewUniqueAmselPerArbiterID(masterAMSels, tileOp, targetArbiter);
1180 if (amselValue == INVALID_AMSEL_VALUE) {

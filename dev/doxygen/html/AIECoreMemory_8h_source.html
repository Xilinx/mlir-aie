31// A half-open [start, start + size) run of bytes. Banks, free runs and a core's
32// data region all use this type.
33struct MemoryRun {
34 int64_t start = 0;
35 int64_t size = 0;
36
37 int64_t end() const { return start + size; }
38 bool contains(int64_t addr) const { return addr >= start && addr < end(); }
39};
40
41// Largest run of free bytes in [0, memSize) whose start is a multiple of
42// `alignBytes`. The occupied half-open intervals may be unsorted and may
43// overlap.
44//
45// AIETargetLdScript grants the core compiler one region for its .data, .rodata
46// and .bss. A core that declares no data_size gets this run as that region.
47//
48// The linker starts .data at a multiple of its strongest section alignment, so
49// an unaligned start loses the difference to padding. A reservation of the
50// exact size then overflows the region.
51inline MemoryRun
52largestFreeRun(int64_t memSize,
53 llvm::SmallVector<std::pair<int64_t, int64_t>> occupied,
54 int64_t alignBytes = 1) {
55 assert(alignBytes > 0 && "alignBytes must be positive");
56 llvm::sort(occupied);
57 MemoryRun best;
58 int64_t cursor = 0;
59 auto consider = [&](int64_t start, int64_t end) {
60 start = llvm::alignTo(start, alignBytes);
61 if (start < end && end - start > best.size)
62 best = {start, end - start};
63 };
64 for (auto &interval : occupied) {
65 // A zero-length interval occupies no bytes, so it must not split a run.
66 if (interval.first == interval.second)
67 continue;
68 consider(cursor, std::min(interval.first, memSize));
69 cursor = std::max(cursor, interval.second);
70 if (cursor >= memSize)
71 return best;
72 }
73 consider(cursor, memSize);
74 return best;
75}

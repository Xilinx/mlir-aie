41/// Shim endpoint lifetimes, including calls to self-contained runtime
42/// sequences. MM2S transfers conservatively remain live until their sequence
43/// returns; a callee's return and a dispatch boundary are themselves taken as
44/// the end of its sends, by the aiex.run and hardware-context contracts, not
45/// by anything this analysis computes. A receiving channel writes whatever
46/// arrives into the task it runs, so a receiving end shares only by explicit
47/// permission or where the array cannot send its words outside its turn: its
48/// senders wait on a gate the host fills only after the end before it is
49/// awaited, and each sequence drains all that gate lets them send.
50class ShimTransferSpans {
51public:
52 /// A shim end a runtime sequence transfers through.
53 struct End {
54 mlir::StringAttr fifo;
55 /// An MM2S end, sending into the array.
56 bool sends;
57 };
58
59 /// A shim input gating a receiving end's senders, per its
60 /// `#aie.consumed_by`.
61 struct Gate {
62 Gate(mlir::StringAttr input, ConsumedByAttr proof)
63 : input(input), proof(proof) {}
64 mlir::StringAttr input;
65 ConsumedByAttr proof;
66 };

70 /// How ends and gates name a shim route end: by its objectFIFO, else by its
71 /// own symbol.
72 static mlir::StringAttr keyOf(RouteEndpointOp endpoint);
73 /// The end a runtime sequence of `device` transfers through when it names
74 /// `symbol`: a shim DMA route end, keyed by keyOf, or, before
75 /// aie-objectfifo-split, an objectFIFO, sending when a shim tile produces
76 /// it.
77 static std::optional<End> endOf(DeviceOp device, mlir::StringAttr symbol);

79 /// `endOf` maps a symbol a runtime sequence transfers through to the
80 /// objectFIFO end it belongs to, or nothing when it belongs to none.
81 /// `gates` holds each receiving end's gates, keyed as `endOf` keys ends;
82 /// without them, a receiving end shares only by explicit permission, which
83 /// is all a caller running before the proofs can know. With
84 /// `plannedOverlay`, the control overlay's channel on each column's shim
85 /// tile is an end too, overlayEnd(), before
86 /// aie-generate-column-control-overlay makes it: it carries the control
87 /// packets to its column and every configuration load, as the overlay's
88 /// own channel does once made.
89 ShimTransferSpans(
90 DeviceOp device,
91 llvm::function_ref<std::optional<End>(mlir::StringAttr)> endOf,
92 std::optional<Gates> gates = std::nullopt, bool plannedOverlay = false);

116 /// The number of runtime sequences `bytes` indexes into.
117 unsigned sequences() const { return spans.size(); }
118 /// The bytes runtime sequence `seq` moves through `end`, nullopt when not
119 /// static or when `end` is not one of its ends.
120 std::optional<int64_t> bytes(unsigned seq, mlir::StringAttr end) const {
121 auto it = spans[seq].find(end);
122 return it == spans[seq].end() ? std::nullopt : it->second.bytes;
123 }
124 /// Whether runtime sequence `seq` transfers through `end` at all, static or
125 /// not; `bytes` alone cannot tell a dynamic end from an absent one.
126 bool names(unsigned seq, mlir::StringAttr end) const {
127 return spans[seq].contains(end);
128 }

146 /// How one shim tile's unpinned ends in one direction share channels, as
147 /// objectFIFO allocation shares them.
148 struct Plan {

155 /// Plans `ends` (keyed as apart() takes keys); `names[i]` is how `shared`'s
156 /// lists name `ends[i]`. Each channel in `shared`, in order, takes the ends
157 /// its list names that are apart from those it already took; the rest
158 /// group as groups() groups them. Indices are into `ends`.
159 Plan plan(llvm::ArrayRef<mlir::StringAttr> ends,
160 llvm::ArrayRef<mlir::StringAttr> names,

173 /// Bytes the sequence moves through the end; nullopt when not static.
174 std::optional<int64_t> bytes = 0;
175 };

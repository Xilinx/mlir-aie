8// A tile's buffer-descriptor ids are one shared pool, but the static tile
9// program and the runtime sequence get their ids from two separate passes that
10// do not see each other. This prepass records those compile-time-pinned runtime
11// ids per tile as an aiex.reserved_bd_ids attribute on each aie.tile.

37struct AIEReserveRuntimeBDIDsPass
38 : xilinx::AIEX::impl::AIEReserveRuntimeBDIDsBase<
39 AIEReserveRuntimeBDIDsPass> {
40
41 void runOnOperation() override {
42 DeviceOp device = getOperation();
43
44 llvm::DenseMap<TileOp, std::set<int32_t>> reserved;
45 device.walk([&](DMAConfigureTaskOp cfg) {
46 TileOp tile = cfg.tryGetTileOp();
47 if (!tile)
48 return;
49 cfg.walk([&](DMABDOp bd) {
50 if (bd.getBdId().has_value())
51 reserved[tile].insert(bd.getBdId().value());
52 });
53 });
54
55 for (auto &[tile, ids] : reserved) {
56 SmallVector<int32_t> sorted(ids.begin(), ids.end());
57 tile->setAttr("aiex.reserved_bd_ids",
58 DenseI32ArrayAttr::get(&getContext(), sorted));
59 }
60 }
61};

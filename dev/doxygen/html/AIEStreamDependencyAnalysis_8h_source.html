75/// One stream from a source tile port to a destination tile port, recovered
76/// from the physical switchbox and shim-mux configuration.
77struct RoutedStream {
78 StreamEndpoint src;
79 StreamEndpoint dst;
80 /// The packet id carried, if any.
81 std::optional<int> packetID;
82 /// The destination stores each packet's header along with its payload.
83 bool keepsPktHeader = false;
84 /// The switchboxes the stream passes, source first, where it is routed.
85 llvm::SmallVector<StreamHop, 8> hops{};
86 /// The stream carries every id that agrees with `packetID` on these bits.
87 int packetMask = ~0;
88
89 /// Whether what a BD sends with header `packet`, or with none, goes on this
90 /// stream.
91 bool carries(std::optional<PacketInfoAttr> packet) const {
92 return !packet || !packetID ||
93 ((packet->getPktId() ^ *packetID) & packetMask) == 0;
94 }
95};

97/// A cycle of waits packet streams can deadlock in, given their routes. A
98/// packet holds the arbiter of every switchbox it has entered until its last
99/// word leaves, so one stuck anywhere holds up whatever needs those arbiters,
100/// or queues behind it on a link.
101struct HoldCycle {
102 enum class Wait {
103 /// `waiting` queues behind `holding` on the port both enter `tile` by.
104 Link,
105 /// `waiting`, or `sharer` it queues behind, needs the arbiter `holding`
106 /// holds at `tile`; `sharer` enters by `sharerInput` and `holding` by
107 /// `holderInput`.
108 Arbiter,
109 /// `waiting` fills its receiver, and draining that waits on `holding`.
110 Drain,
111 };
112 struct Step {
113 Wait wait;
114 size_t waiting;
115 size_t sharer;
116 size_t holding;
117 TileID tile;
118 Port sharerInput;
119 Port holderInput;
120 int arbiter;
121 /// The wait holds however the arbiters are planned, as `sharer` and
122 /// `holding` go into a receiver they share at `tile`, or have one id and
123 /// go on as one from there.
124 bool forced = false;
125 };
126 llvm::SmallVector<Step, 4> steps;
127};

202/// Which agent waits on which. An agent is a core, one DMA channel, or the
203/// controller of a tile, which sends the task-complete tokens the host waits
204/// for on its TileControl port. P waits on Q when P acquires a lock Q
205/// releases, when P sends a stream Q receives or the reverse, or when P is a
206/// runtime-issued channel the host issues only after waiting on Q. Waiting on
207/// a channel waits on the controllers of its column too, whose tokens tell
208/// the host it is done. A channel with no program in the design may wait on
209/// anything on its tile. A receiving channel that takes in all it is sent
210/// before its locks run out waits on no lock. Program order within an agent
211/// is not modeled.
212class StreamWaitGraph {
213public:
214 /// A core, a DMA channel, or a tile's controller. A core's or controller's
215 /// `dma` names only its tile.
216 struct Agent {
217 enum class Kind { Channel, Core, Controller };
218 TileDMAChannel dma;
219 Kind kind = Kind::Channel;
220 bool onShim = false;
221
222 static Agent core(TileID tile) {
223 return {{tile, DMAChannelDir::S2MM, 0}, Kind::Core};
224 }
225 static Agent controller(TileID tile) {
226 return {{tile, DMAChannelDir::S2MM, 0}, Kind::Controller};
227 }
228 static Agent channel(const TileDMAChannel &dma) { return {dma}; }
229 /// Each agent at a stream endpoint: the core, DMA channel or controller
230 /// pushing data into (`sending`) or pulling data out of the fabric there.
231 static std::optional<Agent> at(const StreamEndpoint &endpoint,
232 bool sending);
233 };
234
235 StreamWaitGraph(DeviceOp device, llvm::ArrayRef<RoutedStream> streams,
236 const StreamVolumeAnalysis &volumes);
237
238 /// The agent pushing data into (`sending`) or pulling data out of the
239 /// fabric at `endpoint`, if the endpoint has one.
240 std::optional<unsigned> agentAt(const StreamEndpoint &endpoint,
241 bool sending) const;
242
243 /// Agents whose progress frees room in `agent` for more incoming data: a
244 /// core frees itself, a receiving channel waits on whoever releases the
245 /// locks it acquires and, if the host issues it, on what the host waits for
246 /// first.
247 llvm::SmallVector<unsigned> drainersOf(unsigned agent) const;
248
249 /// A shortest chain of waits from an agent in `from` to one in `targets`
250 /// that avoids `avoid`, both ends included; empty when there is none.
251 llvm::SmallVector<unsigned> waitChain(llvm::ArrayRef<unsigned> from,
252 llvm::ArrayRef<unsigned> targets,
253 llvm::ArrayRef<unsigned> avoid) const;
254
255 /// Whether the design programs this agent, so its waits are known rather
256 /// than assumed.
257 bool isModeled(unsigned id) const { return modeled.contains(id); }
258
259 const Agent &getAgent(unsigned id) const { return agents[id]; }
260 std::string describe(unsigned id) const;
261
262private:
263 enum class EdgeKind { Lock, Stream, Host };
264 struct Edge {
265 unsigned to;
266 EdgeKind kind;
267 };
268
269 unsigned getOrCreate(const Agent &agent);
270 std::optional<unsigned> lookup(const Agent &agent) const;
271 void addEdge(unsigned from, unsigned to, EdgeKind kind);

276 llvm::DenseSet<unsigned> modeled;
277};

316/// The streams a device asks for or already routes, and which pairs of them
317/// must not share an arbiter or a link. The analysis runs on the first query.
318class StreamConflicts {
319public:
320 explicit StreamConflicts(DeviceOp device);
321 StreamConflicts(const StreamConflicts &) = delete;
322 StreamConflicts &operator=(const StreamConflicts &) = delete;
323
324 llvm::ArrayRef<RoutedStream> getStreams() const { return streams; }
325
326 /// The streams the flow ops ask for, which come first in getStreams().
327 llvm::ArrayRef<RoutedStream> getRequestedStreams() const {
328 return llvm::ArrayRef(streams).take_front(numRequested);
329 }
330
331 /// Whether `s` and `t` can deadlock if they share an arbiter or a link.
332 /// Streams from one source are serialized there anyway, and streams into
333 /// one destination already wait on each other there, so neither conflicts.
334 /// The packets one source sends with one id move down every branch as one,
335 /// so the same holds for any two streams of such trees.
336 bool conflict(size_t s, size_t t) const;
337
338 /// Whether a routing that puts packet streams `s` and `t` on one arbiter
339 /// anywhere has a hold cycle: they conflict, or the tree of one, stuck at a
340 /// receiver, waits on the other through trees that each block the next.
341 bool mustSeparate(size_t s, size_t t) const;
342
343 /// Why `s` and `t` conflict. Requires mustSeparate(s, t), or (s, t) from
344 /// unavoidable().
345 std::string explain(size_t s, size_t t) const;
346
347 /// The requested streams where the first can hold up the second however
348 /// they are routed, as one source or receiver already orders them. Only
349 /// pairs with a packet stream count: circuits take no arbiter, so a wait
350 /// between two of them is the design's, not a hazard of any routing. Nor do
351 /// pairs that need StreamDeadlockAnalysis::assumptions: the routing cannot
352 /// help those either, and the design may well not do what is assumed.
353 llvm::SmallVector<std::pair<size_t, size_t>> unavoidable() const;
354
355 /// A cycle of waits the packet streams can deadlock in when routed along
356 /// `routes`, indexed like getStreams(), that the routing of some requested
357 /// stream takes part in; nullopt when there is none. Waits between trees
358 /// at the master port of a receiver they share hold however they are
359 /// routed, so they count only with `forcedWaits`. With `definite`, drains
360 /// that need StreamDeadlockAnalysis::assumptions do not count.
361 std::optional<HoldCycle>
362 holdCycle(llvm::ArrayRef<llvm::SmallVector<StreamHop, 8>> routes,
363 bool forcedWaits = false, bool definite = false) const;
364
365 /// The waits of `cycle`, one sentence each.
366 std::string explain(const HoldCycle &cycle) const;
367
368private:
369 bool blocks(size_t s, size_t t) const;
370 bool related(size_t s, size_t t) const;
371 /// What makes a packet stream one with the others of its tree: the same
372 /// source, the same id, and whether a flow op asks for it.
373 using TreeKey = std::tuple<TileID, Port, int, bool>;
374 /// The tree stream `s` belongs to; nullopt for a circuit stream.
375 std::optional<TreeKey> treeKey(size_t s) const;
376 /// The packet trees tree `a` waits on through trees that each block the
377 /// next, each with the pair of streams, the first of the tree before it,
378 /// that it was reached by.
379 const llvm::DenseMap<size_t, std::pair<size_t, size_t>> &
380 waitsFrom(size_t a) const;
381 const StreamDeadlockAnalysis &getAnalysis() const;
382
383 DeviceOp device;
384 std::vector<RoutedStream> streams;
385 size_t numRequested;
386 /// The streams of each tree, and the tree of each stream.
387 std::vector<llvm::SmallVector<size_t, 2>> treeMembers;
388 std::vector<size_t> treeOf;
389 mutable std::vector<
390 std::optional<llvm::DenseMap<size_t, std::pair<size_t, size_t>>>>
391 waits;
392 mutable std::optional<StreamDeadlockAnalysis> analysis;
393};
